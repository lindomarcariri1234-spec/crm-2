// lib/email/send-reservation-confirmation.ts

import { Resend } from 'resend'
import { ReservationConfirmationEmail } from './templates/reservation-confirmation'
import { prisma } from '@/lib/prisma'
import { format } from 'date-fns'
import { ptBR } from 'date-fns/locale'

const resend = new Resend(process.env.RESEND_API_KEY)

interface SendReservationConfirmationParams {
  reservationId: string
}

export async function sendReservationConfirmation({
  reservationId,
}: SendReservationConfirmationParams) {
  try {
    // Buscar reserva completa
    const reservation = await prisma.reservation.findUnique({
      where: { id: reservationId },
      include: {
        client: true,
        trip: true,
        seats: {
          orderBy: { seat_number: 'asc' },
        },
        tenant: true,
        payments: true,
      },
    })

    if (!reservation) {
      throw new Error('Reserva não encontrada')
    }

    // Formatar data
    const departureDate = format(
      new Date(reservation.departure_date),
      "dd 'de' MMMM 'de' yyyy 'às' HH:mm",
      { locale: ptBR }
    )

    // Calcular duração (exemplo: "6 dias / 5 noites")
    const duration = `${reservation.trip.duration_days} dias / ${reservation.trip.duration_days - 1} noites`

    // Extrair assentos
    const seats = reservation.seats.map((seat) => seat.seat_number)

    // Calcular valor pago
    const amountPaid = reservation.payments?.reduce(
      (sum, payment) => sum + Number(payment.amount),
      0
    ) || 0

    // URLs
    const baseUrl = process.env.NEXT_PUBLIC_APP_URL
    const voucherUrl = `${baseUrl}/voucher/${reservation.voucher_code}`
    const consultUrl = `${baseUrl}/consultar-pedido?numero=${reservation.reservation_number}`
    const whatsappUrl = `https://wa.me/${reservation.tenant.phone.replace(/\D/g, '')}?text=Olá! Fiz uma reserva (${reservation.reservation_number}) e gostaria de mais informações.`

    // Enviar email
    const { data, error } = await resend.emails.send({
      from: `${reservation.tenant.name} <reservas@${process.env.EMAIL_DOMAIN}>`,
      to: [reservation.client_email],
      subject: `✅ Reserva Confirmada - ${reservation.reservation_number}`,
      react: ReservationConfirmationEmail({
        reservationNumber: reservation.reservation_number,
        voucherCode: reservation.voucher_code,
        clientName: reservation.client_name,
        clientCpf: reservation.client_cpf,
        clientEmail: reservation.client_email,
        clientPhone: reservation.client_phone,
        tripTitle: reservation.trip.title,
        destination: reservation.trip.destination,
        departureDate,
        duration,
        seats,
        totalAmount: Number(reservation.total_amount),
        amountPaid,
        amountPending: Number(reservation.amount_pending),
        paymentMethod: reservation.payment_method,
        paymentStatus: reservation.payment_status as 'pending' | 'partial' | 'paid',
        agencyName: reservation.tenant.name,
        agencyLogo: reservation.tenant.logo_url || `${baseUrl}/logo.png`,
        agencyPhone: reservation.tenant.phone,
        agencyEmail: reservation.tenant.email,
        agencyWebsite: baseUrl,
        voucherUrl,
        consultUrl,
        whatsappUrl,
      }),
    })

    if (error) {
      console.error('Erro ao enviar email:', error)
      throw error
    }

    console.log('✅ Email enviado com sucesso:', data)

    // Registrar envio no banco
    await prisma.emailLog.create({
      data: {
        tenant_id: reservation.tenant_id,
        reservation_id: reservation.id,
        recipient: reservation.client_email,
        subject: `Reserva Confirmada - ${reservation.reservation_number}`,
        type: 'reservation_confirmation',
        status: 'sent',
        sent_at: new Date(),
      },
    })

    return { success: true, data }
  } catch (error) {
    console.error('Erro ao enviar email de confirmação:', error)
    
    // Registrar erro no banco
    await prisma.emailLog.create({
      data: {
        tenant_id: reservation?.tenant_id || '',
        reservation_id: reservationId,
        recipient: reservation?.client_email || '',
        subject: `Reserva Confirmada - ${reservation?.reservation_number}`,
        type: 'reservation_confirmation',
        status: 'failed',
        error_message: error instanceof Error ? error.message : 'Erro desconhecido',
      },
    })

    return { success: false, error }
  }
}

// prisma/schema.prisma

model EmailLog {
  id              String    @id @default(cuid())
  tenant_id       String
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  
  reservation_id  String?
  reservation     Reservation? @relation(fields: [reservation_id], references: [id])
  
  recipient       String    // Email do destinatário
  subject         String    // Assunto do email
  type            String    // reservation_confirmation, payment_reminder, etc.
  
  status          String    // sent, failed, bounced
  error_message   String?   // Mensagem de erro (se falhou)
  
  sent_at         DateTime?
  opened_at       DateTime? // Se foi aberto
  clicked_at      DateTime? // Se clicou em algum link
  
  created_at      DateTime  @default(now())
  
  @@index([tenant_id])
  @@index([reservation_id])
  @@index([recipient])
  @@map("email_logs")
}
