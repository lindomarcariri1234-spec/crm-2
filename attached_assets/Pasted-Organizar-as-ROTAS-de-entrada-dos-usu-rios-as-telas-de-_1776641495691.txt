// middleware.ts

import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'
import { getToken } from 'next-auth/jwt'

export async function middleware(request: NextRequest) {
  const token = await getToken({ req: request })

  // Proteger rotas /admin/*
  if (request.nextUrl.pathname.startsWith('/admin')) {
    if (!token) {
      return NextResponse.redirect(new URL('/auth/login', request.url))
    }

    // Verificar se é super admin
    if (token.role !== 'super_admin') {
      return NextResponse.redirect(new URL('/dashboard', request.url))
    }
  }

  return NextResponse.next()
}

export const config = {
  matcher: ['/admin/:path*'],
}

// app/api/admin/dashboard/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { startOfMonth, endOfMonth, subMonths } from 'date-fns'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session || session.user.role !== 'super_admin') {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    // ═══════════════════════════════════════════════════════════
    // MÉTRICAS PRINCIPAIS
    // ═══════════════════════════════════════════════════════════

    // Total de Tenants
    const totalTenants = await prisma.tenant.count()

    // Tenants por Status
    const tenantsByStatus = await prisma.tenant.groupBy({
      by: ['subscription_status'],
      _count: { id: true },
    })

    // MRR Total (Monthly Recurring Revenue)
    const mrrData = await prisma.tenant.aggregate({
      where: {
        subscription_status: { in: ['active', 'trial'] },
      },
      _sum: { mrr: true },
    })

    const totalMRR = mrrData._sum.mrr?.toNumber() || 0

    // ARR (Annual Recurring Revenue)
    const totalARR = totalMRR * 12

    // Novos Tenants este mês
    const currentMonthStart = startOfMonth(new Date())
    const currentMonthEnd = endOfMonth(new Date())

    const newTenantsThisMonth = await prisma.tenant.count({
      where: {
        created_at: {
          gte: currentMonthStart,
          lte: currentMonthEnd,
        },
      },
    })

    // Churn este mês
    const churnedThisMonth = await prisma.tenant.count({
      where: {
        churn_date: {
          gte: currentMonthStart,
          lte: currentMonthEnd,
        },
      },
    })

    // Taxa de Churn
    const churnRate = totalTenants > 0 ? (churnedThisMonth / totalTenants) * 100 : 0

    // Lifetime Value médio
    const avgLTV = await prisma.tenant.aggregate({
      _avg: { lifetime_value: true },
    })

    // Total de Usuários na plataforma
    const totalUsers = await prisma.user.count()

    // Total de Clientes
    const totalCustomers = await prisma.customer.count()

    // Total de Viagens
    const totalTrips = await prisma.trip.count()

    // Faturas Pendentes
    const pendingInvoices = await prisma.invoice.count({
      where: { status: 'pending' },
    })

    const overdueInvoices = await prisma.invoice.count({
      where: { status: 'overdue' },
    })

    // Receita Total
    const totalRevenue = await prisma.invoice.aggregate({
      where: { status: 'paid' },
      _sum: { total: true },
    })

    // ═══════════════════════════════════════════════════════════
    // CRESCIMENTO MENSAL (12 MESES)
    // ═══════════════════════════════════════════════════════════

    const monthlyGrowth = []

    for (let i = 11; i >= 0; i--) {
      const date = subMonths(new Date(), i)
      const monthStart = startOfMonth(date)
      const monthEnd = endOfMonth(date)

      // Novos tenants
      const newTenants = await prisma.tenant.count({
        where: {
          created_at: {
            gte: monthStart,
            lte: monthEnd,
          },
        },
      })

      // Churn
      const churned = await prisma.tenant.count({
        where: {
          churn_date: {
            gte: monthStart,
            lte: monthEnd,
          },
        },
      })

      // MRR do mês
      const monthlyMRR = await prisma.tenant.aggregate({
        where: {
          subscription_status: 'active',
          subscription_started_at: {
            lte: monthEnd,
          },
        },
        _sum: { mrr: true },
      })

      // Receita do mês
      const monthlyRevenue = await prisma.invoice.aggregate({
        where: {
          status: 'paid',
          paid_date: {
            gte: monthStart,
            lte: monthEnd,
          },
        },
        _sum: { total: true },
      })

      monthlyGrowth.push({
        month: date.toISOString(),
        newTenants,
        churned,
        mrr: monthlyMRR._sum.mrr?.toNumber() || 0,
        revenue: monthlyRevenue._sum.total?.toNumber() || 0,
      })
    }

    // ═══════════════════════════════════════════════════════════
    // DISTRIBUIÇÃO POR PLANO
    // ═══════════════════════════════════════════════════════════

    const tenantsByPlan = await prisma.tenant.groupBy({
      by: ['subscription_plan_id'],
      _count: { id: true },
      where: {
        subscription_plan_id: { not: null },
      },
    })

    const plansWithNames = await Promise.all(
      tenantsByPlan.map(async (item) => {
        const plan = await prisma.subscriptionPlan.findUnique({
          where: { id: item.subscription_plan_id! },
          select: { name: true, monthly_price: true },
        })

        return {
          plan: plan?.name || 'Desconhecido',
          count: item._count.id,
          price: plan?.monthly_price?.toNumber() || 0,
        }
      })
    )

    // ═══════════════════════════════════════════════════════════
    // TOP 10 TENANTS POR RECEITA
    // ═══════════════════════════════════════════════════════════

    const topTenants = await prisma.tenant.findMany({
      orderBy: { lifetime_value: 'desc' },
      take: 10,
      select: {
        id: true,
        name: true,
        slug: true,
        subscription_status: true,
        mrr: true,
        lifetime_value: true,
        created_at: true,
        subscription_plan: {
          select: { name: true },
        },
      },
    })

    // ═══════════════════════════════════════════════════════════
    // ATIVIDADE RECENTE
    // ═══════════════════════════════════════════════════════════

    const recentActivity = await prisma.systemLog.findMany({
      orderBy: { created_at: 'desc' },
      take: 20,
      include: {
        tenant: {
          select: { name: true },
        },
        user: {
          select: { name: true, email: true },
        },
      },
    })

    return NextResponse.json({
      metrics: {
        totalTenants,
        tenantsByStatus,
        totalMRR,
        totalARR,
        newTenantsThisMonth,
        churnedThisMonth,
        churnRate,
        avgLTV: avgLTV._avg.lifetime_value?.toNumber() || 0,
        totalUsers,
        totalCustomers,
        totalTrips,
        pendingInvoices,
        overdueInvoices,
        totalRevenue: totalRevenue._sum.total?.toNumber() || 0,
      },
      monthlyGrowth,
      tenantsByPlan: plansWithNames,
      topTenants,
      recentActivity,
    })
  } catch (error) {
    console.error('Erro ao buscar dashboard:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar dados' },
      { status: 500 }
    )
  }
}

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session || session.user.role !== 'super_admin') {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    const { searchParams } = new URL(req.url)
    const page = parseInt(searchParams.get('page') || '1')
    const limit = parseInt(searchParams.get('limit') || '20')
    const search = searchParams.get('search') || ''
    const status = searchParams.get('status') || 'all'
    const plan = searchParams.get('plan') || 'all'

    const skip = (page - 1) * limit

    const where: any = {}

    if (search) {
      where.OR = [
        { name: { contains: search, mode: 'insensitive' } },
        { slug: { contains: search, mode: 'insensitive' } },
        { cnpj: { contains: search } },
      ]
    }

    if (status !== 'all') {
      where.subscription_status = status
    }

    if (plan !== 'all') {
      where.subscription_plan_id = plan
    }

    const [tenants, total] = await Promise.all([
      prisma.tenant.findMany({
        where,
        skip,
        take: limit,
        orderBy: { created_at: 'desc' },
        include: {
          subscription_plan: {
            select: { name: true },
          },
          _count: {
            select: {
              users: true,
              customers: true,
              trips: true,
            },
          },
        },
      }),
      prisma.tenant.count({ where }),
    ])

    return NextResponse.json({
      tenants,
      pagination: {
        total,
        page,
        limit,
        totalPages: Math.ceil(total / limit),
      },
    })
  } catch (error) {
    console.error('Erro ao buscar tenants:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar tenants' },
      { status: 500 }
    )
  }
}

// Atualizar Status do Tenant
export async function PATCH(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session || session.user.role !== 'super_admin') {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    const body = await req.json()
    const { tenantId, action, reason, planId } = body

    if (action === 'suspend') {
      await prisma.tenant.update({
        where: { id: tenantId },
        data: {
          subscription_status: 'suspended',
          is_suspended: true,
          suspension_reason: reason,
        },
      })

      // Log
      await prisma.systemLog.create({
        data: {
          tenant_id: tenantId,
          user_id: session.user.id,
          action: 'suspend_tenant',
          entity_type: 'tenant',
          entity_id: tenantId,
          description: `Tenant suspenso. Motivo: ${reason}`,
          severity: 'warning',
          category: 'tenant',
        },
      })

      return NextResponse.json({ success: true })
    }

    if (action === 'activate') {
      await prisma.tenant.update({
        where: { id: tenantId },
        data: {
          subscription_status: 'active',
          is_suspended: false,
          is_active: true,
          suspension_reason: null,
        },
      })

      // Log
      await prisma.systemLog.create({
        data: {
          tenant_id: tenantId,
          user_id: session.user.id,
          action: 'activate_tenant',
          entity_type: 'tenant',
          entity_id: tenantId,
          description: 'Tenant ativado',
          severity: 'info',
          category: 'tenant',
        },
      })

      return NextResponse.json({ success: true })
    }

    if (action === 'change_plan') {
      const plan = await prisma.subscriptionPlan.findUnique({
        where: { id: planId },
      })

      if (!plan) {
        return NextResponse.json({ error: 'Plano não encontrado' }, { status: 404 })
      }

      await prisma.tenant.update({
        where: { id: tenantId },
        data: {
          subscription_plan_id: planId,
          mrr: plan.monthly_price,
        },
      })

      // Log
      await prisma.systemLog.create({
        data: {
          tenant_id: tenantId,
          user_id: session.user.id,
          action: 'change_plan',
          entity_type: 'tenant',
          entity_id: tenantId,
          description: `Plano alterado para ${plan.name}`,
          severity: 'info',
          category: 'billing',
        },
      })

      return NextResponse.json({ success: true })
    }

    return NextResponse.json({ error: 'Ação inválida' }, { status: 400 })
  } catch (error) {
    console.error('Erro ao atualizar tenant:', error)
    return NextResponse.json(
      { error: 'Erro ao atualizar tenant' },
      { status: 500 }
    )
  }
}

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'

import { prisma } from '@/lib/prisma'
import { startOfMonth, endOfMonth, addMonths } from 'date-fns'

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session || session.user.role !== 'super_admin') {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    // Buscar tenants ativos que precisam de fatura
    const tenants = await prisma.tenant.findMany({
      where: {
        subscription_status: 'active',
        subscription_plan_id: { not: null },
        OR: [
          { next_invoice_date: { lte: new Date() } },
          { next_invoice_date: null },
        ],
      },
      include: {
        subscription_plan: true,
      },
    })

    const invoicesCreated = []
    const errors = []

    for (const tenant of tenants) {
      try {
        if (!tenant.subscription_plan) continue

        // Gerar número da fatura
        const invoiceCount = await prisma.invoice.count()
        const invoiceNumber = `INV-${new Date().getFullYear()}-${String(
          invoiceCount + 1
        ).padStart(6, '0')}`

        // Calcular valores
        const subtotal = tenant.subscription_plan.monthly_price
        const tax = subtotal.toNumber() * 0.0
        const total = subtotal.toNumber() + tax

        // Período
        const periodStart = startOfMonth(new Date())
        const periodEnd = endOfMonth(new Date())
        const dueDate = addMonths(new Date(), 0)

        const invoice = await prisma.invoice.create({
          data: {
            tenant_id: tenant.id,
            invoice_number: invoiceNumber,
            subtotal,
            discount: 0,
            tax,
            total,
            period_start: periodStart,
            period_end: periodEnd,
            issue_date: new Date(),
            due_date: dueDate,
            status: 'pending',
          },
        })

        // Atualizar tenant
        await prisma.tenant.update({
          where: { id: tenant.id },
          data: {
            last_invoice_date: new Date(),
            next_invoice_date: addMonths(new Date(), 1),
          },
        })

        invoicesCreated.push({
          tenantName: tenant.name,
          invoiceNumber,
          total,
        })

        // Log
        await prisma.systemLog.create({
          data: {
            tenant_id: tenant.id,
            user_id: session.user.id,
            action: 'create_invoice_batch',
            entity_type: 'invoice',
            entity_id: invoice.id,
            description: `Fatura ${invoiceNumber} gerada automaticamente`,
            severity: 'info',
            category: 'billing',
          },
        })
      } catch (error) {
        errors.push({
          tenantName: tenant.name,
          error: error instanceof Error ? error.message : 'Erro desconhecido',
        })
      }
    }

    return NextResponse.json({
      success: true,
      invoicesCreated: invoicesCreated.length,
      errors: errors.length,
      details: {
        invoicesCreated,
        errors,
      },
    })
  } catch (error) {
    console.error('Erro ao gerar faturas em lote:', error)
    return NextResponse.json(
      { error: 'Erro ao gerar faturas' },
      { status: 500 }
    )
  }
}

// app/api/admin/metrics/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { startOfMonth, endOfMonth, subMonths, differenceInDays } from 'date-fns'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session || session.user.role !== 'super_admin') {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    const { searchParams } = new URL(req.url)
    const months = parseInt(searchParams.get('months') || '12')

    // ═══════════════════════════════════════════════════════════
    // CRESCIMENTO MENSAL
    // ═══════════════════════════════════════════════════════════

    const monthlyData = []

    for (let i = months - 1; i >= 0; i--) {
      const date = subMonths(new Date(), i)
      const monthStart = startOfMonth(date)
      const monthEnd = endOfMonth(date)

      // Novos tenants
      const newTenants = await prisma.tenant.count({
        where: {
          created_at: { gte: monthStart, lte: monthEnd },
        },
      })

      // Tenants ativos no final do mês
      const activeTenants = await prisma.tenant.count({
        where: {
          created_at: { lte: monthEnd },
          subscription_status: { in: ['active', 'trial'] },
          OR: [
            { churn_date: null },
            { churn_date: { gt: monthEnd } },
          ],
        },
      })

      // Churn
      const churned = await prisma.tenant.count({
        where: {
          churn_date: { gte: monthStart, lte: monthEnd },
        },
      })

      // MRR
      const mrrData = await prisma.tenant.aggregate({
        where: {
          subscription_status: 'active',
          subscription_started_at: { lte: monthEnd },
          OR: [
            { churn_date: null },
            { churn_date: { gt: monthEnd } },
          ],
        },
        _sum: { mrr: true },
      })

      // Receita do mês
      const revenue = await prisma.invoice.aggregate({
        where: {
          status: 'paid',
          paid_date: { gte: monthStart, lte: monthEnd },
        },
        _sum: { total: true },
      })

      // Novos usuários
      const newUsers = await prisma.user.count({
        where: {
          created_at: { gte: monthStart, lte: monthEnd },
        },
      })

      // Novos clientes
      const newCustomers = await prisma.customer.count({
        where: {
          created_at: { gte: monthStart, lte: monthEnd },
        },
      })

      // Taxa de churn
      const churnRate = activeTenants > 0 ? (churned / activeTenants) * 100 : 0

      monthlyData.push({
        month: date.toISOString(),
        newTenants,
        activeTenants,
        churned,
        churnRate,
        mrr: mrrData._sum.mrr?.toNumber() || 0,
        revenue: revenue._sum.total?.toNumber() || 0,
        newUsers,
        newCustomers,
      })
    }

    // ═══════════════════════════════════════════════════════════
    // COHORT ANALYSIS
    // ═══════════════════════════════════════════════════════════

    const cohorts = []

    for (let i = 11; i >= 0; i--) {
      const cohortDate = subMonths(new Date(), i)
      const cohortStart = startOfMonth(cohortDate)
      const cohortEnd = endOfMonth(cohortDate)

      // Tenants que começaram neste mês
      const cohortTenants = await prisma.tenant.findMany({
        where: {
          created_at: { gte: cohortStart, lte: cohortEnd },
        },
        select: {
          id: true,
          churn_date: true,
        },
      })

      const cohortSize = cohortTenants.length

      if (cohortSize === 0) continue

      // Calcular retenção mês a mês
      const retention = []

      for (let month = 0; month <= i; month++) {
        const checkDate = endOfMonth(subMonths(new Date(), i - month))

        const retained = cohortTenants.filter((t) => {
          return !t.churn_date || new Date(t.churn_date) > checkDate
        }).length

        retention.push({
          month,
          retained,
          percentage: (retained / cohortSize) * 100,
        })
      }

      cohorts.push({
        cohortMonth: cohortDate.toISOString(),
        cohortSize,
        retention,
      })
    }

    // ═══════════════════════════════════════════════════════════
    // CUSTOMER LIFETIME VALUE (LTV)
    // ═══════════════════════════════════════════════════════════

    const allTenants = await prisma.tenant.findMany({
      select: {
        created_at: true,
        churn_date: true,
        lifetime_value: true,
      },
    })

    const ltvData = allTenants.map((tenant) => {
      const lifetimeDays = tenant.churn_date
        ? differenceInDays(new Date(tenant.churn_date), new Date(tenant.created_at))
        : differenceInDays(new Date(), new Date(tenant.created_at))

      const lifetimeMonths = lifetimeDays / 30

      return {
        lifetimeMonths,
        ltv: tenant.lifetime_value.toNumber(),
      }
    })

    const avgLTV =
      ltvData.reduce((sum, t) => sum + t.ltv, 0) / (ltvData.length || 1)

    const avgLifetimeMonths =
      ltvData.reduce((sum, t) => sum + t.lifetimeMonths, 0) / (ltvData.length || 1)

    // ═══════════════════════════════════════════════════════════
    // DISTRIBUIÇÃO GEOGRÁFICA
    // ═══════════════════════════════════════════════════════════

    const geographicDistribution = await prisma.tenant.groupBy({
      by: ['state'],
      _count: { id: true },
      where: {
        state: { not: null },
      },
      orderBy: {
        _count: { id: 'desc' },
      },
    })

    // ═══════════════════════════════════════════════════════════
    // ENGAGEMENT METRICS
    // ═══════════════════════════════════════════════════════════

    const currentMonthStart = startOfMonth(new Date())

    // Tenants ativos (com atividade no último mês)
    const activeTenantsThisMonth = await prisma.systemLog.groupBy({
      by: ['tenant_id'],
      where: {
        created_at: { gte: currentMonthStart },
        tenant_id: { not: null },
      },
      _count: { id: true },
    })

    // Taxa de adoção de features
    const tenantsWithGoogleCalendar = await prisma.user.count({
      where: {
        google_calendar_enabled: true,
      },
      distinct: ['tenant_id'],
    })

    const tenantsWithWhatsApp = await prisma.tenant.count({
      where: {
        whatsapp_enabled: true,
      },
    })

    const totalActiveTenants = await prisma.tenant.count({
      where: {
        subscription_status: { in: ['active', 'trial'] },
      },
    })

    return NextResponse.json({
      monthlyData,
      cohorts,
      ltv: {
        average: avgLTV,
        averageLifetimeMonths: avgLifetimeMonths,
        distribution: ltvData,
      },
      geographic: geographicDistribution,
      engagement: {
        activeTenantsThisMonth: activeTenantsThisMonth.length,
        totalActiveTenants,
        engagementRate:
          (activeTenantsThisMonth.length / (totalActiveTenants || 1)) * 100,
        featureAdoption: {
          googleCalendar: {
            count: tenantsWithGoogleCalendar,
            percentage: (tenantsWithGoogleCalendar / (totalActiveTenants || 1)) * 100,
          },
          whatsapp: {
            count: tenantsWithWhatsApp,
            percentage: (tenantsWithWhatsApp / (totalActiveTenants || 1)) * 100,
          },
        },
      },
    })
  } catch (error) {
    console.error('Erro ao buscar métricas:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar métricas' },
      { status: 500 }
    )
  }
}

import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session || session.user.role !== 'super_admin') {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    const { searchParams } = new URL(req.url)
    const page = parseInt(searchParams.get('page') || '1')
    const limit = parseInt(searchParams.get('limit') || '50')
    const search = searchParams.get('search') || ''
    const role = searchParams.get('role') || 'all'
    const tenantId = searchParams.get('tenantId')

    const skip = (page - 1) * limit

    const where: any = {}

    if (search) {
      where.OR = [
        { name: { contains: search, mode: 'insensitive' } },
        { email: { contains: search, mode: 'insensitive' } },
      ]
    }

    if (role !== 'all') {
      where.role = role
    }

    if (tenantId) {
      where.tenant_id = tenantId
    }

    const [users, total] = await Promise.all([
      prisma.user.findMany({
        where,
        skip,
        take: limit,
        orderBy: { created_at: 'desc' },
        include: {
          tenant: {
            select: {
              name: true,
              slug: true,
            },
          },
        },
      }),
      prisma.user.count({ where }),
    ])

    // Estatísticas
    const stats = await prisma.user.groupBy({
      by: ['role'],
      _count: { id: true },
    })

    return NextResponse.json({
      users,
      stats,
      pagination: {
        total,
        page,
        limit,
        totalPages: Math.ceil(total / limit),
      },
    })
  } catch (error) {
    console.error('Erro ao buscar usuários:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar usuários' },

    )

                </>
              ) : (
                'Alterar Plano'
              )}

                </span>
                <span className="text-gray-600">/mês</span>
              </div>
              {plan.yearly_price && (
                <p className="text-sm text-gray-500 mt-1">
                  ou {formatCurrency(plan.yearly_price.toNumber())}/ano
                </p>
              )}
            </div>

            {/* Descrição */}
            {plan.description && (
              <p className="text-sm text-gray-600 mb-4">{plan.description}</p>
            )}

            {/* Limites */}
            <div className="space-y-3 mb-6">
              <div className="flex items-center gap-2 text-sm">
                <Users className="h-4 w-4 text-gray-400" />
                <span>Até {plan.max_users} usuários</span>
              </div>
              <div className="flex items-center gap-2 text-sm">
                <UserPlus className="h-4 w-4 text-gray-400" />
                <span>Até {plan.max_customers} clientes</span>
              </div>
              <div className="flex items-center gap-2 text-sm">
                <Bus className="h-4 w-4 text-gray-400" />
                <span>Até {plan.max_trips} viagens</span>
              </div>
              <div className="flex items-center gap-2 text-sm">
                <HardDrive className="h-4 w-4 text-gray-400" />

              </div>
            </div>

            {/* Features */}
            {plan.features && Array.isArray(plan.features) && plan.features.length > 0 && (
              <div className="mb-6">
                <p className="text-sm font-semibold mb-2">Features:</p>
                <ul className="space-y-2">
                  {plan.features.map((feature: string, index: number) => (
                    <li key={index} className="flex items-start gap-2 text-sm">
                      <Check className="h-4 w-4 text-green-600 mt-0.5 flex-shrink-0" />
                      <span>{feature}</span>
                    </li>
                  ))}
                </ul>
              </div>
            )}

            {/* Info */}
            <div className="pt-4 border-t space-y-2 text-xs text-gray-500">
              <p>Trial: {plan.trial_days} dias</p>
              <p>Tenants ativos: {plan._count.tenants}</p>
            </div>

            {/* Ações */}
            <div className="mt-6 flex gap-2">
              <Button
                variant="outline"
                size="sm"
                className="flex-1"
                onClick={() => handleOpenModal(plan)}
              >
                <Edit className="h-4 w-4 mr-2" />
                Editar

          </Card>
        ))}
      </div>

      {/* Modal de Criação/Edição */}
      <Dialog open={isModalOpen} onOpenChange={setIsModalOpen}>
        <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
          <DialogHeader>
            <DialogTitle>
              {editingPlan ? 'Editar Plano' : 'Novo Plano'}
            </DialogTitle>
            <DialogDescription>

            </DialogDescription>
          </DialogHeader>

          <div className="space-y-4 py-4">
            {/* Informações Básicas */}
            <div className="grid grid-cols-2 gap-4">
              <div>
                <Label htmlFor="name">Nome *</Label>
                <Input
                  id="name"
                  value={formData.name}
                  onChange={(e) =>
                    setFormData({ ...formData, name: e.target.value })
                  }
                  placeholder="Ex: Starter"
                />
              </div>

              <div>
                <Label htmlFor="slug">Slug *</Label>
                <Input
                  id="slug"
                  value={formData.slug}
                  onChange={(e) =>
                    setFormData({ ...formData, slug: e.target.value })
                  }
                  placeholder="Ex: starter"
                />
              </div>
            </div>

            <div>
              <Label htmlFor="description">Descrição</Label>
              <Textarea
                id="description"
                value={formData.description}
                onChange={(e) =>
                  setFormData({ ...formData, description: e.target.value })
                }
                placeholder="Descrição do plano..."
                rows={3}
              />
            </div>

            {/* Preços */}
            <div className="grid grid-cols-3 gap-4">
              <div>
                <Label htmlFor="monthly_price">Preço Mensal *</Label>
                <Input
                  id="monthly_price"
                  type="number"
                  step="0.01"
                  value={formData.monthly_price}
                  onChange={(e) =>
                    setFormData({ ...formData, monthly_price: e.target.value })

                  placeholder="0.00"
                />
              </div>

              <div>
                <Label htmlFor="yearly_price">Preço Anual</Label>
                <Input
                  id="yearly_price"
                  type="number"
                  step="0.01"
                  value={formData.yearly_price}
                  onChange={(e) =>
                    setFormData({ ...formData, yearly_price: e.target.value })
                  }
                  placeholder="0.00"
                />
              </div>

              <div>
                <Label htmlFor="setup_fee">Taxa de Setup</Label>
                <Input
                  id="setup_fee"
                  type="number"
                  step="0.01"
                  value={formData.setup_fee}

                    setFormData({ ...formData, setup_fee: e.target.value })
                  }

              <div>
                <Label htmlFor="max_storage_gb">Armazenamento (GB) *</Label>
                <Input
                  id="max_storage_gb"
                  type="number"
                  value={formData.max_storage_gb}
                  onChange={(e) =>
                    setFormData({
                      ...formData,
                      max_storage_gb: e.target.value,
                    })
                  }
                />
              </div>
            </div>

              </div>

              <div>
                <Label htmlFor="display_order">Ordem de Exibição</Label>
                <Input
                  id="display_order"
                  type="number"
                  value={formData.display_order}
                  onChange={(e) =>
                    setFormData({ ...formData, display_order: e.target.value })
                  }
                />
              </div>
            </div>

            {/* Switches */}
            <div className="flex items-center justify-between p-4 border rounded-lg">
              <Label htmlFor="is_active">Plano Ativo</Label>
              <Switch
                id="is_active"
                checked={formData.is_active}
                onCheckedChange={(checked) =>
                  setFormData({ ...formData, is_active: checked })
                }
              />
            </div>

            <div className="flex items-center justify-between p-4 border rounded-lg">
              <Label htmlFor="is_visible">Visível para Novas Assinaturas</Label>
              <Switch
                id="is_visible"
                checked={formData.is_visible}
                onCheckedChange={(checked) =>
                  setFormData({ ...formData, is_visible: checked })
                }
              />
            </div>
          </div>

          <DialogFooter>
            <Button
              variant="outline"
              onClick={() => setIsModalOpen(false)}
            >
              Cancelar
            </Button>
            <Button onClick={handleSave} disabled={isProcessing}>
              {isProcessing ? (
                <>
                  <Loader2 className="h-4 w-4 mr-2 animate-spin" />
                  Salvando...
                </>
              ) : (
                'Salvar Plano'
              )}
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </div>
  )
}

'use client'

import { useState, useEffect } from 'react'
import { Card } from '@/components/ui/card'
import { Button } from '@/components/ui/button'
import { Badge } from '@/components/ui/badge'
import { Input } from '@/components/ui/input'
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from '@/components/ui/select'
import {
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
} from '@/components/ui/table'
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog'
import { Label } from '@/components/ui/label'
import {
  DollarSign,
  Search,
  FileText,
  CheckCircle,
  XCircle,
  Clock,
  AlertCircle,
  Loader2,
  Download,
  Send,
  Zap,
} from 'lucide-react'
import { toast } from 'sonner'
import { format } from 'date-fns'
import { ptBR } from 'date-fns/locale'

export default function BillingPage() {
  const [isLoading, setIsLoading] = useState(true)
  const [invoices, setInvoices] = useState<any[]>([])
  const [stats, setStats] = useState<any>(null)
  const [pagination, setPagination] = useState<any>(null)
  const [searchTerm, setSearchTerm] = useState('')
  const [statusFilter, setStatusFilter] = useState('all')
  const [isGeneratingBatch, setIsGeneratingBatch] = useState(false)

  // Modal de atualização de status
  const [updateModal, setUpdateModal] = useState<{
    open: boolean
    invoice: any
  }>({
    open: false,
    invoice: null,
  })
  const [newStatus, setNewStatus] = useState('')
  const [paymentMethod, setPaymentMethod] = useState('')
  const [isProcessing, setIsProcessing] = useState(false)

  useEffect(() => {
    loadData()
  }, [statusFilter])

  const loadData = async () => {
    try {
      setIsLoading(true)
      const params = new URLSearchParams({ status: statusFilter })
      const response = await fetch(`/api/admin/billing?${params}`)
      const data = await response.json()

      setInvoices(data.invoices)
      setStats(data.stats)
      setPagination(data.pagination)
    } catch (error) {
      console.error('Erro ao carregar faturas:', error)
      toast.error('Erro ao carregar faturas')
    } finally {
      setIsLoading(false)
    }
  }

  const handleGenerateBatch = async () => {
    if (
      !confirm(
        'Deseja gerar faturas para todos os tenants ativos? Esta ação não pode ser desfeita.'
      )
    ) {
      return
    }

    try {
      setIsGeneratingBatch(true)
      toast.info('Gerando faturas em lote...')

      const response = await fetch('/api/admin/billing/generate-batch', {
        method: 'POST',
      })

      const data = await response.json()

      if (response.ok) {
        toast.success(
          `${data.invoicesCreated} faturas geradas com sucesso!`
        )
        if (data.errors > 0) {
          toast.warning(`${data.errors} erros durante a geração`)
        }
        loadData()
      } else {
        toast.error('Erro ao gerar faturas')
      }
    } catch (error) {
      console.error('Erro:', error)
      toast.error('Erro ao gerar faturas')
    } finally {
      setIsGeneratingBatch(false)
    }
  }

  const handleUpdateStatus = async () => {
    if (!newStatus) {
      toast.error('Selecione um status')
      return
    }

    try {
      setIsProcessing(true)

      const response = await fetch('/api/admin/billing', {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          invoiceId: updateModal.invoice.id,
          status: newStatus,
          paymentMethod: newStatus === 'paid' ? paymentMethod : undefined,
        }),
      })

      if (response.ok) {
        toast.success('Status atualizado com sucesso!')
        setUpdateModal({ open: false, invoice: null })
        setNewStatus('')
        setPaymentMethod('')
        loadData()
      } else {
        toast.error('Erro ao atualizar status')
      }
    } catch (error) {
      console.error('Erro:', error)
      toast.error('Erro ao atualizar status')
    } finally {
      setIsProcessing(false)
    }
  }

  const formatCurrency = (value: number) => {
    return new Intl.NumberFormat('pt-BR', {
      style: 'currency',
      currency: 'BRL',
    }).format(value)
  }

  const statusColors: Record<string, string> = {
    pending: 'bg-yellow-100 text-yellow-800',
    paid: 'bg-green-100 text-green-800',
    overdue: 'bg-red-100 text-red-800',
    cancelled: 'bg-gray-100 text-gray-800',
  }

  const statusIcons: Record<string, any> = {
    pending: Clock,
    paid: CheckCircle,
    overdue: AlertCircle,
    cancelled: XCircle,
  }

  const filteredInvoices = invoices.filter(
    (invoice) =>
      invoice.tenant.name.toLowerCase().includes(searchTerm.toLowerCase()) ||
      invoice.invoice_number.toLowerCase().includes(searchTerm.toLowerCase())
  )

  return (
    <div className="space-y-6">
      {/* Header */}
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-3xl font-bold text-gray-900">Faturamento</h1>
          <p className="text-gray-600 mt-1">
            Gestão de faturas e pagamentos da plataforma
          </p>
        </div>
        <Button onClick={handleGenerateBatch} disabled={isGeneratingBatch}>
          {isGeneratingBatch ? (
            <>
              <Loader2 className="h-4 w-4 mr-2 animate-spin" />
              Gerando...
            </>
          ) : (
            <>
              <Zap className="h-4 w-4 mr-2" />
              Gerar Faturas em Lote
            </>
          )}
        </Button>
      </div>

      {/* Estatísticas */}
      {stats && (
        <div className="grid grid-cols-1 md:grid-cols-4 gap-6">
          {stats.map((stat: any) => {
            const StatusIcon = statusIcons[stat.status] || FileText
            const total = stat._sum.total || 0
            const count = stat._count.id

            return (
              <Card key={stat.status} className="p-6">
                <div className="flex items-center justify-between mb-4">
                  <div
                    className={`p-3 rounded-lg ${
                      stat.status === 'paid'
                        ? 'bg-green-100'
                        : stat.status === 'overdue'
                        ? 'bg-red-100'
                        : stat.status === 'pending'
                        ? 'bg-yellow-100'
                        : 'bg-gray-100'
                    }`}
                  >
                    <StatusIcon
                      className={`h-6 w-6 ${
                        stat.status === 'paid'
                          ? 'text-green-600'
                          : stat.status === 'overdue'
                          ? 'text-red-600'
                          : stat.status === 'pending'
                          ? 'text-yellow-600'
                          : 'text-gray-600'
                      }`}
                    />
                  </div>
                  <Badge className={statusColors[stat.status]}>
                    {count} {count === 1 ? 'fatura' : 'faturas'}
                  </Badge>
                </div>
                <p className="text-sm text-gray-600 mb-1 capitalize">
                  {stat.status === 'paid' && 'Pagas'}
                  {stat.status === 'pending' && 'Pendentes'}
                  {stat.status === 'overdue' && 'Vencidas'}
                  {stat.status === 'cancelled' && 'Canceladas'}
                </p>
                <p className="text-3xl font-bold">
                  {formatCurrency(total)}
                </p>
              </Card>
            )
          })}
        </div>
      )}

      {/* Filtros */}
      <Card className="p-6">
        <div className="flex flex-col md:flex-row gap-4">
          <div className="flex-1 relative">
            <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 h-4 w-4 text-gray-400" />
            <Input
              placeholder="Buscar por agência ou número da fatura..."
              value={searchTerm}
              onChange={(e) => setSearchTerm(e.target.value)}
              className="pl-10"
            />
          </div>

          <Select value={statusFilter} onValueChange={setStatusFilter}>
            <SelectTrigger className="w-full md:w-48">
              <SelectValue placeholder="Status" />
            </SelectTrigger>
            <SelectContent>
              <SelectItem value="all">Todos Status</SelectItem>
              <SelectItem value="pending">Pendentes</SelectItem>
              <SelectItem value="paid">Pagas</SelectItem>
              <SelectItem value="overdue">Vencidas</SelectItem>
              <SelectItem value="cancelled">Canceladas</SelectItem>
            </SelectContent>
          </Select>
        </div>
      </Card>

      {/* Tabela de Faturas */}
      <Card className="p-6">
        {isLoading ? (
          <div className="flex items-center justify-center py-12">
            <Loader2 className="h-8 w-8 animate-spin text-blue-600" />
          </div>
        ) : (
          <div className="overflow-x-auto">
            <Table>
              <TableHeader>
                <TableRow>
                  <TableHead>Número</TableHead>
                  <TableHead>Agência</TableHead>
                  <TableHead>Período</TableHead>
                  <TableHead className="text-right">Subtotal</TableHead>
                  <TableHead className="text-right">Desconto</TableHead>
                  <TableHead className="text-right">Total</TableHead>
                  <TableHead>Vencimento</TableHead>
                  <TableHead>Status</TableHead>
                  <TableHead className="text-right">Ações</TableHead>
                </TableRow>
              </TableHeader>
              <TableBody>
                {filteredInvoices.map((invoice) => {
                  const StatusIcon = statusIcons[invoice.status]
                  const isOverdue =
                    invoice.status === 'pending' &&
                    new Date(invoice.due_date) < new Date()

                  return (
                    <TableRow key={invoice.id}>
                      <TableCell>
                        <div className="flex items-center gap-2">
                          <FileText className="h-4 w-4 text-gray-400" />
                          <span className="font-mono text-sm">
                            {invoice.invoice_number}
                          </span>
                        </div>
                      </TableCell>
                      <TableCell>
                        <div>
                          <p className="font-semibold">{invoice.tenant.name}</p>
                          <p className="text-xs text-gray-500">
                            @{invoice.tenant.slug}
                          </p>
                        </div>
                      </TableCell>
                      <TableCell>
                        <div className="text-sm">
                          <p>
                            {format(
                              new Date(invoice.period_start),
                              'dd/MM/yyyy',
                              { locale: ptBR }
                            )}
                          </p>
                          <p className="text-gray-500">
                            até{' '}
                            {format(
                              new Date(invoice.period_end),
                              'dd/MM/yyyy',
                              { locale: ptBR }
                            )}
                          </p>
                        </div>
                      </TableCell>
                      <TableCell className="text-right">
                        {formatCurrency(invoice.subtotal.toNumber())}
                      </TableCell>
                      <TableCell className="text-right">
                        {invoice.discount.toNumber() > 0 ? (
                          <span className="text-green-600">
                            -{formatCurrency(invoice.discount.toNumber())}
                          </span>
                        ) : (
                          '-'
                        )}
                      </TableCell>
                      <TableCell className="text-right font-bold text-blue-600">
                        {formatCurrency(invoice.total.toNumber())}
                      </TableCell>
                      <TableCell>
                        <div className="text-sm">
                          <p
                            className={
                              isOverdue ? 'text-red-600 font-semibold' : ''

                          >
                            {format(
                              new Date(invoice.due_date),

                          {invoice.paid_date && (
                            <p className="text-gray-500">
                              Pago em{' '}

                                new Date(invoice.paid_date),

                                { locale: ptBR }

                          >

                  <h3 className="text-xl font-bold">{flag.name}</h3>
                  {flag.is_enabled ? (
                    <Badge className="bg-green-600">
                      <CheckCircle className="h-3 w-3 mr-1" />
                      Ativa
                    </Badge>
                  ) : (
                    <Badge variant="secondary">
                      <XCircle className="h-3 w-3 mr-1" />
                      Inativa
                    </Badge>
                  )}
                </div>

                <p className="text-sm text-gray-600 mb-3 font-mono bg-gray-100 px-2 py-1 rounded inline-block">
                  {flag.key}
                </p>

                {flag.description && (
                  <p className="text-sm text-gray-700 mb-4">
                    {flag.description}
                  </p>
                )}

                {/* Rollout */}
                <div className="flex items-center gap-6 mb-3">
                  <div className="flex items-center gap-2">
                    <Percent className="h-4 w-4 text-gray-400" />
                    <span className="text-sm text-gray-600">
                      Rollout: <strong>{flag.rollout_percentage}%</strong>
                    </span>
                  </div>

                  {flag.enabled_tenants && flag.enabled_tenants.length > 0 && (
                    <div className="flex items-center gap-2">
                      <Users className="h-4 w-4 text-gray-400" />
                      <span className="text-sm text-gray-600">
                        {flag.enabled_tenants.length} tenant(s) específico(s)
                      </span>
                    </div>
                  )}
                </div>

                {/* Datas */}
                <div className="flex items-center gap-6 text-xs text-gray-500">
                  <span>
                    Criada em:{' '}
                    {format(new Date(flag.created_at), 'dd/MM/yyyy', {
                      locale: ptBR,
                    })}
                  </span>
                  <span>
                    Atualizada em:{' '}
                    {format(new Date(flag.updated_at), 'dd/MM/yyyy HH:mm', {
                      locale: ptBR,
                    })}
                  </span>
                </div>
              </div>

              {/* Ações */}
              <div className="flex items-center gap-3">
                <Switch
                  checked={flag.is_enabled}
                  onCheckedChange={() => handleToggle(flag)}
                />
                <Button
                  variant="outline"
                  size="sm"
                  onClick={() => handleOpenModal(flag)}
                >
                  <Edit className="h-4 w-4" />
                </Button>
              </div>
            </div>
          </Card>
        ))}
      </div>

      {/* Modal de Criação/Edição */}
      <Dialog open={isModalOpen} onOpenChange={setIsModalOpen}>
        <DialogContent className="max-w-2xl">
          <DialogHeader>
            <DialogTitle>
              {editingFlag ? 'Editar Feature Flag' : 'Nova Feature Flag'}
            </DialogTitle>
            <DialogDescription>
              Configure uma funcionalidade experimental da plataforma
            </DialogDescription>
          </DialogHeader>

          <div className="space-y-4 py-4">
            <div className="grid grid-cols-2 gap-4">
              <div>
                <Label htmlFor="name">Nome *</Label>
                <Input
                  id="name"
                  value={formData.name}
                  onChange={(e) =>
                    setFormData({ ...formData, name: e.target.value })
                  }
                  placeholder="Ex: Nova Interface"
                />
              </div>

              <div>
                <Label htmlFor="key">Key (Identificador) *</Label>
                <Input
                  id="key"
                  value={formData.key}
                  onChange={(e) =>
                    setFormData({ ...formData, key: e.target.value })
                  }
                  placeholder="Ex: new_ui_enabled"
                  disabled={!!editingFlag}
                />
              </div>
            </div>

            <div>
              <Label htmlFor="description">Descrição</Label>
              <Textarea
                id="description"
                value={formData.description}
                onChange={(e) =>
                  setFormData({ ...formData, description: e.target.value })
                }
                placeholder="Descreva o que esta feature flag controla..."
                rows={3}
              />
            </div>

            <div className="flex items-center justify-between p-4 border rounded-lg">
              <div>
                <Label htmlFor="is_enabled">Feature Ativa</Label>
                <p className="text-xs text-gray-500">
                  Ativar esta funcionalidade para os tenants
                </p>
              </div>
              <Switch
                id="is_enabled"
                checked={formData.is_enabled}
                onCheckedChange={(checked) =>
                  setFormData({ ...formData, is_enabled: checked })
                }
              />
            </div>

            <div>
              <Label htmlFor="rollout">
                Rollout Percentage: {formData.rollout_percentage}%
              </Label>
              <Slider
                id="rollout"
                value={[formData.rollout_percentage]}
                onValueChange={(value) =>
                  setFormData({ ...formData, rollout_percentage: value[0] })
                }
                max={100}
                step={5}
                className="mt-2"
              />
              <p className="text-xs text-gray-500 mt-2">
                Porcentagem de tenants que terão acesso a esta feature
              </p>
            </div>

            <div>
              <Label htmlFor="tenants">Tenants Específicos (IDs)</Label>
              <Textarea
                id="tenants"
                value={formData.enabled_tenants.join(', ')}
                onChange={(e) =>
                  setFormData({
                    ...formData,
                    enabled_tenants: e.target.value
                      .split(',')
                      .map((id) => id.trim())
                      .filter(Boolean),
                  })
                }
                placeholder="tenant-id-1, tenant-id-2, tenant-id-3"
                rows={2}
              />
              <p className="text-xs text-gray-500 mt-1">
                IDs dos tenants que terão acesso garantido (separados por vírgula)
              </p>
            </div>
          </div>

          <DialogFooter>
            <Button variant="outline" onClick={() => setIsModalOpen(false)}>
              Cancelar
            </Button>
            <Button onClick={handleSave} disabled={isProcessing}>
              {isProcessing ? (
                <>
                  <Loader2 className="h-4 w-4 mr-2 animate-spin" />
                  Salvando...
                </>
              ) : (
                'Salvar Feature Flag'
              )}
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </div>
  )
}
