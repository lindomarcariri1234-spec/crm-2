COMPONENTE DE SELEÇÃO DE ASSENTOS COM UI/UX APRIMORADA

// components/seats/enhanced-seat-selector.tsx

'use client'

import { useState, useEffect } from 'react'
import { Card } from '@/components/ui/card'
import { Button } from '@/components/ui/button'
import { Badge } from '@/components/ui/badge'
import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
import {
  ArrowUp,
  ArrowDown,
  User,
  Check,
  X,
  Info,
  Loader2,
  Armchair,
  MapPin,
  Sparkles,
  Users,
  DollarSign,
} from 'lucide-react'
import { toast } from 'sonner'
import { cn } from '@/lib/utils'
import { motion, AnimatePresence } from 'framer-motion'

interface Seat {
  id: string
  seat_number: string
  deck: string
  row: number
  column: number
  seat_type: string
  is_available: boolean
  is_blocked: boolean
  price_modifier?: number
  reservations: any[]
}

interface EnhancedSeatSelectorProps {
  tripId: string
  maxSeats?: number
  onSeatsSelected: (seats: Seat[]) => void
  pricePerPerson: number
}

export function EnhancedSeatSelector({
  tripId,
  maxSeats = 10,
  onSeatsSelected,
  pricePerPerson,
}: EnhancedSeatSelectorProps) {
  const [isLoading, setIsLoading] = useState(true)
  const [layout, setLayout] = useState<any>(null)
  const [stats, setStats] = useState<any>(null)
  const [selectedSeats, setSelectedSeats] = useState<Seat[]>([])
  const [currentDeck, setCurrentDeck] = useState<'upper' | 'lower'>('upper')
  const [hoveredSeat, setHoveredSeat] = useState<string | null>(null)

  useEffect(() => {
    loadSeats()
  }, [tripId])

  const loadSeats = async () => {
    try {
      setIsLoading(true)
      const response = await fetch(`/api/public/packages/${tripId}/seats`)
      const data = await response.json()
      setLayout(data.layout)
      setStats(data.stats)
    } catch (error) {
      console.error('Erro ao carregar assentos:', error)
      toast.error('Erro ao carregar assentos')
    } finally {
      setIsLoading(false)
    }
  }

  const handleSeatClick = (seat: Seat) => {
    if (seat.reservations.length > 0 || seat.is_blocked) {
      toast.error('Este assento não está disponível', {
        description: 'Escolha outro assento disponível',
        icon: '🚫',
      })
      return
    }

    const isSelected = selectedSeats.find((s) => s.id === seat.id)

    if (isSelected) {
      const newSelection = selectedSeats.filter((s) => s.id !== seat.id)
      setSelectedSeats(newSelection)
      onSeatsSelected(newSelection)
      toast.success(`Assento ${seat.seat_number} removido`, {
        icon: '✓',
      })
    } else {
      if (selectedSeats.length >= maxSeats) {
        toast.error(`Máximo de ${maxSeats} assentos`, {
          description: 'Remova um assento para selecionar outro',
          icon: '⚠️',
        })
        return
      }

      const newSelection = [...selectedSeats, seat]
      setSelectedSeats(newSelection)
      onSeatsSelected(newSelection)
      toast.success(`Assento ${seat.seat_number} selecionado!`, {
        description: calculateSeatPrice(seat).toLocaleString('pt-BR', {
          style: 'currency',
          currency: 'BRL',
        }),
        icon: '🎉',
      })
    }
  }

  const getSeatStatus = (seat: Seat) => {
    if (seat.reservations.length > 0) return 'occupied'
    if (seat.is_blocked) return 'blocked'
    if (selectedSeats.find((s) => s.id === seat.id)) return 'selected'
    return 'available'
  }

  const getSeatColor = (status: string, seatType: string) => {
    if (status === 'occupied') {
      return 'bg-gradient-to-br from-red-400 to-red-500 text-white cursor-not-allowed border-red-600'
    }
    if (status === 'blocked') {
      return 'bg-gradient-to-br from-gray-300 to-gray-400 text-gray-600 cursor-not-allowed border-gray-500'
    }
    if (status === 'selected') {
      return 'bg-gradient-to-br from-green-400 to-green-500 text-white border-green-600 shadow-xl scale-110 ring-4 ring-green-300'
    }
    if (seatType === 'premium') {
      return 'bg-gradient-to-br from-yellow-300 to-yellow-400 text-yellow-900 hover:from-yellow-400 hover:to-yellow-500 cursor-pointer border-yellow-500 hover:scale-105 hover:shadow-lg'
    }
    if (seatType === 'table') {
      return 'bg-gradient-to-br from-cyan-300 to-cyan-400 text-cyan-900 hover:from-cyan-400 hover:to-cyan-500 cursor-pointer border-cyan-500 hover:scale-105 hover:shadow-lg'
    }
    return 'bg-gradient-to-br from-blue-400 to-blue-500 text-white hover:from-blue-500 hover:to-blue-600 cursor-pointer border-blue-600 hover:scale-105 hover:shadow-lg'
  }

  const calculateSeatPrice = (seat: Seat) => {
    const modifier = seat.price_modifier || 0
    return pricePerPerson * (1 + modifier / 100)
  }

  const formatCurrency = (value: number) => {
    return new Intl.NumberFormat('pt-BR', {
      style: 'currency',
      currency: 'BRL',
    }).format(value)
  }

  const renderDeck = (seats: Seat[], deckName: string) => {
    if (!seats || seats.length === 0) return null

    const maxRow = Math.max(...seats.map((s) => s.row))
    const maxCol = Math.max(...seats.map((s) => s.column))

    const grid: (Seat | null)[][] = Array.from({ length: maxRow + 1 }, () =>
      Array(maxCol + 1).fill(null)
    )

    seats.forEach((seat) => {
      grid[seat.row][seat.column] = seat
    })

    const deckLabel = deckName === 'upper' ? 'Piso Superior' : 'Piso Inferior'
    const deckIcon = deckName === 'upper' ? ArrowUp : ArrowDown
    const deckColor = deckName === 'upper' ? 'from-blue-600 to-blue-700' : 'from-purple-600 to-purple-700'

    return (
      <motion.div
        initial={{ opacity: 0, y: 20 }}
        animate={{ opacity: 1, y: 0 }}
        exit={{ opacity: 0, y: -20 }}
        transition={{ duration: 0.3 }}
        className="space-y-4"
      >
        {/* Header do Deck */}
        <div className={cn(
          'flex items-center justify-between p-5 bg-gradient-to-r text-white rounded-2xl shadow-lg',
          `${deckColor}`
        )}>
          <div className="flex items-center gap-3">
            {deckName === 'upper' ? (
              <ArrowUp className="h-6 w-6" />
            ) : (
              <ArrowDown className="h-6 w-6" />
            )}
            <div>
              <h3 className="font-bold text-lg">{deckLabel}</h3>
              <p className="text-xs text-white/80">
                {seats.filter((s) => s.reservations.length === 0).length} assentos disponíveis
              </p>
            </div>
          </div>
          <Badge variant="secondary" className="bg-white/20 text-white border-white/30 text-sm px-3 py-1">
            {seats.length} lugares
          </Badge>
        </div>

        {/* Grid de Assentos */}
        <div className="p-8 bg-gradient-to-b from-gray-50 via-white to-gray-50 rounded-2xl border-2 border-gray-200 shadow-inner">
          {/* Motorista (apenas piso superior) */}
          {deckName === 'upper' && (
            <motion.div
              initial={{ scale: 0 }}
              animate={{ scale: 1 }}
              className="flex justify-end mb-6"
            >
              <div className="px-4 py-2 bg-gradient-to-r from-gray-800 to-gray-900 text-white rounded-xl shadow-lg flex items-center gap-2">
                <User className="h-4 w-4" />
                <span className="font-bold text-sm">MOTORISTA</span>
              </div>
            </motion.div>
          )}

          {/* Banheiro (apenas piso inferior) */}
          {deckName === 'lower' && (
            <motion.div
              initial={{ scale: 0 }}
              animate={{ scale: 1 }}
              className="flex justify-center mb-6"
            >
              <div className="px-6 py-3 bg-gradient-to-r from-blue-600 to-blue-700 text-white rounded-xl shadow-lg flex items-center gap-2">
                <span className="text-lg">🚻</span>
                <span className="font-bold">BANHEIRO</span>
              </div>
            </motion.div>
          )}

          <div className="flex flex-col gap-3">
            {grid.map((row, rowIndex) => (
              <div key={rowIndex} className="flex justify-center gap-3">
                {row.map((seat, colIndex) => {
                  if (!seat) {
                    return (
                      <div
                        key={`empty-${rowIndex}-${colIndex}`}
                        className="w-16 h-16 flex items-center justify-center"
                      >
                        {colIndex === Math.floor(maxCol / 2) && (
                          <div className="flex flex-col items-center gap-1">
                            <div className="w-1 h-12 bg-gradient-to-b from-gray-200 via-gray-300 to-gray-200 rounded-full" />
                            <span className="text-[10px] text-gray-400 font-bold">CORREDOR</span>
                          </div>
                        )}
                      </div>
                    )
                  }

                  const status = getSeatStatus(seat)
                  const seatPrice = calculateSeatPrice(seat)
                  const isHovered = hoveredSeat === seat.id

                  return (
                    <motion.div
                      key={seat.id}
                      className="relative group"
                      whileHover={{ scale: 1.05 }}
                      whileTap={{ scale: 0.95 }}
                    >
                      <button
                        onClick={() => handleSeatClick(seat)}
                        onMouseEnter={() => setHoveredSeat(seat.id)}
                        onMouseLeave={() => setHoveredSeat(null)}
                        disabled={status === 'occupied' || status === 'blocked'}
                        className={cn(
                          'w-16 h-16 rounded-xl flex flex-col items-center justify-center font-bold text-sm transition-all duration-200 border-2 relative overflow-hidden',
                          getSeatColor(status, seat.seat_type)
                        )}
                      >
                        {/* Ícone do assento */}
                        <Armchair className="h-6 w-6 mb-1" />

                        {/* Número do assento */}
                        <span className="text-xs font-bold">
                          {status === 'selected' && '✓'}
                          {status === 'occupied' && '✗'}
                          {(status === 'available' || status === 'blocked') &&
                            seat.seat_number}
                        </span>

                        {/* Efeito de brilho */}
                        {status === 'available' && (
                          <div className="absolute inset-0 bg-gradient-to-tr from-white/0 via-white/20 to-white/0 opacity-0 group-hover:opacity-100 transition-opacity" />
                        )}
                      </button>

                      {/* Tooltip com preço */}
                      <AnimatePresence>
                        {isHovered && status === 'available' && (
                          <motion.div
                            initial={{ opacity: 0, y: 10 }}
                            animate={{ opacity: 1, y: 0 }}
                            exit={{ opacity: 0, y: 10 }}
                            className="absolute bottom-full left-1/2 transform -translate-x-1/2 mb-2 px-3 py-2 bg-gray-900 text-white text-xs rounded-lg shadow-xl z-20 whitespace-nowrap"
                          >
                            <div className="flex flex-col items-center gap-1">
                              <span className="font-bold text-sm">
                                {formatCurrency(seatPrice)}
                              </span>
                              {seat.price_modifier && seat.price_modifier !== 0 && (
                                <span className="text-yellow-300 text-[10px]">
                                  ({seat.price_modifier > 0 ? '+' : ''}
                                  {seat.price_modifier}%)
                                </span>
                              )}
                            </div>
                            {/* Seta do tooltip */}
                            <div className="absolute top-full left-1/2 transform -translate-x-1/2 -mt-1">
                              <div className="w-0 h-0 border-l-4 border-r-4 border-t-4 border-transparent border-t-gray-900" />
                            </div>
                          </motion.div>
                        )}
                      </AnimatePresence>

                      {/* Badge de tipo especial */}
                      {seat.seat_type === 'premium' && status === 'available' && (
                        <div className="absolute -top-1 -right-1">
                          <Badge className="text-[10px] px-1.5 py-0 bg-yellow-500 text-yellow-900 border-yellow-600 shadow-md">
                            <Sparkles className="h-2.5 w-2.5 mr-0.5" />
                            VIP
                          </Badge>
                        </div>
                      )}

                      {seat.seat_type === 'table' && status === 'available' && (
                        <div className="absolute -top-1 -right-1">
                          <Badge className="text-[10px] px-1.5 py-0 bg-cyan-500 text-white border-cyan-600 shadow-md">
                            MESA
                          </Badge>
                        </div>
                      )}
                    </motion.div>
                  )
                })}
              </div>
            ))}
          </div>

          {/* Entrada (apenas piso superior) */}
          {deckName === 'upper' && (
            <motion.div
              initial={{ scale: 0 }}
              animate={{ scale: 1 }}
              className="mt-6 flex justify-center"
            >
              <div className="px-6 py-2 bg-gradient-to-r from-yellow-400 to-yellow-500 text-yellow-900 font-bold rounded-xl shadow-lg flex items-center gap-2">
                <MapPin className="h-4 w-4" />
                ← ENTRADA
              </div>
            </motion.div>
          )}
        </div>
      </motion.div>
    )
  }

  if (isLoading) {
    return (
      <div className="flex flex-col items-center justify-center p-16 space-y-4">
        <Loader2 className="h-12 w-12 animate-spin text-blue-600" />
        <p className="text-gray-600 font-medium">Carregando assentos...</p>
      </div>
    )
  }

  const isDoubleDecker =
    layout?.upperDeck?.length > 0 && layout?.lowerDeck?.length > 0

  return (
    <div className="space-y-6">
      {/* Estatísticas */}
      {stats && (
        <div className="grid grid-cols-3 gap-4">
          <Card className="p-4 bg-gradient-to-br from-blue-500 to-blue-600 text-white">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-sm text-blue-100 mb-1">Total</p>
                <p className="text-3xl font-bold">{stats.totalSeats}</p>
              </div>
              <Armchair className="h-10 w-10 text-blue-200" />
            </div>
          </Card>

          <Card className="p-4 bg-gradient-to-br from-green-500 to-green-600 text-white">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-sm text-green-100 mb-1">Disponíveis</p>
                <p className="text-3xl font-bold">{stats.availableSeats}</p>
              </div>
              <Check className="h-10 w-10 text-green-200" />
            </div>
          </Card>

          <Card className="p-4 bg-gradient-to-br from-orange-500 to-orange-600 text-white">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-sm text-orange-100 mb-1">Ocupação</p>
                <p className="text-3xl font-bold">
                  {stats.occupancyRate.toFixed(0)}%
                </p>
              </div>
              <Users className="h-10 w-10 text-orange-200" />
            </div>
          </Card>
        </div>
      )}

      {/* Legenda */}
      <Card className="p-5 bg-gradient-to-r from-gray-50 to-gray-100 border-2 border-gray-200">
        <div className="flex items-center justify-between flex-wrap gap-4">
          <div className="flex items-center gap-6">
            <div className="flex items-center gap-2">
              <div className="w-10 h-10 bg-gradient-to-br from-blue-400 to-blue-500 rounded-lg shadow-md flex items-center justify-center">
                <Armchair className="h-5 w-5 text-white" />
              </div>
              <span className="text-sm font-semibold text-gray-700">Disponível</span>
            </div>
            <div className="flex items-center gap-2">
              <div className="w-10 h-10 bg-gradient-to-br from-green-400 to-green-500 rounded-lg shadow-md flex items-center justify-center">
                <Check className="h-5 w-5 text-white" />
              </div>
              <span className="text-sm font-semibold text-gray-700">Selecionado</span>
            </div>
            <div className="flex items-center gap-2">
              <div className="w-10 h-10 bg-gradient-to-br from-red-400 to-red-500 rounded-lg shadow-md flex items-center justify-center">
                <X className="h-5 w-5 text-white" />
              </div>
              <span className="text-sm font-semibold text-gray-700">Ocupado</span>
            </div>
            <div className="flex items-center gap-2">
              <div className="w-10 h-10 bg-gradient-to-br from-yellow-300 to-yellow-400 rounded-lg shadow-md flex items-center justify-center">
                <Sparkles className="h-4 w-4 text-yellow-900" />
              </div>
              <span className="text-sm font-semibold text-gray-700">VIP (+10%)</span>
            </div>
          </div>

          <div className="flex items-center gap-2 px-4 py-2 bg-white rounded-lg shadow-sm border border-gray-200">
            <User className="h-5 w-5 text-blue-600" />
            <span className="font-bold text-gray-900">
              {selectedSeats.length} / {maxSeats}
            </span>
            <span className="text-sm text-gray-600">selecionado(s)</span>
          </div>
        </div>
      </Card>

      {/* Layout dos Assentos com Abas */}
      {isDoubleDecker ? (
        <Tabs value={currentDeck} onValueChange={(v: any) => setCurrentDeck(v)}>
          <TabsList className="grid w-full grid-cols-2 h-14 bg-gradient-to-r from-gray-100 to-gray-200 p-1.5 rounded-2xl shadow-md">
            <TabsTrigger
              value="upper"
              className="flex items-center gap-3 text-base font-bold data-[state=active]:bg-white data-[state=active]:shadow-lg data-[state=active]:text-blue-600 rounded-xl transition-all"
            >
              <ArrowUp className="h-5 w-5" />
              <span>Piso Superior</span>
              {layout.upperDeck && (
                <Badge
                  variant="secondary"
                  className="ml-2 text-xs px-2 py-0.5 bg-blue-100 text-blue-700 border-blue-300"
                >
                  {layout.upperDeck.filter((s: Seat) => s.reservations.length === 0).length}
                </Badge>
              )}
            </TabsTrigger>
            <TabsTrigger
              value="lower"
              className="flex items-center gap-3 text-base font-bold data-[state=active]:bg-white data-[state=active]:shadow-lg data-[state=active]:text-purple-600 rounded-xl transition-all"
            >
              <ArrowDown className="h-5 w-5" />
              <span>Piso Inferior</span>
              {layout.lowerDeck && (
                <Badge
                  variant="secondary"
                  className="ml-2 text-xs px-2 py-0.5 bg-purple-100 text-purple-700 border-purple-300"
                >
                  {layout.lowerDeck.filter((s: Seat) => s.reservations.length === 0).length}
                </Badge>
              )}
            </TabsTrigger>
          </TabsList>

          <TabsContent value="upper" className="mt-6">
            {renderDeck(layout.upperDeck, 'upper')}
          </TabsContent>

          <TabsContent value="lower" className="mt-6">
            {renderDeck(layout.lowerDeck, 'lower')}
          </TabsContent>
        </Tabs>
      ) : (
        <Card className="p-6">{renderDeck(layout.singleDeck || [], 'single')}</Card>
      )}

      {/* Assentos Selecionados */}
      <AnimatePresence>
        {selectedSeats.length > 0 && (
          <motion.div
            initial={{ opacity: 0, y: 20 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: -20 }}
          >
            <Card className="p-6 bg-gradient-to-br from-green-50 to-emerald-50 border-2 border-green-300 shadow-lg">
              <h4 className="font-bold text-xl mb-4 flex items-center gap-2 text-green-800">
                <Check className="h-6 w-6 text-green-600" />
                Seus Assentos Selecionados
              </h4>

              <div className="space-y-3">
                {selectedSeats.map((seat, index) => {
                  const seatPrice = calculateSeatPrice(seat)
                  return (
                    <motion.div
                      key={seat.id}
                      initial={{ opacity: 0, x: -20 }}
                      animate={{ opacity: 1, x: 0 }}
                      transition={{ delay: index * 0.1 }}
                      className="flex items-center justify-between p-4 bg-white rounded-xl border-2 border-green-200 shadow-sm hover:shadow-md transition-shadow"
                    >
                      <div className="flex items-center gap-4">
                        <div className="w-12 h-12 bg-gradient-to-br from-green-500 to-green-600 rounded-xl flex items-center justify-center text-white font-bold shadow-md">
                          {seat.seat_number}
                        </div>
                        <div>
                          <p className="font-bold text-gray-900">
                            Assento {seat.seat_number}
                          </p>
                          <p className="text-sm text-gray-600 flex items-center gap-2">
                            {seat.deck === 'upper' ? (
                              <>
                                <ArrowUp className="h-3 w-3" />
                                Piso Superior
                              </>
                            ) : (
                              <>
                                <ArrowDown className="h-3 w-3" />
                                Piso Inferior
                              </>
                            )}
                            {seat.seat_type === 'premium' && (
                              <Badge className="ml-2 bg-yellow-400 text-yellow-900 text-xs">
                                VIP
                              </Badge>
                            )}
                            {seat.seat_type === 'table' && (
                              <Badge className="ml-2 bg-cyan-400 text-cyan-900 text-xs">
                                Mesa
                              </Badge>
                            )}
                          </p>
                        </div>
                      </div>

                      <div className="text-right">
                        <p className="text-xl font-bold text-green-600">
                          {formatCurrency(seatPrice)}
                        </p>
                        {seat.price_modifier && seat.price_modifier !== 0 && (
                          <p className="text-xs text-gray-600">
                            ({seat.price_modifier > 0 ? '+' : ''}
                            {seat.price_modifier}%)
                          </p>
                        )}
                      </div>
                    </motion.div>
                  )

// app/api/public/packages/[id]/seats/route.ts

import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'

export async function GET(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const tripId = params.id

    // Buscar viagem com veículo
    const trip = await prisma.trip.findUnique({
      where: { id: tripId },
      include: {
        vehicle: {
          include: {
            bus_model: true,
          },
        },
      },
    })

    if (!trip) {
      return NextResponse.json({ error: 'Viagem não encontrada' }, { status: 404 })
    }

    if (!trip.vehicle?.bus_model) {
      return NextResponse.json(
        { error: 'Veículo não configurado' },
        { status: 400 }
      )
    }

    // Buscar todas as reservas de assentos desta viagem
    const seatReservations = await prisma.seatReservation.findMany({
      where: {
        trip_id: tripId,
        reservation: {
          status: {
            in: ['pending', 'confirmed'],
          },
        },
      },
      include: {
        reservation: {
          select: {
            id: true,
            status: true,
            customer: {
              select: {
                name: true,
              },
            },
          },
        },
      },
    })

    // Gerar layout de assentos
    const busModel = trip.vehicle.bus_model
    const config = busModel.config as any

    const isDoubleDecker = busModel.bus_type === 'DOUBLE_DECKER'

    let upperDeckSeats: any[] = []
    let lowerDeckSeats: any[] = []
    let singleDeckSeats: any[] = []

    // Processar piso superior
    if (config.upperDeck) {
      config.upperDeck.layout.forEach((row: string[], rowIndex: number) => {
        row.forEach((cell: string, colIndex: number) => {
          if (cell !== 'AISLE' && cell !== 'X') {
            const seatNumber = cell.replace(/[PS]/g, '')
            const isPremium = cell.startsWith('P')
            const isTable = cell.startsWith('S')

            const reservations = seatReservations.filter(
              (r) => r.seat_number === seatNumber && r.floor === 'upper'
            )

            upperDeckSeats.push({
              id: `upper-${seatNumber}`,
              seat_number: seatNumber,
              deck: 'upper',
              row: rowIndex,
              column: colIndex,
              seat_type: isPremium ? 'premium' : isTable ? 'table' : 'standard',
              is_available: reservations.length === 0,
              is_blocked: false,
              price_modifier: isPremium ? 10 : isTable ? 5 : 0,
              reservations: reservations.map((r) => ({
                id: r.id,
                status: r.reservation.status,
                customer_name: r.reservation.customer.name,
              })),
            })
          }
        })
      })
    }

    // Processar piso inferior
    if (config.lowerDeck) {
      config.lowerDeck.layout.forEach((row: string[], rowIndex: number) => {
        row.forEach((cell: string, colIndex: number) => {
          if (cell !== 'AISLE' && cell !== 'X') {
            const seatNumber = cell.replace(/[PS]/g, '')
            const isPremium = cell.startsWith('P')
            const isTable = cell.startsWith('S')

            const reservations = seatReservations.filter(
              (r) => r.seat_number === seatNumber && r.floor === 'lower'
            )

            lowerDeckSeats.push({
              id: `lower-${seatNumber}`,
              seat_number: seatNumber,
              deck: 'lower',
              row: rowIndex,
              column: colIndex,
              seat_type: isPremium ? 'premium' : isTable ? 'table' : 'standard',
              is_available: reservations.length === 0,
              is_blocked: false,
              price_modifier: isPremium ? 10 : isTable ? 5 : 0,
              reservations: reservations.map((r) => ({
                id: r.id,
                status: r.reservation.status,
                customer_name: r.reservation.customer.name,
              })),
            })
          }
        })
      })
    }

    // Calcular estatísticas
    const allSeats = isDoubleDecker
      ? [...upperDeckSeats, ...lowerDeckSeats]
      : singleDeckSeats

    const totalSeats = allSeats.length
    const occupiedSeats = allSeats.filter((s) => s.reservations.length > 0).length
    const availableSeats = totalSeats - occupiedSeats
    const occupancyRate = (occupiedSeats / totalSeats) * 100

    return NextResponse.json({
      layout: {
        upperDeck: upperDeckSeats,
        lowerDeck: lowerDeckSeats,
        singleDeck: singleDeckSeats,
      },
      stats: {
        totalSeats,
        occupiedSeats,
        availableSeats,
        occupancyRate,
      },
      busModel: {
        name: busModel.name,
        type: busModel.bus_type,
        isDoubleDecker,
      },
    })
  } catch (error) {
    console.error('Erro ao buscar assentos:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar assentos' },
      { status: 500 }
    )
  }
}

  })

  if (!tenant) {
    notFound()
  }

  const trip = await prisma.trip.findUnique({

      destination: true,
      vehicle: {
        include: {
          bus_model: true,
        },
      },
    },
  })

  if (!trip) {
    notFound()
  }

  return <SeatSelectionClient trip={trip} tenant={tenant} />
