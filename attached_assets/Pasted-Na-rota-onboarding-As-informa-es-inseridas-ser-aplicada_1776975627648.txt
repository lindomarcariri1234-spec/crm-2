// app/api/onboarding/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { z } from 'zod'

const onboardingStep1Schema = z.object({
  name: z.string().min(3, 'Nome deve ter no mínimo 3 caracteres'),
  slug: z
    .string()
    .min(3)
    .regex(/^[a-z0-9-]+$/, 'Slug deve conter apenas letras minúsculas, números e hífens'),
  cnpj: z.string().optional(),
  phone: z.string().min(10, 'Telefone inválido'),
  email: z.string().email('Email inválido').optional(),
  address: z.string().optional(),
  city: z.string().optional(),
  state: z.string().optional(),
  zipcode: z.string().optional(),
})

const onboardingStep2Schema = z.object({
  plan_type: z.enum(['free', 'basic', 'pro', 'enterprise']),
})

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session || session.user.role !== 'admin') {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    const body = await req.json()
    const { step } = body

    // ═══════════════════════════════════════════════════════════
    // STEP 1: DADOS DA AGÊNCIA
    // ═══════════════════════════════════════════════════════════

    if (step === 1) {
      const validatedData = onboardingStep1Schema.parse(body.data)

      // Verificar se slug já existe
      const existingTenant = await prisma.tenant.findUnique({
        where: { slug: validatedData.slug },
      })

      if (existingTenant && existingTenant.id !== session.user.tenantId) {
        return NextResponse.json(
          { error: 'Este slug já está em uso' },
          { status: 400 }
        )
      }

      // Atualizar tenant
      const tenant = await prisma.tenant.update({
        where: { id: session.user.tenantId },
        data: {
          ...validatedData,
          whatsapp: validatedData.phone, // Usar mesmo número para WhatsApp
          onboarding_step: 2,
          store_enabled: true,
          store_title: `${validatedData.name} - Pacotes de Viagem`,
          store_description: `Encontre os melhores pacotes de viagem com ${validatedData.name}. Destinos incríveis com os melhores preços!`,
        },
      })

      // Log
      await prisma.systemLog.create({
        data: {
          tenant_id: session.user.tenantId,
          user_id: session.user.id,
          category: 'onboarding',
          action: 'step_1_completed',
          severity: 'info',
          description: 'Dados da agência configurados',
          metadata: { tenantId: tenant.id },
        },
      })

      return NextResponse.json({ tenant, nextStep: 2 })
    }

    // ═══════════════════════════════════════════════════════════
    // STEP 2: ESCOLHA DO PLANO
    // ═══════════════════════════════════════════════════════════

    if (step === 2) {
      const validatedData = onboardingStep2Schema.parse(body.data)

      // Calcular data de expiração baseado no plano
      let plan_expires_at = null
      if (validatedData.plan_type !== 'free') {
        plan_expires_at = new Date()
        plan_expires_at.setDate(plan_expires_at.getDate() + 30) // 30 dias de trial
      }

      // Atualizar tenant
      const tenant = await prisma.tenant.update({
        where: { id: session.user.tenantId },
        data: {
          plan_type: validatedData.plan_type,
          plan_expires_at,
          onboarding_completed: true,
          onboarding_step: 3,
        },
      })

      // ═══════════════════════════════════════════════════════════
      // CRIAR DADOS INICIAIS (SEED)
      // ═══════════════════════════════════════════════════════════

      await seedInitialData(session.user.tenantId)

      // Log
      await prisma.systemLog.create({
        data: {
          tenant_id: session.user.tenantId,
          user_id: session.user.id,
          category: 'onboarding',
          action: 'onboarding_completed',
          severity: 'info',
          description: 'Onboarding concluído com sucesso',
          metadata: { tenantId: tenant.id, plan: validatedData.plan_type },
        },
      })

      return NextResponse.json({
        tenant,
        completed: true,
        message: 'Onboarding concluído com sucesso!',
      })
    }

    return NextResponse.json({ error: 'Step inválido' }, { status: 400 })
  } catch (error) {
    console.error('Erro no onboarding:', error)

    if (error instanceof z.ZodError) {
      return NextResponse.json(
        { error: error.errors[0].message },
        { status: 400 }
      )
    }

    return NextResponse.json(
      { error: 'Erro ao processar onboarding' },
      { status: 500 }
    )
  }
}

// app/api/onboarding/check-slug/route.ts

import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'

export async function GET(req: Request) {
  try {
    const { searchParams } = new URL(req.url)
    const slug = searchParams.get('slug')

    if (!slug) {
      return NextResponse.json({ error: 'Slug não informado' }, { status: 400 })
    }

    const existingTenant = await prisma.tenant.findUnique({
      where: { slug },
    })

    return NextResponse.json({ available: !existingTenant })
  } catch (error) {
    console.error('Erro ao verificar slug:', error)
    return NextResponse.json(
      { error: 'Erro ao verificar slug' },
      { status: 500 }
    )
  }
}

// middleware.ts

import { withAuth } from 'next-auth/middleware'
import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'

export default withAuth(
  async function middleware(req) {
    const token = req.nextauth.token
    const path = req.nextUrl.pathname

    // Rotas públicas que não precisam de verificação
    const publicPaths = [
      '/loja',
      '/auth/login',
      '/auth/register',
      '/auth/forgot-password',
      '/api/public',
    ]

    const isPublicPath = publicPaths.some((publicPath) =>
      path.startsWith(publicPath)
    )

    if (isPublicPath) {
      return NextResponse.next()
    }

    // Se não está autenticado, redirecionar para login
    if (!token) {
      return NextResponse.redirect(new URL('/auth/login', req.url))
    }

    // Verificar se o onboarding foi concluído
    if (token.role === 'admin' && path !== '/onboarding') {
      try {
        const tenant = await prisma.tenant.findUnique({
          where: { id: token.tenantId as string },
          select: { onboarding_completed: true },
        })

        if (!tenant?.onboarding_completed) {
          return NextResponse.redirect(new URL('/onboarding', req.url))
        }
      } catch (error) {
        console.error('Erro ao verificar onboarding:', error)
      }
    }

    // Se está no onboarding mas já completou, redirecionar para dashboard
    if (path === '/onboarding' && token.role === 'admin') {
      try {
        const tenant = await prisma.tenant.findUnique({
          where: { id: token.tenantId as string },
          select: { onboarding_completed: true },
        })

        if (tenant?.onboarding_completed) {
          return NextResponse.redirect(new URL('/dashboard', req.url))
        }
      } catch (error) {
        console.error('Erro ao verificar onboarding:', error)
      }
    }

    return NextResponse.next()
  },
  {
    callbacks: {
      authorized: ({ token }) => !!token,
    },
  }
)

import { useEffect, useState } from 'react'
import { useSession } from 'next-auth/react'

export function useTenantSync() {
  const { data: session, update } = useSession()
  const [tenant, setTenant] = useState<any>(null)
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    if (session?.user?.tenantId) {
      loadTenant()
    }
  }, [session?.user?.tenantId])

  const loadTenant = async () => {
    try {
      setIsLoading(true)
      const response = await fetch('/api/tenant/current')
      const data = await response.json()
      setTenant(data.tenant)
    } catch (error) {
      console.error('Erro ao carregar tenant:', error)
    } finally {
      setIsLoading(false)
    }
  }

  const syncTenant = async () => {
    await loadTenant()
    await update() // Atualizar sessão
  }

  return {
    tenant,
    isLoading,
    syncTenant,
  }
}

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function GET() {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    const tenant = await prisma.tenant.findUnique({
      where: { id: session.user.tenantId },
      select: {
        id: true,
        name: true,
        slug: true,
        cnpj: true,
        phone: true,
        whatsapp: true,
        email: true,
        website: true,
        address: true,
        city: true,
        state: true,
        zipcode: true,
        country: true,
        logo_url: true,
        banner_url: true,
        primary_color: true,
        secondary_color: true,
        store_enabled: true,
        store_title: true,
        store_description: true,
        store_meta_tags: true,
        facebook_url: true,
        instagram_url: true,
        twitter_url: true,
        youtube_url: true,
        onboarding_completed: true,
        onboarding_step: true,
        plan_type: true,
        plan_expires_at: true,
        is_active: true,
        created_at: true,
      },
    })

    if (!tenant) {
      return NextResponse.json({ error: 'Tenant não encontrado' }, { status: 404 })
    }

    return NextResponse.json({ tenant })
  } catch (error) {
    console.error('Erro ao buscar tenant:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar tenant' },
      { status: 500 }
    )
  }
}

export async function PATCH(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session || session.user.role !== 'admin') {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    const body = await req.json()

    const tenant = await prisma.tenant.update({
      where: { id: session.user.tenantId },
      data: {
        ...body,
        updated_at: new Date(),
      },
    })

    // Log
    await prisma.systemLog.create({
      data: {
        tenant_id: session.user.tenantId,
        user_id: session.user.id,
        category: 'tenant',
        action: 'tenant_updated',
        severity: 'info',
        description: 'Configurações da agência atualizadas',
        metadata: { changes: body },
      },
    })

    return NextResponse.json({ tenant })
  } catch (error) {
    console.error('Erro ao atualizar tenant:', error)
    return NextResponse.json(
      { error: 'Erro ao atualizar tenant' },
      { status: 500 }
    )
  }
}

import { notFound } from 'next/navigation'
import { prisma } from '@/lib/prisma'
import { Header } from '@/components/public/header'
import { Footer } from '@/components/public/footer'

export async function generateMetadata({
  params,
}: {
  params: { slug: string }
}) {
  const tenant = await prisma.tenant.findUnique({
    where: { slug: params.slug },
  })

  if (!tenant) {
    return {
      title: 'Agência não encontrada',
    }
  }

  return {
    title: tenant.store_title || tenant.name,
    description: tenant.store_description,
    openGraph: {
      title: tenant.store_title || tenant.name,
      description: tenant.store_description,
      images: tenant.banner_url ? [tenant.banner_url] : [],
    },
  }
}

export default async function StoreLayout({
  children,
  params,
}: {
  children: React.ReactNode
  params: { slug: string }
}) {
  const tenant = await prisma.tenant.findUnique({
    where: { slug: params.slug },
  })

  if (!tenant || !tenant.store_enabled) {
    notFound()
  }

  return (
    <div className="min-h-screen flex flex-col">
      <Header tenant={tenant} />
      <main className="flex-1">{children}</main>
      <Footer tenant={tenant} />
    </div>
  )
}

    logo_url?: string
    phone?: string
    email?: string
    primary_color?: string
  }
}

  const pathname = usePathname()
  const [mobileMenuOpen, setMobileMenuOpen] = useState(false)

  const navigation = [

    { name: 'Pacotes', href: `/loja/${tenant.slug}/pacotes` },
    { name: 'Destinos', href: `/loja/${tenant.slug}/destinos` },
    { name: 'Sobre', href: `/loja/${tenant.slug}/sobre` },
    { name: 'Contato', href: `/loja/${tenant.slug}/contato` },
  ]

  return (
    <header className="sticky top-0 z-50 bg-white border-b shadow-sm">
      {/* Top Bar */}
      <div
        className="text-white"
        style={{ backgroundColor: tenant.primary_color || '#3B82F6' }}
      >
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
          <div className="flex items-center justify-between h-10 text-sm">
            <div className="flex items-center gap-6">
              {tenant.phone && (

                  href={`tel:${tenant.phone}`}

                >
                  <Phone className="h-4 w-4" />
                  <span className="hidden sm:inline">{tenant.phone}</span>
                </a>
              )}
              {tenant.email && (
                <a
                  href={`mailto:${tenant.email}`}
                  className="flex items-center gap-2 hover:opacity-80 transition"
                >
                  <Mail className="h-4 w-4" />
                  <span className="hidden sm:inline">{tenant.email}</span>
                </a>
              )}
            </div>
          </div>
        </div>
      </div>

      {/* Main Header */}
      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
        <div className="flex items-center justify-between h-16">
          {/* Logo */}
          <Link href={`/loja/${tenant.slug}`} className="flex items-center gap-3">
            {tenant.logo_url ? (

                src={tenant.logo_url}
                alt={tenant.name}
                width={40}
                height={40}

              />

      </div>
    </header>
