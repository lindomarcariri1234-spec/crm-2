model Expense {
  id              String    @id @default(cuid())
  tenant_id       String
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  
  // Dados da despesa
  description     String
  category        String    // transporte, equipe, marketing, operacional, etc
  type            String    // fixed, variable
  amount          Decimal   @db.Decimal(10, 2)
  
  // Relacionamento com viagem (opcional)
  trip_id         String?
  trip            Trip?     @relation(fields: [trip_id], references: [id], onDelete: SetNull)
  
  // Pagamento
  status          String    @default("pending")  // pending, paid, overdue
  due_date        DateTime?
  paid_date       DateTime?
  payment_method  String?   // cash, credit_card, debit_card, pix, bank_transfer
  
  // Fornecedor
  supplier_name   String?
  supplier_doc    String?   // CPF/CNPJ
  
  // Auditoria
  created_by      String
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  @@index([tenant_id])
  @@index([tenant_id, status])
  @@index([tenant_id, category])
  @@index([trip_id])
  @@map("expenses")
}

// app/api/dashboard/financial/overview/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const { searchParams } = new URL(req.url)
    const period = searchParams.get('period') || '30' // dias

    const startDate = new Date()
    startDate.setDate(startDate.getDate() - parseInt(period))

    // ═══════════════════════════════════════════════════════════
    // BUSCAR DADOS
    // ═══════════════════════════════════════════════════════════

    // 1. VIAGENS
    const trips = await prisma.trip.findMany({
      where: {
        tenant_id: session.user.tenantId,
        created_at: { gte: startDate },
      },
      select: {
        id: true,
        title: true,
        status: true,
        total_revenue: true,
        total_cost: true,
        total_profit: true,
        profit_margin: true,
        departure_date: true,
        max_passengers: true,
        created_at: true,
      },
    })

    // 2. DESPESAS
    const expenses = await prisma.expense.findMany({
      where: {
        tenant_id: session.user.tenantId,
        created_at: { gte: startDate },
      },
      select: {
        id: true,
        description: true,
        category: true,
        type: true,
        amount: true,
        status: true,
        due_date: true,
        paid_date: true,
        trip_id: true,
      },
    })

    // 3. RECEITAS
    const revenues = await prisma.revenue.findMany({
      where: {
        tenant_id: session.user.tenantId,
        created_at: { gte: startDate },
      },
      select: {
        id: true,
        description: true,
        amount: true,
        status: true,
        due_date: true,
        received_date: true,
        trip_id: true,
      },
    })

    // 4. COMISSÕES
    const commissions = await prisma.commission.findMany({
      where: {
        tenant_id: session.user.tenantId,
        created_at: { gte: startDate },
      },
      select: {
        id: true,
        type: true,
        description: true,
        amount: true,
        status: true,
        beneficiary_name: true,
      },
    })

    // 5. RESERVAS
    const reservations = await prisma.reservation.findMany({
      where: {
        tenant_id: session.user.tenantId,
        created_at: { gte: startDate },
      },
      select: {
        id: true,
        status: true,
        total_amount: true,
        trip: {
          select: {
            title: true,
          },
        },
      },
    })

    // ═══════════════════════════════════════════════════════════
    // CALCULAR TOTAIS
    // ═══════════════════════════════════════════════════════════

    // RECEITAS
    const totalRevenue = revenues.reduce(
      (sum, r) => sum + Number(r.amount),
      0
    )
    const receivedRevenue = revenues
      .filter((r) => r.status === 'received')
      .reduce((sum, r) => sum + Number(r.amount), 0)
    const pendingRevenue = revenues
      .filter((r) => r.status === 'pending')
      .reduce((sum, r) => sum + Number(r.amount), 0)

    // DESPESAS
    const totalExpenses = expenses.reduce(
      (sum, e) => sum + Number(e.amount),
      0
    )
    const paidExpenses = expenses
      .filter((e) => e.status === 'paid')
      .reduce((sum, e) => sum + Number(e.amount), 0)
    const pendingExpenses = expenses
      .filter((e) => e.status === 'pending')
      .reduce((sum, e) => sum + Number(e.amount), 0)

    // COMISSÕES
    const totalCommissions = commissions.reduce(
      (sum, c) => sum + Number(c.amount),
      0
    )
    const paidCommissions = commissions
      .filter((c) => c.status === 'paid')
      .reduce((sum, c) => sum + Number(c.amount), 0)
    const pendingCommissions = commissions
      .filter((c) => c.status === 'pending')
      .reduce((sum, c) => sum + Number(c.amount), 0)

    // LUCRO
    const totalProfit = trips.reduce(
      (sum, t) => sum + Number(t.total_profit || 0),
      0
    )

    // FLUXO DE CAIXA
    const cashFlow = receivedRevenue - paidExpenses - paidCommissions

    // ═══════════════════════════════════════════════════════════
    // ANÁLISES
    // ═══════════════════════════════════════════════════════════

    // Despesas por categoria
    const expensesByCategory = expenses.reduce((acc: any, expense) => {
      const category = expense.category
      if (!acc[category]) {
        acc[category] = 0
      }
      acc[category] += Number(expense.amount)
      return acc
    }, {})

    // Despesas por tipo
    const expensesByType = {
      fixed: expenses
        .filter((e) => e.type === 'fixed')
        .reduce((sum, e) => sum + Number(e.amount), 0),
      variable: expenses
        .filter((e) => e.type === 'variable')
        .reduce((sum, e) => sum + Number(e.amount), 0),
    }

    // Viagens mais lucrativas
    const topProfitableTrips = trips
      .filter((t) => t.total_profit)
      .sort((a, b) => Number(b.total_profit) - Number(a.total_profit))
      .slice(0, 5)

    // Tendência mensal
    const monthlyData = await getMonthlyTrend(
      session.user.tenantId,
      startDate
    )

    // ═══════════════════════════════════════════════════════════
    // RETORNAR DADOS
    // ═══════════════════════════════════════════════════════════

    return NextResponse.json({
      summary: {
        totalRevenue,
        receivedRevenue,
        pendingRevenue,
        totalExpenses,
        paidExpenses,
        pendingExpenses,
        totalCommissions,
        paidCommissions,
        pendingCommissions,
        totalProfit,
        cashFlow,
      },
      trips: {
        total: trips.length,
        active: trips.filter((t) => t.status === 'active').length,
        completed: trips.filter((t) => t.status === 'completed').length,
        topProfitable: topProfitableTrips,
      },
      expenses: {
        total: expenses.length,
        byCategory: expensesByCategory,
        byType: expensesByType,
        pending: expenses.filter((e) => e.status === 'pending').length,
      },
      revenues: {
        total: revenues.length,
        pending: revenues.filter((r) => r.status === 'pending').length,
      },
      commissions: {
        total: commissions.length,
        pending: commissions.filter((c) => c.status === 'pending').length,
      },
      reservations: {
        total: reservations.length,
        confirmed: reservations.filter((r) => r.status === 'confirmed').length,
        pending: reservations.filter((r) => r.status === 'pending').length,
      },
      trends: monthlyData,
    })
  } catch (error) {
    console.error('Erro ao buscar dados financeiros:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar dados financeiros' },
      { status: 500 }
    )
  }
}

// Função auxiliar para tendência mensal
async function getMonthlyTrend(tenantId: string, startDate: Date) {
  const months = []
  const currentDate = new Date()

  for (let i = 5; i >= 0; i--) {
    const monthStart = new Date(
      currentDate.getFullYear(),
      currentDate.getMonth() - i,
      1
    )
    const monthEnd = new Date(
      currentDate.getFullYear(),
      currentDate.getMonth() - i + 1,
      0
    )

    const [revenues, expenses, trips] = await Promise.all([
      prisma.revenue.findMany({
        where: {
          tenant_id: tenantId,
          created_at: { gte: monthStart, lte: monthEnd },
        },
      }),
      prisma.expense.findMany({
        where: {
          tenant_id: tenantId,
          created_at: { gte: monthStart, lte: monthEnd },
        },
      }),
      prisma.trip.findMany({
        where: {
          tenant_id: tenantId,
          created_at: { gte: monthStart, lte: monthEnd },
        },
      }),
    ])

    const monthRevenue = revenues.reduce(
      (sum, r) => sum + Number(r.amount),
      0
    )
    const monthExpenses = expenses.reduce(
      (sum, e) => sum + Number(e.amount),
      0
    )
    const monthProfit = trips.reduce(
      (sum, t) => sum + Number(t.total_profit || 0),
      0
    )

    months.push({
      month: monthStart.toLocaleDateString('pt-BR', { month: 'short' }),
      revenue: monthRevenue,
      expenses: monthExpenses,
      profit: monthProfit,
    })
  }

  return months
}

                    <p className="font-semibold">{expense.description}</p>

                      </span>
                    )}
                    {expense.trip && (
                      <span>Viagem: {expense.trip.title}</span>
                    )}

                </div>
              </div>

              <div className="flex items-center gap-4">
                <div className="text-right">
                  <p className="font-bold text-lg text-orange-600">
                    {formatCurrency(expense.amount)}
                  </p>
                  <Badge
                    variant={
                      expense.status === 'paid'
                        ? 'default'
                        : expense.status === 'overdue'
                        ? 'destructive'
                        : 'secondary'
                    }
                    className="mt-1"
                  >
                    {expense.status === 'paid'
                      ? '✅ Pago'
                      : expense.status === 'overdue'
                      ? '⚠️ Vencido'
                      : '⏳ Pendente'}
                  </Badge>
                </div>

                <div className="flex items-center gap-2">
                  {expense.status === 'pending' && (
                    <Button
                      size="sm"
                      variant="outline"
                      onClick={() => handleMarkAsPaid(expense)}
                    >
                      <CheckCircle className="h-4 w-4" />
                    </Button>
                  )}
                  <Button
                    size="sm"
                    variant="outline"
                    onClick={() => handleEdit(expense)}
                  >
                    <Edit className="h-4 w-4" />
                  </Button>
                  <Button
                    size="sm"
                    variant="outline"
                    onClick={() => handleDelete(expense.id)}
                  >
                    <Trash2 className="h-4 w-4 text-red-600" />
                  </Button>
                </div>
              </div>
            </div>
          ))}

          {filteredExpenses.length === 0 && (
            <div className="text-center py-12 text-gray-500">
              <TrendingDown className="h-12 w-12 mx-auto mb-3 opacity-50" />
              <p className="font-semibold mb-1">Nenhuma despesa encontrada</p>
              <p className="text-sm">
                {searchTerm || filterCategory !== 'all' || filterStatus !== 'all'
                  ? 'Tente ajustar os filtros'
                  : 'Cadastre sua primeira despesa'}
              </p>
            </div>
          )}
        </div>
      </Card>
    </div>
  )
}

  status: string
  beneficiary_name: string
  beneficiary_doc: string | null
  due_date: string | null

  trip_id: string | null

  }

const COMMISSION_TYPES = {
  sale: '💰 Comissão de Venda',
  referral: '🤝 Indicação',
  agent: '👤 Agente',
  partner: '🤝 Parceiro',
}

const PAYMENT_METHODS = {
  cash: '💵 Dinheiro',
  pix: '📱 PIX',
  bank_transfer: '🏦 Transferência',
}

export default function CommissionsPage() {
  const [commissions, setCommissions] = useState<Commission[]>([])
  const [isLoading, setIsLoading] = useState(true)
  const [isDialogOpen, setIsDialogOpen] = useState(false)
  const [isSaving, setIsSaving] = useState(false)
  const [editingCommission, setEditingCommission] = useState<Commission | null>(
    null
  )

  // Filtros
  const [searchTerm, setSearchTerm] = useState('')
  const [filterType, setFilterType] = useState('all')
  const [filterStatus, setFilterStatus] = useState('all')

  // Formulário
  const [formData, setFormData] = useState({
    type: 'sale',
    description: '',
    amount: '',
    percentage: '',
    status: 'pending',
    beneficiary_name: '',
    beneficiary_doc: '',
    due_date: '',
    paid_date: '',
    payment_method: '',
    trip_id: '',
  })

    loadCommissions()
  }, [])

    try {
      setIsLoading(true)
      const response = await fetch('/api/dashboard/commissions')
      const data = await response.json()

      if (response.ok) {
        setCommissions(data.commissions || [])
      }
    } catch (error) {
      console.error('Erro ao carregar comissões:', error)
      toast.error('Erro ao carregar comissões')
    } finally {
      setIsLoading(false)
    }
  }

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()

    if (!formData.description || !formData.beneficiary_name || !formData.amount) {
      toast.error('Preencha todos os campos obrigatórios')
      return
    }

    try {
      setIsSaving(true)

      const url = editingCommission
        ? `/api/dashboard/commissions/${editingCommission.id}`
        : '/api/dashboard/commissions'

      const method = editingCommission ? 'PUT' : 'POST'

      const response = await fetch(url, {
        method,
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(formData),
      })

      const data = await response.json()

      if (response.ok) {
        toast.success(
          editingCommission
            ? 'Comissão atualizada com sucesso!'
            : 'Comissão cadastrada com sucesso!'
        )
        setIsDialogOpen(false)
        resetForm()
        loadCommissions()
      } else {
        toast.error(data.error || 'Erro ao salvar comissão')

    } catch (error) {
      console.error('Erro ao salvar comissão:', error)
      toast.error('Erro ao salvar comissão')
    } finally {
      setIsSaving(false)
    }

  const pendingCommissions = filteredCommissions

          {/* Busca */}
          <div className="md:col-span-2">
            <div className="relative">
              <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 h-4 w-4 text-gray-400" />
              <Input
                placeholder="Buscar por descrição ou beneficiário..."
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
                className="pl-10"
              />
            </div>
          </div>

          {/* Filtro Tipo */}
          <div>
            <Select value={filterType} onValueChange={setFilterType}>
              <SelectTrigger>
                <SelectValue placeholder="Tipo" />
              </SelectTrigger>
              <SelectContent>
                <SelectItem value="all">Todos Tipos</SelectItem>
                {Object.entries(COMMISSION_TYPES).map(([key, label]) => (
                  <SelectItem key={key} value={key}>
                    {label}
                  </SelectItem>
                ))}
              </SelectContent>
            </Select>
          </div>

          {/* Filtro Status */}
          <div>
            <Select value={filterStatus} onValueChange={setFilterStatus}>
              <SelectTrigger>
                <SelectValue placeholder="Status" />
              </SelectTrigger>
              <SelectContent>
                <SelectItem value="all">Todos Status</SelectItem>
                <SelectItem value="pending">⏳ Pendente</SelectItem>
                <SelectItem value="paid">✅ Pago</SelectItem>
                <SelectItem value="cancelled">❌ Cancelado</SelectItem>
              </SelectContent>
            </Select>
          </div>
        </div>
      </Card>

      {/* Lista de Comissões */}
      <Card className="p-6">
        <div className="space-y-3">
          {filteredCommissions.map((commission) => (
            <div
              key={commission.id}
              className="flex items-center justify-between p-4 bg-gray-50 rounded-lg border hover:shadow-md transition"
            >
              <div className="flex items-center gap-4 flex-1">
                <div
                  className={`p-3 rounded-lg ${
                    commission.status === 'paid'
                      ? 'bg-green-100'
                      : commission.status === 'cancelled'
                      ? 'bg-red-100'
                      : 'bg-purple-100'
                  }`}
                >
                  {commission.status === 'paid' ? (
                    <CheckCircle className="h-5 w-5 text-green-600" />
                  ) : commission.status === 'cancelled' ? (
                    <Trash2 className="h-5 w-5 text-red-600" />
                  ) : (
                    <Clock className="h-5 w-5 text-purple-600" />
                  )}
                </div>

                <div className="flex-1">
                  <div className="flex items-center gap-2 mb-1">
                    <p className="font-semibold">{commission.description}</p>
                    <Badge variant="outline" className="text-xs">
                      {COMMISSION_TYPES[commission.type as keyof typeof COMMISSION_TYPES]}
                    </Badge>
                    {commission.percentage && (
                      <Badge variant="secondary" className="text-xs">
                        {commission.percentage}%
                      </Badge>
                    )}
                  </div>
                  <div className="flex items-center gap-4 text-sm text-gray-600">
                    <span className="flex items-center gap-1">
                      <Users className="h-3 w-3" />
                      {commission.beneficiary_name}
                    </span>
                    {commission.trip && (
                      <span>Viagem: {commission.trip.title}</span>
                    )}
                  </div>
                </div>
              </div>

              <div className="flex items-center gap-4">
                <div className="text-right">
                  <p className="font-bold text-lg text-purple-600">
                    {formatCurrency(commission.amount)}
                  </p>
                  <Badge
                    variant={
                      commission.status === 'paid'
                        ? 'default'
                        : commission.status === 'cancelled'
                        ? 'destructive'
                        : 'secondary'
                    }
                    className="mt-1"
                  >
                    {commission.status === 'paid'
                      ? '✅ Pago'
                      : commission.status === 'cancelled'
                      ? '❌ Cancelado'
                      : '⏳ Pendente'}
                  </Badge>
                </div>

                <div className="flex items-center gap-2">
                  {commission.status === 'pending' && (
                    <Button
                      size="sm"
                      variant="outline"
                      onClick={() => handleMarkAsPaid(commission)}
                    >
                      <CheckCircle className="h-4 w-4" />
                    </Button>
                  )}
                  <Button
                    size="sm"
                    variant="outline"
                    onClick={() => handleEdit(commission)}
                  >
                    <Edit className="h-4 w-4" />
                  </Button>
                  <Button
                    size="sm"
                    variant="outline"
                    onClick={() => handleDelete(commission.id)}
                  >
                    <Trash2 className="h-4 w-4 text-red-600" />
                  </Button>
                </div>
              </div>
            </div>
          ))}

          {filteredCommissions.length === 0 && (
            <div className="text-center py-12 text-gray-500">
              <Award className="h-12 w-12 mx-auto mb-3 opacity-50" />
              <p className="font-semibold mb-1">Nenhuma comissão encontrada</p>
              <p className="text-sm">
                {searchTerm || filterType !== 'all' || filterStatus !== 'all'
                  ? 'Tente ajustar os filtros'
                  : 'Cadastre sua primeira comissão'}
              </p>
            </div>
          )}
        </div>
      </Card>
    </div>
  )
}

// app/api/dashboard/expenses/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const expenses = await prisma.expense.findMany({
      where: {
        tenant_id: session.user.tenantId,
      },
      include: {
        trip: {
          select: {
            title: true,
          },
        },
      },
      orderBy: {
        created_at: 'desc',
      },
    })

    return NextResponse.json({ expenses })
  } catch (error) {
    console.error('Erro ao buscar despesas:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar despesas' },
      { status: 500 }
    )
  }
}

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const body = await req.json()

    const expense = await prisma.expense.create({
      data: {
        tenant_id: session.user.tenantId,
        created_by: session.user.id,
        description: body.description,
        category: body.category,
        type: body.type,
        amount: parseFloat(body.amount),
        status: body.status,
        due_date: body.due_date ? new Date(body.due_date) : null,
        paid_date: body.paid_date ? new Date(body.paid_date) : null,
        payment_method: body.payment_method || null,
        supplier_name: body.supplier_name || null,
        supplier_doc: body.supplier_doc || null,
        trip_id: body.trip_id || null,
      },
    })

    return NextResponse.json({
      success: true,
      expense,
      message: 'Despesa cadastrada com sucesso!',
    })
  } catch (error: any) {
    console.error('Erro ao criar despesa:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao criar despesa' },
      { status: 500 }
    )
  }
}
