// lib/commissions/calculate-commission.ts

import { prisma } from '@/lib/prisma'

interface CalculateCommissionParams {
  tenantId: string
  userId: string
  saleAmount: number
  reservationId?: string
}

interface CommissionResult {
  commissionType: string
  commissionRate: number
  commissionAmount: number
  saleAmount: number
}

/**
 * Calcula comissão do vendedor baseado em suas configurações
 */
export async function calculateCommission({
  tenantId,
  userId,
  saleAmount,
}: CalculateCommissionParams): Promise<CommissionResult> {
  // Buscar configurações do vendedor
  const user = await prisma.user.findUnique({
    where: { id: userId },
  })

  if (!user) {
    throw new Error('Vendedor não encontrado')
  }

  let commissionAmount = 0
  let commissionRate = Number(user.commission_rate)
  let commissionType = user.commission_type

  switch (user.commission_type) {
    case 'percentage':
      // Comissão percentual
      commissionAmount = (saleAmount * commissionRate) / 100
      break

    case 'fixed':
      // Comissão fixa por venda
      commissionAmount = Number(user.commission_fixed)
      commissionRate = (commissionAmount / saleAmount) * 100
      break

    case 'hybrid':
      // Comissão híbrida (% + fixo)
      const percentageAmount = (saleAmount * commissionRate) / 100
      const fixedAmount = Number(user.commission_fixed)
      commissionAmount = percentageAmount + fixedAmount
      break

    default:
      commissionAmount = 0
  }

  return {
    commissionType,
    commissionRate,
    commissionAmount: Math.round(commissionAmount * 100) / 100,
    saleAmount,
  }
}

/**
 * Cria registro de comissão
 */
export async function createCommission({
  tenantId,
  userId,
  reservationId,
  saleAmount,
  commissionType,
  commissionRate,
  commissionAmount,
  notes,
}: {
  tenantId: string
  userId: string
  reservationId: string
  saleAmount: number
  commissionType: string
  commissionRate: number
  commissionAmount: number
  notes?: string
}) {
  const commission = await prisma.commission.create({
    data: {
      tenant_id: tenantId,
      user_id: userId,
      reservation_id: reservationId,
      sale_amount: saleAmount,
      commission_type: commissionType,
      commission_rate: commissionRate,
      commission_amount: commissionAmount,
      status: 'pending',
      notes,
    },
  })

  // Atualizar reserva com dados de comissão
  await prisma.reservation.update({
    where: { id: reservationId },
    data: {
      commission_rate: commissionRate,
      commission_amount: commissionAmount,
    },
  })

  return commission
}

import { prisma } from '@/lib/prisma'

/**
 * Atualiza metas do vendedor após nova venda
 */
export async function updateSellerGoals({
  tenantId,
  userId,
  saleAmount,
}: {
  tenantId: string
  userId: string
  saleAmount: number
}) {
  const now = new Date()
  const year = now.getFullYear()
  const month = now.getMonth() + 1
  const quarter = Math.ceil(month / 3)

  // ═══════════════════════════════════════════════════════════
  // ATUALIZAR META MENSAL
  // ═══════════════════════════════════════════════════════════
  
  const monthlyGoal = await prisma.salesGoal.findFirst({
    where: {
      tenant_id: tenantId,
      user_id: userId,
      period_type: 'monthly',
      year,
      month,
    },
  })

  if (monthlyGoal) {
    const newAchievedAmount = Number(monthlyGoal.achieved_amount) + saleAmount
    const newAchievedQuantity = monthlyGoal.achieved_quantity + 1
    const progressPercentage = (newAchievedAmount / Number(monthlyGoal.goal_amount)) * 100

    await prisma.salesGoal.update({
      where: { id: monthlyGoal.id },
      data: {
        achieved_amount: newAchievedAmount,
        achieved_quantity: newAchievedQuantity,
        progress_percentage: Math.min(progressPercentage, 100),
        status: progressPercentage >= 100 ? 'completed' : 'active',
      },
    })
  } else {
    // Criar meta mensal se não existir
    const user = await prisma.user.findUnique({
      where: { id: userId },
    })

    if (user?.monthly_goal) {
      await prisma.salesGoal.create({
        data: {
          tenant_id: tenantId,
          user_id: userId,
          period_type: 'monthly',
          year,
          month,
          goal_amount: user.monthly_goal,
          achieved_amount: saleAmount,
          achieved_quantity: 1,
          progress_percentage: (saleAmount / Number(user.monthly_goal)) * 100,
        },
      })
    }
  }

  // ═══════════════════════════════════════════════════════════
  // ATUALIZAR META TRIMESTRAL
  // ═══════════════════════════════════════════════════════════
  
  const quarterlyGoal = await prisma.salesGoal.findFirst({
    where: {
      tenant_id: tenantId,
      user_id: userId,
      period_type: 'quarterly',
      year,
      quarter,
    },
  })

  if (quarterlyGoal) {
    const newAchievedAmount = Number(quarterlyGoal.achieved_amount) + saleAmount
    const progressPercentage = (newAchievedAmount / Number(quarterlyGoal.goal_amount)) * 100

    await prisma.salesGoal.update({
      where: { id: quarterlyGoal.id },
      data: {
        achieved_amount: newAchievedAmount,
        achieved_quantity: quarterlyGoal.achieved_quantity + 1,
        progress_percentage: Math.min(progressPercentage, 100),
        status: progressPercentage >= 100 ? 'completed' : 'active',
      },
    })
  }

  // ═══════════════════════════════════════════════════════════
  // ATUALIZAR META ANUAL
  // ═══════════════════════════════════════════════════════════
  
  const annualGoal = await prisma.salesGoal.findFirst({
    where: {
      tenant_id: tenantId,
      user_id: userId,
      period_type: 'annual',
      year,
    },
  })

  if (annualGoal) {
    const newAchievedAmount = Number(annualGoal.achieved_amount) + saleAmount
    const progressPercentage = (newAchievedAmount / Number(annualGoal.goal_amount)) * 100

    await prisma.salesGoal.update({
      where: { id: annualGoal.id },
      data: {
        achieved_amount: newAchievedAmount,
        achieved_quantity: annualGoal.achieved_quantity + 1,
        progress_percentage: Math.min(progressPercentage, 100),
        status: progressPercentage >= 100 ? 'completed' : 'active',
      },
    })
  }
}

/**
 * Atualiza ranking de vendedores
 */
export async function updateSellerRanking({
  tenantId,
  userId,
}: {
  tenantId: string
  userId: string
}) {
  const now = new Date()
  const year = now.getFullYear()
  const month = now.getMonth() + 1

  // Calcular totais do vendedor no mês
  const monthlyStats = await prisma.reservation.aggregate({
    where: {
      tenant_id: tenantId,
      seller_id: userId,
      created_at: {
        gte: new Date(year, month - 1, 1),
        lt: new Date(year, month, 1),
      },
      status: {
        in: ['confirmed', 'completed'],
      },
    },
    _sum: {
      total_amount: true,
      commission_amount: true,
    },
    _count: true,
  })

  const totalSales = Number(monthlyStats._sum.total_amount || 0)
  const totalCommission = Number(monthlyStats._sum.commission_amount || 0)
  const totalReservations = monthlyStats._count

  // Calcular score (pode ser customizado)
  const rankScore = totalSales + totalReservations * 100

  // Atualizar ou criar ranking
  await prisma.sellerRanking.upsert({
    where: {
      tenant_id_user_id_period_type_year_month_quarter: {
        tenant_id: tenantId,
        user_id: userId,
        period_type: 'monthly',
        year,
        month,
        quarter: null,
      },
    },
    create: {
      tenant_id: tenantId,
      user_id: userId,
      period_type: 'monthly',
      year,
      month,
      total_sales: totalSales,
      total_reservations: totalReservations,
      total_commission: totalCommission,
      rank_score: rankScore,
      rank_position: 0, // Será calculado depois
    },
    update: {
      total_sales: totalSales,
      total_reservations: totalReservations,
      total_commission: totalCommission,
      rank_score: rankScore,
    },
  })

  // Recalcular posições do ranking
  await recalculateRankingPositions(tenantId, year, month)
}

async function recalculateRankingPositions(
  tenantId: string,
  year: number,
  month: number
) {
  const rankings = await prisma.sellerRanking.findMany({
    where: {
      tenant_id: tenantId,
      period_type: 'monthly',
      year,
      month,
    },
    orderBy: {
      rank_score: 'desc',
    },
  })

  for (let i = 0; i < rankings.length; i++) {
    await prisma.sellerRanking.update({
      where: { id: rankings[i].id },
      data: { rank_position: i + 1 },
    })
  }
}

API DE DADOS PARA GRÁFICO DE VENDAS

// app/api/dashboard/seller/sales-chart/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { subDays, format } from 'date-fns'
import { ptBR } from 'date-fns/locale'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const { searchParams } = new URL(req.url)
    const userId = searchParams.get('userId') || session.user.id

    // Últimos 30 dias
    const endDate = new Date()
    const startDate = subDays(endDate, 30)

    // Buscar vendas por dia
    const sales = await prisma.reservation.findMany({
      where: {
        tenant_id: session.user.tenantId,
        seller_id: userId,
        created_at: {
          gte: startDate,
          lte: endDate,
        },
        status: {
          in: ['pending', 'confirmed', 'completed'],
        },
      },
      select: {
        created_at: true,
        total_amount: true,
      },
    })

    // Agrupar por dia
    const dailySales: Record<string, number> = {}

    // Inicializar todos os dias com 0
    for (let i = 0; i <= 30; i++) {
      const date = subDays(endDate, i)
      const dateKey = format(date, 'yyyy-MM-dd')
      dailySales[dateKey] = 0
    }

    // Somar vendas por dia
    sales.forEach((sale) => {
      const dateKey = format(new Date(sale.created_at), 'yyyy-MM-dd')
      dailySales[dateKey] += Number(sale.total_amount)
    })

    // Formatar dados para o gráfico
    const chartData = Object.entries(dailySales)
      .sort(([a], [b]) => a.localeCompare(b))
      .map(([date, amount]) => ({
        date: format(new Date(date), 'dd/MM', { locale: ptBR }),
        amount: Math.round(amount * 100) / 100,
      }))

    return NextResponse.json({ data: chartData })
  } catch (error) {
    console.error('Erro ao buscar dados do gráfico:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar dados' },
      { status: 500 }
    )
  }
}

API DE DADOS PARA GRÁFICO DE COMISSÕES
// app/api/dashboard/seller/commissions-chart/route.ts

import { NextResponse } from 'next/request'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const { searchParams } = new URL(req.url)
    const userId = searchParams.get('userId') || session.user.id

    const currentYear = new Date().getFullYear()

    // Buscar comissões do ano atual agrupadas por mês e status
    const commissions = await prisma.commission.groupBy({
      by: ['status'],
      where: {
        tenant_id: session.user.tenantId,
        user_id: userId,
        created_at: {
          gte: new Date(currentYear, 0, 1),
          lte: new Date(currentYear, 11, 31, 23, 59, 59),
        },
      },
      _sum: {
        commission_amount: true,
      },
    })

    // Buscar comissões por mês
    const monthlyCommissions = await prisma.$queryRaw<any[]>`
      SELECT 
        EXTRACT(MONTH FROM created_at) as month,
        status,
        SUM(commission_amount) as total
      FROM commissions
      WHERE 
        tenant_id = ${session.user.tenantId}
        AND user_id = ${userId}
        AND EXTRACT(YEAR FROM created_at) = ${currentYear}
      GROUP BY EXTRACT(MONTH FROM created_at), status
      ORDER BY month
    `

    // Formatar dados para o gráfico
    const months = [
      'Jan', 'Fev', 'Mar', 'Abr', 'Mai', 'Jun',
      'Jul', 'Ago', 'Set', 'Out', 'Nov', 'Dez',
    ]

    const chartData = months.map((monthName, index) => {
      const monthNumber = index + 1
      const monthData = monthlyCommissions.filter(
        (c) => Number(c.month) === monthNumber
      )

      return {
        month: monthName,
        pending: Number(
          monthData.find((c) => c.status === 'pending')?.total || 0
        ),
        approved: Number(
          monthData.find((c) => c.status === 'approved')?.total || 0
        ),
        paid: Number(
          monthData.find((c) => c.status === 'paid')?.total || 0
        ),
      }
    })

    return NextResponse.json({ data: chartData })
  } catch (error) {
    console.error('Erro ao buscar dados do gráfico:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar dados' },
      { status: 500 }
    )
  }
}

'use client'

import { useEffect, useState } from 'react'
import { Card } from '@/components/ui/card'
import { Badge } from '@/components/ui/badge'
import { Progress } from '@/components/ui/progress'
import {
  TrendingUp,
  TrendingDown,
  DollarSign,
  ShoppingCart,
  Users,
  Target,
  Award,
  Percent,
} from 'lucide-react'

interface PerformanceReportProps {
  userId: string
}

export function PerformanceReport({ userId }: PerformanceReportProps) {
  const [data, setData] = useState<any>(null)
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    loadData()
  }, [userId])

  const loadData = async () => {
    try {
      const response = await fetch(
        `/api/dashboard/seller/performance?userId=${userId}`
      )
      const result = await response.json()
      setData(result)
    } catch (error) {
      console.error('Erro ao carregar relatório:', error)
    } finally {
      setIsLoading(false)
    }
  }

  if (isLoading) {
    return <div>Carregando...</div>
  }

  return (
    <div className="space-y-6">
      {/* Cards de KPIs */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6">
        {/* Total de Vendas */}
        <Card className="p-6">
          <div className="flex items-center justify-between mb-4">
            <div className="w-12 h-12 bg-blue-100 rounded-lg flex items-center justify-center">
              <DollarSign className="h-6 w-6 text-blue-600" />
            </div>
            {data.comparison.sales > 0 ? (
              <Badge className="bg-green-500">
                <TrendingUp className="h-3 w-3 mr-1" />
                +{data.comparison.sales.toFixed(1)}%
              </Badge>
            ) : (
              <Badge className="bg-red-500">
                <TrendingDown className="h-3 w-3 mr-1" />
                {data.comparison.sales.toFixed(1)}%
              </Badge>
            )}
          </div>
          <p className="text-sm text-gray-600 mb-1">Total de Vendas</p>
          <p className="text-2xl font-bold">
            R$ {data.totalSales.toLocaleString('pt-BR', { minimumFractionDigits: 2 })}
          </p>
        </Card>

        {/* Número de Reservas */}
        <Card className="p-6">
          <div className="flex items-center justify-between mb-4">
            <div className="w-12 h-12 bg-purple-100 rounded-lg flex items-center justify-center">
              <ShoppingCart className="h-6 w-6 text-purple-600" />
            </div>
            {data.comparison.reservations > 0 ? (
              <Badge className="bg-green-500">
                <TrendingUp className="h-3 w-3 mr-1" />
                +{data.comparison.reservations.toFixed(1)}%
              </Badge>
            ) : (
              <Badge className="bg-red-500">
                <TrendingDown className="h-3 w-3 mr-1" />
                {data.comparison.reservations.toFixed(1)}%
              </Badge>
            )}
          </div>
          <p className="text-sm text-gray-600 mb-1">Reservas</p>
          <p className="text-2xl font-bold">{data.totalReservations}</p>
        </Card>

        {/* Ticket Médio */}
        <Card className="p-6">
          <div className="flex items-center justify-between mb-4">
            <div className="w-12 h-12 bg-green-100 rounded-lg flex items-center justify-center">
              <Target className="h-6 w-6 text-green-600" />
            </div>
            {data.comparison.avgTicket > 0 ? (
              <Badge className="bg-green-500">
                <TrendingUp className="h-3 w-3 mr-1" />
                +{data.comparison.avgTicket.toFixed(1)}%
              </Badge>
            ) : (
              <Badge className="bg-red-500">
                <TrendingDown className="h-3 w-3 mr-1" />
                {data.comparison.avgTicket.toFixed(1)}%
              </Badge>
            )}
          </div>
          <p className="text-sm text-gray-600 mb-1">Ticket Médio</p>
          <p className="text-2xl font-bold">
            R$ {data.avgTicket.toLocaleString('pt-BR', { minimumFractionDigits: 2 })}
          </p>
        </Card>

        {/* Taxa de Conversão */}
        <Card className="p-6">
          <div className="flex items-center justify-between mb-4">
            <div className="w-12 h-12 bg-orange-100 rounded-lg flex items-center justify-center">
              <Percent className="h-6 w-6 text-orange-600" />
            </div>
            {data.comparison.conversion > 0 ? (
              <Badge className="bg-green-500">
                <TrendingUp className="h-3 w-3 mr-1" />
                +{data.comparison.conversion.toFixed(1)}%
              </Badge>
            ) : (
              <Badge className="bg-red-500">
                <TrendingDown className="h-3 w-3 mr-1" />
                {data.comparison.conversion.toFixed(1)}%
              </Badge>
            )}
          </div>
          <p className="text-sm text-gray-600 mb-1">Taxa de Conversão</p>
          <p className="text-2xl font-bold">{data.conversionRate.toFixed(1)}%</p>
        </Card>
      </div>

      {/* Progresso de Metas */}
      <Card className="p-6">
        <h3 className="text-xl font-bold mb-6">Progresso de Metas</h3>

        <div className="space-y-6">
          {data.goals.map((goal: any) => (
            <div key={goal.id}>
              <div className="flex items-center justify-between mb-2">
                <div>
                  <p className="font-semibold">{goal.title}</p>
                  <p className="text-sm text-gray-600">
                    R$ {goal.achieved.toLocaleString('pt-BR', { minimumFractionDigits: 2 })} de R$ {goal.target.toLocaleString('pt-BR', { minimumFractionDigits: 2 })}
                  </p>
                </div>
                <Badge
                  className={
                    goal.progress >= 100
                      ? 'bg-green-500'
                      : goal.progress >= 75
                      ? 'bg-blue-500'
                      : goal.progress >= 50
                      ? 'bg-yellow-500'
                      : 'bg-red-500'
                  }
                >
                  {goal.progress.toFixed(1)}%
                </Badge>
              </div>
              <Progress value={Math.min(goal.progress, 100)} className="h-3" />
            </div>
          ))}
        </div>
      </Card>

      {/* Top Produtos/Viagens */}
      <Card className="p-6">
        <h3 className="text-xl font-bold mb-6">Top 5 Viagens Mais Vendidas</h3>

        <div className="space-y-4">
          {data.topTrips.map((trip: any, index: number) => (
            <div
              key={trip.id}
              className="flex items-center justify-between p-4 bg-gray-50 rounded-lg"
            >
              <div className="flex items-center gap-4">
                <div className="w-10 h-10 bg-blue-600 text-white rounded-full flex items-center justify-center font-bold">
                  {index + 1}
                </div>
                <div>
                  <p className="font-semibold">{trip.title}</p>
                  <p className="text-sm text-gray-600">
                    {trip.count} venda{trip.count !== 1 ? 's' : ''}
                  </p>
                </div>
              </div>

              <div className="text-right">
                <p className="font-bold text-green-600">
                  R$ {trip.total.toLocaleString('pt-BR', { minimumFractionDigits: 2 })}
                </p>
                <p className="text-xs text-gray-500">
                  Comissão: R$ {trip.commission.toLocaleString('pt-BR', { minimumFractionDigits: 2 })}
                </p>
              </div>
            </div>
          ))}
        </div>
      </Card>
    </div>
  )
}

API DE RELATÓRIO DE PERFORMANCE
// app/api/dashboard/seller/performance/route.ts

import { NextResponse } from 'next/response'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { startOfMonth, endOfMonth, subMonths } from 'date-fns'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const { searchParams } = new URL(req.url)
    const userId = searchParams.get('userId') || session.user.id

    const now = new Date()
    const currentMonthStart = startOfMonth(now)
    const currentMonthEnd = endOfMonth(now)
    const lastMonthStart = startOfMonth(subMonths(now, 1))
    const lastMonthEnd = endOfMonth(subMonths(now, 1))

    // ═══════════════════════════════════════════════════════════
    // DADOS DO MÊS ATUAL
    // ═══════════════════════════════════════════════════════════

    const currentMonth = await prisma.reservation.aggregate({
      where: {
        tenant_id: session.user.tenantId,
        seller_id: userId,
        created_at: {
          gte: currentMonthStart,
          lte: currentMonthEnd,
        },
        status: {
          in: ['pending', 'confirmed', 'completed'],
        },
      },
      _sum: {
        total_amount: true,
      },
      _count: true,
    })

    const totalSales = Number(currentMonth._sum.total_amount || 0)
    const totalReservations = currentMonth._count
    const avgTicket = totalReservations > 0 ? totalSales / totalReservations : 0

    // ═══════════════════════════════════════════════════════════
    // DADOS DO MÊS PASSADO (COMPARAÇÃO)
    // ═══════════════════════════════════════════════════════════

    const lastMonth = await prisma.reservation.aggregate({
      where: {
        tenant_id: session.user.tenantId,
        seller_id: userId,
        created_at: {
          gte: lastMonthStart,
          lte: lastMonthEnd,
        },
        status: {
          in: ['pending', 'confirmed', 'completed'],
        },
      },
      _sum: {
        total_amount: true,
      },
      _count: true,
    })

    const lastMonthSales = Number(lastMonth._sum.total_amount || 0)
    const lastMonthReservations = lastMonth._count
    const lastMonthAvgTicket =
      lastMonthReservations > 0 ? lastMonthSales / lastMonthReservations : 0

    // Calcular variações percentuais
    const salesComparison =
      lastMonthSales > 0
        ? ((totalSales - lastMonthSales) / lastMonthSales) * 100
        : 0

    const reservationsComparison =
      lastMonthReservations > 0
        ? ((totalReservations - lastMonthReservations) / lastMonthReservations) * 100
        : 0

    const avgTicketComparison =
      lastMonthAvgTicket > 0
        ? ((avgTicket - lastMonthAvgTicket) / lastMonthAvgTicket) * 100
        : 0

    // ═══════════════════════════════════════════════════════════
    // METAS
    // ═══════════════════════════════════════════════════════════

    const goals = await prisma.salesGoal.findMany({
      where: {
        tenant_id: session.user.tenantId,
        user_id: userId,
        year: now.getFullYear(),
        month: now.getMonth() + 1,
      },
    })

    const goalsFormatted = goals.map((goal) => ({
      id: goal.id,
      title:
        goal.period_type === 'monthly'
          ? 'Meta Mensal'
          : goal.period_type === 'quarterly'
          ? 'Meta Trimestral'
          : 'Meta Anual',
      target: Number(goal.goal_amount),
      achieved: Number(goal.achieved_amount),
      progress: Number(goal.progress_percentage),
    }))

    // ═══════════════════════════════════════════════════════════
    // TOP VIAGENS
    // ═══════════════════════════════════════════════════════════

    const topTrips = await prisma.reservation.groupBy({
      by: ['trip_id'],
      where: {
        tenant_id: session.user.tenantId,
        seller_id: userId,
        created_at: {
          gte: currentMonthStart,
          lte: currentMonthEnd,
        },
        status: {
          in: ['pending', 'confirmed', 'completed'],
        },
      },
      _sum: {
        total_amount: true,
        commission_amount: true,
      },
      _count: true,
      orderBy: {
        _sum: {
          total_amount: 'desc',
        },
      },
      take: 5,
    })

    const topTripsWithDetails = await Promise.all(
      topTrips.map(async (trip) => {
        const tripDetails = await prisma.trip.findUnique({
          where: { id: trip.trip_id },
          select: { title: true },
        })

        return {
          id: trip.trip_id,
          title: tripDetails?.title || 'Viagem',
          count: trip._count,
          total: Number(trip._sum.total_amount || 0),
          commission: Number(trip._sum.commission_amount || 0),
        }
      })
    )

    return NextResponse.json({
      totalSales,
      totalReservations,
      avgTicket,
      conversionRate: 0, // Calcular baseado em leads/deals
      comparison: {
        sales: salesComparison,
        reservations: reservationsComparison,
        avgTicket: avgTicketComparison,
        conversion: 0,
      },
      goals: goalsFormatted,
      topTrips: topTripsWithDetails,
    })
  } catch (error) {
    console.error('Erro ao gerar relatório:', error)
    return NextResponse.json(
      { error: 'Erro ao gerar relatório' },
      { status: 500 }
    )
  }
}
