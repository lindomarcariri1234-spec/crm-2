// lib/referral/generate-code.ts

import { prisma } from '@/lib/prisma'

export async function generateReferralCode(
  clientName: string,
  tenantId: string
): Promise<string> {
  // Gerar código base a partir do nome
  const nameParts = clientName.trim().split(' ')
  const firstName = nameParts[0].toUpperCase()
  const year = new Date().getFullYear()
  
  let baseCode = `${firstName}${year}`
  let code = baseCode
  let suffix = 1

  // Verificar se o código já existe
  while (true) {
    const existing = await prisma.client.findFirst({
      where: {
        tenant_id: tenantId,
        referral_code: code,
      },
    })

    if (!existing) {
      break
    }

    // Se existir, adicionar sufixo
    code = `${baseCode}${suffix}`
    suffix++
  }

  return code
}

// lib/referral/tracking.ts

import { cookies } from 'next/headers'
import { prisma } from '@/lib/prisma'
import { v4 as uuidv4 } from 'uuid'

interface TrackingData {
  referralCode: string
  utmSource?: string
  utmMedium?: string
  utmCampaign?: string
  utmContent?: string
  utmTerm?: string
  ipAddress?: string
  userAgent?: string
  landingPage?: string
}

export async function trackReferralVisit(
  tenantId: string,
  data: TrackingData
) {
  const cookieStore = cookies()
  
  // Buscar ou criar cookie ID
  let cookieId = cookieStore.get('referral_tracking_id')?.value
  
  if (!cookieId) {
    cookieId = uuidv4()
    cookieStore.set('referral_tracking_id', cookieId, {
      maxAge: 60 * 60 * 24 * 30, // 30 dias
      httpOnly: true,
      secure: process.env.NODE_ENV === 'production',
      sameSite: 'lax',
    })
  }

  // Salvar código de indicação no cookie
  cookieStore.set('referral_code', data.referralCode, {
    maxAge: 60 * 60 * 24 * 30, // 30 dias
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'lax',
  })

  // Detectar dispositivo
  const userAgent = data.userAgent || ''
  const deviceType = detectDeviceType(userAgent)
  const browser = detectBrowser(userAgent)
  const os = detectOS(userAgent)

  // Buscar tracking existente
  const existingTracking = await prisma.referralTracking.findUnique({
    where: { cookie_id: cookieId },
  })

  if (existingTracking) {
    // Atualizar tracking existente
    await prisma.referralTracking.update({
      where: { cookie_id: cookieId },
      data: {
        last_visit: new Date(),
        visits_count: { increment: 1 },
        pages_visited: {
          push: data.landingPage || '/',
        },
      },
    })
  } else {
    // Criar novo tracking
    await prisma.referralTracking.create({
      data: {
        tenant_id: tenantId,
        cookie_id: cookieId,
        referral_code: data.referralCode,
        ip_address: data.ipAddress,
        user_agent: userAgent,
        device_type: deviceType,
        browser,
        os,
        utm_source: data.utmSource,
        utm_medium: data.utmMedium,
        utm_campaign: data.utmCampaign,
        utm_content: data.utmContent,
        utm_term: data.utmTerm,
        pages_visited: [data.landingPage || '/'],
      },
    })
  }

  // Atualizar referral
  await prisma.referral.updateMany({
    where: {
      tenant_id: tenantId,
      code: data.referralCode,
      cookie_id: null,
    },
    data: {
      cookie_id: cookieId,
      first_visit: new Date(),
      visits_count: { increment: 1 },
      ip_address: data.ipAddress,
      user_agent: userAgent,
      landing_page: data.landingPage,
      utm_source: data.utmSource,
      utm_medium: data.utmMedium,
      utm_campaign: data.utmCampaign,
    },
  })

  return cookieId
}

// Utilitários de detecção
function detectDeviceType(userAgent: string): string {
  if (/mobile/i.test(userAgent)) return 'mobile'
  if (/tablet/i.test(userAgent)) return 'tablet'
  return 'desktop'
}

function detectBrowser(userAgent: string): string {
  if (/chrome/i.test(userAgent)) return 'Chrome'
  if (/firefox/i.test(userAgent)) return 'Firefox'
  if (/safari/i.test(userAgent)) return 'Safari'
  if (/edge/i.test(userAgent)) return 'Edge'
  return 'Unknown'
}

function detectOS(userAgent: string): string {
  if (/windows/i.test(userAgent)) return 'Windows'
  if (/mac/i.test(userAgent)) return 'MacOS'
  if (/linux/i.test(userAgent)) return 'Linux'
  if (/android/i.test(userAgent)) return 'Android'
  if (/ios/i.test(userAgent)) return 'iOS'
  return 'Unknown'
}

    utm_source?: string
    utm_medium?: string
    utm_campaign?: string
    utm_content?: string
    utm_term?: string
  }

export default async function ReferralPage({
  params,
  searchParams,
}: ReferralPageProps) {
  const { code } = params
  const headersList = headers()
  
  // Buscar indicação
  const referral = await prisma.referral.findUnique({
    where: { code: code.toUpperCase() },
    include: { referrer: true },
  })

    redirect('/loja')
  }

  // Verificar expiração
  if (referral.expires_at && new Date() > referral.expires_at) {
    await prisma.referral.update({
      where: { id: referral.id },
      data: { status: 'expired', is_active: false },
    })
    redirect('/loja')
  }

  // Rastrear visita
  await trackReferralVisit(referral.tenant_id, {
    referralCode: code.toUpperCase(),
    utmSource: searchParams.utm_source,
    utmMedium: searchParams.utm_medium,
    utmCampaign: searchParams.utm_campaign,
    utmContent: searchParams.utm_content,
    utmTerm: searchParams.utm_term,
    ipAddress: headersList.get('x-forwarded-for') || undefined,
    userAgent: headersList.get('user-agent') || undefined,

  })

// app/api/public/validate-referral/route.ts

import { NextResponse } from 'next/response'
import { prisma } from '@/lib/prisma'
import { cookies } from 'next/headers'

export async function POST(req: Request) {
  try {
    const { code } = await req.json()
    const cookieStore = cookies()

    if (!code) {
      return NextResponse.json(
        { error: 'Código não informado' },
        { status: 400 }
      )
    }

    // Buscar referral
    const referral = await prisma.referral.findUnique({
      where: { code: code.toUpperCase() },
      include: { referrer: true },
    })

    if (!referral) {
      return NextResponse.json(
        { error: 'Código de indicação inválido' },
        { status: 404 }
      )
    }

    // Verificar se está ativo
    if (!referral.is_active) {
      return NextResponse.json(
        { error: 'Código de indicação inativo' },
        { status: 400 }
      )
    }

    // Verificar expiração
    if (referral.expires_at && new Date() > referral.expires_at) {
      await prisma.referral.update({
        where: { id: referral.id },
        data: { status: 'expired', is_active: false },
      })

      return NextResponse.json(
        { error: 'Código de indicação expirado' },
        { status: 400 }
      )
    }

    // Verificar auto-indicação (se tiver cookie do indicador)
    const referrerCookieId = cookieStore.get('client_id')?.value
    if (referrerCookieId === referral.referrer_id) {
      return NextResponse.json(
        { error: 'Você não pode usar seu próprio código de indicação' },
        { status: 400 }
      )
    }

    // Salvar código no cookie
    cookieStore.set('referral_code', code.toUpperCase(), {
      maxAge: 60 * 60 * 24 * 30, // 30 dias
      httpOnly: true,
      secure: process.env.NODE_ENV === 'production',
      sameSite: 'lax',
    })

    return NextResponse.json({
      success: true,
      referral: {
        code: referral.code,
        referrer_name: referral.referrer.name,
        discount_type: referral.discount_type,
        discount_value: referral.discount_value,
      },
    })
  } catch (error: any) {
    console.error('Erro ao validar código:', error)
    return NextResponse.json(
      { error: 'Erro ao validar código' },
      { status: 500 }
    )
  }
}

APLICAÇÃO DE DESCONTO E BÔNUS

// lib/referral/apply-benefits.ts

import { prisma } from '@/lib/prisma'

interface ApplyBenefitsParams {
  referralCode: string
  reservationId: string
  totalAmount: number
  tenantId: string
}

export async function applyReferralBenefits({
  referralCode,
  reservationId,
  totalAmount,
  tenantId,
}: ApplyBenefitsParams) {
  // Buscar referral
  const referral = await prisma.referral.findUnique({
    where: { code: referralCode },
    include: { referrer: true },
  })

  if (!referral) {
    throw new Error('Código de indicação não encontrado')
  }

  // Buscar configurações
  const settings = await prisma.referralSettings.findUnique({
    where: { tenant_id: tenantId },
  })

  if (!settings || !settings.is_enabled) {
    throw new Error('Programa de indicações não está ativo')
  }

  // Calcular desconto para o indicado
  let discountAmount = 0

  if (referral.discount_type === 'percentage') {
    discountAmount = (totalAmount * Number(referral.discount_value)) / 100
  } else {
    discountAmount = Number(referral.discount_value)
  }

  // Aplicar desconto máximo se configurado
  if (settings.max_discount && discountAmount > Number(settings.max_discount)) {
    discountAmount = Number(settings.max_discount)
  }

  // Calcular bônus para o indicador
  let bonusAmount = 0

  if (settings.bonus_type === 'percentage') {
    bonusAmount = (totalAmount * Number(settings.bonus_value)) / 100
  } else if (settings.bonus_type === 'credit') {
    bonusAmount = Number(settings.bonus_value)
  }

  // Atualizar referral
  await prisma.referral.update({
    where: { id: referral.id },
    data: {
      status: 'completed',
      converted_at: new Date(),
      reservation_id: reservationId,
      discount_applied: true,
      discount_amount: discountAmount,
      bonus_value: bonusAmount,
    },
  })

  // Atualizar estatísticas do indicador
  await prisma.client.update({
    where: { id: referral.referrer_id },
    data: {
      successful_referrals: { increment: 1 },
      referral_earnings: { increment: bonusAmount },
    },
  })

  // Criar crédito para o indicador (se configurado)
  if (settings.bonus_type === 'credit') {
    await prisma.clientCredit.create({
      data: {
        tenant_id: tenantId,
        client_id: referral.referrer_id,
        amount: bonusAmount,
        type: 'referral_bonus',
        description: `Bônus por indicação - Reserva ${reservationId}`,
        expires_at: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000), // 1 ano
        status: 'active',
      },
    })
  }

  // Atualizar tracking
  const cookieId = referral.cookie_id
  if (cookieId) {
    await prisma.referralTracking.updateMany({
      where: { cookie_id: cookieId },
      data: {
        converted: true,
        converted_at: new Date(),
        reservation_id: reservationId,
      },
    })
  }

  return {
    discountAmount,
    bonusAmount,
  }
}

export default async function ReferralPage() {
  const session = await getServerSession(authOptions)

  if (!session) {
    return null
  }

  // Buscar estatísticas
  const stats = await prisma.referral.groupBy({
    by: ['status'],
    where: { tenant_id: session.user.tenantId },
    _count: true,
    _sum: {
      discount_amount: true,
      bonus_value: true,
    },
  })

  // Buscar top indicadores
  const topReferrers = await prisma.client.findMany({
    where: {
      tenant_id: session.user.tenantId,
      successful_referrals: { gt: 0 },
    },
    orderBy: { successful_referrals: 'desc' },
    take: 10,
    select: {
      id: true,
      name: true,
      email: true,
      referral_code: true,
      total_referrals: true,
      successful_referrals: true,
      referral_earnings: true,
    },
  })

  // Buscar indicações recentes
  const recentReferrals = await prisma.referral.findMany({
    where: { tenant_id: session.user.tenantId },
    include: {
      referrer: true,
      referred: true,
      reservation: true,
    },
    orderBy: { created_at: 'desc' },
    take: 20,
  })

  return (
    <div className="container mx-auto py-8 px-4">
      <div className="mb-8">
        <h1 className="text-3xl font-bold">Programa de Indicações</h1>
        <p className="text-gray-600">
          Gerencie e acompanhe o programa de indicações da sua agência
        </p>
      </div>

      {/* Estatísticas */}
      <ReferralStats stats={stats} />

      {/* Top Indicadores */}
      <div className="mt-8">
        <h2 className="text-2xl font-bold mb-4">🏆 Top Indicadores</h2>
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
          {topReferrers.map((referrer, index) => (
            <Card key={referrer.id} className="p-4">
              <div className="flex items-center gap-3">
                <div className="text-3xl">
                  {index === 0 ? '🥇' : index === 1 ? '🥈' : index === 2 ? '🥉' : '🎖️'}
                </div>
                <div className="flex-1">
                  <p className="font-semibold">{referrer.name}</p>
                  <p className="text-sm text-gray-600">
                    Código: {referrer.referral_code}
                  </p>
                </div>
                <div className="text-right">
                  <p className="text-2xl font-bold text-green-600">
                    {referrer.successful_referrals}
                  </p>
                  <p className="text-xs text-gray-600">indicações</p>
                </div>
              </div>
              <div className="mt-3 pt-3 border-t">
                <p className="text-sm text-gray-600">
                  Ganhou:{' '}
                  <span className="font-semibold text-green-600">
                    R$ {Number(referrer.referral_earnings).toFixed(2)}
                  </span>
                </p>
              </div>
            </Card>
          ))}
        </div>
      </div>

      {/* Lista de Indicações */}
      <ReferralList referrals={recentReferrals} />

      {/* Configurações */}
      <ReferralSettings />
    </div>
  )
}
