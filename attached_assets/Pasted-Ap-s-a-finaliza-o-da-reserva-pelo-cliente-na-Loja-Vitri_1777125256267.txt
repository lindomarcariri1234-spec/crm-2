// lib/utils/password-generator.ts

import crypto from 'crypto'
import bcrypt from 'bcryptjs'

/**
 * Gera uma senha aleatória segura
 */
export function generateTempPassword(): string {
  const length = 8
  const charset = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789!@#$%'
  let password = ''

  // Garantir pelo menos 1 maiúscula, 1 minúscula, 1 número e 1 caractere especial
  password += 'ABCDEFGHIJKLMNOPQRSTUVWXYZ'[Math.floor(Math.random() * 26)]
  password += 'abcdefghijklmnopqrstuvwxyz'[Math.floor(Math.random() * 26)]
  password += '0123456789'[Math.floor(Math.random() * 10)]
  password += '!@#$%'[Math.floor(Math.random() * 5)]

  // Completar o resto
  for (let i = password.length; i < length; i++) {
    password += charset[Math.floor(Math.random() * charset.length)]
  }

  // Embaralhar
  return password
    .split('')
    .sort(() => Math.random() - 0.5)
    .join('')
}

/**
 * Hash de senha
 */
export async function hashPassword(password: string): Promise<string> {
  return bcrypt.hash(password, 10)
}

/**
 * Verificar senha
 */
export async function verifyPassword(
  password: string,
  hashedPassword: string
): Promise<boolean> {
  return bcrypt.compare(password, hashedPassword)
}

API - CRIAR RESERVA COM CONTA AUTOMÁTICA
// app/api/public/reservations/route.ts

import { NextResponse } from 'next/request'
import { prisma } from '@/lib/prisma'
import { generateTempPassword, hashPassword } from '@/lib/utils/password-generator'
import { sendReservationConfirmationEmail } from '@/lib/services/email-service'
import { z } from 'zod'

const reservationSchema = z.object({
  tripId: z.string(),
  customer: z.object({
    name: z.string().min(1),
    email: z.string().email(),
    phone: z.string(),
    cpf: z.string(),
  }),
  passengers: z.array(
    z.object({
      seatId: z.string(),
      seatNumber: z.string(),
      name: z.string(),
      cpf: z.string(),
      age: z.string().optional(),
    })
  ),
  seats: z.array(z.string()),
  totalPrice: z.number(),
  paymentMethod: z.string(),
  notes: z.string().optional(),
})

export async function POST(req: Request) {
  try {
    const body = await req.json()
    const validatedData = reservationSchema.parse(body)

    // Buscar viagem
    const trip = await prisma.trip.findUnique({
      where: { id: validatedData.tripId },
      include: {
        destination: true,
        tenant: true,
      },
    })

    if (!trip) {
      return NextResponse.json({ error: 'Viagem não encontrada' }, { status: 404 })
    }

    // Verificar disponibilidade dos assentos
    const seats = await prisma.seatLayout.findMany({
      where: {
        id: { in: validatedData.seats },
        trip_id: validatedData.tripId,
      },
      include: { reservations: true },
    })

    const occupiedSeats = seats.filter(
      (seat) => seat.reservations.length > 0 || seat.is_blocked
    )

    if (occupiedSeats.length > 0) {
      return NextResponse.json(
        {
          error: `Assentos não disponíveis: ${occupiedSeats
            .map((s) => s.seat_number)
            .join(', ')}`,
        },
        { status: 400 }
      )
    }

    // ═══════════════════════════════════════════════════════════
    // BUSCAR OU CRIAR CLIENTE
    // ═══════════════════════════════════════════════════════════

    let customer = await prisma.customer.findFirst({
      where: {
        tenant_id: trip.tenant_id,
        email: validatedData.customer.email,
      },
      include: { user: true },
    })

    let tempPassword: string | null = null
    let userCreated = false

    if (!customer) {
      // Cliente não existe - criar novo
      customer = await prisma.customer.create({
        data: {
          tenant_id: trip.tenant_id,
          name: validatedData.customer.name,
          email: validatedData.customer.email,
          phone: validatedData.customer.phone,
          cpf: validatedData.customer.cpf,
        },
      })
    }

    // ═══════════════════════════════════════════════════════════
    // CRIAR CONTA DE ACESSO (SE NÃO EXISTIR)
    // ═══════════════════════════════════════════════════════════

    if (!customer.has_account || !customer.user_id) {
      // Gerar senha provisória
      tempPassword = generateTempPassword()
      const hashedPassword = await hashPassword(tempPassword)

      // Criar usuário
      const user = await prisma.user.create({
        data: {
          tenant_id: trip.tenant_id,
          name: validatedData.customer.name,
          email: validatedData.customer.email,
          password: hashedPassword,
          role: 'client',
          is_temp_password: true,
          must_change_password: true,
        },
      })

      // Atualizar customer
      await prisma.customer.update({
        where: { id: customer.id },
        data: {
          has_account: true,
          user_id: user.id,
        },
      })

      customer.user_id = user.id
      userCreated = true
    }

    // ═══════════════════════════════════════════════════════════
    // CRIAR RESERVA
    // ═══════════════════════════════════════════════════════════

    const result = await prisma.$transaction(async (tx) => {
      // Criar reserva
      const reservation = await tx.reservation.create({
        data: {
          tenant_id: trip.tenant_id,
          trip_id: validatedData.tripId,
          customer_id: customer!.id,
          passengers_count: validatedData.passengers.length,
          passengers_data: validatedData.passengers,
          total_price: validatedData.totalPrice,
          paid_amount: 0,
          status: 'pending',
          payment_status: 'pending',
          notes: validatedData.notes,
        },
      })

      // Criar reservas de assentos
      await tx.seatReservation.createMany({
        data: validatedData.passengers.map((passenger) => ({
          reservation_id: reservation.id,
          seat_layout_id: passenger.seatId,
          passenger_name: passenger.name,
          passenger_cpf: passenger.cpf,
          passenger_age: passenger.age ? parseInt(passenger.age) : null,
        })),
      })

      // Atualizar assentos disponíveis
      await tx.trip.update({
        where: { id: validatedData.tripId },
        data: {
          available_seats: { decrement: validatedData.seats.length },
        },
      })

      // Criar receita
      await tx.revenue.create({
        data: {
          tenant_id: trip.tenant_id,
          customer_id: customer!.id,
          reservation_id: reservation.id,
          amount: validatedData.totalPrice,
          description: `Reserva #${reservation.id.slice(0, 8)} - ${trip.name}`,
          category: 'sale',
          status: 'pending',
          due_date: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000),
          payment_method: validatedData.paymentMethod,
        },
      })

      return reservation
    })

    // ═══════════════════════════════════════════════════════════
    // ENVIAR EMAIL DE CONFIRMAÇÃO
    // ═══════════════════════════════════════════════════════════

    if (userCreated && tempPassword) {
      const loginUrl = `${process.env.NEXT_PUBLIC_APP_URL}/meu-perfil`

      try {
        await sendReservationConfirmationEmail({
          customerName: validatedData.customer.name,
          customerEmail: validatedData.customer.email,
          reservationId: result.id,
          tripName: trip.name,
          departureDate: trip.departure_date.toISOString(),
          returnDate: trip.return_date.toISOString(),
          totalPrice: validatedData.totalPrice,
          seats: validatedData.passengers.map((p) => p.seatNumber),
          tempPassword,
          loginUrl,
        })
      } catch (emailError) {
        console.error('Erro ao enviar email:', emailError)
        // Não falhar a reserva se o email falhar
      }
    }

    // Buscar reserva completa
    const fullReservation = await prisma.reservation.findUnique({
      where: { id: result.id },
      include: {
        customer: true,
        trip: { include: { destination: true } },
        seat_reservations: { include: { seat_layout: true } },
      },
    })

    return NextResponse.json({
      success: true,
      reservation: fullReservation,
      accountCreated: userCreated,
      tempPassword: userCreated ? tempPassword : null,
      message: userCreated
        ? 'Reserva criada! Verifique seu email para acessar seu perfil.'
        : 'Reserva criada com sucesso!',
    })
  } catch (error) {
    console.error('Erro ao criar reserva:', error)

    if (error instanceof z.ZodError) {
      return NextResponse.json({ error: error.errors[0].message }, { status: 400 })
    }

    return NextResponse.json({ error: 'Erro ao criar reserva' }, { status: 500 })
  }
}

      setError('Erro ao fazer login. Tente novamente.')
    } finally {
      setIsLoading(false)
    }
  }

  return (
    <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-blue-50 to-indigo-100 p-4">
      <Card className="w-full max-w-md p-8 shadow-2xl">
        {/* Header */}
        <div className="text-center mb-8">
          <div className="mx-auto w-16 h-16 bg-gradient-to-br from-blue-500 to-purple-600 rounded-full flex items-center justify-center mb-4">
            <Lock className="h-8 w-8 text-white" />
          </div>
          <h1 className="text-3xl font-bold text-gray-900 mb-2">Meu Perfil</h1>
          <p className="text-gray-600">Acesse sua conta para ver suas reservas</p>
        </div>

        {/* Error Alert */}
        {error && (
          <Alert variant="destructive" className="mb-6">
            <AlertCircle className="h-4 w-4" />
            <AlertDescription>{error}</AlertDescription>
          </Alert>
        )}

        {/* Form */}
        <form onSubmit={handleSubmit} className="space-y-6">
          <div>
            <Label htmlFor="email">Email</Label>
            <div className="relative mt-1">
              <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-5 w-5 text-gray-400" />
              <Input
                id="email"
                type="email"
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                placeholder="seu@email.com"
                className="pl-10"
                required
                disabled={isLoading}
              />
            </div>
          </div>

          <div>
            <Label htmlFor="password">Senha</Label>
            <div className="relative mt-1">
              <Lock className="absolute left-3 top-1/2 -translate-y-1/2 h-5 w-5 text-gray-400" />
              <Input
                id="password"
                type="password"
                value={password}
                onChange={(e) => setPassword(e.target.value)}
                placeholder="••••••••"
                className="pl-10"
                required
                disabled={isLoading}
              />
            </div>
          </div>

          <Button
            type="submit"
            className="w-full bg-gradient-to-r from-blue-600 to-purple-600 hover:from-blue-700 hover:to-purple-700"
            disabled={isLoading}
          >
            {isLoading ? (
              <>
                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
                Entrando...
              </>
            ) : (
              'Entrar'
            )}
          </Button>
        </form>

        {/* Links */}
        <div className="mt-6 text-center space-y-2">
          <Link
            href="/meu-perfil/esqueci-senha"
            className="text-sm text-blue-600 hover:underline block"
          >
            Esqueci minha senha
          </Link>
          <Link
            href="/loja"
            className="text-sm text-gray-600 hover:underline block"
          >
            ← Voltar para a loja
          </Link>
        </div>

        {/* Info */}
        <Alert className="mt-6 bg-blue-50 border-blue-200">
          <AlertCircle className="h-4 w-4 text-blue-600" />
          <AlertDescription className="text-sm text-blue-900">
            <strong>Primeira vez?</strong> Ao fazer uma reserva, criamos automaticamente
            uma conta para você. Verifique seu email!
          </AlertDescription>
        </Alert>
      </Card>
    </div>
  )
}
