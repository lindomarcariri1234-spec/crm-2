// lib/utils/trip-duration.ts

import { differenceInHours, differenceInDays } from 'date-fns'

interface DurationResult {
  days: number
  hours: number
  formatted: string
  formattedShort: string
}

/**
 * Calcula a duração precisa de uma viagem
 * considerando data/hora de embarque e desembarque
 */
export function calculateTripDuration(
  departureDate: Date,
  departureTime: string, // "20:00"
  returnDate: Date,
  returnTime: string // "02:00"
): DurationResult {
  // Criar timestamps completos
  const [depHour, depMinute] = departureTime.split(':').map(Number)
  const [retHour, retMinute] = returnTime.split(':').map(Number)

  const departure = new Date(departureDate)
  departure.setHours(depHour, depMinute, 0, 0)

  const returnDateTime = new Date(returnDate)
  returnDateTime.setHours(retHour, retMinute, 0, 0)

  // Calcular diferença total em horas
  const totalHours = differenceInHours(returnDateTime, departure)

  // Separar dias e horas
  const days = Math.floor(totalHours / 24)
  const hours = totalHours % 24

  // Formatar texto
  let formatted = ''
  let formattedShort = ''

  if (days > 0 && hours > 0) {
    formatted = `${days} ${days === 1 ? 'dia' : 'dias'} e ${hours} ${
      hours === 1 ? 'hora' : 'horas'
    }`
    formattedShort = `${days}d ${hours}h`
  } else if (days > 0) {
    formatted = `${days} ${days === 1 ? 'dia' : 'dias'}`
    formattedShort = `${days}d`
  } else {
    formatted = `${hours} ${hours === 1 ? 'hora' : 'horas'}`
    formattedShort = `${hours}h`
  }

  return {
    days,
    hours,
    formatted,
    formattedShort,
  }
}

/**
 * Versão simplificada para itinerários
 */
export function getItineraryDuration(itinerary: {
  departure_date: Date
  departure_time: string
  return_date: Date
  return_time: string
}): string {
  const duration = calculateTripDuration(
    itinerary.departure_date,
    itinerary.departure_time,
    itinerary.return_date,
    itinerary.return_time
  )

  return duration.formatted
}

/**
 * Versão curta para badges e cards
 */
export function getItineraryDurationShort(itinerary: {
  departure_date: Date
  departure_time: string
  return_date: Date
  return_time: string
}): string {
  const duration = calculateTripDuration(
    itinerary.departure_date,
    itinerary.departure_time,
    itinerary.return_date,
    itinerary.return_time
  )

  return duration.formattedShort
}
