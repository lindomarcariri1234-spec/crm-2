// TABELA DE ASSENTOS RESERVADOS
// ═══════════════════════════════════════════════════════════════

model ReservationSeat {
  id              String    @id @default(cuid())
  
  // RELACIONAMENTOS
  reservation_id  String
  reservation     Reservation @relation(fields: [reservation_id], references: [id], onDelete: Cascade)
  
  trip_id         String
  trip            Trip      @relation(fields: [trip_id], references: [id])
  
  itinerary_id    String
  itinerary       Itinerary @relation(fields: [itinerary_id], references: [id])
  
  // DADOS DO ASSENTO
  seat_number     String    // "8", "15", "31"
  passenger_name  String?   // Nome do passageiro (se diferente do titular)
  passenger_cpf   String?   // CPF do passageiro
  
  // STATUS
  status          String    @default("reserved") // reserved, confirmed, checked_in, cancelled
  
  // CHECK-IN
  checked_in      Boolean   @default(false)
  checked_in_at   DateTime?
  checked_in_by   String?   // ID do usuário que fez check-in
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  @@unique([trip_id, itinerary_id, seat_number])
  @@index([reservation_id])
  @@index([trip_id, itinerary_id])
  @@map("reservation_seats")
}

import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { eventEmitter } from '@/lib/events/event-emitter'
import { createDealFromReservation } from '@/lib/deals/create-from-reservation'

export async function POST(req: Request) {
  try {
    const body = await req.json()

    const {
      trip_id,
      itinerary_id,
      name,
      cpf,
      email,
      phone,
      birthdate,
      passengers_count,
      selected_seats, // ["8"] ou ["8", "15", "31"]
      referral_code,
      coupon_code,
      referral_discount,
      coupon_discount,
      payment_method,
      observations,
    } = body

    // Validações
    if (!trip_id || !itinerary_id || !name || !cpf || !email || !phone) {
      return NextResponse.json(
        { error: 'Campos obrigatórios faltando' },
        { status: 400 }
      )
    }

    if (!selected_seats || selected_seats.length === 0) {
      return NextResponse.json(
        { error: 'Nenhum assento selecionado' },
        { status: 400 }
      )
    }

    if (selected_seats.length !== passengers_count) {
      return NextResponse.json(
        { error: 'Quantidade de assentos não corresponde ao número de passageiros' },
        { status: 400 }
      )
    }

    // Buscar viagem
    const trip = await prisma.trip.findUnique({
      where: { id: trip_id },
      include: { itineraries: true },
    })

    if (!trip) {
      return NextResponse.json(
        { error: 'Viagem não encontrada' },
        { status: 404 }
      )
    }

    const itinerary = trip.itineraries.find((it) => it.id === itinerary_id)

    if (!itinerary) {
      return NextResponse.json(
        { error: 'Itinerário não encontrado' },
        { status: 404 }
      )
    }

    // Verificar se assentos estão disponíveis
    const occupiedSeats = await prisma.reservationSeat.findMany({
      where: {
        trip_id,
        itinerary_id,
        seat_number: { in: selected_seats },
        status: { in: ['reserved', 'confirmed', 'checked_in'] },
      },
    })

    if (occupiedSeats.length > 0) {
      return NextResponse.json(
        {
          error: 'Um ou mais assentos já estão ocupados',
          occupied: occupiedSeats.map((s) => s.seat_number),
        },
        { status: 400 }
      )
    }

    // Calcular valores
    const basePrice = trip.promotional_price || trip.price_per_person
    const subtotal = basePrice * passengers_count
    const totalDiscount = (referral_discount || 0) + (coupon_discount || 0)
    const totalAmount = subtotal - totalDiscount

    // Criar ou atualizar cliente
    let client = await prisma.client.findFirst({
      where: {
        tenant_id: trip.tenant_id,
        OR: [{ cpf }, { email }],
      },
    })

    if (!client) {
      client = await prisma.client.create({
        data: {
          tenant_id: trip.tenant_id,
          name,
          cpf,
          email,
          phone,
          birthdate: new Date(birthdate),
          source: 'website',
          segment: 'individual',
          status: 'active',
        },
      })
    } else {
      client = await prisma.client.update({
        where: { id: client.id },
        data: { name, email, phone, birthdate: new Date(birthdate) },
      })
    }

    // Gerar número de reserva e voucher
    const reservationNumber = await generateReservationNumber(trip.tenant_id)
    const voucherCode = generateVoucherCode()

    // Criar resumo dos assentos
    const seatsSummary = selected_seats.sort((a, b) => parseInt(a) - parseInt(b)).join(', ')

    // ═══════════════════════════════════════════════════════════
    // CRIAR RESERVA COM ASSENTOS
    // ═══════════════════════════════════════════════════════════

    const reservation = await prisma.reservation.create({
      data: {
        tenant_id: trip.tenant_id,
        reservation_number: reservationNumber,
        voucher_code: voucherCode,
        trip_id,
        itinerary_id,
        client_id: client.id,
        client_name: name,
        client_cpf: cpf,
        client_email: email,
        client_phone: phone,
        total_passengers: passengers_count,
        adults_count: passengers_count,
        subtotal_amount: subtotal,
        discount_amount: totalDiscount,
        total_amount: totalAmount,
        amount_pending: totalAmount,
        payment_method,
        payment_status: 'pending',
        status: 'pending',
        source: 'website',
        observations,
        departure_date: itinerary.departure_date,
        seats_summary: seatsSummary, // ✅ "8" ou "8, 15, 31"
      },
    })

    // ═══════════════════════════════════════════════════════════
    // CRIAR ASSENTOS INDIVIDUAIS
    // ═══════════════════════════════════════════════════════════

    for (const seatNumber of selected_seats) {
      await prisma.reservationSeat.create({
        data: {
          reservation_id: reservation.id,
          trip_id,
          itinerary_id,
          seat_number: seatNumber,
          passenger_name: name, // Nome do titular (pode ser diferente depois)
          passenger_cpf: cpf,
          status: 'reserved',
        },
      })
    }

    // Processar referral
    if (referral_code) {
      const referral = await prisma.referral.findFirst({
        where: { code: referral_code, status: 'active' },
      })

      if (referral) {
        await prisma.reservation.update({
          where: { id: reservation.id },
          data: { referral_id: referral.id, referral_code },
        })

        await applyReferralBenefits({
          referralCode: referral_code,
          reservationId: reservation.id,
          totalAmount,
          tenantId: trip.tenant_id,
        })
      }
    }

    // Atualizar vagas do itinerário
    await prisma.itinerary.update({
      where: { id: itinerary_id },
      data: { available_seats: { decrement: passengers_count } },
    })

    // ═══════════════════════════════════════════════════════════
    // CRIAR DEAL NO PIPELINE COM INFORMAÇÕES DOS ASSENTOS
    // ═══════════════════════════════════════════════════════════

    const deal = await createDealFromReservation({
      tenantId: trip.tenant_id,
      clientId: client.id,
      reservationId: reservation.id,
      tripId: trip_id,
      totalAmount,
      seatsSummary, // ✅ Passar assentos
      tripTitle: trip.title,
      departureDate: itinerary.departure_date,
    })

    // Vincular deal à reserva
    await prisma.reservation.update({
      where: { id: reservation.id },
      data: { deal_id: deal.id },
    })

    // Emitir eventos
    eventEmitter.emitEvent('reservation.created', {
      tenant_id: trip.tenant_id,
      timestamp: new Date(),
      data: reservation,
    })

    eventEmitter.emitEvent('deal.created', {
      tenant_id: trip.tenant_id,
      timestamp: new Date(),
      data: deal,
    })

    // Buscar reserva completa
    const fullReservation = await prisma.reservation.findUnique({
      where: { id: reservation.id },
      include: {
        seats: true,
        client: true,
        deal: true,
        trip: true,
        itinerary: true,
      },
    })

    return NextResponse.json({
      success: true,
      reservation: fullReservation,
    })
  } catch (error: any) {
    console.error('Erro ao criar reserva:', error)
    return NextResponse.json(
      { error: 'Erro ao criar reserva' },
      { status: 500 }
    )
  }
}

// Utilitários
async function generateReservationNumber(tenantId: string): Promise<string> {
  const lastReservation = await prisma.reservation.findFirst({
    where: { tenant_id: tenantId },
    orderBy: { created_at: 'desc' },
    select: { reservation_number: true },
  })

  const lastNumber = lastReservation
    ? parseInt(lastReservation.reservation_number.split('-').pop() || '0')
    : 0

  return `RES-${new Date().getFullYear()}-${String(lastNumber + 1).padStart(5, '0')}`
}

function generateVoucherCode(): string {
  const characters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
  let code = 'VC-'

  for (let i = 0; i < 6; i++) {
    code += characters.charAt(Math.floor(Math.random() * characters.length))
  }

  return code
}

// lib/deals/create-from-reservation.ts

import { prisma } from '@/lib/prisma'

interface CreateDealFromReservationParams {
  tenantId: string
  clientId: string
  reservationId: string
  tripId: string
  totalAmount: number
  seatsSummary: string // ✅ "8" ou "8, 15, 31"
  tripTitle: string
  departureDate: Date
}

export async function createDealFromReservation({
  tenantId,
  clientId,
  reservationId,
  tripId,
  totalAmount,
  seatsSummary,
  tripTitle,
  departureDate,
}: CreateDealFromReservationParams) {
  // Buscar pipeline padrão
  const pipeline = await prisma.pipeline.findFirst({
    where: {
      tenant_id: tenantId,
      is_default: true,
    },
    include: {
      stages: {
        where: { slug: 'vitrine' }, // ✅ Estágio "Vitrine"
        orderBy: { order: 'asc' },
      },
    },
  })

  if (!pipeline) {
    throw new Error('Pipeline padrão não encontrado')
  }

  // Buscar estágio "Vitrine"
  let vitrineStage = pipeline.stages[0]

  if (!vitrineStage) {
    vitrineStage = await prisma.pipelineStage.findFirst({
      where: {
        tenant_id: tenantId,
        pipeline_id: pipeline.id,
        order: 2, // Segundo estágio
      },
    })
  }

  if (!vitrineStage) {
    throw new Error('Estágio "Vitrine" não encontrado')
  }

  // Buscar viagem
  const trip = await prisma.trip.findUnique({
    where: { id: tripId },
    select: { title: true, destination: true },
  })

  // Formatar informações dos assentos
  const seatsCount = seatsSummary.split(',').length
  const seatsInfo = seatsCount === 1
    ? `Assento ${seatsSummary}`
    : `Assentos ${seatsSummary}`

  // Criar deal
  const deal = await prisma.deal.create({
    data: {
      tenant_id: tenantId,
      client_id: clientId,
      pipeline_id: pipeline.id,
      stage_id: vitrineStage.id,
      title: trip ? `${trip.title} - ${trip.destination}` : 'Reserva Web',
      value: totalAmount,
      expected_close_date: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000),
      probability: 70,
      source: 'website',
      source_url: '/loja',
      auto_created: true,
      reservation_id: reservationId,
      status: 'open',
      
      // ✅ INFORMAÇÕES SINCRONIZADAS
      seats_info: seatsInfo, // "Assento 8" ou "Assentos 8, 15, 31"
      trip_info: trip ? `${trip.title} - ${trip.destination}` : null,
      departure_date: departureDate,
    },
  })

  // Criar atividade automática
  await prisma.activity.create({
    data: {
      tenant_id: tenantId,
      client_id: clientId,
      deal_id: deal.id,
      type: 'note',
      title: 'Reserva criada via site',
      description: `Cliente realizou reserva através da loja pública.\n\n📋 Reserva: ${reservationId}\n💺 ${seatsInfo}\n📅 Saída: ${departureDate.toLocaleDateString('pt-BR')}`,
      completed: true,
      completed_at: new Date(),
    },
  })

  return deal
}

// app/(dashboard)/dashboard/checkin/[tripId]/page.tsx

import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { CheckInList } from '@/components/dashboard/checkin/checkin-list'

export default async function CheckInPage({
  params,
}: {
  params: { tripId: string }
}) {
  const session = await getServerSession(authOptions)

  if (!session) {
    return null
  }

  const trip = await prisma.trip.findUnique({
    where: { id: params.tripId },
    include: {
      itineraries: {
        where: { status: 'open' },
        orderBy: { departure_date: 'asc' },
      },
    },
  })

  if (!trip) {
    return <div>Viagem não encontrada</div>
  }

  // Buscar reservas com assentos
  const reservations = await prisma.reservation.findMany({
    where: {
      tenant_id: session.user.tenantId,
      trip_id: params.tripId,
      status: { in: ['confirmed', 'pending'] },
    },
    include: {
      client: true,
      seats: {
        orderBy: { seat_number: 'asc' },
      },
    },
    orderBy: { created_at: 'desc' },
  })

  return (
    <div className="container mx-auto py-8 px-4">
      <div className="mb-8">
        <h1 className="text-3xl font-bold mb-2">Check-in de Embarque</h1>
        <p className="text-gray-600">{trip.title}</p>
      </div>

      <CheckInList reservations={reservations} trip={trip} />
    </div>
  )
}

'use client'

import { Card } from '@/components/ui/card'
import { Badge } from '@/components/ui/badge'
import { Button } from '@/components/ui/button'
import { Checkbox } from '@/components/ui/checkbox'
import { Armchair, Check, X } from 'lucide-react'
import { useState } from 'react'
import { useRouter } from 'next/navigation'

interface CheckInListProps {
  reservations: any[]
  trip: any
}

export function CheckInList({ reservations, trip }: CheckInListProps) {
  const router = useRouter()
  const [checkedIn, setCheckedIn] = useState<Set<string>>(new Set())

  const handleCheckIn = async (reservationId: string, seatId: string) => {
    try {
      const response = await fetch(`/api/dashboard/checkin`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          reservation_id: reservationId,
          seat_id: seatId,
        }),
      })

      if (response.ok) {
        setCheckedIn((prev) => new Set(prev).add(seatId))
        router.refresh()
      }
    } catch (error) {
      alert('Erro ao fazer check-in')
    }
  }

  return (
    <div className="space-y-4">
      {reservations.map((reservation) => (
        <Card key={reservation.id} className="p-6">
          <div className="flex items-start justify-between mb-4">
            <div>
              <h3 className="text-lg font-bold">{reservation.client_name}</h3>
              <p className="text-sm text-gray-600">CPF: {reservation.client_cpf}</p>
              <p className="text-sm text-gray-600">
                Voucher: {reservation.voucher_code}
              </p>
            </div>

            <Badge
              className={
                reservation.status === 'confirmed'
                  ? 'bg-green-100 text-green-800'
                  : 'bg-yellow-100 text-yellow-800'
              }
            >
              {reservation.status === 'confirmed' ? 'Confirmado' : 'Pendente'}
            </Badge>
          </div>

          {/* ═══════════════════════════════════════════════════════ */}
          {/* LISTA DE ASSENTOS PARA CHECK-IN */}
          {/* ═══════════════════════════════════════════════════════ */}
          <div className="space-y-2">
            <p className="text-sm font-semibold text-gray-700 mb-2">
              Assentos:
            </p>
            {reservation.seats?.map((seat: any) => (
              <div
                key={seat.id}
                className={`
                  flex items-center justify-between p-3 rounded-lg border-2
                  ${
                    seat.checked_in
                      ? 'bg-green-50 border-green-200'
                      : 'bg-gray-50 border-gray-200'
                  }
                `}
              >
                <div className="flex items-center gap-3">
                  <Armchair
                    className={`h-5 w-5 ${
                      seat.checked_in ? 'text-green-600' : 'text-orange-600'
                    }`}
                  />
                  <div>
                    <p className="font-semibold">Assento {seat.seat_number}</p>
                    {seat.passenger_name && (
                      <p className="text-sm text-gray-600">
                        {seat.passenger_name}
                      </p>
                    )}
                  </div>
                </div>

                {seat.checked_in ? (
                  <Badge className="bg-green-500">
                    <Check className="h-3 w-3 mr-1" />
                    Check-in Feito
                  </Badge>
                ) : (
                  <Button
                    size="sm"
                    onClick={() => handleCheckIn(reservation.id, seat.id)}
                    className="bg-blue-600 hover:bg-blue-700"
                  >
                    Fazer Check-in
                  </Button>
                )}
              </div>
            ))}
          </div>
        </Card>
      ))}
    </div>
  )
}
