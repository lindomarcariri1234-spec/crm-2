O **VisiteCRM**, é um Sistema SaaS CRM COMPLETO, ROBUSTO e ESCALÁVEL
para agências de turismo especializadas em excursões.
Contém Sistema de Reservas COMPLETO, ROBUSTO e sofisticado com reserva criado pelo vendedor, pela Agência e pelo próprio cliente na loja.  seleção de poltronas, passageiros, voucher PDF, QR Code entre outras funcionalidades.

Implementar sistema completo de reservas com fluxo em etapas,
gestão de passageiros e integrações.

GESTÃO DE RESERVAS

3.1 Fluxo de Reserva (4 Passos)

PASSO 1: SELEÇÃO DE VIAGEM
├── Dropdown com viagens disponíveis
├── Filtros (destino, data, preço)
├── Informações da viagem
└── Botão "Próximo"



3.2 Status de Reserva

Fluxo de Status:
Pré-reserva → Confirmada → Paga → Em Viagem → Finalizada
                ↓
            Cancelada → Reembolsada



MÓDULO 4: GESTÃO FINANCEIRA

4.1 Contas a Receber

Tabela:
┌─────────────┬──────────┬────────┬──────────┬──────────┐
│ Viagem      │ Cliente  │ Valor  │ Recebido │ Pendente │
├─────────────┼──────────┼────────┼──────────┼──────────┤


Funcionalidades:
├── Filtros (período, status, viagem)


4.2 Comissões

Cálculo Automático:
Comissão = Valor da Reserva × Percentual de Comissão


Exemplo:
Reserva: R$ 5.000
Comissão: 5%
Valor da Comissão: R$ 250

model Reservation {

  // RELACIONAMENTOS PRINCIPAIS
  // ─────────────────────────────────────────────────────────────
  trip_id         String
  trip            Trip      @relation(fields: [trip_id], references: [id], onDelete: Restrict)
  
  client_id       String
  client          Client    @relation(fields: [client_id], references: [id], onDelete: Restrict)
  
  seller_id       String?
  seller          User?     @relation("SellerReservations", fields: [seller_id], references: [id])

  // ─────────────────────────────────────────────────────────────

  boarding_location BoardingLocation? @relation(fields: [boarding_location_id], references: [id])

  // ─────────────────────────────────────────────────────────────
  subtotal_amount Decimal   @db.Decimal(10, 2) // Valor base

  coupon_discount Decimal   @default(0) @db.Decimal(10, 2) // Desconto cupom
  referral_discount Decimal @default(0) @db.Decimal(10, 2) // Desconto indicação

// app/(dashboard)/dashboard/reservas/nova/page.tsx

import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { ReservationWizard } from '@/components/dashboard/reservations/reservation-wizard'
import { redirect } from 'next/navigation'

export default async function NewReservationPage() {
  const session = await getServerSession(authOptions)

  if (!session) {
    redirect('/login')
  }

  // Buscar dados necessários
  const [trips, clients, boardingLocations, coupons] = await Promise.all([
    // Viagens ativas com vagas disponíveis
    prisma.trip.findMany({
      where: {
        tenant_id: session.user.tenantId,
        status: 'active',
        departure_date: {
          gte: new Date(),
        },
      },
      include: {
        boarding_locations: {
          include: {
            location: true,
          },
          orderBy: {
            order: 'asc',
          },
        },
      },
      orderBy: {
        departure_date: 'asc',
      },
    }),

    // Clientes ativos
    prisma.client.findMany({
      where: {
        tenant_id: session.user.tenantId,
      },
      orderBy: {
        name: 'asc',
      },
      select: {
        id: true,
        name: true,
        email: true,
        phone: true,
        cpf: true,
        referral_code: true,
      },
    }),

    // Locais de embarque
    prisma.boardingLocation.findMany({
      where: {
        tenant_id: session.user.tenantId,
        is_active: true,
      },
      orderBy: {
        name: 'asc',
      },
    }),

    // Cupons ativos
    prisma.coupon.findMany({
      where: {
        tenant_id: session.user.tenantId,
        is_active: true,
        OR: [
          { valid_until: null },
          { valid_until: { gte: new Date() } },
        ],
      },
    }),
  ])

  return (
    <div className="container mx-auto py-8 px-4">
      <div className="mb-6">
        <h1 className="text-3xl font-bold">Nova Reserva</h1>
        <p className="text-gray-600">
          Crie uma nova reserva seguindo os passos abaixo
        </p>
      </div>

      <ReservationWizard
        trips={trips}
        clients={clients}
        boardingLocations={boardingLocations}
        coupons={coupons}
        sellerId={session.user.id}
        tenantId={session.user.tenantId}
      />
    </div>
  )
}

  }>

export function ReservationWizard({
  trips,
  clients,
  boardingLocations,
  coupons,
  sellerId,
  tenantId,
}: ReservationWizardProps) {
  const { toast } = useToast()
  const router = useRouter()

  const [currentStep, setCurrentStep] = useState(1)
  const [loading, setLoading] = useState(false)

  const [reservationData, setReservationData] = useState<ReservationData>({
    trip_id: '',
    client_id: '',
    boarding_location_id: '',
    seller_id: sellerId,
    passengers: [],
    seats: [],
    subtotal_amount: 0,
    discount_amount: 0,
    coupon_discount: 0,
    referral_discount: 0,
    total_amount: 0,
    amount_paid: 0,
    amount_pending: 0,
    payment_method: 'pix',
    installments: 1,
    status: 'pending',
  })

  const steps = [
    { number: 1, title: 'Viagem', component: Step1TripSelection },
    { number: 2, title: 'Passageiros', component: Step2Passengers },
    { number: 3, title: 'Assentos', component: Step3SeatSelection },
    { number: 4, title: 'Pagamento', component: Step4Payment },
    { number: 5, title: 'Confirmação', component: Step5Confirmation },
  ]

  const CurrentStepComponent = steps[currentStep - 1].component

  const handleNext = () => {
    // Validações por passo
    if (currentStep === 1) {
      if (!reservationData.trip_id || !reservationData.client_id) {
        toast({
          title: 'Campos obrigatórios',
          description: 'Selecione a viagem e o cliente',
          variant: 'destructive',
        })
        return
      }
    }

    if (currentStep === 2) {
      if (reservationData.passengers.length === 0) {
        toast({
          title: 'Adicione passageiros',
          description: 'É necessário adicionar pelo menos um passageiro',
          variant: 'destructive',
        })
        return
      }
    }

    if (currentStep === 3) {
      if (reservationData.seats.length !== reservationData.passengers.length) {
        toast({
          title: 'Selecione os assentos',
          description: 'Selecione um assento para cada passageiro',
          variant: 'destructive',
        })
        return
      }
    }

    if (currentStep < 5) {
      setCurrentStep(currentStep + 1)
    }
  }

  const handleBack = () => {
    if (currentStep > 1) {
      setCurrentStep(currentStep - 1)
    }
  }

  const handleSubmit = async () => {
    setLoading(true)
    try {
      const response = await fetch('/api/dashboard/reservations', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          ...reservationData,
          tenant_id: tenantId,
        }),
      })

      const data = await response.json()

      if (!response.ok) {
        throw new Error(data.error || 'Erro ao criar reserva')
      }

      toast({
        title: 'Reserva criada com sucesso! 🎉',
        description: `Voucher: ${data.reservation.voucher_code}`,
      })

      router.push(`/dashboard/reservas/${data.reservation.id}`)
    } catch (error: any) {
      toast({
        title: 'Erro ao criar reserva',
        description: error.message,
        variant: 'destructive',
      })
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="max-w-5xl mx-auto">
      {/* Progress Steps */}
      <div className="mb-8">
        <div className="flex items-center justify-between">
          {steps.map((step, index) => (
            <div key={step.number} className="flex items-center flex-1">
              <div className="flex flex-col items-center flex-1">
                <div
                  className={`w-10 h-10 rounded-full flex items-center justify-center font-bold transition-colors ${
                    currentStep > step.number
                      ? 'bg-green-500 text-white'
                      : currentStep === step.number
                      ? 'bg-blue-500 text-white'
                      : 'bg-gray-200 text-gray-600'
                  }`}
                >
                  {currentStep > step.number ? (
                    <Check className="h-5 w-5" />
                  ) : (
                    step.number
                  )}
                </div>
                <span
                  className={`text-sm mt-2 font-medium ${
                    currentStep >= step.number
                      ? 'text-gray-900'
                      : 'text-gray-500'
                  }`}
                >
                  {step.title}
                </span>
              </div>
              {index < steps.length - 1 && (
                <div
                  className={`h-1 flex-1 mx-2 transition-colors ${
                    currentStep > step.number ? 'bg-green-500' : 'bg-gray-200'
                  }`}
                />
              )}
            </div>
          ))}
        </div>
      </div>

      {/* Step Content */}
      <Card>
        <CardContent className="pt-6">
          <CurrentStepComponent
            data={reservationData}
            setData={setReservationData}
            trips={trips}
            clients={clients}
            boardingLocations={boardingLocations}
            coupons={coupons}
          />
        </CardContent>
      </Card>

      {/* Navigation Buttons */}
      <div className="flex justify-between mt-6">
        <Button
          variant="outline"
          onClick={handleBack}
          disabled={currentStep === 1 || loading}
        >
          <ArrowLeft className="h-4 w-4 mr-2" />
          Voltar
        </Button>

        {currentStep < 5 ? (
          <Button onClick={handleNext} disabled={loading}>
            Próximo
            <ArrowRight className="h-4 w-4 ml-2" />
          </Button>
        ) : (
          <Button onClick={handleSubmit} disabled={loading}>
            {loading ? 'Criando...' : 'Finalizar Reserva'}
            <Check className="h-4 w-4 ml-2" />
          </Button>
        )}
      </div>
    </div>
  )
}

import { useState } from 'react'

import { Card } from '@/components/ui/card'
import { Textarea } from '@/components/ui/textarea'
import { Plus, Trash2, User, Users } from 'lucide-react'
import { Badge } from '@/components/ui/badge'

interface Step2Props {
  data: any
  setData: (data: any) => void
  clients: any[]
  boardingLocations: any[]
}

interface Passenger {
  name: string
  cpf: string
  birth_date: string
  category: 'adult' | 'child' | 'infant'
  age?: number
  boarding_location_id?: string
  special_needs?: string
  phone?: string
  email?: string
  is_primary: boolean
}

export function Step2Passengers({
  data,
  setData,
  clients,
  boardingLocations,
}: Step2Props) {
  const [passengers, setPassengers] = useState<Passenger[]>(
    data.passengers.length > 0
      ? data.passengers
      : [
          {
            name: data.client?.name || '',
            cpf: data.client?.cpf || '',
            birth_date: '',
            category: 'adult',
            boarding_location_id: data.boarding_location_id,
            special_needs: '',
            phone: data.client?.phone || '',
            email: data.client?.email || '',
            is_primary: true,
          },
        ]
  )

  const calculateAge = (birthDate: string) => {
    if (!birthDate) return null
    const today = new Date()
    const birth = new Date(birthDate)
    let age = today.getFullYear() - birth.getFullYear()
    const monthDiff = today.getMonth() - birth.getMonth()
    if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birth.getDate())) {
      age--
    }
    return age
  }

  const getCategoryByAge = (age: number | null) => {
    if (!age) return 'adult'
    if (age < 2) return 'infant'
    if (age < 12) return 'child'
    return 'adult'
  }

  const handlePassengerChange = (index: number, field: string, value: any) => {
    const updatedPassengers = [...passengers]
    updatedPassengers[index] = {
      ...updatedPassengers[index],
      [field]: value,
    }

    // Auto-calcular idade e categoria
    if (field === 'birth_date') {
      const age = calculateAge(value)
      updatedPassengers[index].age = age || undefined
      updatedPassengers[index].category = getCategoryByAge(age)
    }

    setPassengers(updatedPassengers)
    setData({ ...data, passengers: updatedPassengers })
  }

  const addPassenger = () => {
    const newPassenger: Passenger = {
      name: '',
      cpf: '',
      birth_date: '',
      category: 'adult',
      boarding_location_id: data.boarding_location_id,
      special_needs: '',
      is_primary: false,
    }
    const updatedPassengers = [...passengers, newPassenger]
    setPassengers(updatedPassengers)
    setData({ ...data, passengers: updatedPassengers })
  }

  const removePassenger = (index: number) => {
    if (passengers[index].is_primary) {
      alert('Não é possível remover o passageiro principal')
      return
    }
    const updatedPassengers = passengers.filter((_, i) => i !== index)
    setPassengers(updatedPassengers)
    setData({ ...data, passengers: updatedPassengers })
  }

  const calculateTotals = () => {
    const adults = passengers.filter((p) => p.category === 'adult').length
    const children = passengers.filter((p) => p.category === 'child').length
    const infants = passengers.filter((p) => p.category === 'infant').length

    const trip = data.trip
    if (!trip) return { adults: 0, children: 0, infants: 0, subtotal: 0 }

    const adultPrice = Number(trip.price_adult || 0)
    const childPrice = Number(trip.price_child || adultPrice * 0.5)
    const infantPrice = Number(trip.price_infant || 0)

    const subtotal = adults * adultPrice + children * childPrice + infants * infantPrice

    return { adults, children, infants, subtotal }
  }

  const totals = calculateTotals()

  // Atualizar subtotal no data
  if (data.subtotal_amount !== totals.subtotal) {
    setData({
      ...data,
      subtotal_amount: totals.subtotal,
      total_amount: totals.subtotal,
      amount_pending: totals.subtotal,
    })
  }

  return (
    <div className="space-y-6">
      <div>
        <h2 className="text-2xl font-bold mb-2">Passageiros</h2>
        <p className="text-gray-600">
          Adicione os dados de todos os passageiros da reserva
        </p>
      </div>

      {/* Resumo */}
      <Card className="p-4 bg-blue-50 border-blue-200">
        <div className="flex items-center justify-between">
          <div className="flex items-center gap-4">
            <Users className="h-8 w-8 text-blue-600" />
            <div>
              <p className="font-bold text-lg">
                {passengers.length} Passageiro(s)
              </p>
              <p className="text-sm text-gray-600">
                {totals.adults} adulto(s) • {totals.children} criança(s) •{' '}
                {totals.infants} bebê(s)
              </p>
            </div>
          </div>
          <div className="text-right">
            <p className="text-sm text-gray-600">Subtotal</p>
            <p className="text-2xl font-bold text-green-600">
              R${' '}
              {totals.subtotal.toLocaleString('pt-BR', {
                minimumFractionDigits: 2,
              })}
            </p>
          </div>
        </div>
      </Card>

      {/* Lista de Passageiros */}
      <div className="space-y-4">
        {passengers.map((passenger, index) => (
          <Card key={index} className="p-4">
            <div className="flex items-start justify-between mb-4">
              <div className="flex items-center gap-2">
                <User className="h-5 w-5 text-gray-600" />
                <h3 className="font-bold">
                  Passageiro {index + 1}
                  {passenger.is_primary && (
                    <Badge className="ml-2 bg-blue-500">Principal</Badge>
                  )}
                </h3>
              </div>
              {!passenger.is_primary && (
                <Button

interface Step3Props {

interface Seat {
  number: string
  row: number
  column: string
  status: 'available' | 'selected' | 'occupied' | 'blocked'
  passenger_name?: string
  passenger_cpf?: string
}

export function Step3SeatSelection({ data, setData }: Step3Props) {
  const [seats, setSeats] = useState<Seat[]>([])
  const [selectedSeats, setSelectedSeats] = useState<any[]>(data.seats || [])
  const [loading, setLoading] = useState(true)

  // Carregar assentos da viagem
  useEffect(() => {
    if (data.trip_id) {
      loadSeats()
    }
  }, [data.trip_id])

  const loadSeats = async () => {
    try {
      const response = await fetch(`/api/dashboard/trips/${data.trip_id}/seats`)
      const result = await response.json()

      if (result.seats) {
        setSeats(result.seats)
      } else {
        // Gerar assentos se não existirem
        generateSeats()
      }
    } catch (error) {
      console.error('Erro ao carregar assentos:', error)
      generateSeats()
    } finally {
      setLoading(false)
    }
  }

  const generateSeats = () => {
    const totalSeats = data.trip?.total_seats || 50
    const layout = data.trip?.seat_layout || '2+2'
    const [left, right] = layout.split('+').map(Number)
    const seatsPerRow = left + right
    const totalRows = Math.ceil(totalSeats / seatsPerRow)

    const generatedSeats: Seat[] = []
    const columns = ['A', 'B', 'C', 'D', 'E', 'F']

    for (let row = 1; row <= totalRows; row++) {
      for (let col = 0; col < seatsPerRow; col++) {
        const seatNumber = `${row}${columns[col]}`
        generatedSeats.push({
          number: seatNumber,
          row,
          column: columns[col],
          status: 'available',
        })
      }
    }

    setSeats(generatedSeats.slice(0, totalSeats))
  }

  const handleSeatClick = (seat: Seat) => {
    if (seat.status === 'occupied' || seat.status === 'blocked') {
      return
    }

    const isSelected = selectedSeats.some((s) => s.seat_number === seat.number)

    if (isSelected) {
      // Desselecionar
      const updated = selectedSeats.filter((s) => s.seat_number !== seat.number)
      setSelectedSeats(updated)
      setData({ ...data, seats: updated })
    } else {
      // Verificar se já selecionou todos os assentos necessários
      if (selectedSeats.length >= data.passengers.length) {
        alert('Você já selecionou assentos para todos os passageiros')
        return
      }

      // Selecionar
      const updated = [
        ...selectedSeats,
        {
          seat_number: seat.number,
          passenger_name: '',
          passenger_cpf: '',
        },
      ]
      setSelectedSeats(updated)
      setData({ ...data, seats: updated })
    }
  }

  const handlePassengerAssignment = (
    seatNumber: string,
    passengerId: number

  const getSeatStatus = (seat: Seat): Seat['status'] => {
    const isSelected = selectedSeats.some((s) => s.seat_number === seat.number)

      case 'selected':
        return 'bg-blue-500 border-blue-600 text-white cursor-pointer'
      case 'occupied':
        return 'bg-orange-500 border-orange-600 text-white cursor-not-allowed'
      case 'blocked':
        return 'bg-gray-400 border-gray-500 text-white cursor-not-allowed'
      default:
        return 'bg-white border-gray-300'
    }
  }

  const layout = data.trip?.seat_layout || '2+2'
  const [leftSeats, rightSeats] = layout.split('+').map(Number)

  if (loading) {
    return (
      <div className="flex items-center justify-center py-12">
        <p className="text-gray-600">Carregando mapa de assentos...</p>
      </div>
    )
  }

  return (
    <div className="space-y-6">
      <div>
        <h2 className="text-2xl font-bold mb-2">Seleção de Assentos</h2>
        <p className="text-gray-600">
          Selecione {data.passengers.length} assento(s) para os passageiros
        </p>
      </div>

      {/* Legenda */}
      <Card className="p-4">
        <div className="flex flex-wrap gap-4 justify-center">

            <div className="w-8 h-8 border-2 border-gray-300 bg-white rounded" />
            <span className="text-sm">Disponível</span>
          </div>

            <div className="w-8 h-8 border-2 border-blue-600 bg-blue-500 rounded" />
            <span className="text-sm">Selecionado</span>
          </div>
          <div className="flex items-center gap-2">
            <div className="w-8 h-8 border-2 border-orange-600 bg-orange-500 rounded" />

                const leftSide = rowSeats.slice(0, leftSeats)

'use client'

import { useState, useEffect } from 'react'

} from 'lucide-react'
import { useToast } from '@/components/ui/use-toast'

interface Step4Props {
  data: any
  setData: (data: any) => void
  coupons: any[]
}

export function Step4Payment({ data, setData, coupons }: Step4Props) {
  const { toast } = useToast()
  const [couponCode, setCouponCode] = useState(data.coupon_code || '')
  const [appliedCoupon, setAppliedCoupon] = useState<any>(null)
  const [referralCode, setReferralCode] = useState(data.referred_by_code || '')
  const [appliedReferral, setAppliedReferral] = useState<any>(null)
  const [loadingCoupon, setLoadingCoupon] = useState(false)
  const [loadingReferral, setLoadingReferral] = useState(false)

  // Calcular valores
  const calculateTotals = () => {
    const subtotal = data.subtotal_amount || 0
    let couponDiscount = 0
    let referralDiscount = 0

    // Desconto do cupom
    if (appliedCoupon) {
      if (appliedCoupon.discount_type === 'percentage') {
        couponDiscount = (subtotal * appliedCoupon.discount_value) / 100
        if (appliedCoupon.max_discount_amount) {
          couponDiscount = Math.min(
            couponDiscount,
            appliedCoupon.max_discount_amount
          )
        }
      } else {
        couponDiscount = appliedCoupon.discount_value
      }
    }

    // Desconto de indicação (5%)
    if (appliedReferral && data.client?.is_first_purchase !== false) {
      referralDiscount = subtotal * 0.05 // 5%
      // Limite máximo de R$ 100
      referralDiscount = Math.min(referralDiscount, 100)
    }

    const totalDiscount = couponDiscount + referralDiscount
    const total = Math.max(subtotal - totalDiscount, 0)
    const amountPaid = data.amount_paid || 0
    const amountPending = Math.max(total - amountPaid, 0)

    return {
      subtotal,
      couponDiscount,
      referralDiscount,
      totalDiscount,
      total,
      amountPaid,
      amountPending,
    }
  }

  const totals = calculateTotals()

  // Atualizar data quando valores mudarem
  useEffect(() => {
    setData({
      ...data,
      subtotal_amount: totals.subtotal,
      coupon_discount: totals.couponDiscount,
      referral_discount: totals.referralDiscount,
      discount_amount: totals.totalDiscount,
      total_amount: totals.total,
      amount_pending: totals.amountPending,
    })
  }, [appliedCoupon, appliedReferral, data.amount_paid])

  // Aplicar cupom
  const handleApplyCoupon = async () => {
    if (!couponCode.trim()) {
      toast({
        title: 'Digite um cupom',
        variant: 'destructive',
      })
      return
    }

    setLoadingCoupon(true)
    try {
      const response = await fetch('/api/dashboard/coupons/validate', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          code: couponCode.toUpperCase(),
          subtotal: totals.subtotal,
        }),
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Cupom inválido')
      }

      setAppliedCoupon(result.coupon)
      setData({
        ...data,
        coupon_code: couponCode.toUpperCase(),
        coupon_id: result.coupon.id,
      })

      toast({
        title: 'Cupom aplicado! 🎉',
        description: `Desconto de ${
          result.coupon.discount_type === 'percentage'
            ? `${result.coupon.discount_value}%`
            : `R$ ${result.coupon.discount_value}`
        }`,
      })
    } catch (error: any) {
      toast({
        title: 'Erro ao aplicar cupom',
        description: error.message,
        variant: 'destructive',
      })
    } finally {
      setLoadingCoupon(false)
    }
  }

  // Remover cupom
  const handleRemoveCoupon = () => {
    setAppliedCoupon(null)
    setCouponCode('')
    setData({
      ...data,
      coupon_code: null,
      coupon_id: null,
      coupon_discount: 0,
    })
  }

  // Aplicar código de indicação
  const handleApplyReferral = async () => {
    if (!referralCode.trim()) {
      toast({
        title: 'Digite um código de indicação',
        variant: 'destructive',
      })
      return
    }

    setLoadingReferral(true)
    try {
      const response = await fetch('/api/dashboard/referrals/validate', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          code: referralCode.toUpperCase(),
          client_id: data.client_id,
        }),
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Código inválido')
      }

      setAppliedReferral(result.referrer)
      setData({
        ...data,
        referred_by_code: referralCode.toUpperCase(),
      })

      toast({
        title: 'Código de indicação aplicado! 🎁',
        description: 'Desconto de 5% aplicado',
      })
    } catch (error: any) {
      toast({
        title: 'Erro ao aplicar código',
        description: error.message,
        variant: 'destructive',
      })
    } finally {
      setLoadingReferral(false)
    }
  }

  // Remover indicação
  const handleRemoveReferral = () => {
    setAppliedReferral(null)
    setReferralCode('')
    setData({
      ...data,
      referred_by_code: null,
      referral_discount: 0,
    })
  }

  // Calcular parcelas
  const calculateInstallment = (installments: number) => {
    if (installments <= 1) return totals.total
    return totals.total / installments
  }

  // Calcular comissão
  const calculateCommission = () => {
    const percentage = data.commission_percentage || 5
    return (totals.total * percentage) / 100
  }

  return (
    <div className="space-y-6">
      <div>
        <h2 className="text-2xl font-bold mb-2">Pagamento e Descontos</h2>
        <p className="text-gray-600">
          Configure os valores e forma de pagamento da reserva
        </p>
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
        {/* Formulário de Pagamento */}
        <div className="lg:col-span-2 space-y-6">
          {/* Cupom de Desconto */}
          <Card className="p-4">
            <div className="flex items-center gap-2 mb-4">
              <Tag className="h-5 w-5 text-purple-600" />
              <h3 className="font-bold">Cupom de Desconto</h3>
            </div>

            {appliedCoupon ? (
              <div className="bg-purple-50 border border-purple-200 rounded-lg p-4">
                <div className="flex items-center justify-between">
                  <div>
                    <p className="font-bold text-purple-900">
                      {appliedCoupon.code}
                    </p>
                    <p className="text-sm text-purple-700">
                      Desconto:{' '}
                      {appliedCoupon.discount_type === 'percentage'
                        ? `${appliedCoupon.discount_value}%`
                        : `R$ ${appliedCoupon.discount_value}`}
                    </p>
                  </div>
                  <Button
                    variant="ghost"
                    size="sm"
                    onClick={handleRemoveCoupon}
                    className="text-purple-600"
                  >
                    Remover
                  </Button>
                </div>
              </div>
            ) : (
              <div className="flex gap-2">
                <Input
                  placeholder="Digite o código do cupom"
                  value={couponCode}
                  onChange={(e) => setCouponCode(e.target.value.toUpperCase())}
                  className="flex-1"
                />
                <Button

            )}
          </Card>

          {/* Código de Indicação */}
          <Card className="p-4">
            <div className="flex items-center gap-2 mb-4">
              <Gift className="h-5 w-5 text-blue-600" />
              <h3 className="font-bold">Código de Indicação</h3>

            {appliedReferral ? (

                  className={`font-bold ${

                  }`}
                >
                  R${' '}
                  {totals.amountPending.toLocaleString('pt-BR', {
                    minimumFractionDigits: 2,
                  })}
                </span>
              </div>

              {/* Parcelamento */}
              {data.installments > 1 && (
                <div className="bg-blue-50 p-3 rounded-lg mt-3">
                  <p className="text-sm text-blue-900 font-medium">
                    {data.installments}x de R${' '}
                    {(data.installment_amount || 0).toLocaleString('pt-BR', {

                  </p>

              )}

              {/* Comissão */}
              {data.commission_percentage > 0 && (

                  <div className="flex justify-between text-sm">
                    <span className="text-gray-600">
                      Comissão ({data.commission_percentage}%):
                    </span>
                    <span className="font-medium text-orange-600">
                      R${' '}
                      {calculateCommission().toLocaleString('pt-BR', {
                        minimumFractionDigits: 2,
                      })}
                    </span>
                  </div>
                </div>
              )}
            </div>
          </Card>
        </div>
      </div>
    </div>
  )
}

                </div>
              </div>

              {passenger.special_needs && (
                <div className="mt-3 pt-3 border-t">
                  <p className="text-sm text-gray-600">Necessidades Especiais:</p>
                  <p className="text-sm">{passenger.special_needs}</p>
                </div>
              )}
            </div>
          ))}
        </div>
      </Card>

      {/* Assentos */}
      <Card className="p-6">
        <div className="flex items-center gap-2 mb-4">
          <Armchair className="h-5 w-5 text-orange-600" />
          <h3 className="font-bold text-lg">
            Assentos Selecionados ({data.seats?.length || 0})
          </h3>
        </div>

        <div className="flex flex-wrap gap-2">
          {data.seats?.map((seat: any, index: number) => (
            <div
              key={index}
              className="border rounded-lg p-3 bg-blue-50 border-blue-200"
            >
              <div className="flex items-center gap-2">
                <Badge className="bg-blue-500">{seat.seat_number}</Badge>
                <span className="text-sm font-medium">{seat.passenger_name}</span>
              </div>
            </div>
          ))}
        </div>
      </Card>

      {/* Pagamento */}
      <Card className="p-6">
        <div className="flex items-center gap-2 mb-4">
          <CreditCard className="h-5 w-5 text-blue-600" />
          <h3 className="font-bold text-lg">Pagamento</h3>
        </div>

        <div className="space-y-4">
          {/* Resumo Financeiro */}
          <div className="bg-gray-50 rounded-lg p-4 space-y-2">
            <div className="flex justify-between text-sm">
              <span className="text-gray-600">Subtotal:</span>
              <span className="font-medium">
                R${' '}
                {(data.subtotal_amount || 0).toLocaleString('pt-BR', {
                  minimumFractionDigits: 2,
                })}
              </span>
            </div>

            {data.coupon_discount > 0 && (
              <div className="flex justify-between text-sm text-purple-600">
                <span className="flex items-center gap-1">
                  <Tag className="h-3 w-3" />
                  Desconto Cupom:
                </span>
                <span className="font-medium">
                  - R${' '}
                  {(data.coupon_discount || 0).toLocaleString('pt-BR', {
                    minimumFractionDigits: 2,
                  })}
                </span>
              </div>
            )}

            {data.referral_discount > 0 && (
              <div className="flex justify-between text-sm text-blue-600">
                <span className="flex items-center gap-1">
                  <Gift className="h-3 w-3" />
                  Desconto Indicação:
                </span>
                <span className="font-medium">
                  - R${' '}
                  {(data.referral_discount || 0).toLocaleString('pt-BR', {
                    minimumFractionDigits: 2,
                  })}
                </span>
              </div>
            )}

            <Separator />

            <div className="flex justify-between font-bold text-lg">
              <span>Total:</span>
              <span className="text-green-600">
                R${' '}
                {(data.total_amount || 0).toLocaleString('pt-BR', {
                  minimumFractionDigits: 2,
                })}
              </span>
            </div>

            <div className="flex justify-between text-sm bg-white p-2 rounded">
              <span className="text-gray-600">Valor Pago:</span>
              <span className="font-medium text-green-600">
                R${' '}
                {(data.amount_paid || 0).toLocaleString('pt-BR', {
                  minimumFractionDigits: 2,
                })}
              </span>
            </div>

            <div
              className={`flex justify-between text-sm p-2 rounded ${
                data.amount_pending > 0 ? 'bg-red-50' : 'bg-green-50'
              }`}
            >
              <span className="text-gray-600">Valor Pendente:</span>
              <span
                className={`font-bold ${
                  data.amount_pending > 0 ? 'text-red-600' : 'text-green-600'
                }`}
              >
                R${' '}
                {(data.amount_pending || 0).toLocaleString('pt-BR', {
                  minimumFractionDigits: 2,
                })}
              </span>
            </div>
          </div>

          {/* Forma de Pagamento */}
          <div className="grid grid-cols-2 gap-4">
            <div>
              <p className="text-sm text-gray-600">Forma de Pagamento</p>
              <p className="font-medium">
                {getPaymentMethodLabel(data.payment_method)}
              </p>
            </div>
            {data.installments > 1 && (
              <div>
                <p className="text-sm text-gray-600">Parcelamento</p>
                <p className="font-medium">
                  {data.installments}x de R${' '}
                  {(data.installment_amount || 0).toLocaleString('pt-BR', {
                    minimumFractionDigits: 2,
                  })}
                </p>
              </div>
            )}
          </div>

          {/* Status */}
          <div>
            <p className="text-sm text-gray-600 mb-2">Status da Reserva</p>
            <Badge className={getStatusColor(data.status)}>
              {getStatusLabel(data.status)}
            </Badge>
          </div>

          {/* Comissão */}
          {data.commission_percentage > 0 && (
            <div className="bg-orange-50 border border-orange-200 rounded-lg p-3">
              <div className="flex items-center justify-between">
                <div className="flex items-center gap-2">
                  <Percent className="h-4 w-4 text-orange-600" />
                  <span className="text-sm font-medium">
                    Comissão ({data.commission_percentage}%)
                  </span>
                </div>
                <span className="font-bold text-orange-600">
                  R${' '}
                  {(data.commission_amount || 0).toLocaleString('pt-BR', {
                    minimumFractionDigits: 2,
                  })}
                </span>
              </div>
            </div>
          )}
        </div>
      </Card>

      {/* Observações */}
      {data.notes && (
        <Card className="p-6">
          <h3 className="font-bold mb-2">Observações</h3>
          <p className="text-sm text-gray-700 whitespace-pre-wrap">{data.notes}</p>
        </Card>
      )}

      {/* Alerta Final */}
      <Card className="p-4 bg-blue-50 border-blue-200">
        <div className="flex items-start gap-3">
          <CheckCircle className="h-5 w-5 text-blue-600 flex-shrink-0 mt-0.5" />
          <div>
            <p className="font-medium text-blue-900">Tudo pronto!</p>
            <p className="text-sm text-blue-700 mt-1">
              Ao clicar em "Finalizar Reserva", um voucher será gerado automaticamente
              e o cliente receberá uma confirmação por WhatsApp e e-mail.
            </p>
          </div>
        </div>
      </Card>
    </div>
  )
}
