model Tenant {
  id              String    @id @default(cuid())
  name            String
  slug            String    @unique
  cnpj            String?   @unique
  email           String
  phone           String?
  logo            String?
  
  // Endereço
  address         String?
  city            String?
  state           String?
  zip_code        String?
  country         String    @default("BR")
  
  // Configurações
  settings        Json?
  
  // Plano e Faturamento
  plan_id         String?
  plan            Plan?     @relation(fields: [plan_id], references: [id])
  plan_expires_at DateTime?
  billing_cycle   String    @default("monthly") // monthly, yearly
  
  // Status
  status          String    @default("active") // active, suspended, cancelled
  is_active       Boolean   @default(true)
  
  // Métricas
  total_users     Int       @default(0)
  total_clients   Int       @default(0)
  total_revenue   Decimal   @default(0) @db.Decimal(10, 2)
  
  // Limites
  max_users       Int       @default(5)
  max_clients     Int       @default(100)
  max_trips       Int       @default(50)
  
  // Datas
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  activated_at    DateTime?
  suspended_at    DateTime?
  cancelled_at    DateTime?
  
  // Relações
  users           User[]
  clients         Client[]
  trips           Trip[]
  reservations    Reservation[]
  invoices        Invoice[]
  subscriptions   Subscription[]
  
  @@map("tenants")
}

model Plan {
  id              String    @id @default(cuid())
  name            String
  slug            String    @unique
  description     String?
  
  // Preços
  price_monthly   Decimal   @db.Decimal(10, 2)
  price_yearly    Decimal   @db.Decimal(10, 2)
  
  // Limites
  max_users       Int
  max_clients     Int
  max_trips       Int
  max_storage_gb  Int
  
  // Features
  features        Json      // Array de features
  
  // Status
  is_active       Boolean   @default(true)
  is_featured     Boolean   @default(false)
  
  // Ordem de exibição
  order           Int       @default(0)
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  tenants         Tenant[]
  subscriptions   Subscription[]
  
  @@map("plans")
}

model Subscription {
  id              String    @id @default(cuid())
  tenant_id       String
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  
  plan_id         String
  plan            Plan      @relation(fields: [plan_id], references: [id])
  
  status          String    // active, cancelled, expired, trial
  
  // Datas
  started_at      DateTime
  expires_at      DateTime?
  cancelled_at    DateTime?
  
  // Faturamento
  billing_cycle   String    // monthly, yearly
  amount          Decimal   @db.Decimal(10, 2)
  
  // Trial
  is_trial        Boolean   @default(false)
  trial_ends_at   DateTime?
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  @@map("subscriptions")
}

model Invoice {
  id              String    @id @default(cuid())
  tenant_id       String
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  
  invoice_number  String    @unique
  
  // Valores
  amount          Decimal   @db.Decimal(10, 2)
  tax             Decimal   @default(0) @db.Decimal(10, 2)
  total           Decimal   @db.Decimal(10, 2)
  
  // Status
  status          String    @default("pending") // pending, paid, overdue, cancelled
  
  // Datas
  issue_date      DateTime  @default(now())

// app/api/admin/metrics/overview/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { startOfMonth, endOfMonth, subMonths } from 'date-fns'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    
    // 🔒 Verificar se é Super Admin
    if (session?.user?.role !== 'superadmin') {
      return NextResponse.json(
        { error: 'Acesso negado' },
        { status: 403 }
      )
    }

    const now = new Date()
    const startOfCurrentMonth = startOfMonth(now)
    const endOfCurrentMonth = endOfMonth(now)
    const startOfLastMonth = startOfMonth(subMonths(now, 1))
    const endOfLastMonth = endOfMonth(subMonths(now, 1))

    // Buscar métricas em paralelo
    const [
      // Total de agências
      totalAgencies,
      activeAgencies,
      suspendedAgencies,
      newAgenciesThisMonth,
      newAgenciesLastMonth,
      
      // Total de usuários
      totalUsers,
      activeUsers,
      newUsersThisMonth,
      
      // Receita
      totalRevenue,
      revenueThisMonth,
      revenueLastMonth,
      
      // MRR (Monthly Recurring Revenue)
      mrr,
      
      // Churn
      cancelledThisMonth,
      
    ] = await Promise.all([
      // Agências
      prisma.tenant.count(),
      prisma.tenant.count({ where: { status: 'active' } }),
      prisma.tenant.count({ where: { status: 'suspended' } }),
      prisma.tenant.count({
        where: {
          created_at: {
            gte: startOfCurrentMonth,
            lte: endOfCurrentMonth,
          },
        },
      }),
      prisma.tenant.count({
        where: {
          created_at: {
            gte: startOfLastMonth,
            lte: endOfLastMonth,
          },
        },
      }),
      
      // Usuários
      prisma.user.count(),
      prisma.user.count({ where: { is_active: true } }),
      prisma.user.count({
        where: {
          created_at: {
            gte: startOfCurrentMonth,
            lte: endOfCurrentMonth,
          },
        },
      }),
      
      // Receita
      prisma.invoice.aggregate({
        where: { status: 'paid' },
        _sum: { total: true },
      }),
      prisma.invoice.aggregate({
        where: {
          status: 'paid',
          paid_at: {
            gte: startOfCurrentMonth,
            lte: endOfCurrentMonth,
          },
        },
        _sum: { total: true },
      }),
      prisma.invoice.aggregate({
        where: {
          status: 'paid',
          paid_at: {
            gte: startOfLastMonth,
            lte: endOfLastMonth,
          },
        },
        _sum: { total: true },
      }),
      
      // MRR
      prisma.subscription.aggregate({
        where: {
          status: 'active',
          billing_cycle: 'monthly',
        },
        _sum: { amount: true },
      }),
      
      // Churn
      prisma.subscription.count({
        where: {
          status: 'cancelled',
          cancelled_at: {
            gte: startOfCurrentMonth,
            lte: endOfCurrentMonth,
          },
        },
      }),
    ])

    // Calcular crescimento
    const agencyGrowth = newAgenciesLastMonth > 0
      ? ((newAgenciesThisMonth - newAgenciesLastMonth) / newAgenciesLastMonth) * 100
      : 0

    const revenueGrowth = Number(revenueLastMonth._sum.total || 0) > 0
      ? ((Number(revenueThisMonth._sum.total || 0) - Number(revenueLastMonth._sum.total || 0)) / 
         Number(revenueLastMonth._sum.total || 0)) * 100
      : 0

    // Calcular churn rate
    const churnRate = activeAgencies > 0
      ? (cancelledThisMonth / activeAgencies) * 100
      : 0

    return NextResponse.json({
      agencies: {
        total: totalAgencies,
        active: activeAgencies,
        suspended: suspendedAgencies,
        new_this_month: newAgenciesThisMonth,
        growth_percentage: agencyGrowth.toFixed(2),
      },
      users: {
        total: totalUsers,
        active: activeUsers,
        new_this_month: newUsersThisMonth,
      },
      revenue: {
        total: Number(totalRevenue._sum.total || 0),
        this_month: Number(revenueThisMonth._sum.total || 0),
        last_month: Number(revenueLastMonth._sum.total || 0),
        growth_percentage: revenueGrowth.toFixed(2),
      },
      mrr: {
        current: Number(mrr._sum.amount || 0),
      },
      churn: {
        rate: churnRate.toFixed(2),
        cancelled_this_month: cancelledThisMonth,
      },
    })
  } catch (error) {
    console.error('Erro ao buscar métricas:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar métricas' },
      { status: 500 }
    )
  }
}

// app/api/admin/agencies/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { z } from 'zod'

const agencySchema = z.object({
  name: z.string().min(3),
  slug: z.string().optional(),
  cnpj: z.string().optional(),
  email: z.string().email(),
  phone: z.string().optional(),
  address: z.string().optional(),
  city: z.string().optional(),
  state: z.string().optional(),
  zip_code: z.string().optional(),
  plan_id: z.string(),
  max_users: z.number().default(5),
  max_clients: z.number().default(100),
  max_trips: z.number().default(50),
})

// GET - Listar todas as agências
export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    
    if (session?.user?.role !== 'superadmin') {
      return NextResponse.json({ error: 'Acesso negado' }, { status: 403 })
    }

    const { searchParams } = new URL(req.url)
    const page = parseInt(searchParams.get('page') || '1')
    const limit = parseInt(searchParams.get('limit') || '20')
    const search = searchParams.get('search') || ''
    const status = searchParams.get('status') || ''
    const planId = searchParams.get('plan_id') || ''

    const skip = (page - 1) * limit

    const where: any = {}

    if (search) {
      where.OR = [
        { name: { contains: search, mode: 'insensitive' } },
        { email: { contains: search, mode: 'insensitive' } },
        { cnpj: { contains: search } },
      ]
    }

    if (status) {
      where.status = status
    }

    if (planId) {
      where.plan_id = planId
    }

    const [agencies, total] = await Promise.all([
      prisma.tenant.findMany({
        where,
        include: {
          plan: true,
          _count: {
            select: {
              users: true,
              clients: true,
              trips: true,
              reservations: true,
            },
          },
        },
        orderBy: { created_at: 'desc' },
        skip,
        take: limit,
      }),
      prisma.tenant.count({ where }),
    ])

    return NextResponse.json({
      agencies,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
    })
  } catch (error) {
    console.error('Erro ao buscar agências:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar agências' },
      { status: 500 }
    )
  }
}

// POST - Criar nova agência
export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    
    if (session?.user?.role !== 'superadmin') {
      return NextResponse.json({ error: 'Acesso negado' }, { status: 403 })
    }

    const body = await req.json()
    const data = agencySchema.parse(body)

    // Gerar slug se não fornecido
    let slug = data.slug || data.name
      .toLowerCase()
      .normalize('NFD')
      .replace(/[\u0300-\u036f]/g, '')
      .replace(/[^a-z0-9]+/g, '-')
      .replace(/(^-|-$)/g, '')

    // Verificar se slug já existe
    let counter = 1
    while (await prisma.tenant.findUnique({ where: { slug } })) {
      slug = `${slug}-${counter}`
      counter++
    }

    // Criar agência
    const agency = await prisma.tenant.create({
      data: {
        ...data,
        slug,
        status: 'active',
        is_active: true,
        activated_at: new Date(),
      },
      include: {
        plan: true,
      },
    })

    // Registrar log
    await prisma.auditLog.create({
      data: {
        user_id: session.user.id,
        action: 'create',
        resource: 'tenant',
        resource_id: agency.id,
        metadata: { agency_name: agency.name },
      },
    })

    return NextResponse.json(agency, { status: 201 })
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json(
        { error: 'Dados inválidos', details: error.errors },
        { status: 400 }
      )
    }

    console.error('Erro ao criar agência:', error)
    return NextResponse.json(
      { error: 'Erro ao criar agência' },
      { status: 500 }
    )
  }
}

// app/api/admin/agencies/[id]/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

// GET - Buscar agência por ID
export async function GET(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)
    
    if (session?.user?.role !== 'superadmin') {
      return NextResponse.json({ error: 'Acesso negado' }, { status: 403 })
    }

    const agency = await prisma.tenant.findUnique({
      where: { id: params.id },
      include: {
        plan: true,
        users: {
          select: {
            id: true,
            name: true,
            email: true,
            role: true,
            is_active: true,
            created_at: true,
          },
        },
        subscriptions: {
          orderBy: { created_at: 'desc' },
          take: 5,
        },
        invoices: {
          orderBy: { created_at: 'desc' },
          take: 10,
        },
        _count: {
          select: {
            users: true,
            clients: true,
            trips: true,
            reservations: true,
          },
        },
      },
    })

    if (!agency) {
      return NextResponse.json(
        { error: 'Agência não encontrada' },
        { status: 404 }
      )
    }

    // Buscar métricas adicionais
    const [totalRevenue, activeTrips] = await Promise.all([
      prisma.reservation.aggregate({
        where: { tenant_id: params.id },
        _sum: { total_value: true },
      }),
      prisma.trip.count({
        where: {
          tenant_id: params.id,
          status: 'ativa',
        },
      }),
    ])

    return NextResponse.json({
      ...agency,
      metrics: {
        total_revenue: Number(totalRevenue._sum.total_value || 0),
        active_trips: activeTrips,
      },
    })
  } catch (error) {
    console.error('Erro ao buscar agência:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar agência' },
      { status: 500 }
    )
  }
}

export async function PATCH(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)
    
    if (session?.user?.role !== 'superadmin') {

    }

    const body = await req.json()

    const agency = await prisma.tenant.update({
      where: { id: params.id },
      data: body,
      include: {

      },
    })

    // Registrar log
    await prisma.auditLog.create({
      data: {
        user_id: session.user.id,
        action: 'update',
        resource: 'tenant',
        resource_id: agency.id,
        metadata: { changes: body },
      },
    })

    return NextResponse.json(agency)
  } catch (error) {
    console.error('Erro ao atualizar agência:', error)
    return NextResponse.json(
      { error: 'Erro ao atualizar agência' },
      { status: 500 }
    )
  }
}

// DELETE - Excluir agência
export async function DELETE(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)
    
    if (session?.user?.role !== 'superadmin') {
      return NextResponse.json({ error: 'Acesso negado' }, { status: 403 })
    }

    // Verificar se agência existe
    const agency = await prisma.tenant.findUnique({
      where: { id: params.id },
    })

    if (!agency) {
      return NextResponse.json(
        { error: 'Agência não encontrada' },
        { status: 404 }
      )
    }

    // Excluir agência (cascade irá excluir todos os dados relacionados)
    await prisma.tenant.delete({
      where: { id: params.id },
    })

    // Registrar log
    await prisma.auditLog.create({
      data: {
        user_id: session.user.id,
        action: 'delete',
        resource: 'tenant',
        resource_id: params.id,
        metadata: { agency_name: agency.name },
      },
    })

    return NextResponse.json({ message: 'Agência excluída com sucesso' })
  } catch (error) {
    console.error('Erro ao excluir agência:', error)
    return NextResponse.json(
      { error: 'Erro ao excluir agência' },
      { status: 500 }
    )
  }
}

interface Agency {

  logo: string | null
  status: string
  plan: {
    name: string
  }
  created_at: Date
  _count: {
    users: number
    clients: number
    trips: number
    reservations: number
  }
}

interface AgencyListProps {
  page: number
  search: string
  status: string
}

export function AgencyList({ page, search, status }: AgencyListProps) {
  const router = useRouter()
  const { toast } = useToast()
  const [agencies, setAgencies] = useState<Agency[]>([])
  const [total, setTotal] = useState(0)
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    loadAgencies()
  }, [page, search, status])

  async function loadAgencies() {
    try {
      setIsLoading(true)
      const params = new URLSearchParams({
        page: page.toString(),
        search,
        status,
      })

      const res = await fetch(`/api/admin/agencies?${params}`)
      const data = await res.json()

      setAgencies(data.agencies)
      setTotal(data.pagination.total)
    } catch (error) {
      toast({
        title: 'Erro',
        description: 'Erro ao carregar agências',
        variant: 'destructive',
      })
    } finally {
      setIsLoading(false)
    }
  }

  async function handleSuspend(id: string) {
    if (!confirm('Tem certeza que deseja suspender esta agência?')) return

    try {
      const res = await fetch(`/api/admin/agencies/${id}/suspend`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          reason: 'Suspensão manual pelo administrador',
        }),
      })

      if (!res.ok) throw new Error()

      toast({
        title: 'Sucesso',
        description: 'Agência suspensa com sucesso',
      })

      loadAgencies()
    } catch (error) {
      toast({
        title: 'Erro',
        description: 'Erro ao suspender agência',
        variant: 'destructive',
      })
    }
  }

  async function handleActivate(id: string) {
    try {
      const res = await fetch(`/api/admin/agencies/${id}/activate`, {
        method: 'POST',
      })

      if (!res.ok) throw new Error()

      toast({
        title: 'Sucesso',
        description: 'Agência ativada com sucesso',
      })

      loadAgencies()
    } catch (error) {
      toast({
        title: 'Erro',
        description: 'Erro ao ativar agência',
        variant: 'destructive',
      })
    }
  }

  async function handleDelete(id: string) {
    if (!confirm('ATENÇÃO: Esta ação irá excluir permanentemente a agência e TODOS os seus dados. Deseja continuar?')) return

    try {
      const res = await fetch(`/api/admin/agencies/${id}`, {
        method: 'DELETE',
      })

      if (!res.ok) throw new Error()

      toast({
        title: 'Sucesso',
        description: 'Agência excluída com sucesso',
      })

      loadAgencies()
    } catch (error) {
      toast({
        title: 'Erro',
        description: 'Erro ao excluir agência',
        variant: 'destructive',
      })
    }
  }

  const statusColors = {
    active: 'bg-green-100 text-green-800',
    suspended: 'bg-red-100 text-red-800',
    cancelled: 'bg-gray-100 text-gray-800',
  }

  const statusLabels = {
    active: 'Ativa',
    suspended: 'Suspensa',
    cancelled: 'Cancelada',
  }

  if (isLoading) {
    return <div className="text-center py-12">Carregando...</div>
  }

  return (
    <div className="space-y-4">
      <div className="bg-white rounded-lg shadow">
        <Table>
          <TableHeader>
            <TableRow>
              <TableHead>Agência</TableHead>
              <TableHead>Plano</TableHead>
              <TableHead>Status</TableHead>
              <TableHead className="text-right">Usuários</TableHead>
              <TableHead className="text-right">Clientes</TableHead>
              <TableHead className="text-right">Viagens</TableHead>
              <TableHead>Criada em</TableHead>
              <TableHead className="w-[50px]"></TableHead>
            </TableRow>
          </TableHeader>
          <TableBody>
            {agencies.map((agency) => (
              <TableRow key={agency.id}>
                <TableCell>
                  <div className="flex items-center gap-3">
                    <Avatar>
                      <AvatarImage src={agency.logo || ''} />
                      <AvatarFallback>
                        {agency.name.substring(0, 2).toUpperCase()}
                      </AvatarFallback>
                    </Avatar>
                    <div>
                      <div className="font-medium">{agency.name}</div>
                      <div className="text-sm text-gray-500">{agency.email}</div>
                    </div>
                  </div>
                </TableCell>
                <TableCell>
                  <Badge variant="outline">{agency.plan.name}</Badge>
                </TableCell>
                <TableCell>
                  <Badge className={statusColors[agency.status as keyof typeof statusColors]}>
                    {statusLabels[agency.status as keyof typeof statusLabels]}
                  </Badge>
                </TableCell>
                <TableCell className="text-right">{agency._count.users}</TableCell>
                <TableCell className="text-right">{agency._count.clients}</TableCell>
                <TableCell className="text-right">{agency._count.trips}</TableCell>
                <TableCell>
                  {format(new Date(agency.created_at), 'dd/MM/yyyy', { locale: ptBR })}

const formSchema = z.object({
  name: z.string().min(3, 'Nome deve ter no mínimo 3 caracteres'),
  email: z.string().email('Email inválido'),
  cnpj: z.string().optional(),
  phone: z.string().optional(),
  address: z.string().optional(),

  state: z.string().optional(),
  zip_code: z.string().optional(),
  plan_id: z.string().min(1, 'Selecione um plano'),
  max_users: z.number().min(1).default(5),
  max_clients: z.number().min(1).default(100),
  max_trips: z.number().min(1).default(50),
})

interface Plan {
  id: string
  name: string
  max_users: number
  max_clients: number
  max_trips: number
}

interface AgencyFormProps {
  plans: Plan[]
  initialData?: any
}

export function AgencyForm({ plans, initialData }: AgencyFormProps) {
  const router = useRouter()
  const { toast } = useToast()
  const [isLoading, setIsLoading] = useState(false)

  const form = useForm<z.infer<typeof formSchema>>({
    resolver: zodResolver(formSchema),
    defaultValues: initialData || {
      name: '',
      email: '',
      cnpj: '',
      phone: '',
      address: '',
      city: '',
      state: '',
      zip_code: '',
      plan_id: '',
      max_users: 5,
      max_clients: 100,
      max_trips: 50,
    },
  })

  async function onSubmit(values: z.infer<typeof formSchema>) {
    try {
      setIsLoading(true)

      const url = initialData
        ? `/api/admin/agencies/${initialData.id}`
        : '/api/admin/agencies'

      const method = initialData ? 'PATCH' : 'POST'

      const res = await fetch(url, {
        method,
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(values),
      })

      if (!res.ok) {
        const error = await res.json()
        throw new Error(error.error)
      }

      const agency = await res.json()

      toast({
        title: 'Sucesso!',
        description: initialData
          ? 'Agência atualizada com sucesso'
          : 'Agência criada com sucesso',
      })

      router.push(`/admin/agencies/${agency.id}`)
      router.refresh()
    } catch (error: any) {
      toast({
        title: 'Erro',
        description: error.message || 'Erro ao salvar agência',
        variant: 'destructive',
      })
    } finally {
      setIsLoading(false)
    }
  }

    const plan = plans.find(p => p.id === planId)

      form.setValue('max_users', plan.max_users)
      form.setValue('max_clients', plan.max_clients)
      form.setValue('max_trips', plan.max_trips)
    }
  }

  return (

      <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-6">

            <CardTitle>Informações Básicas</CardTitle>
          </CardHeader>
          <CardContent className="space-y-4">
            <FormField
              control={form.control}
              name="name"
              render={({ field }) => (
                <FormItem>
                  <FormLabel>Nome da Agência *</FormLabel>
                  <FormControl>
                    <Input placeholder="Visite Turismo" {...field} />
                  </FormControl>
                  <FormMessage />
                </FormItem>
              )}
            />

            <div className="grid grid-cols-2 gap-4">
              <FormField
                control={form.control}
                name="email"
                render={({ field }) => (
                  <FormItem>
                    <FormLabel>Email *</FormLabel>
                    <FormControl>
                      <Input
                        type="email"
                        placeholder="contato@agencia.com"
                        {...field}
                      />
                    </FormControl>
                    <FormMessage />
                  </FormItem>
                )}
              />

              <FormField
                control={form.control}
                name="phone"
                render={({ field }) => (
                  <FormItem>
                    <FormLabel>Telefone</FormLabel>
                    <FormControl>
                      <Input placeholder="(11) 99999-9999" {...field} />
                    </FormControl>
                    <FormMessage />
                  </FormItem>
                )}
              />
            </div>

            <FormField
              control={form.control}
              name="cnpj"
              render={({ field }) => (
                <FormItem>
                  <FormLabel>CNPJ</FormLabel>
                  <FormControl>
                    <Input placeholder="00.000.000/0000-00" {...field} />
                  </FormControl>
                  <FormMessage />
                </FormItem>
              )}
            />
          </CardContent>
        </Card>

        {/* Endereço */}

              control={form.control}

  const [health, setHealth] = useState<SystemHealth | null>(null)
  const [isLoading, setIsLoading] = useState(true)

    loadHealth()
    const interval = setInterval(loadHealth, 30000) // Atualizar a cada 30s
    return () => clearInterval(interval)
  }, [])

  async function loadHealth() {
    try {
      const res = await fetch('/api/admin/system/health')
      const data = await res.json()
      setHealth(data)

      setIsLoading(false)

        return <XCircle className="h-5 w-5 text-red-600" />

API: MRR (Monthly Recurring Revenue)
// app/api/admin/metrics/mrr/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { startOfMonth, endOfMonth, subMonths, format } from 'date-fns'
import { ptBR } from 'date-fns/locale'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    
    if (session?.user?.role !== 'superadmin') {
      return NextResponse.json({ error: 'Acesso negado' }, { status: 403 })
    }

    const { searchParams } = new URL(req.url)
    const months = parseInt(searchParams.get('months') || '12')

    const data = []

    for (let i = months - 1; i >= 0; i--) {
      const date = subMonths(new Date(), i)
      const start = startOfMonth(date)
      const end = endOfMonth(date)

      // Buscar assinaturas ativas no período
      const subscriptions = await prisma.subscription.findMany({
        where: {
          status: 'active',
          started_at: {
            lte: end,
          },
          OR: [
            { expires_at: null },
            { expires_at: { gte: start } },
          ],
        },
        include: {
          plan: true,
        },
      })

      // Calcular MRR
      let mrr = 0
      let newMRR = 0
      let churnedMRR = 0
      let expansionMRR = 0

      subscriptions.forEach((sub) => {
        const amount = Number(sub.amount)
        
        // Normalizar para mensal
        const monthlyAmount = sub.billing_cycle === 'yearly' 
          ? amount / 12 
          : amount

        mrr += monthlyAmount

        // New MRR (assinaturas iniciadas neste mês)
        if (sub.started_at >= start && sub.started_at <= end) {
          newMRR += monthlyAmount
        }
      })

      // Churn MRR (assinaturas canceladas neste mês)
      const cancelledSubs = await prisma.subscription.findMany({
        where: {
          status: 'cancelled',
          cancelled_at: {
            gte: start,
            lte: end,
          },
        },
      })

      cancelledSubs.forEach((sub) => {
        const amount = Number(sub.amount)
        const monthlyAmount = sub.billing_cycle === 'yearly' 
          ? amount / 12 
          : amount
        churnedMRR += monthlyAmount
      })

      data.push({
        month: format(date, 'MMM/yy', { locale: ptBR }),
        mrr: Math.round(mrr),
        new_mrr: Math.round(newMRR),
        churned_mrr: Math.round(churnedMRR),
        expansion_mrr: Math.round(expansionMRR),
        net_mrr: Math.round(newMRR + expansionMRR - churnedMRR),
      })
    }

    return NextResponse.json(data)
  } catch (error) {
    console.error('Erro ao calcular MRR:', error)
    return NextResponse.json(
      { error: 'Erro ao calcular MRR' },
      { status: 500 }
    )
  }
}

// app/api/admin/metrics/churn/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { startOfMonth, endOfMonth, subMonths, format } from 'date-fns'
import { ptBR } from 'date-fns/locale'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    
    if (session?.user?.role !== 'superadmin') {
      return NextResponse.json({ error: 'Acesso negado' }, { status: 403 })
    }

    const { searchParams } = new URL(req.url)
    const months = parseInt(searchParams.get('months') || '12')

    const data = []

    for (let i = months - 1; i >= 0; i--) {
      const date = subMonths(new Date(), i)
      const start = startOfMonth(date)
      const end = endOfMonth(date)

      // Agências ativas no início do mês
      const activeAtStart = await prisma.tenant.count({
        where: {
          created_at: { lt: start },
          OR: [
            { cancelled_at: null },
            { cancelled_at: { gte: start } },
          ],
        },
      })

      // Agências canceladas durante o mês
      const cancelled = await prisma.tenant.count({
        where: {
          status: 'cancelled',
          cancelled_at: {
            gte: start,
            lte: end,
          },
        },
      })

      // Calcular churn rate
      const churnRate = activeAtStart > 0 
        ? (cancelled / activeAtStart) * 100 
        : 0

      // Revenue Churn
      const [cancelledRevenue, totalRevenue] = await Promise.all([
        prisma.subscription.aggregate({
          where: {
            status: 'cancelled',
            cancelled_at: {
              gte: start,
              lte: end,
            },
          },
          _sum: { amount: true },
        }),
        prisma.subscription.aggregate({
          where: {
            status: 'active',
            started_at: { lte: start },
          },
          _sum: { amount: true },
        }),
      ])

      const revenueChurnRate = Number(totalRevenue._sum.amount || 0) > 0
        ? (Number(cancelledRevenue._sum.amount || 0) / Number(totalRevenue._sum.amount || 0)) * 100
        : 0

      data.push({
        month: format(date, 'MMM/yy', { locale: ptBR }),
        customer_churn_rate: Number(churnRate.toFixed(2)),
        revenue_churn_rate: Number(revenueChurnRate.toFixed(2)),
        cancelled_count: cancelled,
        active_count: activeAtStart,
      })
    }

    return NextResponse.json(data)
  } catch (error) {
    console.error('Erro ao calcular churn:', error)
    return NextResponse.json(
      { error: 'Erro ao calcular churn' },
      { status: 500 }
    )
  }
}

          },
          _sum: { amount: true },
        })

        return {
          name: plan?.name || 'Sem plano',
          revenue: Number(subs._sum.amount || 0),
        }
      })
    )

    // Crescimento mensal
    const monthlyGrowth = []
    for (let i = 5; i >= 0; i--) {
      const date = subMonths(now, i)
      const start = startOfMonth(date)
      const end = endOfMonth(date)

      const [newAgencies, cancelled] = await Promise.all([
        prisma.tenant.count({
          where: {
            created_at: { gte: start, lte: end },
          },
        }),
        prisma.tenant.count({
          where: {
            status: 'cancelled',
            cancelled_at: { gte: start, lte: end },
          },
        }),
      ])

      monthlyGrowth.push({
        month: start.toLocaleDateString('pt-BR', { month: 'short' }),
        new_agencies: newAgencies,
        cancelled,
      })
    }

    return NextResponse.json({
      arr: Math.round(arr),
      arr_growth: 15, // Calcular vs período anterior
      mrr: Math.round(mrr),
      mrr_growth: 12,
      cac: Math.round(cac),
      ltv: Math.round(ltv),
      ltv_cac_ratio: cac > 0 ? Number((ltv / cac).toFixed(2)) : 0,
      agencies_by_plan: plansData,
      revenue_by_plan: revenueByPlan,
      monthly_growth: monthlyGrowth,
    })
  } catch (error) {
    console.error('Erro ao buscar dashboard SaaS:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar dashboard SaaS' },
      { status: 500 }
    )
  }
}

export async function GET(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)
    
    if (session?.user?.role !== 'superadmin') {
      return NextResponse.json({ error: 'Acesso negado' }, { status: 403 })
    }

    const tenantId = params.id

    // Buscar dados dos últimos 6 meses
    const data = []
    for (let i = 5; i >= 0; i--) {
      const date = subMonths(new Date(), i)
      const start = startOfMonth(date)
      const end = endOfMonth(date)

      const [
        newClients,
        newReservations,
        revenue,
        activeUsers,
      ] = await Promise.all([
        prisma.client.count({
          where: {
            tenant_id: tenantId,
            created_at: { gte: start, lte: end },
          },
        }),
        prisma.reservation.count({
          where: {
            tenant_id: tenantId,
            created_at: { gte: start, lte: end },
          },
        }),
        prisma.reservation.aggregate({
          where: {
            tenant_id: tenantId,
            created_at: { gte: start, lte: end },
          },
          _sum: { total_value: true },
        }),
        prisma.user.count({
          where: {
            tenant_id: tenantId,
            is_active: true,
            last_login: { gte: start, lte: end },
          },
        }),
      ])

      data.push({
        month: start.toLocaleDateString('pt-BR', { month: 'short', year: '2-digit' }),
        new_clients: newClients,
        new_reservations: newReservations,
        revenue: Number(revenue._sum.total_value || 0),
        active_users: activeUsers,
      })
    }

    // Totais gerais
    const [
      totalClients,
      totalReservations,
      totalRevenue,
      totalUsers,
      activeTrips,
    ] = await Promise.all([
      prisma.client.count({ where: { tenant_id: tenantId } }),
      prisma.reservation.count({ where: { tenant_id: tenantId } }),
      prisma.reservation.aggregate({
        where: { tenant_id: tenantId },
        _sum: { total_value: true },
      }),
      prisma.user.count({ where: { tenant_id: tenantId } }),
      prisma.trip.count({
        where: {
          tenant_id: tenantId,
          status: 'ativa',
        },
      }),
    ])

    return NextResponse.json({
      monthly_data: data,
      totals: {
        clients: totalClients,
        reservations: totalReservations,
        revenue: Number(totalRevenue._sum.total_value || 0),
        users: totalUsers,
        active_trips: activeTrips,
      },
    })

// middleware.ts

import { withAuth } from 'next-auth/middleware'
import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'

// Rate limiting simples (em produção usar Redis)
const rateLimit = new Map<string, { count: number; resetTime: number }>()

function checkRateLimit(ip: string): boolean {
  const now = Date.now()
  const limit = rateLimit.get(ip)

  if (!limit || now > limit.resetTime) {
    rateLimit.set(ip, {
      count: 1,
      resetTime: now + 60000, // 1 minuto
    })
    return true
  }

  if (limit.count >= 100) { // 100 requests por minuto
    return false
  }

  limit.count++
  return true
}

export default withAuth(
  function middleware(req) {
    const token = req.nextauth.token
    const pathname = req.nextUrl.pathname

    // Rate limiting
    const ip = req.ip || req.headers.get('x-forwarded-for') || 'unknown'
    if (!checkRateLimit(ip)) {
      return new NextResponse('Too Many Requests', { status: 429 })
    }

    // Verificar autenticação
    if (!token) {
      return NextResponse.redirect(new URL('/login', req.url))
    }

    // Proteção Super Admin
    if (pathname.startsWith('/admin')) {
      if (token.role !== 'superadmin') {
        return NextResponse.redirect(new URL('/unauthorized', req.url))
      }
    }

    // Proteção Dashboard (Agência/Vendedor)
    if (pathname.startsWith('/dashboard')) {
      if (!['superadmin', 'agencia', 'vendedor'].includes(token.role as string)) {
        return NextResponse.redirect(new URL('/unauthorized', req.url))
      }
    }

    // Proteção Portal (Cliente)
    if (pathname.startsWith('/portal')) {
      if (token.role !== 'cliente') {
        return NextResponse.redirect(new URL('/unauthorized', req.url))
      }
    }

    // Logs de auditoria para ações sensíveis
    if (pathname.includes('/suspend') || pathname.includes('/delete')) {
      // Registrar log assíncrono
      fetch(`${process.env.NEXT_PUBLIC_APP_URL}/api/audit/log`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          user_id: token.id,
          action: pathname.split('/').pop(),
          path: pathname,
          ip,
          timestamp: new Date().toISOString(),
        }),
      }).catch(console.error)
    }

    return NextResponse.next()
  },
  {
    callbacks: {
      authorized: ({ token }) => !!token,
    },
  }
)

// lib/api-middleware.ts

import { getServerSession } from 'next-auth'
import { authOptions } from './auth'
import { NextResponse } from 'next/server'
import { hasPermission } from './rbac'

export async function withAuth(
  handler: (req: Request, context: any) => Promise<Response>,
  options?: {
    requiredRole?: string[]
    requiredPermission?: string
  }
) {
  return async (req: Request, context: any) => {
    const session = await getServerSession(authOptions)

    // Verificar autenticação
    if (!session?.user) {
      return NextResponse.json(
        { error: 'Não autenticado' },
        { status: 401 }
      )
    }

    // Verificar role
    if (options?.requiredRole) {
      if (!options.requiredRole.includes(session.user.role)) {
        return NextResponse.json(
          { error: 'Acesso negado' },
          { status: 403 }
        )
      }
    }

    // Verificar permissão
    if (options?.requiredPermission) {
      if (!hasPermission(session.user.role, options.requiredPermission)) {
        return NextResponse.json(
          { error: 'Sem permissão' },
          { status: 403 }
        )
      }
    }

    // Executar handler
    return handler(req, context)
  }
}

Criptografia de Dados Sensíveis
// lib/encryption.ts

import crypto from 'crypto'

const ENCRYPTION_KEY = process.env.ENCRYPTION_KEY! // 32 bytes
const IV_LENGTH = 16

/**
 * Criptografar dados sensíveis
 */
export function encrypt(text: string): string {
  const iv = crypto.randomBytes(IV_LENGTH)
  const cipher = crypto.createCipheriv(
    'aes-256-cbc',
    Buffer.from(ENCRYPTION_KEY),
    iv
  )

  let encrypted = cipher.update(text)
  encrypted = Buffer.concat([encrypted, cipher.final()])

  return iv.toString('hex') + ':' + encrypted.toString('hex')
}

/**
 * Descriptografar dados
 */
export function decrypt(text: string): string {
  const parts = text.split(':')
  const iv = Buffer.from(parts.shift()!, 'hex')
  const encryptedText = Buffer.from(parts.join(':'), 'hex')

  const decipher = crypto.createDecipheriv(
    'aes-256-cbc',
    Buffer.from(ENCRYPTION_KEY),
    iv
  )

  let decrypted = decipher.update(encryptedText)
  decrypted = Buffer.concat([decrypted, decipher.final()])

  return decrypted.toString()
}

/**
 * Hash de senha (usar bcrypt na prática)
 */
export function hashPassword(password: string): string {
  return crypto
    .createHash('sha256')
    .update(password + process.env.PASSWORD_SALT)
    .digest('hex')
}
