// app/api/dashboard/metrics/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { startOfDay, endOfDay, addDays, startOfMonth, endOfMonth, subMonths } from 'date-fns'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const tenantId = session.user.tenantId

    // ═══════════════════════════════════════════════════════════
    // DATAS PARA FILTROS
    // ═══════════════════════════════════════════════════════════
    
    const today = new Date()
    const todayStart = startOfDay(today)
    const todayEnd = endOfDay(today)
    const next3Days = addDays(today, 3)
    const currentMonthStart = startOfMonth(today)
    const currentMonthEnd = endOfMonth(today)

    // ═══════════════════════════════════════════════════════════
    // 1. FATURAMENTO TOTAL
    // ═══════════════════════════════════════════════════════════
    
    const totalRevenue = await prisma.revenue.aggregate({
      where: {
        tenant_id: tenantId,
        status: 'received',
      },
      _sum: {
        amount: true,
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 2. TOTAL DE CLIENTES
    // ═══════════════════════════════════════════════════════════
    
    const totalCustomers = await prisma.customer.count({
      where: {
        tenant_id: tenantId,
        status: 'active',
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 3. RECEITA TOTAL (INCLUINDO PENDENTE)
    // ═══════════════════════════════════════════════════════════
    
    const totalRevenueAll = await prisma.revenue.aggregate({
      where: {
        tenant_id: tenantId,
      },
      _sum: {
        amount: true,
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 4. TOTAL DESPESAS
    // ═══════════════════════════════════════════════════════════
    
    const totalExpenses = await prisma.expense.aggregate({
      where: {
        tenant_id: tenantId,
        status: 'paid',
      },
      _sum: {
        amount: true,
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 5. LUCRO LÍQUIDO
    // ═══════════════════════════════════════════════════════════
    
    const netProfit =
      (totalRevenue._sum.amount?.toNumber() || 0) -
      (totalExpenses._sum.amount?.toNumber() || 0)

    // ═══════════════════════════════════════════════════════════
    // 6. MARGEM %
    // ═══════════════════════════════════════════════════════════
    
    const margin =
      totalRevenue._sum.amount?.toNumber() || 0 > 0
        ? (netProfit / (totalRevenue._sum.amount?.toNumber() || 1)) * 100
        : 0

    // ═══════════════════════════════════════════════════════════
    // 7. RECEBIDOS HOJE
    // ═══════════════════════════════════════════════════════════
    
    const receivedToday = await prisma.revenue.aggregate({
      where: {
        tenant_id: tenantId,
        status: 'received',
        received_date: {
          gte: todayStart,
          lte: todayEnd,
        },
      },
      _sum: {
        amount: true,
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 8. A RECEBER NOS PRÓXIMOS 3 DIAS
    // ═══════════════════════════════════════════════════════════
    
    const toReceiveNext3Days = await prisma.revenue.aggregate({
      where: {
        tenant_id: tenantId,
        status: 'pending',
        due_date: {
          gte: todayStart,
          lte: next3Days,
        },
      },
      _sum: {
        amount: true,
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 9. RESERVAS REALIZADAS HOJE
    // ═══════════════════════════════════════════════════════════
    
    const reservationsToday = await prisma.reservation.count({
      where: {
        tenant_id: tenantId,
        created_at: {
          gte: todayStart,
          lte: todayEnd,
        },
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 10. TICKET MÉDIO POR RESERVA
    // ═══════════════════════════════════════════════════════════
    
    const reservationsWithRevenue = await prisma.reservation.findMany({
      where: {
        tenant_id: tenantId,
        status: 'confirmed',
      },
      select: {
        total_amount: true,
      },
    })

    const avgTicket =
      reservationsWithRevenue.length > 0
        ? reservationsWithRevenue.reduce(
            (sum, r) => sum + (r.total_amount?.toNumber() || 0),
            0
          ) / reservationsWithRevenue.length
        : 0

    // ═══════════════════════════════════════════════════════════
    // 11. TOTAL A RECEBER
    // ═══════════════════════════════════════════════════════════
    
    const totalToReceive = await prisma.revenue.aggregate({
      where: {
        tenant_id: tenantId,
        status: 'pending',
      },
      _sum: {
        amount: true,
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 12. TOTAL A PAGAR
    // ═══════════════════════════════════════════════════════════
    
    const totalToPay = await prisma.expense.aggregate({
      where: {
        tenant_id: tenantId,
        status: 'pending',
      },
      _sum: {
        amount: true,
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 13. TOTAL DE VIAGENS ATIVAS
    // ═══════════════════════════════════════════════════════════
    
    const activeTrips = await prisma.trip.count({
      where: {
        tenant_id: tenantId,
        status: 'active',
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 14. TOTAL DE CLIENTES ATIVOS
    // ═══════════════════════════════════════════════════════════
    
    const activeCustomers = await prisma.customer.count({
      where: {
        tenant_id: tenantId,
        status: 'active',
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 15. JÁ RECEBIDO (VIAGENS ATIVAS)
    // ═══════════════════════════════════════════════════════════
    
    const receivedFromActiveTrips = await prisma.revenue.aggregate({
      where: {
        tenant_id: tenantId,
        status: 'received',
        trip: {
          status: 'active',
        },
      },
      _sum: {
        amount: true,
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 16. PENDENTE (VIAGENS ATIVAS)
    // ═══════════════════════════════════════════════════════════
    
    const pendingFromActiveTrips = await prisma.revenue.aggregate({
      where: {
        tenant_id: tenantId,
        status: 'pending',
        trip: {
          status: 'active',
        },
      },
      _sum: {
        amount: true,
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 17. NPS MÉDIO
    // ═══════════════════════════════════════════════════════════
    
    const npsResponses = await prisma.nPSResponse.aggregate({
      where: {
        tenant_id: tenantId,
      },
      _avg: {
        score: true,
      },
    })

    const npsScore = npsResponses._avg.score?.toNumber() || 0

    // Calcular NPS real (% promotores - % detratores)
    const allNPS = await prisma.nPSResponse.findMany({
      where: {
        tenant_id: tenantId,
      },
      select: {
        score: true,
      },
    })

    const promoters = allNPS.filter((n) => n.score >= 9).length
    const detractors = allNPS.filter((n) => n.score <= 6).length
    const total = allNPS.length

    const nps = total > 0 ? ((promoters - detractors) / total) * 100 : 0

    // ═══════════════════════════════════════════════════════════
    // RETORNAR TODAS AS MÉTRICAS
    // ═══════════════════════════════════════════════════════════

    return NextResponse.json({
      kpis: {
        totalRevenue: totalRevenue._sum.amount?.toNumber() || 0,
        totalCustomers,
        totalRevenueAll: totalRevenueAll._sum.amount?.toNumber() || 0,
        totalExpenses: totalExpenses._sum.amount?.toNumber() || 0,
        netProfit,
        margin,
        receivedToday: receivedToday._sum.amount?.toNumber() || 0,
        toReceiveNext3Days: toReceiveNext3Days._sum.amount?.toNumber() || 0,
        reservationsToday,
        avgTicket,
        totalToReceive: totalToReceive._sum.amount?.toNumber() || 0,
        totalToPay: totalToPay._sum.amount?.toNumber() || 0,
        activeTrips,
        activeCustomers,
        receivedFromActiveTrips:
          receivedFromActiveTrips._sum.amount?.toNumber() || 0,
        pendingFromActiveTrips:
          pendingFromActiveTrips._sum.amount?.toNumber() || 0,
        nps,
        npsScore,
      },
    })
  } catch (error) {
    console.error('Erro ao buscar métricas:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar métricas' },
      { status: 500 }
    )
  }
}


API DE GRÁFICOS

// app/api/dashboard/charts/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { startOfMonth, endOfMonth, subMonths, format } from 'date-fns'
import { ptBR } from 'date-fns/locale'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const tenantId = session.user.tenantId

    // ═══════════════════════════════════════════════════════════
    // 1. RECEITA VS DESPESAS (ÚLTIMOS 6 MESES)
    // ═══════════════════════════════════════════════════════════

    const revenueVsExpenses = []

    for (let i = 5; i >= 0; i--) {
      const date = subMonths(new Date(), i)
      const monthStart = startOfMonth(date)
      const monthEnd = endOfMonth(date)

      const revenue = await prisma.revenue.aggregate({
        where: {
          tenant_id: tenantId,
          status: 'received',
          received_date: {
            gte: monthStart,
            lte: monthEnd,
          },
        },
        _sum: {
          amount: true,
        },
      })

      const expenses = await prisma.expense.aggregate({
        where: {
          tenant_id: tenantId,
          status: 'paid',
          paid_date: {
            gte: monthStart,
            lte: monthEnd,
          },
        },
        _sum: {
          amount: true,
        },
      })

      revenueVsExpenses.push({
        month: format(date, 'MMM/yy', { locale: ptBR }),
        revenue: revenue._sum.amount?.toNumber() || 0,
        expenses: expenses._sum.amount?.toNumber() || 0,
        profit:
          (revenue._sum.amount?.toNumber() || 0) -
          (expenses._sum.amount?.toNumber() || 0),
      })
    }

    // ═══════════════════════════════════════════════════════════
    // 2. ORIGEM DOS CLIENTES
    // ═══════════════════════════════════════════════════════════

    const customersBySource = await prisma.customer.groupBy({
      by: ['source'],
      where: {
        tenant_id: tenantId,
      },
      _count: {
        id: true,
      },
    })

    const customerOrigins = customersBySource.map((item) => ({
      source: item.source || 'Não informado',
      count: item._count.id,
    }))

    // ═══════════════════════════════════════════════════════════
    // 3. DESTINOS MAIS RESERVADOS
    // ═══════════════════════════════════════════════════════════

    const topDestinations = await prisma.reservation.groupBy({
      by: ['trip_id'],
      where: {
        tenant_id: tenantId,
        status: 'confirmed',
      },
      _count: {
        id: true,
      },
      orderBy: {
        _count: {
          id: 'desc',
        },
      },
      take: 10,
    })

    const destinationsWithNames = await Promise.all(
      topDestinations.map(async (item) => {
        const trip = await prisma.trip.findUnique({
          where: { id: item.trip_id },
          select: { title: true, destination: true },
        })

        return {
          destination: trip?.destination || trip?.title || 'Desconhecido',
          count: item._count.id,
        }
      })
    )

    // ═══════════════════════════════════════════════════════════
    // 4. TAXA DE CANCELAMENTO (VIAGENS)
    // ═══════════════════════════════════════════════════════════

    const totalTrips = await prisma.trip.count({
      where: { tenant_id: tenantId },
    })

    const cancelledTrips = await prisma.trip.count({
      where: {
        tenant_id: tenantId,
        status: 'cancelled',
      },
    })

    const tripCancellationRate =
      totalTrips > 0 ? (cancelledTrips / totalTrips) * 100 : 0

    // ═══════════════════════════════════════════════════════════
    // 5. TAXA DE CANCELAMENTO (RESERVAS)
    // ═══════════════════════════════════════════════════════════

    const totalReservations = await prisma.reservation.count({
      where: { tenant_id: tenantId },
    })

    const cancelledReservations = await prisma.reservation.count({
      where: {
        tenant_id: tenantId,
        status: 'cancelled',
      },
    })

    const reservationCancellationRate =
      totalReservations > 0
        ? (cancelledReservations / totalReservations) * 100
        : 0

    // ═══════════════════════════════════════════════════════════
    // 6. MÉDIA DE RESERVA POR VIAGEM
    // ═══════════════════════════════════════════════════════════

    const tripsWithReservations = await prisma.trip.findMany({
      where: {
        tenant_id: tenantId,
        status: 'active',
      },
      include: {
        reservations: {
          where: {
            status: 'confirmed',
          },
        },
      },
    })

    const avgReservationsPerTrip =
      tripsWithReservations.length > 0
        ? tripsWithReservations.reduce(
            (sum, trip) => sum + trip.reservations.length,
            0
          ) / tripsWithReservations.length
        : 0

    // ═══════════════════════════════════════════════════════════
    // 7. VIAGENS POR MÊS (ÚLTIMOS 12 MESES)
    // ═══════════════════════════════════════════════════════════

    const tripsByMonth = []

    for (let i = 11; i >= 0; i--) {
      const date = subMonths(new Date(), i)
      const monthStart = startOfMonth(date)
      const monthEnd = endOfMonth(date)

      const count = await prisma.trip.count({
        where: {
          tenant_id: tenantId,
          created_at: {
            gte: monthStart,
            lte: monthEnd,
          },
        },
      })

      tripsByMonth.push({
        month: format(date, 'MMM/yy', { locale: ptBR }),
        count,
      })
    }

    // ═══════════════════════════════════════════════════════════
    // 8. RESERVAS POR MÊS (ÚLTIMOS 12 MESES)
    // ═══════════════════════════════════════════════════════════

    const reservationsByMonth = []

    for (let i = 11; i >= 0; i--) {
      const date = subMonths(new Date(), i)
      const monthStart = startOfMonth(date)
      const monthEnd = endOfMonth(date)

      const count = await prisma.reservation.count({
        where: {
          tenant_id: tenantId,
          created_at: {
            gte: monthStart,
            lte: monthEnd,
          },
        },
      })

      reservationsByMonth.push({
        month: format(date, 'MMM/yy', { locale: ptBR }),
        count,
      })
    }

    // ═══════════════════════════════════════════════════════════
    // 9. PASSAGEIROS EMBARCADOS POR MÊS (ÚLTIMOS 12 MESES)
    // ═══════════════════════════════════════════════════════════

    const passengersByMonth = []

    for (let i = 11; i >= 0; i--) {
      const date = subMonths(new Date(), i)
      const monthStart = startOfMonth(date)
      const monthEnd = endOfMonth(date)

      const count = await prisma.passenger.count({
        where: {
          reservation: {
            tenant_id: tenantId,
          },
          checked_in: true,
          checked_in_at: {
            gte: monthStart,
            lte: monthEnd,
          },
        },
      })

      passengersByMonth.push({
        month: format(date, 'MMM/yy', { locale: ptBR }),
        count,
      })
    }

    // ═══════════════════════════════════════════════════════════
    // 10. PONTOS DE EMBARQUE (TOP 10)
    // ═══════════════════════════════════════════════════════════

    const topBoardingPoints = await prisma.passenger.groupBy({
      by: ['boarding_location_id'],
      where: {
        reservation: {
          tenant_id: tenantId,
        },
        boarding_location_id: {
          not: null,
        },
      },
      _count: {
        id: true,
      },
      orderBy: {
        _count: {
          id: 'desc',
        },
      },
      take: 10,
    })

    const boardingPointsWithNames = await Promise.all(
      topBoardingPoints.map(async (item) => {
        const location = await prisma.boardingLocation.findUnique({
          where: { id: item.boarding_location_id! },
          select: { name: true, city: true, state: true },
        })

        return {
          location: location
            ? `${location.name} - ${location.city}/${location.state}`
            : 'Desconhecido',
          count: item._count.id,
        }
      })
    )

    // ═══════════════════════════════════════════════════════════
    // RETORNAR TODOS OS GRÁFICOS
    // ═══════════════════════════════════════════════════════════

    return NextResponse.json({
      charts: {
        revenueVsExpenses,
        customerOrigins,
        topDestinations: destinationsWithNames,
        tripCancellationRate,
        reservationCancellationRate,
        avgReservationsPerTrip,
        tripsByMonth,
        reservationsByMonth,
        passengersByMonth,
        topBoardingPoints: boardingPointsWithNames,
      },
    })
  } catch (error) {
    console.error('Erro ao buscar gráficos:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar gráficos' },
      { status: 500 }
    )
  }
}

import { prisma } from '@/lib/prisma'

  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const tenantId = session.user.tenantId

    // ═══════════════════════════════════════════════════════════
    // FUNIL: LEAD → VITRINE → CLIENTE → PAGAMENTO
    // ═══════════════════════════════════════════════════════════

    // 1. LEADS (todos os clientes cadastrados)
    const totalLeads = await prisma.customer.count({
      where: {
        tenant_id: tenantId,
      },
    })

    // 2. VITRINE (clientes que visualizaram viagens)
    const vitrineCustomers = await prisma.customer.count({
      where: {
        tenant_id: tenantId,
        reservations: {
          some: {},
        },
      },
    })

    // 3. CLIENTES (clientes com reservas confirmadas)
    const activeCustomers = await prisma.customer.count({
      where: {
        tenant_id: tenantId,
        reservations: {
          some: {
            status: 'confirmed',
          },
        },
      },
    })

    // 4. PAGAMENTO (clientes que pagaram)
    const paidCustomers = await prisma.customer.count({
      where: {
        tenant_id: tenantId,
        reservations: {
          some: {
            status: 'confirmed',
            revenues: {
              some: {
                status: 'received',
              },
            },
          },
        },
      },
    })

    // ═══════════════════════════════════════════════════════════
    // FUNIL POR CANAL
    // ═══════════════════════════════════════════════════════════

    const channels = await prisma.customer.groupBy({
      by: ['source'],
      where: {
        tenant_id: tenantId,
      },
      _count: {
        id: true,
      },
    })

    const funnelByChannel = await Promise.all(
      channels.map(async (channel) => {
        const source = channel.source || 'Não informado'

        const leads = channel._count.id

        const vitrine = await prisma.customer.count({
          where: {
            tenant_id: tenantId,
            source,
            reservations: {
              some: {},
            },
          },
        })

        const clients = await prisma.customer.count({
          where: {
            tenant_id: tenantId,
            source,
            reservations: {
              some: {
                status: 'confirmed',
              },
            },
          },
        })

        const paid = await prisma.customer.count({
          where: {
            tenant_id: tenantId,
            source,
            reservations: {
              some: {
                status: 'confirmed',
                revenues: {
                  some: {
                    status: 'received',
                  },
                },
              },
            },
          },
        })

        // Calcular ticket médio por canal
        const reservations = await prisma.reservation.findMany({
          where: {
            tenant_id: tenantId,
            customer: {
              source,
            },
            status: 'confirmed',
          },
          select: {
            total_amount: true,
          },
        })

        const avgTicket =
          reservations.length > 0
            ? reservations.reduce(
                (sum, r) => sum + (r.total_amount?.toNumber() || 0),
                0
              ) / reservations.length
            : 0

        return {
          channel: source,
          leads,
          vitrine,
          clients,
          paid,
          avgTicket,
          conversionRate: leads > 0 ? (paid / leads) * 100 : 0,
        }
      })
    )

    // ═══════════════════════════════════════════════════════════
    // RETORNAR FUNIL
    // ═══════════════════════════════════════════════════════════

    return NextResponse.json({
      funnel: {
        total: {
          leads: totalLeads,
          vitrine: vitrineCustomers,
          clients: activeCustomers,
          paid: paidCustomers,
          conversionRate:
            totalLeads > 0 ? (paidCustomers / totalLeads) * 100 : 0,
        },
        byChannel: funnelByChannel,
      },
    })
  } catch (error) {
    console.error('Erro ao buscar funil:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar funil' },
      { status: 500 }
    )
  }
}

  // Calcular saúde do negócio
  const calculateHealth = () => {
    let score = 0
    const factors = []

    // Margem de lucro
    if (kpis.margin >= 30) {
      score += 25
      factors.push({ name: 'Margem de Lucro', status: 'good', value: `${kpis.margin.toFixed(1)}%` })
    } else if (kpis.margin >= 15) {
      score += 15
      factors.push({ name: 'Margem de Lucro', status: 'warning', value: `${kpis.margin.toFixed(1)}%` })
    } else {
      score += 5
      factors.push({ name: 'Margem de Lucro', status: 'bad', value: `${kpis.margin.toFixed(1)}%` })
    }

    // NPS
    if (kpis.nps >= 50) {
      score += 25
      factors.push({ name: 'NPS', status: 'good', value: kpis.nps.toFixed(0) })
    } else if (kpis.nps >= 0) {
      score += 15
      factors.push({ name: 'NPS', status: 'warning', value: kpis.nps.toFixed(0) })
    } else {
      score += 5
      factors.push({ name: 'NPS', status: 'bad', value: kpis.nps.toFixed(0) })
    }

    // Taxa de ocupação média
    const avgOccupancy = (kpis.activeTrips > 0 ? (charts.avgReservationsPerTrip / 44) * 100 : 0)
    if (avgOccupancy >= 70) {
      score += 25
      factors.push({ name: 'Taxa de Ocupação', status: 'good', value: `${avgOccupancy.toFixed(1)}%` })
    } else if (avgOccupancy >= 50) {
      score += 15
      factors.push({ name: 'Taxa de Ocupação', status: 'warning', value: `${avgOccupancy.toFixed(1)}%` })
    } else {
      score += 5
      factors.push({ name: 'Taxa de Ocupação', status: 'bad', value: `${avgOccupancy.toFixed(1)}%` })
    }

    // Taxa de cancelamento
    if (charts.reservationCancellationRate < 15) {
      score += 25
      factors.push({ name: 'Taxa de Cancelamento', status: 'good', value: `${charts.reservationCancellationRate.toFixed(1)}%` })
    } else if (charts.reservationCancellationRate < 25) {
      score += 15
      factors.push({ name: 'Taxa de Cancelamento', status: 'warning', value: `${charts.reservationCancellationRate.toFixed(1)}%` })
    } else {
      score += 5
      factors.push({ name: 'Taxa de Cancelamento', status: 'bad', value: `${charts.reservationCancellationRate.toFixed(1)}%` })
    }

    return { score, factors }
  }

  const health = calculateHealth()

  // Recomendações estratégicas
  const getRecommendations = () => {
    const recommendations = []

    if (kpis.margin < 20) {
      recommendations.push({
        title: 'Melhorar Margem de Lucro',

        priority: 'high',
        icon: TrendingUp,

        description: 'Implemente políticas de cancelamento mais flexíveis e melhore a comunicação.',
        priority: 'medium',
        icon: AlertCircle,
      })

    if (charts.avgReservationsPerTrip < 20) {
      recommendations.push({
        title: 'Aumentar Ocupação das Viagens',
        description: 'Invista em marketing digital e parcerias estratégicas.',
        priority: 'medium',
        icon: Target,
      })

        title: 'Expandir Base de Clientes',
        description: 'Crie campanhas de aquisição e programas de indicação.',
        priority: 'high',
        icon: Zap,
      })
    }

    return recommendations
  }

  const recommendations = getRecommendations()

  return (

      {/* Resumo do Negócio */}
      <Card className="p-6">
        <div className="mb-6">

          <p className="text-sm text-gray-600">Diagnóstico geral da operação</p>
        </div>

        {/* Score de Saúde */}

            <span className="text-sm font-semibold">Saúde do Negócio</span>
            <span className="text-2xl font-bold text-blue-600">{health.score}/100</span>
          </div>

          <p className="text-xs text-gray-500 mt-2">

// app/api/dashboard/trips/upcoming/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const tenantId = session.user.tenantId

    const trips = await prisma.trip.findMany({
      where: {
        tenant_id: tenantId,
        status: 'active',
        departure_date: {
          gte: new Date(),
        },
      },
      include: {
        reservations: {
          where: {
            status: 'confirmed',
          },
        },
      },
      orderBy: {
        departure_date: 'asc',
      },
      take: 10,
    })

    const tripsWithStats = trips.map((trip) => ({
      id: trip.id,
      name: trip.name,
      title: trip.title,
      destination: trip.destination,
      departure_date: trip.departure_date,
      total_capacity: trip.total_capacity,
      reservations_count: trip.reservations.length,
      occupancy_rate:
        trip.total_capacity && trip.total_capacity > 0
          ? Math.round((trip.reservations.length / trip.total_capacity) * 100)
          : 0,
    }))

    return NextResponse.json({ trips: tripsWithStats })
  } catch (error) {
    console.error('Erro ao buscar viagens:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar viagens' },
      { status: 500 }
    )
  }
}

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const tenantId = session.user.tenantId

    const payments = await prisma.revenue.findMany({
      where: {
        tenant_id: tenantId,
        status: 'pending',
      },
      include: {
        customer: {
          select: {
            name: true,
          },
        },
      },
      orderBy: {
        due_date: 'asc',
      },
      take: 10,
    })

    const paymentsFormatted = payments.map((payment) => ({
      id: payment.id,
      description: payment.description || 'Pagamento pendente',
      amount: payment.amount.toNumber(),
      due_date: payment.due_date,
      customer_name: payment.customer?.name || 'Cliente não informado',
      status: payment.status,
    }))

    return NextResponse.json({ payments: paymentsFormatted })
  } catch (error) {
    console.error('Erro ao buscar pagamentos:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar pagamentos' },
      { status: 500 }
    )
  }
}

// app/api/dashboard/export/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import PDFDocument from 'pdfkit'
import { format } from 'date-fns'
import { ptBR } from 'date-fns/locale'

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const body = await req.json()
    const { dateRange } = body

    // Buscar dados do dashboard
    const kpisResponse = await fetch(
      `${process.env.NEXTAUTH_URL}/api/dashboard/metrics`,
      {
        headers: {
          cookie: req.headers.get('cookie') || '',
        },
      }
    )
    const kpisData = await kpisResponse.json()

    // Gerar PDF
    const doc = new PDFDocument({ size: 'A4', margin: 50 })
    const chunks: Buffer[] = []

    doc.on('data', (chunk) => chunks.push(chunk))

    // Título
    doc
      .fontSize(24)
      .font('Helvetica-Bold')
      .text('Relatório do Dashboard', { align: 'center' })

    doc.moveDown()

    doc
      .fontSize(12)
      .font('Helvetica')
      .text(
        `Período: ${dateRange === 'all' ? 'Todo período' : `Últimos ${dateRange} dias`}`,
        { align: 'center' }
      )

    doc
      .text(`Gerado em: ${format(new Date(), "dd/MM/yyyy 'às' HH:mm", { locale: ptBR })}`, {
        align: 'center',
      })

    doc.moveDown(2)

    // KPIs
    doc.fontSize(16).font('Helvetica-Bold').text('Indicadores Principais')

    doc.moveDown()

    const kpis = [
      ['Faturamento Total', `R$ ${kpisData.kpis.totalRevenue.toFixed(2)}`],
      ['Total de Clientes', kpisData.kpis.totalCustomers],
      ['Lucro Líquido', `R$ ${kpisData.kpis.netProfit.toFixed(2)}`],
      ['Margem de Lucro', `${kpisData.kpis.margin.toFixed(2)}%`],
      ['NPS Score', kpisData.kpis.nps.toFixed(0)],
    ]

    doc.fontSize(12).font('Helvetica')

    kpis.forEach(([label, value]) => {
      doc.text(`${label}: ${value}`)
      doc.moveDown(0.5)
    })

    doc.end()

    const pdfBuffer = await new Promise<Buffer>((resolve) => {
      doc.on('end', () => resolve(Buffer.concat(chunks)))
    })

    return new NextResponse(pdfBuffer, {
      headers: {
        'Content-Type': 'application/pdf',
        'Content-Disposition': `attachment; filename="relatorio-${format(new Date(), 'yyyy-MM-dd')}.pdf"`,
      },
    })
  } catch (error) {
    console.error('Erro ao exportar relatório:', error)
    return NextResponse.json(
      { error: 'Erro ao exportar relatório' },
      { status: 500 }
    )
  }
}
