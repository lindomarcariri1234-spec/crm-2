ATUALIZAR O SISTEMA DE LAYOUT DE ASSENTOS - IMPLEMENTAR

📋 VISÃO GERAL DO SISTEMA
Objetivo:
Implementar um sistema completo de layout de assentos que suporte:

Ônibus de Repartição Única - Layout simples (Convencional, Executivo, Semi-Leito)
Ônibus Double Decker - Layout com 2 andares (Piso Superior + Piso Inferior)
Comportamento:

Criação: Layouts separados por andar (quando Double Decker)
Apresentação: Visualização unificada com tabs (Superior/Inferior)
Disponível em: Sistema interno (agência/vendedor) + Loja pública (vitrine)

Mantenha os Layouts já criado no sistema: Micro-ônibus (2x1), Van 15 (1+2) e Ônibus 46 (2x2)

SCHEMA PRISMA

// prisma/schema.prisma

enum BusType {
  CONVENTIONAL    // Convencional (2x2 - 50 lugares)
  EXECUTIVE       // Executivo (2x2 - 46 lugares)
  SEMI_LEITO      // Semi-Leito (2x2 amplo - 44 lugares)
  LEITO           // Leito (2x1 - 28 lugares)
  DOUBLE_DECKER   // Double Decker (2 andares - 55/56 lugares)
}

model BusModel {
  id              String   @id @default(cuid())
  tenant_id       String
  
  name            String   // Ex: "Ônibus Executivo Turismo"
  bus_type        BusType
  total_seats     Int
  
  // Configuração de andares (apenas para Double Decker)
  has_upper_deck  Boolean  @default(false)
  has_lower_deck  Boolean  @default(false)
  upper_deck_seats Int?    // Ex: 46 assentos
  lower_deck_seats Int?    // Ex: 10 assentos
  
  // Layout visual (JSON com matriz de assentos)
  config          Json     @db.JsonB
  
  is_active       Boolean  @default(true)
  created_at      DateTime @default(now())
  updated_at      DateTime @updatedAt

  tenant Tenant @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  trips  Trip[]

  @@index([tenant_id])
  @@index([bus_type])
  @@map("bus_models")
}

model SeatLayout {
  id              String   @id @default(cuid())
  trip_id         String
  bus_model_id    String?
  
  // Identificação do assento
  deck            String   // 'upper' (andar de cima), 'lower' (andar de baixo), 'single' (único)
  seat_number     String   // Número do assento (1, 2, 3...)
  row             Int      // Linha no grid
  column          Int      // Coluna no grid
  
  // Tipo e características
  seat_type       String   @default("standard") // 'standard', 'premium', 'table', 'accessible'
  is_available    Boolean  @default(true)
  is_blocked      Boolean  @default(false)
  
  // Modificador de preço (ex: +10% para VIP, +5% para mesa)
  price_modifier  Decimal? @db.Decimal(5, 2)
  
  created_at      DateTime @default(now())
  updated_at      DateTime @updatedAt

  trip            Trip     @relation(fields: [trip_id], references: [id], onDelete: Cascade)
  reservations    SeatReservation[]

  @@unique([trip_id, deck, seat_number])
  @@index([trip_id])
  @@index([deck])
  @@map("seat_layouts")
}

model SeatReservation {
  id              String   @id @default(cuid())
  reservation_id  String
  seat_layout_id  String
  
  passenger_name  String
  passenger_cpf   String?
  passenger_age   Int?
  
  created_at      DateTime @default(now())

  reservation Reservation @relation(fields: [reservation_id], references: [id], onDelete: Cascade)
  seat_layout SeatLayout  @relation(fields: [seat_layout_id], references: [id], onDelete: Cascade)

  @@index([reservation_id])
  @@index([seat_layout_id])
  @@map("seat_reservations")
}

API - GERAÇÃO AUTOMÁTICA DE LAYOUTS
// app/api/trips/[tripId]/generate-seats/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function POST(
  req: Request,
  { params }: { params: { tripId: string } }
) {
  try {
    const session = await getServerSession(authOptions)

    if (!session || session.user.role !== 'admin') {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    const { busModelId } = await req.json()

    const trip = await prisma.trip.findUnique({
      where: { id: params.tripId },
    })

    if (!trip) {
      return NextResponse.json({ error: 'Viagem não encontrada' }, { status: 404 })
    }

    const busModel = await prisma.busModel.findUnique({
      where: { id: busModelId },
    })

    if (!busModel) {
      return NextResponse.json({ error: 'Modelo não encontrado' }, { status: 404 })
    }

    // Deletar layouts existentes
    await prisma.seatLayout.deleteMany({
      where: { trip_id: params.tripId },
    })

    const config = busModel.config as any
    const seatsToCreate: any[] = []

    // ═══════════════════════════════════════════════════════════
    // DOUBLE DECKER (2 ANDARES)
    // ═══════════════════════════════════════════════════════════
    if (busModel.bus_type === 'DOUBLE_DECKER') {
      // PISO SUPERIOR
      if (config.upperDeck) {
        const upperLayout = config.upperDeck.layout

        upperLayout.forEach((row: string[], rowIndex: number) => {
          row.forEach((cell: string, colIndex: number) => {
            if (cell && cell !== 'X' && cell !== 'AISLE') {
              const seatNumber = cell.replace(/[PS]/g, '')
              const seatType = cell.startsWith('P')
                ? 'premium'
                : cell.startsWith('S')
                ? 'table'
                : 'standard'

              seatsToCreate.push({
                trip_id: params.tripId,
                bus_model_id: busModelId,
                deck: 'upper',
                seat_number: seatNumber,
                row: rowIndex,
                column: colIndex,
                seat_type: seatType,
                is_available: true,
                is_blocked: false,
                price_modifier: seatType === 'premium' ? 10 : seatType === 'table' ? 5 : 0,
              })
            }
          })
        })
      }

      // PISO INFERIOR
      if (config.lowerDeck) {
        const lowerLayout = config.lowerDeck.layout

        lowerLayout.forEach((row: string[], rowIndex: number) => {
          row.forEach((cell: string, colIndex: number) => {
            if (cell && cell !== 'X' && cell !== 'AISLE') {
              const seatNumber = cell.replace(/[PS]/g, '')
              const seatType = cell.startsWith('S') ? 'table' : 'standard'

              seatsToCreate.push({
                trip_id: params.tripId,
                bus_model_id: busModelId,
                deck: 'lower',
                seat_number: seatNumber,
                row: rowIndex,
                column: colIndex,
                seat_type: seatType,
                is_available: true,
                is_blocked: false,
                price_modifier: seatType === 'table' ? 5 : 0,
              })
            }
          })
        })
      }
    } else {
      // ═══════════════════════════════════════════════════════════
      // REPARTIÇÃO ÚNICA
      // ═══════════════════════════════════════════════════════════
      const layout = config.singleDeck?.layout

      if (layout) {
        layout.forEach((row: string[], rowIndex: number) => {
          row.forEach((cell: string, colIndex: number) => {
            if (cell && cell !== 'X' && cell !== 'AISLE') {
              const seatNumber = cell.replace(/[P]/g, '')
              const seatType = cell.startsWith('P') ? 'premium' : 'standard'

              seatsToCreate.push({
                trip_id: params.tripId,
                bus_model_id: busModelId,
                deck: 'single',
                seat_number: seatNumber,
                row: rowIndex,
                column: colIndex,
                seat_type: seatType,
                is_available: true,
                is_blocked: false,
                price_modifier: seatType === 'premium' ? 10 : 0,
              })
            }
          })
        })
      }
    }

    // Criar todos os assentos
    await prisma.seatLayout.createMany({
      data: seatsToCreate,
    })

    // Atualizar viagem
    await prisma.trip.update({
      where: { id: params.tripId },
      data: {
        bus_model_id: busModelId,
        total_seats: seatsToCreate.length,
        available_seats: seatsToCreate.length,
      },
    })

    return NextResponse.json({
      success: true,
      seatsCreated: seatsToCreate.length,
      breakdown: {
        upperDeck: seatsToCreate.filter((s) => s.deck === 'upper').length,
        lowerDeck: seatsToCreate.filter((s) => s.deck === 'lower').length,
        singleDeck: seatsToCreate.filter((s) => s.deck === 'single').length,
      },
    })
  } catch (error) {
    console.error('Erro ao gerar layout:', error)
    return NextResponse.json(
      { error: 'Erro ao gerar layout' },
      { status: 500 }
    )
  }
}
