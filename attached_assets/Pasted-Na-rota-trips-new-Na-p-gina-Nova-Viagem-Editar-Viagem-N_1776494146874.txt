// prisma/schema.prisma

model Trip {
  id              String    @id @default(cuid())
  tenant_id       String
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  
  // ═══════════════════════════════════════════════════════════
  // DADOS BÁSICOS
  // ═══════════════════════════════════════════════════════════
  
  // Nome da viagem
  name            String    // Nome da viagem (ex: "Excursão Nordeste 2026")
  title           String    // Título/Destino (ex: "Nordeste Brasileiro")
  description     String?   @db.Text
  
  // ORIGEM
  origin_city     String?   // Cidade de origem
  origin_state    String?   // Estado (UF) de origem
  
  // DESTINO
  destination     String    // Cidade de destino
  destination_state String? // Estado (UF) de destino
  
  // Imagens
  image_url       String?   @db.Text
  image_key       String?
  gallery_images  Json?     @default("[]")
  
  // ═══════════════════════════════════════════════════════════
  // DATAS E HORÁRIOS
  // ═══════════════════════════════════════════════════════════
  
  departure_date  DateTime? // Data e hora de saída
  return_date     DateTime? // Data e hora de retorno
  duration_days   Int?
  
  // ═══════════════════════════════════════════════════════════
  // VEÍCULO E CAPACIDADE
  // ═══════════════════════════════════════════════════════════
  
  seat_layout     String?   // Layout de assentos (ex: "2x2", "2x1", "executivo")
  total_capacity  Int?      // Capacidade total de assentos (autopreenchido)
  
  // PREÇOS
  price_adult     Decimal   @db.Decimal(10, 2)
  price_child     Decimal?  @db.Decimal(10, 2)
  price_senior    Decimal?  @db.Decimal(10, 2)
  price_launch    Decimal?  @db.Decimal(10, 2)
  price_early     Decimal?  @db.Decimal(10, 2)
  price_standard  Decimal?  @db.Decimal(10, 2)
  price_last      Decimal?  @db.Decimal(10, 2)
  
  // CAPACIDADE
  max_passengers  Int?
  min_passengers  Int?      @default(4)
  
  // CUSTOS
  fixed_costs     Json?     @default("[]")
  variable_costs  Json?     @default("[]")
  discounts       Json?     @default("[]")
  taxes           Json?     @default("[]")
  financial_fees  Json?     @default("[]")
  
  // TOTAIS CALCULADOS
  total_fixed_cost      Decimal?  @db.Decimal(10, 2)
  total_variable_cost   Decimal?  @db.Decimal(10, 2)
  total_cost            Decimal?  @db.Decimal(10, 2)
  total_revenue         Decimal?  @db.Decimal(10, 2)
  total_profit          Decimal?  @db.Decimal(10, 2)
  profit_margin         Decimal?  @db.Decimal(5, 2)
  
  // TRANSPORTE
  vehicle_type    String?
  vehicle_plate   String?
  driver_name     String?
  tour_guide      String?
  trip_organizer  String?
  
  // STATUS
  status          String    @default("active")
  
  // AUDITORIA
  created_by      String
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  // RELACIONAMENTOS
  reservations    Reservation[]
  expenses        Expense[]
  revenues        Revenue[]
  commissions     Commission[]
  
  @@index([tenant_id])
  @@index([tenant_id, status])
  @@map("trips")
}

export default function NewTripPage() {
  const router = useRouter()
  const [isSaving, setIsSaving] = useState(false)
  const [activeTab, setActiveTab] = useState('basico')

  const [formData, setFormData] = useState({
    // NOME E DESCRIÇÃO
    name: '',
    title: '',
    description: '',
    
    // ORIGEM
    origin_city: '',
    origin_state: '',
    
    // DESTINO
    destination: '',
    destination_state: '',
    
    // IMAGENS
    image_url: '',
    image_key: '',
    gallery_images: [] as GalleryImage[],
    
    // DATAS E HORÁRIOS
    departure_date: '',
    departure_time: '',
    return_date: '',
    return_time: '',
    duration_days: '',
    
    // VEÍCULO E CAPACIDADE
    seat_layout: '',
    total_capacity: '',
    
    // PREÇOS
    price_adult: '',
    price_child: '',
    price_senior: '',
    price_launch: '',
    price_early: '',
    price_standard: '',
    price_last: '',
    
    // CAPACIDADE
    min_passengers: '10',
    max_passengers: '45',
    
    // CUSTOS
    fixed_costs: [] as Array<{
      id: string
      category: string
      description: string
      value: number
    }>,
    variable_costs: [] as Array<{
      id: string
      category: string
      description: string
      value_per_pax: number
    }>,
    discounts: [] as Array<{
      id: string
      type: string
      description: string
      value: number
    }>,
    taxes: [] as Array<{
      id: string
      name: string
      percentage: number
      base: string
    }>,
    financial_fees: [] as Array<{
      id: string
      type: string
      description: string
      percentage: number
    }>,
    
    // TRANSPORTE
    vehicle_type: '',
    vehicle_plate: '',
    driver_name: '',
    tour_guide: '',
    trip_organizer: '',
  })

  // Autopreenchimento da capacidade baseado no layout
  useEffect(() => {
    if (formData.seat_layout && formData.seat_layout !== 'custom') {
      const layout = SEAT_LAYOUTS.find((l) => l.value === formData.seat_layout)
      if (layout) {
        setFormData((prev) => ({
          ...prev,
          total_capacity: layout.capacity.toString(),
          max_passengers: layout.capacity.toString(),
        }))
      }
    }
  }, [formData.seat_layout])

  const validateForm = () => {
    if (!formData.name) {
      toast.error('Nome da viagem é obrigatório')
      setActiveTab('basico')
      return false
    }

    if (!formData.destination) {
      toast.error('Cidade de destino é obrigatória')
      setActiveTab('basico')
      return false
    }

    if (!formData.departure_date) {
      toast.error('Data de saída é obrigatória')
      setActiveTab('basico')
      return false
    }

    if (!formData.total_capacity) {
      toast.error('Capacidade total é obrigatória')
      setActiveTab('basico')
      return false
    }

    if (!formData.image_url) {
      toast.error('Imagem de capa é obrigatória')
      setActiveTab('midia')
      return false

    if (!formData.price_adult) {
      toast.error('Preço adulto é obrigatório')
      setActiveTab('precos')
      return false
    }

    return true

  const handleSubmit = async () => {
    if (!validateForm()) {
      return

    try {
      setIsSaving(true)

      // Combinar data e hora
      const departureDateTime =
        formData.departure_date && formData.departure_time
          ? `${formData.departure_date}T${formData.departure_time}`
          : formData.departure_date

      const returnDateTime =
        formData.return_date && formData.return_time
          ? `${formData.return_date}T${formData.return_time}`
          : formData.return_date

      const dataToSend = {
        ...formData,
        departure_date: departureDateTime,
        return_date: returnDateTime,
      }

      console.log('📤 Enviando dados:', dataToSend)

      const response = await fetch('/api/dashboard/trips', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(dataToSend),
      })

      const data = await response.json()

      if (response.ok) {
        console.log('✅ Viagem criada:', data.trip)
        toast.success('Viagem criada com sucesso!')
        router.push(`/dashboard/trips/${data.trip.id}`)
      } else {
        console.error('❌ Erro:', data)
        toast.error(data.error || 'Erro ao criar viagem')
      }
    } catch (error) {
      console.error('❌ Erro ao criar viagem:', error)
      toast.error('Erro ao criar viagem')
    } finally {
      setIsSaving(false)
    }

        {/* ═══════════════════════════════════════════════════════════ */}

        <TabsContent value="basico" className="space-y-6">
          <Card className="p-6">
            <div className="space-y-6">

              {/* Nome da Viagem */}

                </Label>
                <Input
                  id="name"
                  value={formData.name}
                  onChange={(e) =>
                    setFormData({ ...formData, name: e.target.value })
                  }
                  placeholder="Ex: Excursão Nordeste 2026"
                  required
                  disabled={isSaving}
                  className="mt-2"
                />
                <p className="text-xs text-gray-500 mt-1">
                  Nome interno da viagem para identificação
                </p>
              </div>

              {/* Descrição */}
              <div>
                <Label htmlFor="description">Descrição</Label>

                  id="description"
                  value={formData.description}
                  onChange={(e) =>
                    setFormData({ ...formData, description: e.target.value })
                  }
                  placeholder="Descreva a viagem, principais atrações, o que está incluído..."
                  rows={5}
                  disabled={isSaving}
                  className="mt-2"
                />
              </div>
            </div>
          </Card>

          <Card className="p-6">
            <div className="space-y-6">
              <h3 className="text-lg font-bold mb-4">📍 Origem e Destino</h3>

                      value={formData.origin_state}

                        setFormData({ ...formData, origin_state: value })

              {formData.total_capacity && (

            </div>
          </Card>

// app/api/dashboard/trips/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { hasPermission } from '@/lib/permissions/check-permissions'

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    if (!hasPermission(session.user.role as any, 'trips.create')) {
      return NextResponse.json(
        { error: 'Sem permissão para criar viagens' },
        { status: 403 }
      )
    }

    const body = await req.json()

    console.log('📥 Dados recebidos:', body)

    // Validações
    if (!body.name) {
      return NextResponse.json(
        { error: 'Nome da viagem é obrigatório' },
        { status: 400 }
      )
    }

    if (!body.destination) {
      return NextResponse.json(
        { error: 'Cidade de destino é obrigatória' },
        { status: 400 }
      )
    }

    if (!body.departure_date) {
      return NextResponse.json(
        { error: 'Data de saída é obrigatória' },
        { status: 400 }
      )
    }

    if (!body.total_capacity) {
      return NextResponse.json(
        { error: 'Capacidade total é obrigatória' },
        { status: 400 }
      )
    }

    if (!body.image_url) {
      return NextResponse.json(
        { error: 'Imagem de capa é obrigatória' },
        { status: 400 }
      )
    }

    if (!body.price_adult) {
      return NextResponse.json(
        { error: 'Preço adulto é obrigatório' },
        { status: 400 }
      )
    }

    // Calcular totais financeiros
    const pricePerPax = parseFloat(body.price_adult)
    const totalPax = parseInt(body.max_passengers) || parseInt(body.total_capacity) || 0

    const totalFixedCost = (body.fixed_costs || []).reduce(
      (sum: number, cost: any) => sum + cost.value,
      0
    )

    const totalVariableCost = (body.variable_costs || []).reduce(
      (sum: number, cost: any) => sum + cost.value_per_pax * totalPax,
      0
    )

    const totalCost = totalFixedCost + totalVariableCost
    const totalRevenue = pricePerPax * totalPax
    const totalProfit = totalRevenue - totalCost
    const profitMargin = totalRevenue > 0 ? (totalProfit / totalRevenue) * 100 : 0

    // Criar viagem
    const trip = await prisma.trip.create({
      data: {
        tenant_id: session.user.tenantId,
        created_by: session.user.id,
        
        // DADOS BÁSICOS
        name: body.name,
        title: body.title || body.name,
        description: body.description || null,
        
        // ORIGEM
        origin_city: body.origin_city || null,
        origin_state: body.origin_state || null,
        
        // DESTINO
        destination: body.destination,
        destination_state: body.destination_state || null,
        
        // IMAGENS
        image_url: body.image_url,
        image_key: body.image_key || null,
        gallery_images: body.gallery_images || [],
        
        // DATAS E HORÁRIOS
        departure_date: body.departure_date
          ? new Date(body.departure_date)
          : null,
        return_date: body.return_date ? new Date(body.return_date) : null,
        duration_days: body.duration_days
          ? parseInt(body.duration_days)
          : null,
        
        // VEÍCULO E CAPACIDADE
        seat_layout: body.seat_layout || null,
        total_capacity: body.total_capacity
          ? parseInt(body.total_capacity)
          : null,
        
        // PREÇOS
        price_adult: parseFloat(body.price_adult),
        price_child: body.price_child ? parseFloat(body.price_child) : null,
        price_senior: body.price_senior ? parseFloat(body.price_senior) : null,
        price_launch: body.price_launch ? parseFloat(body.price_launch) : null,
        price_early: body.price_early ? parseFloat(body.price_early) : null,
        price_standard: body.price_standard
          ? parseFloat(body.price_standard)
          : null,
        price_last: body.price_last ? parseFloat(body.price_last) : null,
        
        // CAPACIDADE
        min_passengers: body.min_passengers
          ? parseInt(body.min_passengers)
          : null,
        max_passengers: body.max_passengers
          ? parseInt(body.max_passengers)
          : parseInt(body.total_capacity) || null,
        
        // CUSTOS
        fixed_costs: body.fixed_costs || [],
        variable_costs: body.variable_costs || [],
        discounts: body.discounts || [],
        taxes: body.taxes || [],
        financial_fees: body.financial_fees || [],
        
        // TOTAIS CALCULADOS
        total_fixed_cost: totalFixedCost,
        total_variable_cost: totalVariableCost,
        total_cost: totalCost,
        total_revenue: totalRevenue,
        total_profit: totalProfit,
        profit_margin: profitMargin,
        
        // TRANSPORTE
        vehicle_type: body.vehicle_type || null,
        vehicle_plate: body.vehicle_plate || null,
        driver_name: body.driver_name || null,
        tour_guide: body.tour_guide || null,
        trip_organizer: body.trip_organizer || null,
        
        // STATUS
        status: 'active',
      },
    })

    console.log('✅ Viagem criada:', trip)

    return NextResponse.json({
      success: true,
      trip,
      message: 'Viagem criada com sucesso!',
      financial: {
        totalCost,
        totalRevenue,
        totalProfit,
        profitMargin,
      },
    })
  } catch (error: any) {
    console.error('❌ Erro ao criar viagem:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao criar viagem' },
      { status: 500 }
    )
  }
}
