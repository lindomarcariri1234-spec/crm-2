  birthday_messages BirthdayMessage[]

  @@map("clients")

// ═══════════════════════════════════════════════════════════════
// TABELA DE MENSAGENS DE ANIVERSÁRIO
// ═══════════════════════════════════════════════════════════════

model BirthdayMessage {
  id              String    @id @default(cuid())
  tenant_id       String
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  
  client_id       String
  client          Client    @relation(fields: [client_id], references: [id], onDelete: Cascade)
  
  // DATA DO ANIVERSÁRIO
  birthday_year   Int       // 2026
  birthday_date   DateTime  // Data completa do aniversário
  
  // CANAIS DE ENVIO
  sent_whatsapp   Boolean   @default(false)
  sent_email      Boolean   @default(false)
  sent_sms        Boolean   @default(false)
  
  // TIMESTAMPS DE ENVIO
  whatsapp_sent_at DateTime?
  email_sent_at    DateTime?
  sms_sent_at      DateTime?
  
  // STATUS DE ENTREGA
  whatsapp_delivered Boolean @default(false)
  email_opened       Boolean @default(false)
  sms_delivered      Boolean @default(false)
  
  // CUPOM GERADO
  coupon_id       String?
  coupon          Coupon?   @relation(fields: [coupon_id], references: [id])
  
  // CONVERSÃO
  converted       Boolean   @default(false) // Cliente fez reserva
  reservation_id  String?
  reservation     Reservation? @relation(fields: [reservation_id], references: [id])
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  @@unique([tenant_id, client_id, birthday_year])
  @@index([tenant_id, birthday_date])
  @@index([client_id])
  @@map("birthday_messages")
}

// ═══════════════════════════════════════════════════════════════
// TABELA DE CUPONS (atualizada)
// ═══════════════════════════════════════════════════════════════

model Coupon {
  id              String    @id @default(cuid())
  tenant_id       String
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  
  code            String    @unique // "ANIVER-LINDOMAR-2026"
  type            String    // percentage, fixed, birthday
  
  // DESCONTO
  discount_type   String    // percentage, fixed
  discount_value  Decimal   // 15 (15%) ou 50 (R$ 50)
  
  // VALIDADE
  valid_from      DateTime
  valid_until     DateTime
  
  // RESTRIÇÕES
  min_amount      Decimal?  // Valor mínimo da compra
  max_uses        Int       @default(1) // Quantas vezes pode usar
  used_count      Int       @default(0)
  
  // CLIENTE ESPECÍFICO (para cupons de aniversário)
  client_id       String?
  client          Client?   @relation(fields: [client_id], references: [id])
  
  // TIPO ESPECIAL
  is_birthday     Boolean   @default(false)
  
  status          String    @default("active") // active, expired, used
  
  // RELACIONAMENTOS
  birthday_messages BirthdayMessage[]
  reservations    Reservation[]
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  @@index([tenant_id])
  @@index([client_id])
  @@index([code])
  @@map("coupons")
}

// app/api/cron/birthday-messages/route.ts

import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { sendBirthdayMessages } from '@/lib/birthday/send-messages'

/**
 * Cron Job que roda todos os dias às 00:00
 * Vercel Cron: https://vercel.com/docs/cron-jobs
 */
export async function GET(req: Request) {
  try {
    // Verificar autenticação do cron (segurança)
    const authHeader = req.headers.get('authorization')
    if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    console.log('🎂 Iniciando envio de mensagens de aniversário...')

    // Buscar todos os tenants ativos
    const tenants = await prisma.tenant.findMany({
      where: {
        status: 'active',
        birthday_messages_enabled: true, // Feature flag
      },
    })

    let totalSent = 0
    const results = []

    for (const tenant of tenants) {
      try {
        const result = await sendBirthdayMessages(tenant.id)
        totalSent += result.sent
        results.push({
          tenant_id: tenant.id,
          tenant_name: tenant.name,
          ...result,
        })
      } catch (error) {
        console.error(`Erro ao processar tenant ${tenant.id}:`, error)
        results.push({
          tenant_id: tenant.id,
          tenant_name: tenant.name,
          error: error instanceof Error ? error.message : 'Erro desconhecido',
        })
      }
    }

    console.log(`✅ Processo concluído. Total de mensagens enviadas: ${totalSent}`)

    return NextResponse.json({
      success: true,
      total_sent: totalSent,
      results,
    })
  } catch (error) {
    console.error('Erro no cron job de aniversários:', error)
    return NextResponse.json(
      { error: 'Erro ao processar aniversários' },
      { status: 500 }
    )
  }
}

// lib/birthday/send-messages.ts

import { prisma } from '@/lib/prisma'
import { startOfDay, endOfDay } from 'date-fns'
import { sendWhatsAppMessage } from '@/lib/whatsapp/send-message'
import { sendEmail } from '@/lib/email/send-email'
import { sendSMS } from '@/lib/sms/send-sms'
import { generateBirthdayCoupon } from './generate-coupon'
import { renderTemplate } from './render-template'

export async function sendBirthdayMessages(tenantId: string) {
  const today = new Date()
  const currentYear = today.getFullYear()
  
  // Buscar clientes que fazem aniversário hoje
  const birthdayClients = await prisma.client.findMany({
    where: {
      tenant_id: tenantId,
      status: 'active',
      // Comparar apenas mês e dia (ignorar ano)
      AND: [
        {
          birthdate: {
            gte: startOfDay(new Date(currentYear, today.getMonth(), today.getDate())),
            lte: endOfDay(new Date(currentYear, today.getMonth(), today.getDate())),
          },
        },
      ],
    },
    include: {
      birthday_messages: {
        where: {
          birthday_year: currentYear,
        },
      },
    },
  })

  console.log(`🎂 Encontrados ${birthdayClients.length} aniversariantes para ${tenantId}`)

  let sent = 0
  let errors = 0

  for (const client of birthdayClients) {
    try {
      // Verificar se já enviou este ano
      if (client.birthday_messages.length > 0) {
        console.log(`⏭️ Já enviado para ${client.name} este ano`)
        continue
      }

      // ═══════════════════════════════════════════════════════════
      // GERAR CUPOM DE ANIVERSÁRIO
      // ═══════════════════════════════════════════════════════════
      
      const coupon = await generateBirthdayCoupon({
        tenantId,
        clientId: client.id,
        clientName: client.name,
        year: currentYear,
      })

      console.log(`🎟️ Cupom gerado: ${coupon.code}`)

      // ═══════════════════════════════════════════════════════════
      // CRIAR REGISTRO DE MENSAGEM
      // ═══════════════════════════════════════════════════════════
      
      const birthdayMessage = await prisma.birthdayMessage.create({
        data: {
          tenant_id: tenantId,
          client_id: client.id,
          birthday_year: currentYear,
          birthday_date: today,
          coupon_id: coupon.id,
        },
      })

      // ═══════════════════════════════════════════════════════════
      // ENVIAR WHATSAPP (PRIORIDADE 1)
      // ═══════════════════════════════════════════════════════════
      
      if (client.whatsapp_opt_in && client.phone) {
        try {
          const whatsappMessage = await renderTemplate({
            tenantId,
            type: 'birthday',
            channel: 'whatsapp',
            variables: {
              name: client.name.split(' ')[0], // Primeiro nome
              coupon_code: coupon.code,
              discount: '15%',
              valid_until: '7 dias',
            },
          })

          await sendWhatsAppMessage({
            to: client.phone,
            message: whatsappMessage,
          })

          await prisma.birthdayMessage.update({
            where: { id: birthdayMessage.id },
            data: {
              sent_whatsapp: true,
              whatsapp_sent_at: new Date(),
            },
          })

          console.log(`✅ WhatsApp enviado para ${client.name}`)
        } catch (error) {
          console.error(`❌ Erro ao enviar WhatsApp para ${client.name}:`, error)
        }
      }

      // ═══════════════════════════════════════════════════════════
      // ENVIAR EMAIL (PRIORIDADE 2)
      // ═══════════════════════════════════════════════════════════
      
      if (client.email_opt_in && client.email) {
        try {
          await sendEmail({
            to: client.email,
            subject: `🎉 Feliz Aniversário, ${client.name.split(' ')[0]}!`,
            template: 'birthday',
            variables: {
              name: client.name,
              coupon_code: coupon.code,
              discount: '15%',
              valid_until: '7 dias',
            },
          })

          await prisma.birthdayMessage.update({
            where: { id: birthdayMessage.id },
            data: {
              sent_email: true,
              email_sent_at: new Date(),
            },
          })

          console.log(`✅ Email enviado para ${client.name}`)
        } catch (error) {
          console.error(`❌ Erro ao enviar email para ${client.name}:`, error)
        }
      }

      // ═══════════════════════════════════════════════════════════
      // ENVIAR SMS (OPCIONAL)
      // ═══════════════════════════════════════════════════════════
      
      if (client.sms_opt_in && client.phone) {
        try {
          const smsMessage = `🎂 Feliz aniversário! Ganhe 15% OFF com o cupom ${coupon.code}. Válido por 7 dias!`

          await sendSMS({
            to: client.phone,
            message: smsMessage,
          })

          await prisma.birthdayMessage.update({
            where: { id: birthdayMessage.id },
            data: {
              sent_sms: true,
              sms_sent_at: new Date(),
            },
          })

          console.log(`✅ SMS enviado para ${client.name}`)
        } catch (error) {
          console.error(`❌ Erro ao enviar SMS para ${client.name}:`, error)
        }
      }

      sent++
    } catch (error) {
      console.error(`❌ Erro ao processar ${client.name}:`, error)
      errors++
    }
  }

  return {
    total: birthdayClients.length,
    sent,
    errors,
    skipped: birthdayClients.length - sent - errors,
  }
}

GERAÇÃO DE CUPOM DE ANIVERSÁRIO

// lib/birthday/generate-coupon.ts

import { prisma } from '@/lib/prisma'
import { addDays } from 'date-fns'

interface GenerateBirthdayCouponParams {
  tenantId: string
  clientId: string
  clientName: string
  year: number
}

export async function generateBirthdayCoupon({
  tenantId,
  clientId,
  clientName,
  year,
}: GenerateBirthdayCouponParams) {
  // Gerar código único
  const firstName = clientName.split(' ')[0].toUpperCase()
  const code = `ANIVER-${firstName}-${year}`

  // Verificar se já existe
  const existing = await prisma.coupon.findUnique({
    where: { code },
  })

  if (existing) {
    return existing
  }

  // Criar cupom
  const coupon = await prisma.coupon.create({
    data: {
      tenant_id: tenantId,
      client_id: clientId,
      code,
      type: 'birthday',
      discount_type: 'percentage',
      discount_value: 15, // 15% de desconto
      valid_from: new Date(),
      valid_until: addDays(new Date(), 7), // Válido por 7 dias
      max_uses: 1, // Uso único
      is_birthday: true,
      status: 'active',
    },
  })

  return coupon
}

// lib/birthday/render-template.ts

import { prisma } from '@/lib/prisma'

interface RenderTemplateParams {
  tenantId: string
  type: string // 'birthday'
  channel: string // 'whatsapp', 'email', 'sms'
  variables: Record<string, string>
}

export async function renderTemplate({
  tenantId,
  type,
  channel,
  variables,
}: RenderTemplateParams) {
  // Buscar template customizado
  let template = await prisma.messageTemplate.findFirst({
    where: {
      tenant_id: tenantId,
      type,
      channel,
      is_active: true,
    },
    orderBy: {
      is_default: 'desc',
    },
  })

  // Se não encontrar, usar template padrão
  if (!template) {
    template = getDefaultTemplate(type, channel)
  }

  // Substituir variáveis
  let message = template.body

  for (const [key, value] of Object.entries(variables)) {
    message = message.replace(new RegExp(`{{${key}}}`, 'g'), value)
  }

  return message
}

async function getMonthStats(tenantId: string, year: number) {
  const messages = await prisma.birthdayMessage.findMany({
    where: {
      tenant_id: tenantId,
      birthday_year: year,
    },
  })

  const converted = messages.filter((m) => m.converted).length
  const conversionRate = messages.length > 0
    ? Math.round((converted / messages.length) * 100)
    : 0

  return {
    total: messages.length,
    converted,
    conversionRate,
  }
}
