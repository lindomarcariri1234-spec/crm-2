🛒 MODELO DE LOJA REFORMULADO - MULTI-TENANT
📋 VISÃO GERAL
Cada Agência (Tenant) terá sua própria loja online independente com:

✅ URL Única: minhaagencia.visitecrm.com.br ou domínio próprio
✅ Catálogo Próprio: Viagens, pacotes, passeios, seguros, etc
✅ Identidade Visual: Logo, cores, banner personalizados
✅ Pagamentos Próprios: Contas Stripe/Mercado Pago da agência
✅ Integração Hurb: Opcional por agência
✅ Multi-Produtos: Não só viagens, mas qualquer serviço turístico

ARQUITETURA REFORMULADA

// ═══════════════════════════════════════════════════════════════
// CONFIGURAÇÕES DA LOJA (1 POR TENANT)
// ═══════════════════════════════════════════════════════════════

model Store {
  id              String    @id @default(cuid())
  tenant_id       String    @unique
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  
  // ─────────────────────────────────────────────────────────────
  // IDENTIDADE VISUAL
  // ─────────────────────────────────────────────────────────────
  name            String                    // Nome da loja
  slug            String    @unique         // URL: slug.visitecrm.com.br
  tagline         String?                   // Slogan
  description     String?   @db.Text        // Descrição da loja
  
  // Imagens
  logo            String?                   // Logo principal
  logo_dark       String?                   // Logo para fundo escuro
  favicon         String?                   // Favicon
  banner_home     String?                   // Banner página inicial
  banner_mobile   String?                   // Banner mobile
  
  // Cores (Hex)
  primary_color   String    @default("#3b82f6")
  secondary_color String    @default("#10b981")
  accent_color    String    @default("#f59e0b")
  
  // ─────────────────────────────────────────────────────────────
  // DOMÍNIO
  // ─────────────────────────────────────────────────────────────
  custom_domain   String?   @unique         // Ex: www.minhaagencia.com.br
  domain_verified Boolean   @default(false)
  ssl_enabled     Boolean   @default(false)
  
  // ─────────────────────────────────────────────────────────────
  // CONTATO
  // ─────────────────────────────────────────────────────────────
  email           String
  phone           String?
  whatsapp        String?
  address         String?
  city            String?
  state           String?
  zip_code        String?
  
  // Redes Sociais
  facebook_url    String?
  instagram_url   String?
  twitter_url     String?
  youtube_url     String?
  linkedin_url    String?
  tiktok_url      String?
  
  // ─────────────────────────────────────────────────────────────
  // SEO
  // ─────────────────────────────────────────────────────────────
  meta_title      String?
  meta_description String?
  meta_keywords   String?
  google_analytics_id String?
  facebook_pixel_id String?
  google_tag_manager_id String?
  
  // ─────────────────────────────────────────────────────────────
  // CONFIGURAÇÕES DE CHECKOUT
  // ─────────────────────────────────────────────────────────────
  require_login   Boolean   @default(false) // Obrigar login para comprar
  guest_checkout  Boolean   @default(true)  // Permitir compra sem cadastro
  
  // Parcelamento
  min_installments Int      @default(1)
  max_installments Int      @default(12)
  installment_fee Decimal   @default(0) @db.Decimal(5, 2) // % de juros
  
  // Valores mínimos
  min_order_value Decimal?  @db.Decimal(10, 2)
  
  // ─────────────────────────────────────────────────────────────
  // MÉTODOS DE PAGAMENTO
  // ─────────────────────────────────────────────────────────────
  payment_methods Json      @default("[]")  // ['credit_card', 'pix', 'boleto', 'debit_card']
  
  // Stripe
  stripe_enabled  Boolean   @default(false)
  stripe_public_key String?
  stripe_secret_key String? @db.Text
  
  // Mercado Pago
  mp_enabled      Boolean   @default(false)
  mp_public_key   String?
  mp_access_token String?   @db.Text
  
  // PIX Manual
  pix_enabled     Boolean   @default(false)
  pix_key         String?
  pix_key_type    String?   // cpf, cnpj, email, phone, random
  
  // Boleto
  boleto_enabled  Boolean   @default(false)
  
  // ─────────────────────────────────────────────────────────────
  // POLÍTICAS E TERMOS
  // ─────────────────────────────────────────────────────────────
  terms_of_service String?  @db.Text
  privacy_policy  String?   @db.Text
  refund_policy   String?   @db.Text
  cancellation_policy String? @db.Text
  
  // URLs externas (se preferir)
  terms_url       String?
  privacy_url     String?
  
  // ─────────────────────────────────────────────────────────────
  // NOTIFICAÇÕES
  // ─────────────────────────────────────────────────────────────
  notification_email String? // Email para receber notificações de vendas
  order_notification_enabled Boolean @default(true)
  
  // ─────────────────────────────────────────────────────────────
  // INTEGRAÇÕES
  // ─────────────────────────────────────────────────────────────
  hurb_enabled    Boolean   @default(false)
  
  // ─────────────────────────────────────────────────────────────
  // STATUS E MANUTENÇÃO
  // ─────────────────────────────────────────────────────────────
  is_active       Boolean   @default(true)
  maintenance_mode Boolean  @default(false)
  maintenance_message String?
  
  // ─────────────────────────────────────────────────────────────
  // ESTATÍSTICAS
  // ─────────────────────────────────────────────────────────────
  total_orders    Int       @default(0)
  total_revenue   Decimal   @default(0) @db.Decimal(12, 2)
  total_visits    Int       @default(0)
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  // Relacionamentos
  categories      StoreCategory[]
  products        StoreProduct[]
  orders          StoreOrder[]
  coupons         StoreCoupon[]
  reviews         StoreReview[]
  pages           StorePage[]
  hurb_integration HurbIntegration?
  
  @@index([slug])
  @@index([custom_domain])
  @@index([tenant_id])
  @@map("stores")
}

// CATEGORIAS DE PRODUTOS
// ═══════════════════════════════════════════════════════════════

model StoreCategory {
  id              String    @id @default(cuid())
  store_id        String
  store           Store     @relation(fields: [store_id], references: [id], onDelete: Cascade)
  
  name            String
  slug            String
  description     String?
  icon            String?   // Ícone ou emoji
  image           String?   // Imagem da categoria
  
  // Hierarquia
  parent_id       String?
  parent          StoreCategory? @relation("CategoryHierarchy", fields: [parent_id], references: [id])
  children        StoreCategory[] @relation("CategoryHierarchy")
  
  // SEO
  meta_title      String?
  meta_description String?
  
  // Ordenação
  order           Int       @default(0)
  
  is_active       Boolean   @default(true)
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  products        StoreProduct[]
  
  @@unique([store_id, slug])
  @@index([store_id])
  @@map("store_categories")
}

// PRODUTOS DA LOJA (MULTI-TIPO)

model StoreProduct {
  id              String    @id @default(cuid())
  store_id        String
  store           Store     @relation(fields: [store_id], references: [id], onDelete: Cascade)
  
  // ─────────────────────────────────────────────────────────────
  // TIPO DE PRODUTO
  // ─────────────────────────────────────────────────────────────
  type            String    // trip, package, tour, insurance, transfer, hotel, flight, service
  
  // ─────────────────────────────────────────────────────────────
  // INFORMAÇÕES BÁSICAS
  // ─────────────────────────────────────────────────────────────
  name            String
  slug            String
  description     String    @db.Text
  short_description String?
  
  // ─────────────────────────────────────────────────────────────
  // CATEGORIA
  // ─────────────────────────────────────────────────────────────
  category_id     String?
  category        StoreCategory? @relation(fields: [category_id], references: [id])
  
  // ─────────────────────────────────────────────────────────────
  // PREÇOS
  // ─────────────────────────────────────────────────────────────
  price           Decimal   @db.Decimal(10, 2)
  compare_price   Decimal?  @db.Decimal(10, 2) // Preço "de"
  cost_price      Decimal?  @db.Decimal(10, 2) // Custo (privado)
  
  // Promoção
  on_sale         Boolean   @default(false)
  sale_price      Decimal?  @db.Decimal(10, 2)
  sale_starts_at  DateTime?
  sale_ends_at    DateTime?
  
  // ─────────────────────────────────────────────────────────────
  // ESTOQUE E DISPONIBILIDADE
  // ─────────────────────────────────────────────────────────────
  track_inventory Boolean   @default(true)
  stock_quantity  Int?
  allow_backorder Boolean   @default(false)
  
  // Para viagens com datas específicas
  has_dates       Boolean   @default(false)
  start_date      DateTime?
  end_date        DateTime?
  
  // ─────────────────────────────────────────────────────────────
  // IMAGENS
  // ─────────────────────────────────────────────────────────────
  images          Json      @default("[]") // Array de URLs
  thumbnail       String?
  gallery         Json?     @default("[]") // Galeria adicional
  
  // ─────────────────────────────────────────────────────────────
  // DETALHES DO PRODUTO
  // ─────────────────────────────────────────────────────────────
  features        Json?     @default("[]") // Array de características
  includes        Json?     @default("[]") // O que está incluído
  excludes        Json?     @default("[]") // O que não está incluído
  requirements    Json?     @default("[]") // Requisitos (ex: documentos)
  
  // Para viagens
  destination     String?
  duration_days   Int?
  duration_nights Int?
  
  // Localização
  city            String?
  state           String?
  country         String?   @default("Brasil")
  
  // ─────────────────────────────────────────────────────────────
  // VARIAÇÕES (Ex: Quarto Single, Double, Triple)
  // ─────────────────────────────────────────────────────────────
  has_variants    Boolean   @default(false)
  variants        Json?     @default("[]") // Array de variações
  
  // ─────────────────────────────────────────────────────────────
  // SEO
  // ─────────────────────────────────────────────────────────────
  meta_title      String?
  meta_description String?
  meta_keywords   String?
  
  // ─────────────────────────────────────────────────────────────
  // RELACIONAMENTOS
  // ─────────────────────────────────────────────────────────────
  // Se for uma viagem do sistema
  trip_id         String?   @unique
  trip            Trip?     @relation(fields: [trip_id], references: [id])
  
  // Se for produto Hurb
  hurb_product_id String?   @unique
  hurb_product    HurbProduct? @relation(fields: [hurb_product_id], references: [id])
  
  // ─────────────────────────────────────────────────────────────
  // DESTAQUE E ORDENAÇÃO
  // ─────────────────────────────────────────────────────────────
  is_featured     Boolean   @default(false)
  order           Int       @default(0)
  
  // ─────────────────────────────────────────────────────────────
  // AVALIAÇÕES
  // ─────────────────────────────────────────────────────────────
  rating_average  Decimal?  @db.Decimal(3, 2)
  rating_count    Int       @default(0)
  
  // ─────────────────────────────────────────────────────────────
  // STATUS
  // ─────────────────────────────────────────────────────────────
  status          String    @default("draft") // draft, active, archived
  published_at    DateTime?
  
  // ─────────────────────────────────────────────────────────────
  // ESTATÍSTICAS
  // ─────────────────────────────────────────────────────────────
  views_count     Int       @default(0)
  sales_count     Int       @default(0)
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  // Relacionamentos
  order_items     StoreOrderItem[]
  reviews         StoreReview[]
  
  @@unique([store_id, slug])
  @@index([store_id, status])
  @@index([store_id, type])
  @@index([category_id])
  @@map("store_products")
}

model StoreOrder {
  id              String    @id @default(cuid())
  store_id        String
  store           Store     @relation(fields: [store_id], references: [id], onDelete: Cascade)
  tenant_id       String    // Para facilitar queries
  
  // ─────────────────────────────────────────────────────────────
  // NÚMERO DO PEDIDO
  // ─────────────────────────────────────────────────────────────
  order_number    String    @unique // Ex: #2024-0001
  
  // ─────────────────────────────────────────────────────────────
  // CLIENTE
  // ─────────────────────────────────────────────────────────────
  client_id       String?
  client          Client?   @relation(fields: [client_id], references: [id])
  
  // Dados do cliente (guest ou cópia)
  customer_name   String
  customer_email  String
  customer_phone  String
  customer_cpf    String?
  customer_address Json?    // Endereço completo
  
  // ─────────────────────────────────────────────────────────────
  // ITENS DO PEDIDO
  // ─────────────────────────────────────────────────────────────
  items           StoreOrderItem[]
  
  // ─────────────────────────────────────────────────────────────
  // VALORES
  // ─────────────────────────────────────────────────────────────
  subtotal        Decimal   @db.Decimal(10, 2)
  discount_amount Decimal   @default(0) @db.Decimal(10, 2)
  tax_amount      Decimal   @default(0) @db.Decimal(10, 2)
  shipping_amount Decimal   @default(0) @db.Decimal(10, 2)
  total_amount    Decimal   @db.Decimal(10, 2)
  
  // ─────────────────────────────────────────────────────────────
  // CUPOM DE DESCONTO
  // ─────────────────────────────────────────────────────────────
  coupon_id       String?
  coupon          StoreCoupon? @relation(fields: [coupon_id], references: [id])
  coupon_code     String?
  
  // ─────────────────────────────────────────────────────────────
  // PAGAMENTO
  // ─────────────────────────────────────────────────────────────
  payment_method  String    // credit_card, pix, boleto, debit_card
  payment_provider String   // stripe, mercadopago, manual
  payment_status  String    @default("pending") // pending, processing, paid, failed, refunded, cancelled
  
  // IDs externos
  payment_intent_id String? // ID no Stripe/MP
  payment_charge_id String? // ID da cobrança
  
  // Parcelamento
  installments    Int       @default(1)
  installment_amount Decimal? @db.Decimal(10, 2)
  
  // PIX
  pix_qr_code     String?   @db.Text
  pix_qr_code_url String?
  pix_copy_paste  String?   @db.Text
  
  // Boleto
  boleto_url      String?
  boleto_barcode  String?
  
  // ─────────────────────────────────────────────────────────────
  // STATUS DO PEDIDO
  // ─────────────────────────────────────────────────────────────
  status          String    @default("pending") 
  // pending, confirmed, processing, completed, cancelled, refunded
  
  fulfillment_status String @default("unfulfilled")
  // unfulfilled, partial, fulfilled
  
  // ─────────────────────────────────────────────────────────────
  // NOTAS
  // ─────────────────────────────────────────────────────────────
  customer_notes  String?   @db.Text
  internal_notes  String?   @db.Text
  
  // ─────────────────────────────────────────────────────────────
  // RASTREAMENTO
  // ─────────────────────────────────────────────────────────────
  ip_address      String?
  user_agent      String?
  
  // ─────────────────────────────────────────────────────────────
  // DATAS IMPORTANTES
  // ─────────────────────────────────────────────────────────────
  paid_at         DateTime?
  confirmed_at    DateTime?
  completed_at    DateTime?
  cancelled_at    DateTime?
  refunded_at     DateTime?
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  // Relacionamentos
  reservations    Reservation[]
  transactions    Transaction[]
  
  @@index([store_id, status])
  @@index([tenant_id])
  @@index([customer_email])
  @@index([order_number])
  @@index([payment_status])
  @@map("store_orders")
}

model StoreOrderItem {

  created_at      DateTime  @default(now())
  
  @@index([order_id])

export async function generateMetadata({ params }: StoreLayoutProps) {
  const store = await prisma.store.findUnique({
    where: { slug: params.slug },
  })

  if (!store) return {}

  return {
    title: store.meta_title || store.name,
    description: store.meta_description || store.description,
    keywords: store.meta_keywords,
    icons: {
      icon: store.favicon || store.logo,
    },
    openGraph: {
      title: store.name,
      description: store.description || '',
      images: [store.banner_home || store.logo || ''],
    },
  }
}

  })

  if (!store) {
    notFound()
  }

  // Buscar produtos em destaque
  const featuredProducts = await prisma.storeProduct.findMany({
    where: {
      store_id: store.id,
      status: 'active',
      is_featured: true,
    },
    orderBy: {
      order: 'asc',
    },
    take: 8,
    include: {
      category: true,
    },
  })

  // Buscar categorias
  const categories = await prisma.storeCategory.findMany({
    where: {
      store_id: store.id,
      is_active: true,
      parent_id: null,
    },
    orderBy: {
      order: 'asc',
    },
    take: 6,
  })

  // Buscar destinos populares
  const destinations = await prisma.storeProduct.groupBy({
    by: ['destination'],
    where: {
      store_id: store.id,
      status: 'active',
      destination: {
        not: null,
      },
    },
    _count: {
      destination: true,
    },
    orderBy: {
      _count: {
        destination: 'desc',
      },
    },
    take: 8,
  })

  // Buscar avaliações
  const reviews = await prisma.storeReview.findMany({
    where: {
      store_id: store.id,

          name: true,
          avatar: true,

      product: {
        select: {
          name: true,

import { prisma } from '@/lib/prisma'
import { notFound } from 'next/navigation'
import { ProductGrid } from '@/components/storefront/products/product-grid'
import { ProductFilters } from '@/components/storefront/products/product-filters'
import { ProductSort } from '@/components/storefront/products/product-sort'
import { Breadcrumb } from '@/components/storefront/breadcrumb'

interface ProductsPageProps {
  params: {
    slug: string
  }
  searchParams: {
    search?: string
    category?: string
    destination?: string
    min_price?: string
    max_price?: string
    sort?: string
    page?: string
  }
}

export default async function ProductsPage({ params, searchParams }: ProductsPageProps) {
  const store = await prisma.store.findUnique({
    where: { slug: params.slug },
  })

  if (!store) {
    notFound()
  }

  // Construir filtros
  const page = parseInt(searchParams.page || '1')
  const limit = 12
  const skip = (page - 1) * limit

  const where: any = {
    store_id: store.id,
    status: 'active',
  }

  // Busca por texto
  if (searchParams.search) {
    where.OR = [
      { name: { contains: searchParams.search, mode: 'insensitive' } },
      { description: { contains: searchParams.search, mode: 'insensitive' } },
      { destination: { contains: searchParams.search, mode: 'insensitive' } },
    ]
  }

  // Filtro por categoria
  if (searchParams.category) {
    where.category_id = searchParams.category
  }

  // Filtro por destino
  if (searchParams.destination) {
    where.destination = searchParams.destination
  }

  // Filtro por preço
  if (searchParams.min_price || searchParams.max_price) {
    where.price = {}
    if (searchParams.min_price) {
      where.price.gte = parseFloat(searchParams.min_price)
    }
    if (searchParams.max_price) {
      where.price.lte = parseFloat(searchParams.max_price)
    }
  }

  // Ordenação
  let orderBy: any = { order: 'asc' }
  
  switch (searchParams.sort) {
    case 'price_asc':
      orderBy = { price: 'asc' }
      break
    case 'price_desc':
      orderBy = { price: 'desc' }
      break
    case 'name_asc':
      orderBy = { name: 'asc' }
      break
    case 'name_desc':
      orderBy = { name: 'desc' }
      break
    case 'newest':
      orderBy = { created_at: 'desc' }
      break
    case 'popular':
      orderBy = { sales_count: 'desc' }
      break
    case 'rating':
      orderBy = { rating_average: 'desc' }
      break
  }

  // Buscar produtos
  const [products, total] = await Promise.all([
    prisma.storeProduct.findMany({
      where,
      orderBy,
      skip,
      take: limit,
      include: {
        category: true,
      },
    }),
    prisma.storeProduct.count({ where }),
  ])

  // Buscar categorias para filtro
  const categories = await prisma.storeCategory.findMany({
    where: {
      store_id: store.id,
      is_active: true,
    },
    orderBy: {
      name: 'asc',
    },
  })

  // Buscar destinos únicos
  const destinationsData = await prisma.storeProduct.findMany({
    where: {
      store_id: store.id,
      status: 'active',
      destination: { not: null },
    },
    select: {
      destination: true,
    },
    distinct: ['destination'],
  })

  const destinations = destinationsData
    .map(d => d.destination)
    .filter(Boolean) as string[]

  // Calcular range de preços
  const priceRange = await prisma.storeProduct.aggregate({
    where: {
      store_id: store.id,
      status: 'active',
    },
    _min: { price: true },
    _max: { price: true },
  })

  const totalPages = Math.ceil(total / limit)

  return (
    <div className="min-h-screen bg-gray-50">
      <div className="container mx-auto px-4 py-8">
        {/* Breadcrumb */}
        <Breadcrumb
          items={[
            { label: 'Início', href: `/${params.slug}` },
            { label: 'Viagens', href: `/${params.slug}/viagens` },
          ]}
        />

        {/* Header */}
        <div className="mt-6 mb-8">
          <h1 className="text-3xl font-bold mb-2">
            {searchParams.search 
              ? `Resultados para "${searchParams.search}"`
              : 'Todas as Viagens'
            }
          </h1>
          <p className="text-gray-600">
            {total} {total === 1 ? 'viagem encontrada' : 'viagens encontradas'}
          </p>
        </div>

        <div className="grid grid-cols-1 lg:grid-cols-4 gap-8">
          {/* Sidebar - Filtros */}
          <aside className="lg:col-span-1">
            <ProductFilters
              categories={categories}
              destinations={destinations}
              priceRange={{
                min: Number(priceRange._min.price || 0),
                max: Number(priceRange._max.price || 10000),
              }}
              currentFilters={searchParams}
              storeSlug={params.slug}
            />
          </aside>

          {/* Products Grid */}
          <div className="lg:col-span-3">
            {/* Sort */}
            <div className="mb-6 flex items-center justify-between">
              <p className="text-sm text-gray-600">
                Mostrando {skip + 1}-{Math.min(skip + limit, total)} de {total}
              </p>
              <ProductSort currentSort={searchParams.sort} storeSlug={params.slug} />
            </div>

            {/* Grid */}
            {products.length > 0 ? (
              <>
                <ProductGrid products={products} storeSlug={params.slug} />

                {/* Pagination */}
                {totalPages > 1 && (
                  <div className="mt-12 flex items-center justify-center gap-2">
                    {Array.from({ length: totalPages }, (_, i) => i + 1).map((pageNum) => (
                      <a
                        key={pageNum}
                        href={`/${params.slug}/viagens?${new URLSearchParams({
                          ...searchParams,
                          page: pageNum.toString(),
                        })}`}
                        className={`px-4 py-2 rounded-lg ${
                          pageNum === page
                            ? 'bg-blue-600 text-white'
                            : 'bg-white text-gray-700 hover:bg-gray-100'
                        }`}
                      >
                        {pageNum}
                      </a>
                    ))}
                  </div>
                )}
              </>
            ) : (
              <div className="text-center py-12">
                <p className="text-gray-600 mb-4">
                  Nenhuma viagem encontrada com os filtros selecionados
                </p>
                <a
                  href={`/${params.slug}/viagens`}
                  className="text-blue-600 hover:underline"
                >
                  Limpar filtros
                </a>
              </div>
            )}
          </div>
        </div>
      </div>
    </div>
  )
}

'use client'

import { useState } from 'react'
import { useRouter, useSearchParams } from 'next/navigation'
import { Button } from '@/components/ui/button'
import { Checkbox } from '@/components/ui/checkbox'
import { Label } from '@/components/ui/label'

interface ProductFiltersProps {
  categories: Array<{ id: string; name: string }>
  destinations: string[]
  priceRange: { min: number; max: number }
  currentFilters: any
  storeSlug: string
}

export function ProductFilters({
  categories,
  destinations,
  priceRange,
  currentFilters,
  storeSlug,
}: ProductFiltersProps) {
  const router = useRouter()
  const searchParams = useSearchParams()

  const [selectedCategories, setSelectedCategories] = useState<string[]>(
    currentFilters.category ? [currentFilters.category] : []
  )
  const [selectedDestinations, setSelectedDestinations] = useState<string[]>(
    currentFilters.destination ? [currentFilters.destination] : []
  )
  const [priceValues, setPriceValues] = useState<[number, number]>([
    parseFloat(currentFilters.min_price || priceRange.min),
    parseFloat(currentFilters.max_price || priceRange.max),
  ])

  const applyFilters = () => {
    const params = new URLSearchParams(searchParams.toString())

    // Limpar filtros anteriores
    params.delete('category')
    params.delete('destination')
    params.delete('min_price')
    params.delete('max_price')
    params.delete('page')

    // Aplicar novos filtros
    if (selectedCategories.length > 0) {
      params.set('category', selectedCategories[0])
    }
    if (selectedDestinations.length > 0) {
      params.set('destination', selectedDestinations[0])
    }
    if (priceValues[0] > priceRange.min) {
      params.set('min_price', priceValues[0].toString())
    }
    if (priceValues[1] < priceRange.max) {
      params.set('max_price', priceValues[1].toString())
    }

  }

  const clearFilters = () => {
    setSelectedCategories([])
    setSelectedDestinations([])
    setPriceValues([priceRange.min, priceRange.max])
    router.push(`/${storeSlug}/viagens`)
  }

  const hasActiveFilters = 
    selectedCategories.length > 0 ||
    selectedDestinations.length > 0 ||
    priceValues[0] > priceRange.min ||
    priceValues[1] < priceRange.max

  return (
    <div className="bg-white rounded-lg p-6 sticky top-24">
      <div className="flex items-center justify-between mb-6">

            onClick={clearFilters}
            className="text-red-600 hover:text-red-700"
          >
            <X className="h-4 w-4 mr-1" />
            Limpar
          </Button>
        )}
      </div>

      {/* Categorias */}
      {categories.length > 0 && (
        <div className="mb-6">
          <h4 className="font-semibold mb-3">Categorias</h4>
          <div className="space-y-2">
            {categories.map((category) => (
              <div key={category.id} className="flex items-center">
                <Checkbox
                  id={`category-${category.id}`}
                  checked={selectedCategories.includes(category.id)}
                  onCheckedChange={(checked) => {
                    if (checked) {
                      setSelectedCategories([category.id])
                    } else {
                      setSelectedCategories([])
                    }
                  }}
                />
                <Label
                  htmlFor={`category-${category.id}`}
                  className="ml-2 text-sm cursor-pointer"
                >
                  {category.name}
                </Label>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* Destinos */}
      {destinations.length > 0 && (
        <div className="mb-6">
          <h4 className="font-semibold mb-3">Destinos</h4>
          <div className="space-y-2 max-h-64 overflow-y-auto">
            {destinations.map((destination) => (
              <div key={destination} className="flex items-center">
                <Checkbox
                  id={`destination-${destination}`}
                  checked={selectedDestinations.includes(destination)}
                  onCheckedChange={(checked) => {
                    if (checked) {
                      setSelectedDestinations([destination])
                    } else {
                      setSelectedDestinations([])
                    }
                  }}
                />
                <Label
                  htmlFor={`destination-${destination}`}
                  className="ml-2 text-sm cursor-pointer"
                >
                  {destination}
                </Label>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* Preço */}
      <div className="mb-6">
        <h4 className="font-semibold mb-3">Faixa de Preço</h4>
        <div className="px-2">
          <Slider
            min={priceRange.min}
            max={priceRange.max}
            step={100}
            value={priceValues}
            onValueChange={(values) => setPriceValues(values as [number, number])}
            className="mb-4"
          />
          <div className="flex items-center justify-between text-sm">
            <span>R$ {priceValues[0].toLocaleString('pt-BR')}</span>
            <span>R$ {priceValues[1].toLocaleString('pt-BR')}</span>
          </div>
        </div>
      </div>

      {/* Apply Button */}
      <Button onClick={applyFilters} className="w-full">
        Aplicar Filtros
      </Button>
    </div>
  )
}

// contexts/cart-context.tsx

'use client'

import { createContext, useContext, useState, useEffect, ReactNode } from 'react'

interface CartItem {
  product_id: string
  name: string
  price: number
  quantity: number
  thumbnail: string | null
  variant?: any
}

interface CartContextType {
  items: CartItem[]
  itemsCount: number
  subtotal: number
  addItem: (item: CartItem) => void
  removeItem: (productId: string) => void
  updateQuantity: (productId: string, quantity: number) => void
  clearCart: () => void
}

const CartContext = createContext<CartContextType | undefined>(undefined)

export function CartProvider({ 
  children, 
  storeId 
}: { 
  children: ReactNode
  storeId: string 
}) {
  const [items, setItems] = useState<CartItem[]>([])

  // Carregar do localStorage
  useEffect(() => {
    const saved = localStorage.getItem(`cart_${storeId}`)
    if (saved) {
      setItems(JSON.parse(saved))
    }
  }, [storeId])

  // Salvar no localStorage
  useEffect(() => {
    localStorage.setItem(`cart_${storeId}`, JSON.stringify(items))
  }, [items, storeId])

  const addItem = (item: CartItem) => {
    setItems((current) => {
      const existingIndex = current.findIndex(
        (i) => i.product_id === item.product_id
      )

      if (existingIndex > -1) {
        const updated = [...current]
        updated[existingIndex].quantity += item.quantity
        return updated
      }

      return [...current, item]
    })
  }

  const removeItem = (productId: string) => {
    setItems((current) => current.filter((i) => i.product_id !== productId))
  }

  const updateQuantity = (productId: string, quantity: number) => {
    if (quantity <= 0) {
      removeItem(productId)
      return
    }

    setItems((current) =>
      current.map((item) =>
        item.product_id === productId ? { ...item, quantity } : item
      )
    )
  }

  const clearCart = () => {
    setItems([])
  }

  const itemsCount = items.reduce((sum, item) => sum + item.quantity, 0)
  const subtotal = items.reduce((sum, item) => sum + item.price * item.quantity, 0)

  return (
    <CartContext.Provider
      value={{
        items,
        itemsCount,
        subtotal,
        addItem,
        removeItem,
        updateQuantity,
        clearCart,
      }}
    >
      {children}
    </CartContext.Provider>
  )
}

export function useCart() {
  const context = useContext(CartContext)
  if (!context) {
    throw new Error('useCart must be used within CartProvider')
  }
  return context
}

                }

// app/api/storefront/[slug]/checkout/route.ts

import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { createStripePaymentIntent } from '@/lib/payments/stripe-client'
import { createMercadoPagoPreference } from '@/lib/payments/mercadopago-client'
import { generateOrderNumber } from '@/lib/utils/order-number'
import { sendOrderConfirmationEmail } from '@/lib/emails/order-confirmation'

export async function POST(
  req: Request,
  { params }: { params: { slug: string } }
) {
  try {
    const body = await req.json()
    const { customer, payment, items, coupon_code, subtotal, discount, total } = body

    // Buscar loja
    const store = await prisma.store.findUnique({
      where: { slug: params.slug },
      include: {
        tenant: true,
      },
    })

    if (!store || !store.is_active) {
      return NextResponse.json(
        { error: 'Loja não encontrada' },
        { status: 404 }
      )
    }

    // Validar estoque dos produtos
    for (const item of items) {
      const product = await prisma.storeProduct.findUnique({
        where: { id: item.product_id },
      })

      if (!product || product.status !== 'active') {
        return NextResponse.json(
          { error: `Produto ${item.name} não está disponível` },
          { status: 400 }
        )
      }

      if (product.track_inventory && product.stock_quantity !== null) {
        if (product.stock_quantity < item.quantity) {
          return NextResponse.json(
            { error: `Estoque insuficiente para ${item.name}` },
            { status: 400 }
          )
        }
      }
    }

    // Validar cupom (se aplicado)
    let couponId = null
    if (coupon_code) {
      const coupon = await prisma.storeCoupon.findUnique({
        where: {
          store_id_code: {
            store_id: store.id,
            code: coupon_code,
          },
        },
      })

      if (!coupon || !coupon.is_active) {
        return NextResponse.json(
          { error: 'Cupom inválido' },
          { status: 400 }
        )
      }

      // Validar validade
      const now = new Date()
      if (now < coupon.starts_at || now > coupon.expires_at) {
        return NextResponse.json(
          { error: 'Cupom expirado' },
          { status: 400 }
        )
      }

      // Validar limite de uso
      if (coupon.usage_limit && coupon.usage_count >= coupon.usage_limit) {
        return NextResponse.json(
          { error: 'Cupom esgotado' },
          { status: 400 }
        )
      }

      couponId = coupon.id
    }

    // Buscar ou criar cliente
    let client = await prisma.client.findFirst({
      where: {
        tenant_id: store.tenant_id,
        email: customer.email,
      },
    })

    if (!client) {
      client = await prisma.client.create({
        data: {
          tenant_id: store.tenant_id,
          name: customer.name,
          email: customer.email,
          phone: customer.phone,
          cpf: customer.cpf,
          address: customer.address.street,
          city: customer.address.city,
          state: customer.address.state,
          zip_code: customer.address.zip_code,
          source: 'store',
        },
      })
    }

    // Gerar número do pedido
    const orderNumber = await generateOrderNumber(store.tenant_id)

    // Criar pedido
    const order = await prisma.storeOrder.create({
      data: {
        store_id: store.id,
        tenant_id: store.tenant_id,
        order_number: orderNumber,
        client_id: client.id,
        customer_name: customer.name,
        customer_email: customer.email,
        customer_phone: customer.phone,
        customer_cpf: customer.cpf,
        customer_address: customer.address,
        subtotal_amount: subtotal,
        discount_amount: discount,
        total_amount: total,
        coupon_id: couponId,
        coupon_code: coupon_code,
        payment_method: payment.method,
        payment_provider: payment.method === 'credit_card' ? 'stripe' : 'mercadopago',
        payment_status: 'pending',
        status: 'pending',
        installments: payment.installments || 1,
        installment_amount: payment.installments ? total / payment.installments : total,
        items: {
          create: items.map((item: any) => ({
            product_id: item.product_id,
            product_name: item.name,
            product_type: 'trip',
            product_image: item.thumbnail,
            price: item.price,
            quantity: item.quantity,
            subtotal: item.price * item.quantity,
            total: item.price * item.quantity,
          })),
        },
      },
      include: {
        items: true,
      },
    })

    // Processar pagamento
    let paymentData: any = {}

    switch (payment.method) {
      case 'credit_card':
        // Stripe
        if (store.stripe_enabled && store.stripe_secret_key) {
          const paymentIntent = await createStripePaymentIntent({
            amount: total,
            currency: 'brl',
            customer_email: customer.email,
            metadata: {
              order_id: order.id,
              store_id: store.id,
              tenant_id: store.tenant_id,
            },
            stripe_secret_key: store.stripe_secret_key,
          })

          paymentData = {
            payment_intent_id: paymentIntent.id,
            client_secret: paymentIntent.client_secret,
          }

          // Atualizar pedido
          await prisma.storeOrder.update({
            where: { id: order.id },
            data: {
              payment_intent_id: paymentIntent.id,
              payment_status: 'processing',
            },
          })
        }
        break

      case 'pix':
      case 'boleto':
        // Mercado Pago
        if (store.mp_enabled && store.mp_access_token) {
          const preference = await createMercadoPagoPreference({
            items: items.map((item: any) => ({
              title: item.name,
              quantity: item.quantity,
              unit_price: item.price,
            })),
            payer: {
              name: customer.name,
              email: customer.email,
              phone: customer.phone,
              identification: {
                type: 'CPF',
                number: customer.cpf,
              },
              address: customer.address,
            },
            payment_methods: {
              excluded_payment_types: [],
              installments: payment.installments || 1,
            },
            back_urls: {
              success: `${process.env.NEXT_PUBLIC_APP_URL}/${store.slug}/confirmacao/${order.id}`,
              failure: `${process.env.NEXT_PUBLIC_APP_URL}/${store.slug}/checkout`,
              pending: `${process.env.NEXT_PUBLIC_APP_URL}/${store.slug}/confirmacao/${order.id}`,
            },
            notification_url: `${process.env.NEXT_PUBLIC_APP_URL}/api/webhooks/mercadopago`,
            external_reference: order.id,
            mp_access_token: store.mp_access_token,
          })

          paymentData = {
            preference_id: preference.id,
            init_point: preference.init_point,
            sandbox_init_point: preference.sandbox_init_point,
          }

          // Se for PIX, gerar QR Code
          if (payment.method === 'pix') {
            // Aqui você pode gerar o QR Code do PIX
            // usando a API do Mercado Pago ou outra solução
          }

          // Atualizar pedido
          await prisma.storeOrder.update({
            where: { id: order.id },
            data: {
              payment_intent_id: preference.id,
            },
          })
        }
        break
    }

    // Atualizar uso do cupom
    if (couponId) {
      await prisma.storeCoupon.update({
        where: { id: couponId },
        data: {
          usage_count: { increment: 1 },
        },
      })
    }

    // Enviar email de confirmação
    await sendOrderConfirmationEmail({
      to: customer.email,
      order_number: orderNumber,
      customer_name: customer.name,
      items,
      total,
      store_name: store.name,
    })

    // Notificar agência
    if (store.order_notification_enabled && store.notification_email) {
      // Enviar email para agência
    }

    // Criar log de auditoria
    await prisma.auditLog.create({
      data: {
        tenant_id: store.tenant_id,
        action: 'create',
        resource: 'store_order',
        resource_id: order.id,
        metadata: {
          order_number: orderNumber,
          total,
          payment_method: payment.method,
        },
      },
    })

    return NextResponse.json({
      success: true,
      order_id: order.id,
      order_number: orderNumber,
      payment: paymentData,
    })
  } catch (error: any) {
    console.error('Erro ao processar checkout:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao processar pedido' },
      { status: 500 }
    )
  }
}

// app/api/storefront/[slug]/coupons/validate/route.ts

import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'

export async function POST(
  req: Request,
  { params }: { params: { slug: string } }
) {
  try {
    const { code, subtotal } = await req.json()

    // Buscar loja
    const store = await prisma.store.findUnique({
      where: { slug: params.slug },
    })

    if (!store) {
      return NextResponse.json(
        { error: 'Loja não encontrada' },
        { status: 404 }
      )
    }

    // Buscar cupom
    const coupon = await prisma.storeCoupon.findUnique({
      where: {
        store_id_code: {
          store_id: store.id,
          code: code.toUpperCase(),
        },
      },
    })

    if (!coupon) {
      return NextResponse.json(
        { error: 'Cupom não encontrado' },
        { status: 404 }
      )
    }

    // Validar status
    if (!coupon.is_active) {
      return NextResponse.json(
        { error: 'Cupom inativo' },
        { status: 400 }
      )
    }

    // Validar validade
    const now = new Date()
    if (now < coupon.starts_at) {
      return NextResponse.json(
        { error: 'Cupom ainda não está válido' },
        { status: 400 }
      )
    }

    if (now > coupon.expires_at) {
      return NextResponse.json(
        { error: 'Cupom expirado' },
        { status: 400 }
      )
    }

    // Validar limite de uso
    if (coupon.usage_limit && coupon.usage_count >= coupon.usage_limit) {
      return NextResponse.json(
        { error: 'Cupom esgotado' },
        { status: 400 }
      )
    }

    // Validar valor mínimo
    if (coupon.min_purchase_amount && subtotal < Number(coupon.min_purchase_amount)) {
      return NextResponse.json(
        {
          error: `Valor mínimo de compra: R$ ${Number(coupon.min_purchase_amount).toLocaleString('pt-BR')}`,
        },
        { status: 400 }
      )
    }

    // Calcular desconto
    let discount = 0

    if (coupon.type === 'percentage') {
      discount = (subtotal * Number(coupon.value)) / 100
    } else if (coupon.type === 'fixed') {
      discount = Number(coupon.value)
    }

    // Aplicar desconto máximo
    if (coupon.max_discount_amount && discount > Number(coupon.max_discount_amount)) {
      discount = Number(coupon.max_discount_amount)
    }

    // Não pode ser maior que o subtotal
    if (discount > subtotal) {
      discount = subtotal
    }

    return NextResponse.json({
      success: true,
      coupon: {
        code: coupon.code,
        type: coupon.type,
        value: Number(coupon.value),
        description: coupon.description,
      },
      discount,
      final_total: subtotal - discount,
    })
  } catch (error: any) {
    console.error('Erro ao validar cupom:', error)
    return NextResponse.json(
      { error: 'Erro ao validar cupom' },
      { status: 500 }
    )
  }
}

import { NextResponse } from 'next/server'
import { headers } from 'next/headers'
import Stripe from 'stripe'
import { prisma } from '@/lib/prisma'
import { sendPaymentConfirmationEmail } from '@/lib/emails/payment-confirmation'

export async function POST(req: Request) {
  try {
    const body = await req.text()
    const signature = headers().get('stripe-signature')

    if (!signature) {
      return NextResponse.json(
        { error: 'Missing signature' },
        { status: 400 }
      )
    }

    // Buscar pedido para obter a secret key da loja
    const event = JSON.parse(body)
    const paymentIntentId = event.data.object.id

    const order = await prisma.storeOrder.findFirst({
      where: {
        payment_intent_id: paymentIntentId,
      },
      include: {
        store: true,
        items: {
          include: {
            product: true,
          },
        },
        client: true,
      },
    })

    if (!order || !order.store.stripe_secret_key) {
      return NextResponse.json(
        { error: 'Order not found' },
        { status: 404 }
      )
    }

    // Verificar assinatura
    const stripe = new Stripe(order.store.stripe_secret_key, {
      apiVersion: '2024-11-20.acacia',
    })

    const webhookSecret = process.env.STRIPE_WEBHOOK_SECRET!
    
    let stripeEvent: Stripe.Event

    try {
      stripeEvent = stripe.webhooks.constructEvent(
        body,
        signature,
        webhookSecret
      )
    } catch (err: any) {
      console.error('Webhook signature verification failed:', err.message)
      return NextResponse.json(
        { error: 'Invalid signature' },
        { status: 400 }
      )
    }

    // Processar evento
    switch (stripeEvent.type) {
      case 'payment_intent.succeeded':
        await handlePaymentSuccess(order)
        break

      case 'payment_intent.payment_failed':
        await handlePaymentFailed(order)
        break

      case 'charge.refunded':
        await handleRefund(order)
        break

      default:
        console.log(`Unhandled event type: ${stripeEvent.type}`)
    }

    return NextResponse.json({ received: true })
  } catch (error: any) {
    console.error('Stripe webhook error:', error)
    return NextResponse.json(
      { error: error.message },
      { status: 500 }
    )
  }
}

async function handlePaymentSuccess(order: any) {
  // Atualizar pedido
  await prisma.storeOrder.update({
    where: { id: order.id },
    data: {
      payment_status: 'paid',
      status: 'confirmed',
      paid_at: new Date(),
      confirmed_at: new Date(),
    },
  })

  // Atualizar estoque
  for (const item of order.items) {
    if (item.product.track_inventory && item.product.stock_quantity !== null) {
      await prisma.storeProduct.update({
        where: { id: item.product_id },
        data: {
          stock_quantity: {
            decrement: item.quantity,
          },
          sales_count: {
            increment: item.quantity,
          },
        },
      })
    }
  }

  // Criar reservas (se for viagem)
  for (const item of order.items) {
    if (item.product.trip_id) {
      await prisma.reservation.create({
        data: {
          tenant_id: order.tenant_id,
          client_id: order.client_id!,
          trip_id: item.product.trip_id,
          store_order_id: order.id,
          num_passengers: item.quantity,
          total_value: item.total,
          status: 'confirmada',
          payment_status: 'pago',
        },
      })
    }
  }

  // Criar transação financeira
  await prisma.transaction.create({
    data: {
      tenant_id: order.tenant_id,
      type: 'receita',
      category: 'venda_online',
      description: `Venda online - Pedido ${order.order_number}`,
      amount: order.total_amount,
      payment_method: order.payment_method,
      status: 'confirmada',
      transaction_date: new Date(),
      client_id: order.client_id,
    },
  })

  // Atualizar estatísticas da loja
  await prisma.store.update({
    where: { id: order.store_id },
    data: {
      total_orders: { increment: 1 },
      total_revenue: { increment: order.total_amount },
    },
  })

  // Enviar email de confirmação
  await sendPaymentConfirmationEmail({
    to: order.customer_email,
    order_number: order.order_number,
    customer_name: order.customer_name,
    total: Number(order.total_amount),
    store_name: order.store.name,
  })

  // Log de auditoria
  await prisma.auditLog.create({
    data: {
      tenant_id: order.tenant_id,
      action: 'payment_confirmed',
      resource: 'store_order',
      resource_id: order.id,
      metadata: {
        order_number: order.order_number,
        amount: Number(order.total_amount),
        payment_method: order.payment_method,
      },
    },
  })
}

export async function syncHurbProducts(integrationId: string) {
  try {
    const integration = await prisma.hurbIntegration.findUnique({
      where: { id: integrationId },
      include: { tenant: true },
    })

    if (!integration || !integration.is_active) {
      throw new Error('Integração não encontrada ou inativa')
    }

    const client = new HurbClient({
      apiKey: integration.api_key,
      apiSecret: integration.api_secret,
      partnerId: integration.partner_id,
    })

    // Buscar produtos da Hurb
    const products = await client.getProducts({
      page: 1,
      limit: 100,
    })

    let syncedCount = 0
    let errorCount = 0

    for (const product of products) {
      try {
        await syncHurbProduct(integrationId, product.id, product)
        syncedCount++
      } catch (error) {
        console.error(`Erro ao sincronizar produto ${product.id}:`, error)
        errorCount++
      }
    }

    // Atualizar última sincronização
    await prisma.hurbIntegration.update({
      where: { id: integrationId },
      data: {
        last_sync_at: new Date(),
        last_sync_status: errorCount > 0 ? 'partial' : 'success',
        last_sync_error: errorCount > 0 
          ? `${errorCount} produtos com erro`
          : null,
      },
    })

    return {
      success: true,
      synced: syncedCount,
      errors: errorCount,
    }
  } catch (error: any) {
    // Atualizar com erro
    await prisma.hurbIntegration.update({
      where: { id: integrationId },
      data: {
        last_sync_at: new Date(),
        last_sync_status: 'error',
        last_sync_error: error.message,
      },
    })

    throw error
  }
}

export class HurbClient {
  private apiKey: string
  private apiSecret: string
  private partnerId: string
  private baseURL: string

  constructor(config: HurbClientConfig) {
    this.apiKey = config.apiKey
    this.apiSecret = config.apiSecret
    this.partnerId = config.partnerId
    this.baseURL = process.env.HURB_API_URL || 'https://api.hurb.com/v1'
  }

  private async request(endpoint: string, options: any = {}) {
    const response = await axios({
      url: `${this.baseURL}${endpoint}`,
      headers: {
        'Authorization': `Bearer ${this.apiKey}`,
        'X-Partner-Id': this.partnerId,
        'Content-Type': 'application/json',
        ...options.headers,
      },
      ...options,
    })

    return response.data
  }

  async getProducts(params: { page?: number; limit?: number } = {}) {
    return this.request('/products', {
      method: 'GET',
      params: {
        page: params.page || 1,
        limit: params.limit || 50,
      },
    })
  }

    return this.request(`/products/${productId}`, {

/**
 * Gera um código único de indicação
 */
export async function generateReferralCode(
  name: string,
  tenantId: string
): Promise<string> {
  // Gerar código base a partir do nome
  const nameBase = name
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '') // Remove acentos
    .toUpperCase()
    .replace(/[^A-Z]/g, '')
    .substring(0, 4)
  
  // Adicionar números aleatórios
  let code = `${nameBase}${Math.floor(1000 + Math.random() * 9000)}`
  
  // Verificar se já existe
  let exists = await prisma.client.findFirst({
    where: {
      tenant_id: tenantId,
      referral_code: code,
    },
  })
  
  // Se existir, gerar novo
  let attempts = 0
  while (exists && attempts < 10) {
    code = `${nameBase}${Math.floor(1000 + Math.random() * 9000)}`
    exists = await prisma.client.findFirst({
      where: {
        tenant_id: tenantId,
        referral_code: code,
      },
    })
    attempts++
  }
  
  // Se ainda existir após 10 tentativas, usar CUID
  if (exists) {
    code = `REF${Math.random().toString(36).substring(2, 10).toUpperCase()}`
  }
  
  return code
}

/**
 * Valida um código de indicação
 */
export async function validateReferralCode(
  code: string,
  tenantId: string,
  excludeClientId?: string
): Promise<{ valid: boolean; client?: any; error?: string }> {
  if (!code || code.trim() === '') {
    return { valid: false, error: 'Código inválido' }
  }
  
  const client = await prisma.client.findFirst({
    where: {
      tenant_id: tenantId,
      referral_code: code.toUpperCase(),
    },
  })
  
  if (!client) {
    return { valid: false, error: 'Código não encontrado' }
  }
  
  // Não pode usar o próprio código
  if (excludeClientId && client.id === excludeClientId) {
    return { valid: false, error: 'Você não pode usar seu próprio código' }
  }
  
  return { valid: true, client }
}

    // Verificar se CPF já existe
    if (cpf) {
      const existingCpf = await prisma.client.findFirst({
        where: {
          tenant_id: store.tenant_id,
          cpf,
        },
      })

      if (existingCpf) {
        return NextResponse.json(
          { error: 'CPF já cadastrado' },
          { status: 400 }
        )
      }
    }

    // Validar código de indicação (se fornecido)
    let referrer = null
    if (referral_code) {
      const validation = await validateReferralCode(
        referral_code,
        store.tenant_id
      )

      if (!validation.valid) {
        return NextResponse.json(
          { error: validation.error },
          { status: 400 }
        )
      }

      referrer = validation.client
    }

    // Gerar código único para o novo cliente
    const newReferralCode = await generateReferralCode(name, store.tenant_id)

    // Hash da senha
    const hashedPassword = await hash(password, 10)

    // Criar cliente
    const client = await prisma.client.create({
      data: {
        tenant_id: store.tenant_id,
        name,
        email,
        cpf,
        phone,
        password: hashedPassword,
        referral_code: newReferralCode,
        referred_by_code: referral_code?.toUpperCase() || null,
        source: 'store',
      },
    })

    // Se tiver indicação, criar registro
    if (referrer) {
      // Buscar configurações do programa
      const settings = await prisma.referralSettings.findUnique({
        where: { tenant_id: store.tenant_id },
      })

      const bonusAmount = settings?.bonus_amount || 50
      const discountPercentage = settings?.discount_percentage || 5

      await prisma.referral.create({
        data: {
          tenant_id: store.tenant_id,
          referrer_id: referrer.id,
          referred_id: client.id,
          bonus_amount: bonusAmount,
          discount_percentage: discountPercentage,
          status: 'pending',
        },
      })

      // Log
      await prisma.auditLog.create({
        data: {
          tenant_id: store.tenant_id,
          action: 'referral_created',
          resource: 'referral',
          resource_id: client.id,
          metadata: {
            referrer_id: referrer.id,
            referrer_name: referrer.name,
            referred_name: client.name,
          },
        },
      })
    }

    return NextResponse.json({
      success: true,
      client: {
        id: client.id,
        name: client.name,
        email: client.email,
        referral_code: client.referral_code,
      },
      has_referral: !!referrer,
    })
  } catch (error: any) {
    console.error('Erro ao criar conta:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao criar conta' },
      { status: 500 }
    )
  }
}

// lib/referral/apply-discount.ts

import { prisma } from '@/lib/prisma'

interface ApplyReferralDiscountParams {
  clientId: string
  tenantId: string
  subtotal: number
}

interface ReferralDiscountResult {
  hasDiscount: boolean
  discountAmount: number
  discountPercentage: number
  finalTotal: number
  referralId?: string
}

export async function applyReferralDiscount({
  clientId,
  tenantId,
  subtotal,
}: ApplyReferralDiscountParams): Promise<ReferralDiscountResult> {
  // Verificar se é primeira compra
  const orderCount = await prisma.storeOrder.count({
    where: {
      client_id: clientId,
      payment_status: 'paid',
    },
  })

  if (orderCount > 0) {
    return {
      hasDiscount: false,
      discountAmount: 0,
      discountPercentage: 0,
      finalTotal: subtotal,
    }
  }

  // Buscar indicação pendente
  const referral = await prisma.referral.findFirst({
    where: {
      tenant_id: tenantId,
      referred_id: clientId,
      status: 'pending',
      discount_used: false,
    },
  })

  if (!referral) {
    return {
      hasDiscount: false,
      discountAmount: 0,
      discountPercentage: 0,
      finalTotal: subtotal,
    }
  }

  // Buscar configurações
  const settings = await prisma.referralSettings.findUnique({
    where: { tenant_id: tenantId },
  })

  // Verificar valor mínimo
  if (settings?.min_purchase_amount && subtotal < Number(settings.min_purchase_amount)) {
    return {
      hasDiscount: false,
      discountAmount: 0,
      discountPercentage: 0,
      finalTotal: subtotal,
    }
  }

  // Calcular desconto
  const discountPercentage = Number(referral.discount_percentage)
  let discountAmount = (subtotal * discountPercentage) / 100

  // Aplicar desconto máximo se configurado
  if (settings?.max_discount_amount && discountAmount > Number(settings.max_discount_amount)) {
    discountAmount = Number(settings.max_discount_amount)
  }

  const finalTotal = subtotal - discountAmount

  return {
    hasDiscount: true,
    discountAmount,
    discountPercentage,
    finalTotal,
    referralId: referral.id,
  }
}

// lib/referral/process-bonus.ts

import { prisma } from '@/lib/prisma'
import { sendReferralBonusEmail } from '@/lib/emails/referral-bonus'

export async function processReferralBonus(orderId: string) {
  try {
    const order = await prisma.storeOrder.findUnique({
      where: { id: orderId },
      include: {
        client: true,
        store: true,
      },
    })

    if (!order) {
      throw new Error('Pedido não encontrado')
    }

    // Só processar se pagamento confirmado
    if (order.payment_status !== 'paid') {
      return { processed: false, reason: 'payment_not_confirmed' }
    }

    // Só processar se for primeira compra com indicação
    if (!order.is_first_order || !order.referral_applied) {
      return { processed: false, reason: 'not_eligible' }
    }

    // Buscar indicação
    const referral = await prisma.referral.findFirst({
      where: {
        conversion_order_id: orderId,
        status: 'pending',
        bonus_paid: false,
      },
      include: {
        referrer: true,
        referred: true,
      },
    })

    if (!referral) {
      return { processed: false, reason: 'referral_not_found' }
    }

    // Creditar bônus para quem indicou
    await prisma.client.update({
      where: { id: referral.referrer_id },
      data: {
        referral_balance: {
          increment: referral.bonus_amount,
        },
      },
    })

    // Atualizar indicação
    await prisma.referral.update({
      where: { id: referral.id },
      data: {
        status: 'converted',
        bonus_paid: true,
        bonus_paid_at: new Date(),
        converted_at: new Date(),
      },
    })

    // Criar transação de crédito
    await prisma.transaction.create({
      data: {
        tenant_id: order.tenant_id,
        type: 'credito',
        category: 'bonus_indicacao',
        description: `Bônus de indicação - ${referral.referred.name}`,
        amount: referral.bonus_amount,
        status: 'confirmada',
        transaction_date: new Date(),
        client_id: referral.referrer_id,
      },
    })

    // Enviar email de notificação
    await sendReferralBonusEmail({
      to: referral.referrer.email,
      referrer_name: referral.referrer.name,
      referred_name: referral.referred.name,
      bonus_amount: Number(referral.bonus_amount),
      new_balance: Number(referral.referrer.referral_balance) + Number(referral.bonus_amount),
    })

    // Log
    await prisma.auditLog.create({
      data: {
        tenant_id: order.tenant_id,
        action: 'referral_bonus_paid',
        resource: 'referral',
        resource_id: referral.id,
        metadata: {
          referrer_id: referral.referrer_id,
          referred_id: referral.referred_id,
          bonus_amount: Number(referral.bonus_amount),
          order_id: orderId,
        },
      },
    })

    return {
      processed: true,
      bonus_amount: Number(referral.bonus_amount),
      referrer_name: referral.referrer.name,
    }
  } catch (error: any) {
    console.error('Erro ao processar bônus de indicação:', error)
    throw error
  }
}

    include: {
      referral_stats: true,
    },
  })

  if (!client) {
    return <div>Cliente não encontrado</div>
  }

  // Buscar estatísticas
  const [totalOrders, pendingOrders, completedOrders, totalSpent] = await Promise.all([
    prisma.storeOrder.count({
      where: {
        client_id: client.id,
      },
    }),
    prisma.storeOrder.count({
      where: {
        client_id: client.id,
        status: 'pending',
      },
    }),
    prisma.storeOrder.count({
      where: {
        client_id: client.id,
        status: 'completed',
      },
    }),
    prisma.storeOrder.aggregate({
      where: {
        client_id: client.id,
        payment_status: 'paid',
      },
      _sum: {
        total_amount: true,
      },
    }),
  ])

  // Buscar últimos pedidos
  const recentOrders = await prisma.storeOrder.findMany({
    where: {
      client_id: client.id,
    },
    orderBy: {
      created_at: 'desc',
    },
    take: 5,
    include: {
      items: {
        take: 1,
      },
    },
  })

  // Buscar indicações
  const referralStats = await prisma.referral.aggregate({
    where: {
      referrer_id: client.id,
    },
    _count: {
      id: true,

    where: { tenant_id: session?.user.tenantId },
  })

  if (!store) {
    return null
  }

  // Data ranges
  const today = new Date()
  const startOfMonth = new Date(today.getFullYear(), today.getMonth(), 1)
  const startOfLastMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1)
  const endOfLastMonth = new Date(today.getFullYear(), today.getMonth(), 0)

  // Current month stats
  const [
    currentMonthOrders,
    currentMonthRevenue,
    pendingOrders,
    totalProducts,
    totalClients,
  ] = await Promise.all([
    // Total de pedidos do mês
    prisma.storeOrder.count({
      where: {
        store_id: store.id,
        created_at: { gte: startOfMonth },
      },
    }),
    // Receita do mês
    prisma.storeOrder.aggregate({
      where: {
        store_id: store.id,
        payment_status: 'paid',
        created_at: { gte: startOfMonth },
      },
      _sum: { total_amount: true },
    }),
    // Pedidos pendentes
    prisma.storeOrder.count({
      where: {
        store_id: store.id,
        status: 'pending',
      },
    }),
    // Total de produtos
    prisma.storeProduct.count({
      where: {
        store_id: store.id,
        status: 'active',
      },
    }),
    // Total de clientes
    prisma.client.count({
      where: {
        tenant_id: store.tenant_id,
        source: 'store',
      },
    }),
  ])

  // Last month stats for comparison
  const [lastMonthOrders, lastMonthRevenue] = await Promise.all([
    prisma.storeOrder.count({
      where: {
        store_id: store.id,
        created_at: {
          gte: startOfLastMonth,
          lte: endOfLastMonth,
        },
      },
    }),
    prisma.storeOrder.aggregate({
      where: {
        store_id: store.id,
        payment_status: 'paid',
        created_at: {
          gte: startOfLastMonth,
          lte: endOfLastMonth,
        },
      },
      _sum: { total_amount: true },
    }),
  ])

  // Calculate growth
  const ordersGrowth = lastMonthOrders > 0
    ? ((currentMonthOrders - lastMonthOrders) / lastMonthOrders) * 100
    : 0

  const revenueGrowth = Number(lastMonthRevenue._sum.total_amount || 0) > 0
    ? ((Number(currentMonthRevenue._sum.total_amount || 0) -
        Number(lastMonthRevenue._sum.total_amount || 0)) /
        Number(lastMonthRevenue._sum.total_amount || 0)) *
      100
    : 0

  const recentOrders = await prisma.storeOrder.findMany({
    where: { store_id: store.id },

    take: 10,
    include: {
      client: true,
      items: {
        take: 1,
      },

  })

      status: 'active',
    },
    orderBy: {
      sales_count: 'desc',
    },
    take: 5,
  })

              Ver Loja
            </Link>
          </Button>

            <Link href="/dashboard/loja/produtos/novo">
              Novo Produto
            </Link>
          </Button>
        </div>

        {/* Total Revenue */}
        <Card>
          <CardContent className="pt-6">
            <div className="flex items-center justify-between mb-4">
              <div className="w-12 h-12 bg-green-100 rounded-full flex items-center justify-center">
                <DollarSign className="h-6 w-6 text-green-600" />
              </div>
              {revenueGrowth !== 0 && (

'use client'

import { useEffect, useState } from 'react'
import {
  LineChart,
  Line,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  ResponsiveContainer,
} from 'recharts'

interface RevenueChartProps {
  storeId: string
}

export function RevenueChart({ storeId }: RevenueChartProps) {
  const [data, setData] = useState<any[]>([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    loadData()
  }, [storeId])

  const loadData = async () => {
    try {
      const response = await fetch(`/api/dashboard/store/revenue?storeId=${storeId}`)
      const result = await response.json()
      setData(result.data || [])
    } catch (error) {
      console.error('Erro ao carregar dados:', error)
    } finally {
      setLoading(false)
    }
  }

  if (loading) {
    return (
      <div className="h-[300px] flex items-center justify-center">
        <p className="text-gray-600">Carregando...</p>
      </div>
    )
  }

  if (data.length === 0) {
    return (
      <div className="h-[300px] flex items-center justify-center">
        <p className="text-gray-600">Nenhum dado disponível</p>
      </div>
    )
  }

  return (
    <ResponsiveContainer width="100%" height={300}>
      <LineChart data={data}>
        <CartesianGrid strokeDasharray="3 3" />
        <XAxis dataKey="date" />
        <YAxis />
        <Tooltip
          formatter={(value: any) =>
            `R$ ${Number(value).toLocaleString('pt-BR', {
              minimumFractionDigits: 2,
            })}`
          }
        />
        <Line
          type="monotone"
          dataKey="revenue"
          stroke="#3b82f6"
          strokeWidth={2}
          dot={{ r: 4 }}
        />
      </LineChart>

  )
}

export default async function OrdersPage({ searchParams }: OrdersPageProps) {
  const session = await getServerSession(authOptions)

  const store = await prisma.store.findUnique({
    where: { tenant_id: session?.user.tenantId },
  })

  if (!store) {
    return null
  }

  // Build filters
  const page = parseInt(searchParams.page || '1')
  const limit = 20
  const skip = (page - 1) * limit

  const where: any = {
    store_id: store.id,
  }

  if (searchParams.search) {
    where.OR = [
      { order_number: { contains: searchParams.search, mode: 'insensitive' } },
      { customer_name: { contains: searchParams.search, mode: 'insensitive' } },
      { customer_email: { contains: searchParams.search, mode: 'insensitive' } },
    ]
  }

  if (searchParams.status && searchParams.status !== 'all') {
    where.status = searchParams.status
  }

  if (searchParams.payment_status && searchParams.payment_status !== 'all') {
    where.payment_status = searchParams.payment_status
  }

  if (searchParams.date_from) {
    where.created_at = {
      ...where.created_at,
      gte: new Date(searchParams.date_from),
    }
  }

  if (searchParams.date_to) {
    where.created_at = {
      ...where.created_at,
      lte: new Date(searchParams.date_to),
    }
  }

  // Fetch orders
  const [orders, total] = await Promise.all([
    prisma.storeOrder.findMany({
      where,
      skip,
      take: limit,
      orderBy: { created_at: 'desc' },
      include: {
        client: true,
        items: {
          include: {
            product: true,
          },
        },
      },
    }),
    prisma.storeOrder.count({ where }),
  ])

    _count: { id: true },

  })

    statusStats.find((s) => s.status === 'cancelled')?._count.id || 0

        return <Clock className="h-4 w-4 text-yellow-600" />
      case 'cancelled':
        return <XCircle className="h-4 w-4 text-red-600" />
      default:
        return null
    }
  }

  const getStatusColor = (status: string) => {
    switch (status) {
      case 'converted':
        return 'bg-green-100 text-green-800'
      case 'pending':
        return 'bg-yellow-100 text-yellow-800'
      case 'cancelled':
        return 'bg-red-100 text-red-800'
      default:
        return 'bg-gray-100 text-gray-800'
    }
  }

  const getStatusLabel = (status: string) => {
    switch (status) {
      case 'converted':
        return 'Convertida'
      case 'pending':
        return 'Pendente'
      case 'cancelled':
        return 'Cancelada'
      default:
        return status
    }
  }

  return (
    <div className="overflow-x-auto">
      <table className="w-full">
        <thead>
          <tr className="border-b">

// app/api/dashboard/referral/chart/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { subDays, format } from 'date-fns'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const { searchParams } = new URL(req.url)
    const days = parseInt(searchParams.get('days') || '30')

    const startDate = subDays(new Date(), days)

    // Buscar indicações por dia
    const referrals = await prisma.referral.findMany({
      where: {
        tenant_id: session.user.tenantId,
        created_at: {
          gte: startDate,
        },
      },
      orderBy: {
        created_at: 'asc',
      },
    })

    // Agrupar por dia
    const dataMap = new Map<string, { total: number; converted: number }>()

    for (let i = 0; i < days; i++) {
      const date = format(subDays(new Date(), days - i - 1), 'yyyy-MM-dd')
      dataMap.set(date, { total: 0, converted: 0 })
    }

    referrals.forEach((referral) => {
      const date = format(new Date(referral.created_at), 'yyyy-MM-dd')
      const current = dataMap.get(date) || { total: 0, converted: 0 }
      
      current.total++
      if (referral.status === 'converted') {
        current.converted++
      }
      
      dataMap.set(date, current)
    })

    // Converter para array
    const data = Array.from(dataMap.entries()).map(([date, values]) => ({
      date: format(new Date(date), 'dd/MM'),
      total: values.total,
      converted: values.converted,
    }))

    return NextResponse.json({ data })
  } catch (error: any) {
    console.error('Erro ao buscar dados do gráfico:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar dados' },
      { status: 500 }
    )
  }
}

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const body = await req.json()
    const {
      is_active,
      bonus_amount,
      discount_percentage,
      min_purchase_amount,
      max_discount_amount,
      share_message,
      terms,
    } = body

    // Validações
    if (bonus_amount < 0) {
      return NextResponse.json(
        { error: 'Bônus não pode ser negativo' },
        { status: 400 }
      )
    }

    if (discount_percentage < 0 || discount_percentage > 100) {
      return NextResponse.json(
        { error: 'Desconto deve estar entre 0 e 100%' },
        { status: 400 }
      )
    }

    // Criar ou atualizar configurações
    const settings = await prisma.referralSettings.upsert({
      where: { tenant_id: session.user.tenantId },
      create: {
        tenant_id: session.user.tenantId,
        is_active,
        bonus_amount,
        discount_percentage,
        min_purchase_amount,
        max_discount_amount,
        share_message,
        terms,
      },
      update: {
        is_active,
        bonus_amount,
        discount_percentage,
        min_purchase_amount,
        max_discount_amount,
        share_message,
        terms,
      },
    })

    // Log de auditoria
    await prisma.auditLog.create({
      data: {
        tenant_id: session.user.tenantId,
        user_id: session.user.id,
        action: 'update',
        resource: 'referral_settings',
        resource_id: settings.id,
        metadata: {
          is_active,
          bonus_amount,
          discount_percentage,
        },
      },
    })

    return NextResponse.json({ success: true, settings })
  } catch (error: any) {
    console.error('Erro ao salvar configurações:', error)
    return NextResponse.json(
      { error: 'Erro ao salvar configurações' },
      { status: 500 }
    )
  }
}

// app/api/dashboard/referral/pay-bonus/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { sendReferralBonusEmail } from '@/lib/emails/referral-bonus'

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const { referral_id } = await req.json()

    // Buscar indicação
    const referral = await prisma.referral.findUnique({
      where: { id: referral_id },
      include: {
        referrer: true,
        referred: true,
      },
    })

    if (!referral) {
      return NextResponse.json(
        { error: 'Indicação não encontrada' },
        { status: 404 }
      )
    }

    // Verificar se já foi pago
    if (referral.bonus_paid) {
      return NextResponse.json(
        { error: 'Bônus já foi pago' },
        { status: 400 }
      )
    }

    // Verificar se está convertida
    if (referral.status !== 'converted') {
      return NextResponse.json(
        { error: 'Indicação ainda não foi convertida' },
        { status: 400 }
      )
    }

    // Creditar bônus
    await prisma.client.update({
      where: { id: referral.referrer_id },
      data: {
        referral_balance: {
          increment: referral.bonus_amount,
        },
      },
    })

    // Atualizar indicação
    await prisma.referral.update({
      where: { id: referral_id },
      data: {
        bonus_paid: true,
        bonus_paid_at: new Date(),
      },
    })

    // Criar transação
    await prisma.transaction.create({
      data: {
        tenant_id: referral.tenant_id,
        type: 'credito',
        category: 'bonus_indicacao',
        description: `Bônus de indicação - ${referral.referred?.name || referral.referee_name}`,
        amount: referral.bonus_amount,
        status: 'confirmada',
        transaction_date: new Date(),
        client_id: referral.referrer_id,
      },
    })

    // Enviar email
    await sendReferralBonusEmail({
      to: referral.referrer.email,
      referrer_name: referral.referrer.name,
      referred_name: referral.referred?.name || referral.referee_name,
      bonus_amount: Number(referral.bonus_amount),
      new_balance:
        Number(referral.referrer.referral_balance) + Number(referral.bonus_amount),
    })

    // Log
    await prisma.auditLog.create({
      data: {
        tenant_id: session.user.tenantId,
        user_id: session.user.id,
        action: 'referral_bonus_paid_manual',
        resource: 'referral',
        resource_id: referral_id,
        metadata: {
          referrer_id: referral.referrer_id,
          bonus_amount: Number(referral.bonus_amount),
        },
      },
    })

    return NextResponse.json({ success: true })
  } catch (error: any) {
    console.error('Erro ao processar pagamento:', error)
    return NextResponse.json(
      { error: 'Erro ao processar pagamento' },
      { status: 500 }
    )
  }
}

// components/dashboard/store/referral-chart.tsx

'use client'

import { useEffect, useState } from 'react'
import {
  LineChart,
  Line,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  Legend,
  ResponsiveContainer,
} from 'recharts'

interface ReferralChartProps {
  tenantId: string
}

export function ReferralChart({ tenantId }: ReferralChartProps) {
  const [data, setData] = useState<any[]>([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    loadData()
  }, [tenantId])

  const loadData = async () => {
    try {
      const response = await fetch('/api/dashboard/referral/chart?days=30')
      const result = await response.json()
      setData(result.data || [])
    } catch (error) {
      console.error('Erro ao carregar dados:', error)
    } finally {
      setLoading(false)
    }
  }

  if (loading) {
    return (
      <div className="h-[300px] flex items-center justify-center">
        <p className="text-gray-600">Carregando...</p>
      </div>
    )
  }

  if (data.length === 0) {
    return (
      <div className="h-[300px] flex items-center justify-center">
        <p className="text-gray-600">Nenhum dado disponível</p>
      </div>
    )
  }

  return (
    <ResponsiveContainer width="100%" height={300}>
      <LineChart data={data}>
        <CartesianGrid strokeDasharray="3 3" />
        <XAxis dataKey="date" />
        <YAxis />
        <Tooltip />
        <Legend />
        <Line
          type="monotone"
          dataKey="total"
          stroke="#3b82f6"
          strokeWidth={2}
          name="Total de Indicações"
        />
        <Line
          type="monotone"
          dataKey="converted"
          stroke="#10b981"
          strokeWidth={2}
          name="Convertidas"
        />
      </LineChart>
    </ResponsiveContainer>
  )
}

// components/dashboard/store/order-timeline.tsx

import { CheckCircle, Clock, Package, Truck, XCircle } from 'lucide-react'
import { formatDistanceToNow } from 'date-fns'
import { ptBR } from 'date-fns/locale'

interface OrderTimelineProps {
  order: {
    created_at: Date
    confirmed_at?: Date | null
    paid_at?: Date | null
    completed_at?: Date | null
    cancelled_at?: Date | null
    status: string
    payment_status: string
  }
}

export function OrderTimeline({ order }: OrderTimelineProps) {
  const events = [
    {
      title: 'Pedido Criado',
      date: order.created_at,
      icon: Package,
      color: 'blue',
      completed: true,
    },
    {
      title: 'Pagamento Confirmado',
      date: order.paid_at,
      icon: CheckCircle,
      color: 'green',
      completed: !!order.paid_at,
    },
    {
      title: 'Pedido Confirmado',
      date: order.confirmed_at,
      icon: CheckCircle,
      color: 'green',
      completed: !!order.confirmed_at,
    },
    {
      title: 'Concluído',
      date: order.completed_at,
      icon: CheckCircle,
      color: 'green',
      completed: !!order.completed_at,
    },
  ]

  // Se cancelado, adicionar evento
  if (order.cancelled_at) {
    events.push({
      title: 'Pedido Cancelado',
      date: order.cancelled_at,
      icon: XCircle,
      color: 'red',
      completed: true,
    })
  }

  return (
    <div className="space-y-4">
      {events.map((event, index) => (
        <div key={index} className="flex gap-4">
          <div className="flex flex-col items-center">
            <div
              className={`w-10 h-10 rounded-full flex items-center justify-center ${
                event.completed
                  ? event.color === 'red'
                    ? 'bg-red-100'
                    : event.color === 'green'
                    ? 'bg-green-100'
                    : 'bg-blue-100'
                  : 'bg-gray-100'
              }`}
            >
              <event.icon
                className={`h-5 w-5 ${
                  event.completed
                    ? event.color === 'red'
                      ? 'text-red-600'
                      : event.color === 'green'
                      ? 'text-green-600'
                      : 'text-blue-600'
                    : 'text-gray-400'
                }`}
              />
            </div>
            {index < events.length - 1 && (
              <div
                className={`w-0.5 h-12 ${
                  event.completed ? 'bg-green-200' : 'bg-gray-200'
                }`}
              />
            )}
          </div>
          <div className="flex-1 pb-8">
            <p
              className={`font-medium ${
                event.completed ? 'text-gray-900' : 'text-gray-400'
              }`}
            >
              {event.title}
            </p>
            {event.date && (
              <p className="text-sm text-gray-600">
                {formatDistanceToNow(new Date(event.date), {
                  addSuffix: true,
                  locale: ptBR,
                })}
              </p>
            )}
          </div>
        </div>
      ))}
    </div>
  )
}
