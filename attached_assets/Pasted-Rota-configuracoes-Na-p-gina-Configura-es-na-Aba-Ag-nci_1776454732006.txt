'use client'

import { useState } from 'react'
import { UploadDropzone } from '@/lib/uploadthing'
import { Card } from '@/components/ui/card'
import { Button } from '@/components/ui/button'
import { Alert, AlertDescription } from '@/components/ui/alert'
import {
  X,
  CheckCircle,
  AlertCircle,
  Loader2,
  Upload,
  Image as ImageIcon,
} from 'lucide-react'
import Image from 'next/image'

interface LogoUploadProps {
  value?: string
  valueKey?: string
  onChange: (url: string, key: string) => void
  disabled?: boolean
  label?: string
  description?: string
}

export function LogoUpload({
  value,
  valueKey,
  onChange,
  disabled,
  label = 'Logo da Agência',
  description = 'Faça upload do logo da sua agência (máx. 2MB)',
}: LogoUploadProps) {
  const [isUploading, setIsUploading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const handleRemove = () => {
    onChange('', '')
    setError(null)
  }

  return (
    <div className="space-y-4">
      {/* Header */}
      <div>
        <label className="block text-sm font-medium mb-2">
          {label}
        </label>
        <p className="text-sm text-gray-600">{description}</p>
      </div>

      {/* Preview do Logo */}
      {value ? (
        <Card className="relative overflow-hidden border-2 border-green-200 bg-green-50">
          {/* Preview da Imagem */}
          <div className="relative h-48 w-full bg-white flex items-center justify-center p-6">
            <div className="relative h-32 w-full">
              <Image
                src={value}
                alt="Logo"
                fill
                className="object-contain"
                priority
              />
            </div>
          </div>

          {/* Botão Remover */}
          <div className="absolute top-3 right-3">
            <Button
              type="button"
              variant="destructive"
              size="sm"
              onClick={handleRemove}
              disabled={disabled || isUploading}
              className="shadow-lg"
            >
              <X className="h-4 w-4 mr-1" />
              Remover
            </Button>
          </div>

          {/* Status */}
          <div className="p-4 bg-green-50 border-t-2 border-green-200">
            <div className="flex items-center gap-2 text-green-700">
              <CheckCircle className="h-5 w-5" />
              <span className="font-semibold">✅ Logo carregado com sucesso</span>
            </div>
          </div>
        </Card>
      ) : (
        <>
          {/* Área de Upload */}
          <Card className="p-6 border-2 border-dashed border-gray-300 hover:border-blue-400 transition bg-gray-50">
            <UploadDropzone
              endpoint="agencyLogo"
              onClientUploadComplete={(res) => {
                if (res && res[0]) {
                  console.log('✅ Upload de logo completo:', res[0])
                  onChange(res[0].url, res[0].key || '')
                  setError(null)
                }
                setIsUploading(false)
              }}
              onUploadError={(error: Error) => {
                console.error('❌ Erro no upload:', error)
                setError(error.message)
                setIsUploading(false)
              }}
              onUploadBegin={() => {
                console.log('🔄 Iniciando upload do logo...')
                setIsUploading(true)
                setError(null)
              }}
              appearance={{
                container: 'w-full cursor-pointer',
                uploadIcon: 'text-blue-600 w-12 h-12',
                label: 'text-blue-600 hover:text-blue-700 font-bold text-base mt-4',
                allowedContent: 'text-gray-500 text-sm mt-2',
                button:
                  'bg-blue-600 hover:bg-blue-700 text-white px-6 py-2 rounded-lg font-semibold transition ut-ready:bg-blue-600 ut-uploading:bg-blue-400',
              }}
              content={{
                label: isUploading
                  ? 'Enviando logo...'
                  : 'Clique ou arraste o logo aqui',
                allowedContent: 'PNG, JPG, JPEG ou WEBP (máx. 2MB)',
                button: isUploading ? (
                  <span className="flex items-center gap-2">
                    <Loader2 className="h-4 w-4 animate-spin" />
                    Enviando...
                  </span>
                ) : (
                  'Selecionar arquivo'
                ),
              }}
            />
          </Card>

          {/* Erro */}
          {error && (
            <Alert variant="destructive">
              <AlertCircle className="h-4 w-4" />
              <AlertDescription>
                <strong>Erro ao fazer upload:</strong> {error}
              </AlertDescription>
            </Alert>
          )}
        </>
      )}

      {/* Loading State */}
      {isUploading && (
        <div className="flex items-center justify-center gap-3 p-4 bg-blue-50 rounded-lg border-2 border-blue-200 animate-pulse">
          <Loader2 className="h-5 w-5 animate-spin text-blue-600" />
          <span className="text-sm font-semibold text-blue-700">
            Fazendo upload do logo... Por favor, aguarde.
          </span>

// app/api/dashboard/settings/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { hasPermission } from '@/lib/permissions/check-permissions'

// GET - Buscar configurações
export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const settings = await prisma.tenant.findUnique({
      where: {
        id: session.user.tenantId,
      },
      select: {
        name: true,
        email: true,
        phone: true,
        whatsapp: true,
        address: true,
        logo_url: true,
        logo_key: true,
        primary_color: true,
        secondary_color: true,
      },
    })

    if (!settings) {
      return NextResponse.json(
        { error: 'Configurações não encontradas' },
        { status: 404 }
      )
    }

    return NextResponse.json({ settings })
  } catch (error) {
    console.error('Erro ao buscar configurações:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar configurações' },
      { status: 500 }
    )
  }
}

// PUT - Atualizar configurações
export async function PUT(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    if (!hasPermission(session.user.role as any, 'settings.edit')) {
      return NextResponse.json(
        { error: 'Sem permissão para editar configurações' },
        { status: 403 }
      )
    }

    const body = await req.json()

    console.log('📝 Atualizando configurações:', body)

    const settings = await prisma.tenant.update({
      where: {
        id: session.user.tenantId,
      },
      data: {
        name: body.name,
        email: body.email,
        phone: body.phone,
        whatsapp: body.whatsapp,
        address: body.address,
        
        // LOGO
        logo_url: body.logo_url,
        logo_key: body.logo_key,
        
        // Personalização
        primary_color: body.primary_color,
        secondary_color: body.secondary_color,
        
        updated_at: new Date(),
      },
    })

    console.log('✅ Configurações atualizadas:', settings)

    return NextResponse.json({
      success: true,
      settings,
      message: 'Configurações atualizadas com sucesso!',
    })
  } catch (error: any) {
    console.error('❌ Erro ao atualizar configurações:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao atualizar configurações' },
      { status: 500 }
    )
  }
}
