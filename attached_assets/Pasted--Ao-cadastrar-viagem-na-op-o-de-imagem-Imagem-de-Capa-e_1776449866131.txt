// app/api/uploadthing/core.ts

import { createUploadthing, type FileRouter } from 'uploadthing/next'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'

const f = createUploadthing()

export const ourFileRouter = {
  // ═══════════════════════════════════════════════════════════
  // UPLOAD DE IMAGEM DE CAPA (1 arquivo, máx 4MB)
  // ═══════════════════════════════════════════════════════════
  tripCoverImage: f({
    image: {
      maxFileSize: '4MB',
      maxFileCount: 1,
    },
  })
    .middleware(async () => {
      // Autenticação
      const session = await getServerSession(authOptions)

      if (!session) {
        throw new Error('Não autorizado')
      }

      console.log('✅ Usuário autenticado:', session.user.name)

      return {
        userId: session.user.id,
        tenantId: session.user.tenantId,
      }
    })
    .onUploadComplete(async ({ metadata, file }) => {
      console.log('✅ Upload de capa completo!')
      console.log('📁 Arquivo:', file.name)
      console.log('🔗 URL:', file.url)
      console.log('👤 Usuário:', metadata.userId)

      return {
        url: file.url,
        key: file.key,
        name: file.name,
      }
    }),

  // ═══════════════════════════════════════════════════════════

      console.log('✅ Upload de galeria completo!')

'use client'

import { UploadButton } from '@/lib/uploadthing'
import { useState } from 'react'

export function SimpleUploadButton() {
  const [uploadedUrl, setUploadedUrl] = useState<string>('')

  return (
    <div className="flex flex-col items-center gap-4 p-8">
      <h2 className="text-2xl font-bold">Upload de Imagem</h2>

      <UploadButton
        endpoint="tripCoverImage"
        onClientUploadComplete={(res) => {
          if (res && res[0]) {
            console.log('✅ Upload completo:', res)
            setUploadedUrl(res[0].url)
            alert('Imagem enviada com sucesso!')
          }
        }}
        onUploadError={(error: Error) => {
          console.error('❌ Erro no upload:', error)
          alert(`Erro: ${error.message}`)
        }}
        appearance={{
          button:
            'bg-blue-600 hover:bg-blue-700 text-white px-6 py-3 rounded-lg font-semibold transition ut-ready:bg-blue-600 ut-uploading:bg-blue-400',
          allowedContent: 'text-gray-500 text-sm',
        }}
      />

      {uploadedUrl && (
        <div className="mt-4 p-4 bg-green-50 border border-green-200 rounded-lg">
          <p className="text-green-700 font-semibold mb-2">
            ✅ Upload bem-sucedido!
          </p>
          <img
            src={uploadedUrl}
            alt="Uploaded"
            className="max-w-md rounded-lg shadow-lg"
          />
          <p className="text-xs text-gray-600 mt-2 break-all">{uploadedUrl}</p>
        </div>
      )}
    </div>
  )

  onChange,
  disabled,
}: CoverImageUploadProps) {

// components/dashboard/trips/gallery-upload.tsx

'use client'

import { useState } from 'react'
import { UploadDropzone } from '@/lib/uploadthing'
import { Card } from '@/components/ui/card'
import { Button } from '@/components/ui/button'
import { Badge } from '@/components/ui/badge'
import { Alert, AlertDescription } from '@/components/ui/alert'
import {
  X,
  CheckCircle,
  AlertCircle,
  Loader2,
  Images,
} from 'lucide-react'
import Image from 'next/image'

interface GalleryImage {
  url: string
  key: string
  order: number
}

interface GalleryUploadProps {
  value: GalleryImage[]
  onChange: (images: GalleryImage[]) => void
  disabled?: boolean
  maxImages?: number
}

export function GalleryUpload({
  value = [],
  onChange,
  disabled,
  maxImages = 3,
}: GalleryUploadProps) {
  const [isUploading, setIsUploading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const handleRemove = (index: number) => {
    const newImages = value.filter((_, i) => i !== index)
    // Reordenar
    const reordered = newImages.map((img, i) => ({ ...img, order: i + 1 }))
    onChange(reordered)
    setError(null)
  }

  const canUploadMore = value.length < maxImages

  return (
    <div className="space-y-4">
      {/* Header */}
      <div>
        <label className="block text-sm font-medium mb-2">
          Galeria de Imagens
        </label>
        <p className="text-sm text-gray-600">
          Faça upload de até {maxImages} imagens para a galeria (máx. 4MB cada)
        </p>
      </div>

      {/* Preview das Imagens */}
      {value.length > 0 && (
        <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
          {value.map((image, index) => (
            <Card
              key={index}
              className="relative overflow-hidden border-2 border-green-200 group"
            >
              <div className="relative h-48 w-full bg-gray-100">
                <Image
                  src={image.url}
                  alt={`Imagem ${index + 1}`}
                  fill
                  className="object-cover"
                />
              </div>

              {/* Overlay com botão remover */}
              <div className="absolute inset-0 bg-black/0 group-hover:bg-black/60 transition flex items-center justify-center">
                <Button
                  type="button"
                  variant="destructive"
                  size="sm"
                  onClick={() => handleRemove(index)}
                  disabled={disabled || isUploading}
                  className="opacity-0 group-hover:opacity-100 transition shadow-xl"
                >
                  <X className="h-4 w-4 mr-1" />
                  Remover
                </Button>
              </div>

              {/* Badge com número */}
              <div className="absolute top-2 left-2">
                <Badge className="bg-blue-600 text-white shadow-lg font-bold">
                  #{index + 1}
                </Badge>
              </div>

              {/* Status */}
              <div className="p-2 bg-green-50 border-t border-green-200">
                <div className="flex items-center gap-2 text-green-700 text-xs">
                  <CheckCircle className="h-3 w-3" />
                  <span className="font-semibold">Carregada</span>
                </div>
              </div>
            </Card>
          ))}
        </div>
      )}

      {/* Área de Upload */}
      {canUploadMore && (
        <>
          <Card className="p-8 border-2 border-dashed border-gray-300 hover:border-blue-400 transition bg-gray-50">
            <UploadDropzone
              endpoint="tripGalleryImages"
              onClientUploadComplete={(res) => {
                if (res) {
                  console.log('✅ Upload de galeria completo:', res)
                  const newImages = res.map((file, index) => ({
                    url: file.url,
                    key: file.key || '',
                    order: value.length + index + 1,
                  }))

                  onChange([...value, ...newImages])
                  setError(null)
                }
                setIsUploading(false)
              }}
              onUploadError={(error: Error) => {
                console.error('❌ Erro no upload:', error)
                setError(error.message)
                setIsUploading(false)
              }}
              onUploadBegin={() => {
                console.log('🔄 Iniciando upload de galeria...')
                setIsUploading(true)
                setError(null)
              }}
              appearance={{
                container: 'w-full cursor-pointer',
                uploadIcon: 'text-blue-600 w-12 h-12',
                label: 'text-blue-600 hover:text-blue-700 font-bold mt-4',
                allowedContent: 'text-gray-500 text-sm mt-2',
                button:
                  'bg-blue-600 hover:bg-blue-700 text-white px-6 py-3 rounded-lg font-semibold transition ut-ready:bg-blue-600 ut-uploading:bg-blue-400',
              }}
              content={{
                label: isUploading
                  ? 'Enviando imagens...'
                  : `Adicionar imagens (${value.length}/${maxImages})`,
                allowedContent: `PNG, JPG, JPEG ou WEBP (máx. 4MB cada) • ${maxImages - value.length} restante${maxImages - value.length !== 1 ? 's' : ''}`,
                button: isUploading ? (
                  <span className="flex items-center gap-2">
                    <Loader2 className="h-4 w-4 animate-spin" />
                    Enviando...
                  </span>
                ) : (
                  'Selecionar arquivos'
                ),
              }}
            />
          </Card>

          {/* Erro */}
          {error && (
            <Alert variant="destructive">
              <AlertCircle className="h-4 w-4" />
              <AlertDescription>
                <strong>Erro ao fazer upload:</strong> {error}
              </AlertDescription>
            </Alert>
          )}
        </>
      )}

      {/* Limite Atingido */}
      {!canUploadMore && (
        <div className="text-center p-6 bg-green-50 rounded-lg border-2 border-green-200">
          <CheckCircle className="h-10 w-10 text-green-600 mx-auto mb-3" />
          <p className="font-bold text-green-700 text-lg">
            ✅ Limite de {maxImages} imagens atingido
          </p>
          <p className="text-sm text-green-600 mt-2">
            Remova uma imagem para adicionar outra
          </p>
        </div>
      )}

      {/* Loading State */}
      {isUploading && (
        <div className="flex items-center justify-center gap-3 p-4 bg-blue-50 rounded-lg border-2 border-blue-200 animate-pulse">
          <Loader2 className="h-5 w-5 animate-spin text-blue-600" />
          <span className="text-sm font-semibold text-blue-700">
            Fazendo upload das imagens... Por favor, aguarde.
          </span>
        </div>
      )}
    </div>
  )
}

// app/api/dashboard/trips/route.ts

import { NextResponse } from 'next/response'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { hasPermission } from '@/lib/permissions/check-permissions'

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    if (!hasPermission(session.user.role as any, 'trips.create')) {
      return NextResponse.json(
        { error: 'Sem permissão para criar viagens' },
        { status: 403 }
      )
    }

    const body = await req.json()

    console.log('📥 Dados recebidos:', body)

    // Validações
    if (!body.title || !body.destination) {
      return NextResponse.json(
        { error: 'Título e destino são obrigatórios' },
        { status: 400 }
      )
    }

    if (!body.image_url) {
      return NextResponse.json(
        { error: 'Imagem de capa é obrigatória' },
        { status: 400 }
      )
    }

    if (!body.price_per_person) {
      return NextResponse.json(
        { error: 'Preço por pessoa é obrigatório' },
        { status: 400 }
      )
    }

    const trip = await prisma.trip.create({
      data: {
        tenant_id: session.user.tenantId,
        created_by: session.user.id,
        
        // Dados básicos
        title: body.title,
        destination: body.destination,
        description: body.description || null,
        
        // IMAGENS
        image_url: body.image_url,
        image_key: body.image_key || null,
        gallery_images: body.gallery_images || [], // JSON array
        
        // Preços
        price_per_person: parseFloat(body.price_per_person),
        promotional_price: body.promotional_price
          ? parseFloat(body.promotional_price)
          : null,
        
        // Datas
        departure_date: body.departure_date
          ? new Date(body.departure_date)
          : null,
        return_date: body.return_date ? new Date(body.return_date) : null,
        
        // Detalhes
        duration_days: body.duration_days
          ? parseInt(body.duration_days)
          : null,
        max_passengers: body.max_passengers
          ? parseInt(body.max_passengers)
          : null,
        
        // Status
        status: 'active',
      },
    })

    console.log('✅ Viagem criada:', trip)

    return NextResponse.json({
      success: true,
      trip,
      message: 'Viagem criada com sucesso!',
    })
  } catch (error: any) {
    console.error('❌ Erro ao criar viagem:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao criar viagem' },
      { status: 500 }
    )
  }
}

// components/dashboard/trips/cover-image-upload-with-preview.tsx

'use client'

import { useState, useRef } from 'react'
import { UploadDropzone } from '@/lib/uploadthing'
import { Card } from '@/components/ui/card'
import { Button } from '@/components/ui/button'
import { Alert, AlertDescription } from '@/components/ui/alert'
import {
  X,
  Upload,
  CheckCircle,
  AlertCircle,
  Loader2,
  Eye,
  Trash2,
} from 'lucide-react'
import Image from 'next/image'

interface CoverImageUploadProps {
  value?: string
  valueKey?: string
  onChange: (url: string, key: string) => void
  disabled?: boolean
  onRemove?: () => Promise<void>
}

export function CoverImageUploadWithPreview({
  value,
  valueKey,
  onChange,
  disabled,
  onRemove,
}: CoverImageUploadProps) {
  const [isUploading, setIsUploading] = useState(false)
  const [isRemoving, setIsRemoving] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const [previewUrl, setPreviewUrl] = useState<string | null>(null)
  const fileInputRef = useRef<HTMLInputElement>(null)

  // Preview local antes do upload
  const handleFileSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return

    // Validar tamanho
    if (file.size > 4 * 1024 * 1024) {
      setError('Arquivo muito grande. Máximo: 4MB')
      return
    }

    // Validar tipo
    if (!file.type.startsWith('image/')) {
      setError('Arquivo deve ser uma imagem')
      return
    }

    // Criar preview local
    const reader = new FileReader()
    reader.onloadend = () => {
      setPreviewUrl(reader.result as string)
      setError(null)
    }
    reader.readAsDataURL(file)
  }

  const handleRemove = async () => {
    if (!value || !valueKey) return

    try {
      setIsRemoving(true)

      // Chamar API para remover do UploadThing
      const response = await fetch('/api/uploadthing/delete', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ key: valueKey }),
      })

      if (response.ok) {
        onChange('', '')
        setPreviewUrl(null)
        setError(null)

        if (onRemove) {
          await onRemove()
        }
      } else {
        const data = await response.json()
        setError(data.error || 'Erro ao remover imagem')
      }
    } catch (error) {
      console.error('❌ Erro ao remover:', error)
      setError('Erro ao remover imagem')
    } finally {
      setIsRemoving(false)
    }
  }

  const clearPreview = () => {
    setPreviewUrl(null)
    if (fileInputRef.current) {
      fileInputRef.current.value = ''
    }
  }

  return (
    <div className="space-y-4">
      {/* Header */}
      <div>
        <label className="block text-sm font-medium mb-2">
          Imagem de Capa <span className="text-red-500">*</span>
        </label>
        <p className="text-sm text-gray-600">
          Faça upload de uma imagem para a capa da viagem (máx. 4MB)
        </p>
      </div>

      {/* Preview da Imagem Carregada */}
      {value && !previewUrl ? (
        <Card className="relative overflow-hidden border-2 border-green-200 bg-green-50">
          <div className="relative h-64 w-full bg-gray-100">
            <Image
              src={value}
              alt="Imagem de capa"
              fill
              className="object-cover"
              priority
            />
          </div>

          {/* Overlay com ações */}
          <div className="absolute inset-0 bg-black/0 hover:bg-black/60 transition flex items-center justify-center gap-3 opacity-0 hover:opacity-100">
            <Button
              type="button"
              variant="secondary"
              size="lg"
              onClick={() => window.open(value, '_blank')}
              disabled={disabled || isRemoving}
              className="shadow-xl"
            >
              <Eye className="h-5 w-5 mr-2" />
              Visualizar
            </Button>

            <Button
              type="button"
              variant="destructive"
              size="lg"
              onClick={handleRemove}
              disabled={disabled || isRemoving}
              className="shadow-xl"
            >
              {isRemoving ? (
                <>
                  <Loader2 className="h-5 w-5 mr-2 animate-spin" />
                  Removendo...
                </>
              ) : (
                <>
                  <Trash2 className="h-5 w-5 mr-2" />
                  Remover
                </>
              )}
            </Button>
          </div>

          {/* Status */}
          <div className="p-4 bg-green-50 border-t-2 border-green-200">
            <div className="flex items-center gap-2 text-green-700">
              <CheckCircle className="h-5 w-5" />
              <span className="font-semibold">
                ✅ Imagem de capa carregada
              </span>
            </div>
          </div>
        </Card>
      ) : previewUrl ? (
        // Preview Local (antes do upload)
        <Card className="relative overflow-hidden border-2 border-blue-200 bg-blue-50">
          <div className="relative h-64 w-full bg-gray-100">
            <Image
              src={previewUrl}
              alt="Preview"
              fill
              className="object-cover"
            />
          </div>

          {/* Badge de Preview */}
          <div className="absolute top-3 left-3">
            <div className="bg-blue-600 text-white px-3 py-1 rounded-full text-sm font-semibold shadow-lg">
              👁️ Preview Local
            </div>
          </div>

          {/* Botão Cancelar Preview */}
          <div className="absolute top-3 right-3">
            <Button
              type="button"
              variant="destructive"
              size="sm"
              onClick={clearPreview}
              disabled={disabled}
              className="shadow-lg"
            >
              <X className="h-4 w-4 mr-1" />
              Cancelar
            </Button>
          </div>

          {/* Status */}
          <div className="p-4 bg-blue-50 border-t-2 border-blue-200">
            <div className="flex items-center gap-2 text-blue-700">
              <Eye className="h-5 w-5" />
              <span className="font-semibold">
                Preview da imagem (não salva ainda)
              </span>
            </div>
          </div>
        </Card>
      ) : (
        <>
          {/* Input de arquivo oculto para preview */}
          <input
            ref={fileInputRef}
            type="file"
            accept="image/*"
            onChange={handleFileSelect}
            className="hidden"
          />

          {/* Botão para Preview Local */}
          <Button
            type="button"
            variant="outline"
            onClick={() => fileInputRef.current?.click()}
            disabled={disabled}
            className="w-full mb-4"

interface GalleryImage {

  name?: string
}

interface GalleryUploadCompleteProps {
  value: GalleryImage[]
  onChange: (images: GalleryImage[]) => void
  disabled?: boolean
  maxImages?: number
  onRemove?: (key: string) => Promise<void>
}

export function GalleryUploadComplete({
  value = [],
  onChange,
  disabled,
  maxImages = 3,
  onRemove,
}: GalleryUploadCompleteProps) {
  const [isUploading, setIsUploading] = useState(false)
  const [removingKey, setRemovingKey] = useState<string | null>(null)
  const [error, setError] = useState<string | null>(null)
  const [previewImages, setPreviewImages] = useState<string[]>([])
  const [lightboxImage, setLightboxImage] = useState<string | null>(null)
  const fileInputRef = useRef<HTMLInputElement>(null)

  // Preview local antes do upload
  const handleFileSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
    const files = Array.from(e.target.files || [])
    if (files.length === 0) return

    const validFiles = files.filter((file) => {
      // Validar tamanho
      if (file.size > 4 * 1024 * 1024) {
        setError(`${file.name}: Arquivo muito grande (máx. 4MB)`)
        return false
      }

      // Validar tipo
      if (!file.type.startsWith('image/')) {
        setError(`${file.name}: Deve ser uma imagem`)
        return false
      }

      return true
    })

    // Criar previews locais
    const newPreviews: string[] = []
    validFiles.forEach((file) => {
      const reader = new FileReader()
      reader.onloadend = () => {
        newPreviews.push(reader.result as string)
        if (newPreviews.length === validFiles.length) {
          setPreviewImages(newPreviews)
          setError(null)
        }
      }
      reader.readAsDataURL(file)
    })
  }

  const handleRemove = async (index: number) => {
    const imageToRemove = value[index]
    if (!imageToRemove) return

    try {
      setRemovingKey(imageToRemove.key)

      // Chamar API para remover do UploadThing
      const response = await fetch('/api/uploadthing/delete', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ key: imageToRemove.key }),
      })

      if (response.ok) {
        const newImages = value.filter((_, i) => i !== index)
        // Reordenar
        const reordered = newImages.map((img, i) => ({ ...img, order: i + 1 }))
        onChange(reordered)
        setError(null)

        if (onRemove) {
          await onRemove(imageToRemove.key)
        }
      } else {
        const data = await response.json()
        setError(data.error || 'Erro ao remover imagem')
      }
    } catch (error) {
      console.error('❌ Erro ao remover:', error)
      setError('Erro ao remover imagem')
    } finally {
      setRemovingKey(null)
    }
  }

  const clearPreviews = () => {
    setPreviewImages([])
    if (fileInputRef.current) {
      fileInputRef.current.value = ''
    }
  }

  const canUploadMore = value.length < maxImages

  return (
    <div className="space-y-4">
      {/* Header */}
      <div>
        <label className="block text-sm font-medium mb-2">

        <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
          {value.map((image, index) => (

              className="relative overflow-hidden border-2 border-green-200 group"
            >
              <div className="relative h-48 w-full bg-gray-100">
                <Image
                  src={image.url}
                  alt={`Imagem ${index + 1}`}
                  fill
                  className="object-cover cursor-pointer"
                  onClick={() => setLightboxImage(image.url)}
                />
              </div>

              {/* Overlay com ações */}
              <div className="absolute inset-0 bg-black/0 group-hover:bg-black/60 transition flex items-center justify-center gap-2">
                <Button
                  type="button"
                  variant="secondary"
                  size="sm"
                  onClick={() => setLightboxImage(image.url)}
                  disabled={disabled}
                  className="opacity-0 group-hover:opacity-100 transition shadow-xl"
                >
                  <ZoomIn className="h-4 w-4" />
                </Button>

                <Button

                <div className="flex items-center gap-2 text-green-700 text-xs">

                  <span className="font-semibold truncate">
                    {image.name || 'Carregada'}
                  </span>

// PUT - Atualizar viagem
export async function PUT(
  req: Request,
  { params }: { params: { tripId: string } }
) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    if (!hasPermission(session.user.role as any, 'trips.edit')) {
      return NextResponse.json(
        { error: 'Sem permissão para editar viagens' },
        { status: 403 }
      )
    }

    const body = await req.json()

    console.log('📝 Atualizando viagem:', params.tripId)
    console.log('📥 Dados recebidos:', body)

    const trip = await prisma.trip.update({
      where: {
        id: params.tripId,
        tenant_id: session.user.tenantId,
      },
      data: {
        // Dados básicos
        title: body.title,
        destination: body.destination,
        description: body.description,
        
        // IMAGENS
        image_url: body.image_url,
        image_key: body.image_key,
        gallery_images: body.gallery_images, // JSON array
        
        // Preços
        price_per_person: body.price_per_person
          ? parseFloat(body.price_per_person)
          : undefined,
        promotional_price: body.promotional_price
          ? parseFloat(body.promotional_price)
          : null,
        
        // Atualização
        updated_at: new Date(),
      },
    })

    console.log('✅ Viagem atualizada:', trip)

    return NextResponse.json({
      success: true,
      trip,
      message: 'Viagem atualizada com sucesso!',
    })
  } catch (error: any) {
    console.error('❌ Erro ao atualizar viagem:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao atualizar viagem' },
      { status: 500 }
    )
  }
}

// DELETE - Excluir viagem
export async function DELETE(
  req: Request,
  { params }: { params: { tripId: string } }
) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    if (!hasPermission(session.user.role as any, 'trips.delete')) {
      return NextResponse.json(
        { error: 'Sem permissão para excluir viagens' },
        { status: 403 }
      )
    }

    // Buscar viagem para pegar as keys das imagens
    const trip = await prisma.trip.findUnique({
      where: {
        id: params.tripId,
        tenant_id: session.user.tenantId,
      },
    })

    if (!trip) {
      return NextResponse.json(
        { error: 'Viagem não encontrada' },
        { status: 404 }
      )
    }

    // Deletar imagens do UploadThing
    const keysToDelete: string[] = []

    if (trip.image_key) {
      keysToDelete.push(trip.image_key)
    }

    if (trip.gallery_images) {
      const gallery = trip.gallery_images as any[]
      gallery.forEach((img) => {
        if (img.key) {
          keysToDelete.push(img.key)
        }
      })
    }

    if (keysToDelete.length > 0) {
      console.log('🗑️ Removendo imagens:', keysToDelete)
      
      // Chamar API de remoção
      await fetch(`${process.env.NEXTAUTH_URL}/api/uploadthing/delete`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ keys: keysToDelete }),
      })
    }

    // Deletar viagem do banco
    await prisma.trip.delete({
      where: { id: params.tripId },
    })

    console.log('✅ Viagem excluída:', params.tripId)

    return NextResponse.json({
      success: true,
      message: 'Viagem excluída com sucesso!',
    })
  } catch (error: any) {
    console.error('❌ Erro ao excluir viagem:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao excluir viagem' },
      { status: 500 }
    )
  }
}

    tripId: string
  }
}

export default async function TripEditPage({ params }: TripEditPageProps) {

  if (!session) {
    redirect('/login')
  }

  const trip = await prisma.trip.findUnique({
    where: {
      id: params.tripId,
      tenant_id: session.user.tenantId,
    },
  })

  if (!trip) {
    notFound()
  }

  return <TripEditClient trip={trip} />

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { UTApi } from 'uploadthing/server'

const utapi = new UTApi()

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const body = await req.json()
    const { key, keys } = body

    // Suporta tanto key única quanto múltiplas keys
    const keysToDelete = keys || (key ? [key] : [])

    if (keysToDelete.length === 0) {
      return NextResponse.json(
        { error: 'Nenhuma key fornecida' },
        { status: 400 }
      )
    }

    console.log('🗑️ Removendo imagens:', keysToDelete)

    // Deletar do UploadThing
    await utapi.deleteFiles(keysToDelete)

    console.log('✅ Imagens removidas com sucesso')

    return NextResponse.json({
      success: true,
      message: `${keysToDelete.length} imagem(ns) removida(s) com sucesso`,
    })
  } catch (error: any) {
    console.error('❌ Erro ao remover imagens:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao remover imagens' },
      { status: 500 }
    )
  }
}
