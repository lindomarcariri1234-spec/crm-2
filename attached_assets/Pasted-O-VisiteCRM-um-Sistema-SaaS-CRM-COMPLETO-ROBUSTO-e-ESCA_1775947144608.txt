O **VisiteCRM**, é um Sistema SaaS CRM COMPLETO, ROBUSTO e ESCALÁVEL
para agências de turismo especializadas em excursões.


Sincronizar/cruzar os dados

Integração conectada: Relacionamento entre, cliente, viagem, reservas, poltronas, embarque, vendedor, agência, pagamento, loja, Papiline, Kanbam, Automação ou seja, todo sistema é relacionado entre si para um perfeito funcionamento.


Contém Sistema de Reservas COMPLETO, ROBUSTO e sofisticado com reserva criado pelo vendedor, pela Agência e pelo próprio cliente na loja.  seleção de poltronas, passageiros, voucher PDF, QR Code entre outras funcionalidades.

Melhorar a UX..

// ═══════════════════════════════════════════════════════════════
// TENANT (Multi-Tenancy)
// ═══════════════════════════════════════════════════════════════

model Tenant {
  id              String    @id @default(cuid())
  name            String
  slug            String    @unique
  domain          String?   @unique
  
  // Configurações
  settings        Json?
  
  // Status
  is_active       Boolean   @default(true)
  subscription_status String @default("trial") // trial, active, suspended, cancelled
  subscription_plan String  @default("basic") // basic, pro, enterprise
  
  // Limites
  max_users       Int       @default(5)
  max_clients     Int       @default(1000)
  max_trips       Int       @default(100)
  
  // Relacionamentos
  users           User[]
  clients         Client[]
  trips           Trip[]
  reservations    Reservation[]
  passengers      Passenger[]
  seats           ReservationSeat[]
  payments        ReservationPayment[]
  boarding_locations BoardingLocation[]
  coupons         Coupon[]
  referrals       Referral[]
  pipelines       Pipeline[]
  deals           Deal[]
  automations     Automation[]
  store_products  StoreProduct[]
  store_orders    StoreOrder[]
  audit_logs      AuditLog[]
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  @@map("tenants")
}

model User {
  id              String    @id @default(cuid())
  tenant_id       String
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  
  name            String
  email           String
  password        String
  avatar          String?
  
  // Papel e Permissões
  role            String    @default("agent") // admin, manager, agent, viewer
  permissions     Json?
  
  // Comissões
  commission_percentage Decimal? @db.Decimal(5, 2)
  
  // Status
  is_active       Boolean   @default(true)
  
  // Relacionamentos
  clients         Client[]  @relation("ClientOwner")
  reservations    Reservation[] @relation("SellerReservations")
  deals           Deal[]
  tasks           Task[]
  activities      Activity[]
  
  // Autenticação
  email_verified  DateTime?
  last_login      DateTime?
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  @@unique([tenant_id, email])
  @@index([tenant_id])
  @@map("users")
}

model Client {
  id              String    @id @default(cuid())
  tenant_id       String
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  
  // Dados Pessoais
  name            String
  email           String?
  phone           String?
  cpf             String?
  rg              String?
  birth_date      DateTime?
  
  // Endereço
  address         String?
  city            String?
  state           String?
  zip_code        String?
  
  // Relacionamento
  owner_id        String?
  owner           User?     @relation("ClientOwner", fields: [owner_id], references: [id])
  
  // Pipeline/Kanban
  pipeline_stage_id String?
  pipeline_stage  PipelineStage? @relation(fields: [pipeline_stage_id], references: [id])
  
  // Tags e Segmentação
  tags            String[]
  segment         String?   // vip, regular, potential
  
  // Score e Engajamento
  lead_score      Int       @default(0)
  engagement_level String   @default("cold") // cold, warm, hot
  
  // Sistema de Indicação
  referral_code   String?   @unique
  referred_by_id  String?
  referred_by     Client?   @relation("ClientReferrals", fields: [referred_by_id], references: [id])
  referrals_made  Client[]  @relation("ClientReferrals")
  
  // Estatísticas
  total_trips     Int       @default(0)
  total_spent     Decimal   @default(0) @db.Decimal(10, 2)
  is_first_purchase Boolean @default(true)
  last_purchase_date DateTime?
  
  // Relacionamentos
  reservations    Reservation[]
  deals           Deal[]
  activities      Activity[]
  referrals_as_referrer Referral[] @relation("ReferrerClient")
  referrals_as_referred Referral[] @relation("ReferredClient")
  store_orders    StoreOrder[]
  
  // Status
  is_active       Boolean   @default(true)
  
  // Observações
  notes           String?   @db.Text
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  @@index([tenant_id])
  @@index([owner_id])
  @@index([pipeline_stage_id])
  @@index([email])
  @@index([phone])
  @@index([cpf])
  @@map("clients")
}

  seats           ReservationSeat[]
  boarding_locations TripBoardingLocation[]
  store_product   StoreProduct?
  
  // Itinerário

  // Observações
  notes           String?   @db.Text
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  @@index([tenant_id])
  @@index([status])
  @@index([departure_date])
  @@map("trips")
}

// ═══════════════════════════════════════════════════════════════
// RESERVAS (Centro de Integração)
// ═══════════════════════════════════════════════════════════════

model Reservation {
  id              String    @id @default(cuid())
  tenant_id       String
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  
  // Códigos
  reservation_number String  @unique
  voucher_code    String    @unique
  
  // Relacionamentos Principais
  trip_id         String
  trip            Trip      @relation(fields: [trip_id], references: [id], onDelete: Restrict)
  
  client_id       String
  client          Client    @relation(fields: [client_id], references: [id], onDelete: Restrict)
  
  seller_id       String?
  seller          User?     @relation("SellerReservations", fields: [seller_id], references: [id])
  
  // Pipeline/Deal
  deal_id         String?   @unique
  deal            Deal?     @relation(fields: [deal_id], references: [id])
  
  // Passageiros e Assentos
  passengers      Passenger[]
  seats           ReservationSeat[]
  total_passengers Int      @default(1)
  adults_count    Int       @default(1)
  children_count  Int       @default(0)
  
  // Embarque
  boarding_location_id String?
  boarding_location BoardingLocation? @relation(fields: [boarding_location_id], references: [id])
  
  // Valores
  subtotal_amount Decimal   @db.Decimal(10, 2)
  discount_amount Decimal   @default(0) @db.Decimal(10, 2)
  coupon_discount Decimal   @default(0) @db.Decimal(10, 2)
  referral_discount Decimal @default(0) @db.Decimal(10, 2)
  total_amount    Decimal   @db.Decimal(10, 2)
  amount_paid     Decimal   @default(0) @db.Decimal(10, 2)
  amount_pending  Decimal   @db.Decimal(10, 2)
  
  // Cupom e Indicação
  coupon_code     String?
  coupon_id       String?
  coupon          Coupon?   @relation(fields: [coupon_id], references: [id])
  
  referral_id     String?   @unique
  referral        Referral? @relation(fields: [referral_id], references: [id])
  referred_by_code String?
  
  // Pagamentos
  payments        ReservationPayment[]
  payment_method  String?
  installments    Int?      @default(1)
  installment_amount Decimal? @db.Decimal(10, 2)
  
  // Comissão
  commission_percentage Decimal? @db.Decimal(5, 2)
  commission_amount Decimal? @db.Decimal(10, 2)
  commission_paid Boolean @default(false)
  
  // Status
  status          String    @default("pending")
  payment_status  String    @default("pending")
  
  // Check-in
  checked_in      Boolean   @default(false)
  checked_in_at   DateTime?
  checked_in_by   String?
  
  completed_at    DateTime?
  cancelled_at    DateTime?
  cancellation_reason String? @db.Text
  
  // Observações
  notes           String?   @db.Text
  internal_notes  String?   @db.Text
  
  // Origem
  source          String    @default("admin") // admin, store, api, pipeline
  
  // Loja
  store_order_id  String?   @unique
  store_order     StoreOrder? @relation(fields: [store_order_id], references: [id])
  
  // Automações
  automation_triggered Boolean @default(false)
  automation_logs Json?
  
  created_at      DateTime  @default(now())
  updated_at      DateTime  @updatedAt
  
  @@index([tenant_id])
  @@index([trip_id])
  @@index([client_id])
  @@index([seller_id])
  @@index([deal_id])
  @@index([status])
  @@index([payment_status])
  @@map("reservations")
}

// lib/events/event-emitter.ts

import { EventEmitter } from 'events'

export type EventType =
  // Clientes
  | 'client.created'
  | 'client.updated'
  | 'client.deleted'
  | 'client.stage_changed'
  
  // Viagens
  | 'trip.created'
  | 'trip.updated'
  | 'trip.published'
  | 'trip.cancelled'
  | 'trip.completed'
  
  // Reservas
  | 'reservation.created'
  | 'reservation.updated'
  | 'reservation.confirmed'
  | 'reservation.cancelled'
  | 'reservation.checked_in'
  | 'reservation.completed'
  
  // Pagamentos
  | 'payment.received'
  | 'payment.pending'
  | 'payment.failed'
  | 'payment.refunded'
  
  // Pipeline/Deals
  | 'deal.created'
  | 'deal.updated'
  | 'deal.stage_changed'
  | 'deal.won'
  | 'deal.lost'
  
  // Poltronas
  | 'seat.reserved'
  | 'seat.released'
  | 'seat.confirmed'
  
  // Embarque
  | 'boarding.checked_in'
  | 'boarding.completed'
  
  // Loja
  | 'store.order_created'
  | 'store.order_paid'
  
  // Indicação
  | 'referral.created'
  | 'referral.completed'
  | 'referral.bonus_paid'

export interface EventPayload {
  tenant_id: string
  user_id?: string
  timestamp: Date
  data: any
  metadata?: Record<string, any>
}

class AppEventEmitter extends EventEmitter {
  private static instance: AppEventEmitter

  private constructor() {
    super()
    this.setMaxListeners(100) // Aumentar limite de listeners
  }

  static getInstance(): AppEventEmitter {
    if (!AppEventEmitter.instance) {
      AppEventEmitter.instance = new AppEventEmitter()
    }
    return AppEventEmitter.instance
  }

  emitEvent(event: EventType, payload: EventPayload): void {
    console.log(`[EVENT] ${event}`, {
      tenant_id: payload.tenant_id,
      timestamp: payload.timestamp,
    })
    
    this.emit(event, payload)
    
    // Emit evento genérico para logging
    this.emit('*', { event, payload })
  }

  onEvent(event: EventType | '*', handler: (payload: EventPayload) => void | Promise<void>): void {
    this.on(event, async (payload) => {
      try {
        await handler(payload)
      } catch (error) {
        console.error(`[EVENT ERROR] ${event}`, error)
      }
    })
  }
}

export const eventEmitter = AppEventEmitter.getInstance()

eventEmitter.onEvent('client.created', async (payload) => {
  const { data } = payload
  
  try {
    // Buscar pipeline padrão
    const defaultPipeline = await prisma.pipeline.findFirst({
      where: {
        tenant_id: payload.tenant_id,
        is_default: true,
      },
      include: {
        stages: {
          orderBy: { order: 'asc' },
          take: 1,
        },
      },
    })

    if (defaultPipeline && defaultPipeline.stages[0]) {
      // Atualizar cliente com primeiro estágio
      await prisma.client.update({
        where: { id: data.id },
        data: {
          pipeline_stage_id: defaultPipeline.stages[0].id,
        },
      })

      console.log(`[SYNC] Cliente ${data.id} adicionado ao pipeline`)
    }
  } catch (error) {
    console.error('[SYNC ERROR] client.created -> pipeline', error)
  }
})

async function updateTripAvailableSeats(tripId: string) {
  const trip = await prisma.trip.findUnique({
    where: { id: tripId },
    select: {
      total_seats: true,
      seats: {
        where: {
          status: { in: ['reserved', 'confirmed'] },
        },
      },
    },
  })

  if (trip) {
    const occupiedSeats = trip.seats.length
    const availableSeats = trip.total_seats - occupiedSeats

    await prisma.trip.update({
      where: { id: tripId },
      data: {
        available_seats: availableSeats,
        status: availableSeats === 0 ? 'full' : 'active',
      },
    })
  }
}

eventEmitter.onEvent('payment.received', async (payload) => {
  const { data } = payload
  
  try {
    const reservation = await prisma.reservation.findUnique({
      where: { id: data.reservation_id },
      select: {
        total_amount: true,
        amount_paid: true,
      },
    })

    if (reservation) {
      const newAmountPaid = Number(reservation.amount_paid) + Number(data.amount)
      const totalAmount = Number(reservation.total_amount)
      const amountPending = totalAmount - newAmountPaid

      let paymentStatus = 'pending'
      let reservationStatus = 'pending'

      if (amountPending <= 0) {
        paymentStatus = 'paid'
        reservationStatus = 'confirmed'
      } else if (newAmountPaid > 0) {
        paymentStatus = 'partial'
        reservationStatus = 'confirmed'
      }

      await prisma.reservation.update({
        where: { id: data.reservation_id },
        data: {
          amount_paid: newAmountPaid,
          amount_pending: Math.max(amountPending, 0),
          payment_status: paymentStatus,
          status: reservationStatus,
        },
      })

      console.log(`[SYNC] Status de pagamento da reserva ${data.reservation_id} atualizado`)

      // Emitir evento de confirmação se totalmente pago
      if (paymentStatus === 'paid') {
        eventEmitter.emitEvent('reservation.confirmed', {
          tenant_id: payload.tenant_id,
          timestamp: new Date(),
          data: { id: data.reservation_id },
        })
      }
    }
  } catch (error) {
    console.error('[SYNC ERROR] payment.received -> reservation', error)
  }
})

          payment_status: 'pending',

          total_passengers: 1,

    const client = await prisma.client.findUnique({
      where: { id: clientId },
      select: { lead_score: true },
    })

    if (client) {
      const newScore = client.lead_score + points

      // Atualizar score e engagement level
      let engagementLevel = 'cold'
      if (newScore >= 100) engagementLevel = 'hot'
      else if (newScore >= 50) engagementLevel = 'warm'

      await prisma.client.update({
        where: { id: clientId },
        data: {
          lead_score: newScore,
          engagement_level: engagementLevel,
        },
      })

      console.log(`[SYNC] Lead score do cliente ${clientId}: ${newScore}`)
    }
  } catch (error) {
    console.error('[SYNC ERROR] updateClientLeadScore', error)
  }
}

async function generateReservationNumber(tenantId: string): Promise<string> {
  const lastReservation = await prisma.reservation.findFirst({
    where: { tenant_id: tenantId },
    orderBy: { created_at: 'desc' },
    select: { reservation_number: true },
  })

  const lastNumber = lastReservation
    ? parseInt(lastReservation.reservation_number.split('-').pop() || '0')
    : 0

  return `RES-${new Date().getFullYear()}-${String(lastNumber + 1).padStart(5, '0')}`
}

function generateVoucherCode(): string {
  const characters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
  let code = 'VC-'

  for (let i = 0; i < 6; i++) {
    code += characters.charAt(Math.floor(Math.random() * characters.length))
  }

  return code
}

import { prisma } from '@/lib/prisma'
import { eventEmitter } from '@/lib/hooks'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'

export async function createReservation(data: any) {
  const session = await getServerSession(authOptions)
  
  if (!session) {
    throw new Error('Não autorizado')
  }

  // Criar reserva
  const reservation = await prisma.reservation.create({
    data: {
      ...data,
      tenant_id: session.user.tenantId,
    },
    include: {
      trip: true,
      client: true,
      passengers: true,
      seats: true,
    },
  })

  // Emitir evento
  eventEmitter.emitEvent('reservation.created', {
    tenant_id: session.user.tenantId,
    user_id: session.user.id,
    timestamp: new Date(),
    data: reservation,
  })

  return reservation
}

export async function confirmReservation(reservationId: string) {
  const session = await getServerSession(authOptions)
  
  if (!session) {
    throw new Error('Não autorizado')
  }

  const reservation = await prisma.reservation.update({
    where: { id: reservationId },
    data: {
      status: 'confirmed',
    },
  })

  // Emitir evento
  eventEmitter.emitEvent('reservation.confirmed', {

export async function moveDealToStage(dealId: string, stageId: string) {
  const session = await getServerSession(authOptions)
  
  if (!session) {
    throw new Error('Não autorizado')
  }

  const deal = await prisma.deal.findUnique({
    where: { id: dealId },
    select: { stage_id: true },
  })

  const oldStageId = deal?.stage_id

  const updatedDeal = await prisma.deal.update({
    where: { id: dealId },
    data: { stage_id: stageId },
    include: {
      stage: true,
      client: true,
    },
  })

  // Emitir evento
  eventEmitter.emitEvent('deal.stage_changed', {
    tenant_id: session.user.tenantId,
    user_id: session.user.id,
    timestamp: new Date(),
    data: {
      ...updatedDeal,
      old_stage_id: oldStageId,
      new_stage_id: stageId,
    },
  })

  // Se moveu para estágio ganho
  if (updatedDeal.stage.is_closed_won) {
    eventEmitter.emitEvent('deal.won', {
      tenant_id: session.user.tenantId,
      user_id: session.user.id,
      timestamp: new Date(),
      data: updatedDeal,
    })
  }

  // Se moveu para estágio perdido
  if (updatedDeal.stage.is_closed_lost) {
    eventEmitter.emitEvent('deal.lost', {
      tenant_id: session.user.tenantId,
      user_id: session.user.id,
      timestamp: new Date(),
      data: updatedDeal,
    })
  }

  return updatedDeal
}

// lib/automation/automation-engine.ts

import { prisma } from '@/lib/prisma'
import { eventEmitter, EventType, EventPayload } from '@/lib/events/event-emitter'
import { executeAction } from './actions'
import { evaluateConditions } from './conditions'

interface AutomationConfig {
  id: string
  name: string
  trigger_type: string
  trigger_config: any
  conditions: any
  actions: any[]
  delay_type: string
  delay_value?: number
  delay_unit?: string
}

class AutomationEngine {
  private static instance: AutomationEngine
  private automations: Map<string, AutomationConfig[]> = new Map()

  private constructor() {
    this.initializeAutomations()
    this.setupEventListeners()
  }

  static getInstance(): AutomationEngine {
    if (!AutomationEngine.instance) {
      AutomationEngine.instance = new AutomationEngine()
    }
    return AutomationEngine.instance
  }

  // Carregar automações do banco
  async initializeAutomations() {
    try {
      const automations = await prisma.automation.findMany({
        where: { is_active: true },
      })

      // Agrupar por trigger_type
      automations.forEach((automation) => {
        const triggerType = automation.trigger_type
        if (!this.automations.has(triggerType)) {
          this.automations.set(triggerType, [])
        }
        this.automations.get(triggerType)!.push(automation as any)
      })

      console.log(`[AUTOMATION] ${automations.length} automações carregadas`)
    } catch (error) {
      console.error('[AUTOMATION] Erro ao carregar automações:', error)
    }
  }

  // Recarregar automações
  async reloadAutomations() {
    this.automations.clear()
    await this.initializeAutomations()
  }

  // Configurar listeners para eventos
  setupEventListeners() {
    // Mapear eventos para triggers
    const eventToTriggerMap: Record<string, string> = {
      'reservation.created': 'reservation_created',
      'reservation.confirmed': 'reservation_confirmed',
      'reservation.cancelled': 'reservation_cancelled',
      'reservation.checked_in': 'reservation_checked_in',
      'payment.received': 'payment_received',
      'payment.failed': 'payment_failed',
      'deal.created': 'deal_created',
      'deal.stage_changed': 'deal_stage_changed',
      'deal.won': 'deal_won',
      'deal.lost': 'deal_lost',
      'client.created': 'client_created',
      'client.updated': 'client_updated',
      'trip.published': 'trip_published',
      'referral.completed': 'referral_completed',
    }

    // Registrar listener para cada evento
    Object.entries(eventToTriggerMap).forEach(([event, trigger]) => {
      eventEmitter.onEvent(event as EventType, async (payload) => {
        await this.processAutomations(trigger, payload)
      })
    })

    console.log('[AUTOMATION] Event listeners configurados')
  }

  // Processar automações para um trigger
  async processAutomations(triggerType: string, payload: EventPayload) {
    const automations = this.automations.get(triggerType) || []

    if (automations.length === 0) {
      return
    }

    console.log(
      `[AUTOMATION] Processando ${automations.length} automação(ões) para ${triggerType}`
    )

    for (const automation of automations) {
      // Verificar se pertence ao mesmo tenant
      if (automation.trigger_config?.tenant_id && 
          automation.trigger_config.tenant_id !== payload.tenant_id) {
        continue
      }

      // Executar automação
      await this.executeAutomation(automation, payload)
    }
  }

  // Executar uma automação
  async executeAutomation(automation: AutomationConfig, payload: EventPayload) {
    const executionId = await this.createExecution(automation.id, payload)

    try {
      await prisma.automationExecution.update({
        where: { id: executionId },
        data: { status: 'running' },
      })

      // Avaliar condições
      if (automation.conditions) {
        const conditionsMet = await evaluateConditions(
          automation.conditions,
          payload.data
        )

        if (!conditionsMet) {
          console.log(
            `[AUTOMATION] Condições não atendidas para ${automation.name}`
          )
          await prisma.automationExecution.update({
            where: { id: executionId },
            data: {
              status: 'completed',
              completed_at: new Date(),
              result: { skipped: true, reason: 'Condições não atendidas' },
            },
          })
          return
        }
      }

      // Aplicar delay se configurado
      if (automation.delay_type === 'delay' && automation.delay_value) {
        const delayMs = this.calculateDelayMs(
          automation.delay_value,
          automation.delay_unit || 'minutes'
        )
        await this.scheduleExecution(
          automation,
          payload,
          executionId,
          delayMs
        )
        return
      }

      // Executar ações
      const results = await this.executeActions(
        automation.actions,
        payload,
        executionId
      )

      // Atualizar execução
      await prisma.automationExecution.update({
        where: { id: executionId },
        data: {
          status: 'completed',
          completed_at: new Date(),
          result: results,
          duration_ms: Date.now() - new Date(payload.timestamp).getTime(),
        },
      })

      // Atualizar estatísticas da automação
      await prisma.automation.update({
        where: { id: automation.id },
        data: {
          execution_count: { increment: 1 },
          success_count: { increment: 1 },
          last_executed: new Date(),
        },
      })

      console.log(`[AUTOMATION] ${automation.name} executada com sucesso`)
    } catch (error: any) {
      console.error(`[AUTOMATION] Erro ao executar ${automation.name}:`, error)

      await prisma.automationExecution.update({
        where: { id: executionId },
        data: {
          status: 'failed',
          completed_at: new Date(),
          error_message: error.message,
        },
      })

      await prisma.automation.update({
        where: { id: automation.id },
        data: {
          error_count: { increment: 1 },
          last_error: error.message,
        },
      })
    }
  }

  // Criar registro de execução
  async createExecution(
    automationId: string,
    payload: EventPayload
  ): Promise<string> {
    const execution = await prisma.automationExecution.create({
      data: {
        automation_id: automationId,
        trigger_data: payload as any,
        status: 'pending',
      },
    })
    return execution.id
  }

  // Executar ações
  async executeActions(
    actions: any[],
    payload: EventPayload,
    executionId: string
  ) {
    const results: any[] = []

    for (const action of actions) {
      try {
        const result = await executeAction(action, payload)
        results.push({
          action: action.type,
          status: 'success',
          result,
        })
      } catch (error: any) {
        results.push({
          action: action.type,
          status: 'error',
          error: error.message,
        })
      }
    }

    return results
  }

  // Calcular delay em milissegundos
  calculateDelayMs(value: number, unit: string): number {
    const multipliers: Record<string, number> = {
      minutes: 60 * 1000,
      hours: 60 * 60 * 1000,
      days: 24 * 60 * 60 * 1000,
    }
    return value * (multipliers[unit] || multipliers.minutes)
  }

  // Agendar execução com delay
  async scheduleExecution(
    automation: AutomationConfig,
    payload: EventPayload,
    executionId: string,
    delayMs: number
  ) {
    console.log(
      `[AUTOMATION] Agendando ${automation.name} para daqui a ${delayMs}ms`
    )

    setTimeout(async () => {
      const results = await this.executeActions(
        automation.actions,
        payload,
        executionId
      )

      await prisma.automationExecution.update({
        where: { id: executionId },
        data: {
          status: 'completed',
          completed_at: new Date(),
          result: results,
        },
      })

      await prisma.automation.update({
        where: { id: automation.id },
        data: {
          execution_count: { increment: 1 },
          success_count: { increment: 1 },
          last_executed: new Date(),
        },
      })
    }, delayMs)
  }
}

  const modelMap: Record<string, any> = {
    client: prisma.client,
    reservation: prisma.reservation,
    deal: prisma.deal,
    trip: prisma.trip,
  }

  const prismaModel = modelMap[model]
  if (!prismaModel) {
    throw new Error(`Model desconhecido: ${model}`)
  }

  const recordIdValue = replaceVariables(record_id, payload.data)
  const fieldValue = replaceVariables(value, payload.data)

  await prismaModel.update({
    where: { id: recordIdValue },
    data: { [field]: fieldValue },
  })

  return { updated: true, model, field }
}

// ═══════════════════════════════════════════════════════════════
// UTILITÁRIO: Substituir Variáveis
// ═══════════════════════════════════════════════════════════════

function replaceVariables(text: string, data: any): string {
  if (!text) return text

  let result = text

  // Substituir variáveis no formato {{variable}}
  const matches = text.match(/\{\{([^}]+)\}\}/g)

  if (matches) {
    matches.forEach((match) => {
      const variable = match.replace(/\{\{|\}\}/g, '').trim()
      const value = getNestedValue(data, variable)
      result = result.replace(match, value || '')
    })
  }

  return result
}

function getNestedValue(obj: any, path: string): any {
  return path.split('.').reduce((current, key) => current?.[key], obj)
}

// lib/automation/conditions.ts

export async function evaluateConditions(
  conditions: any,
  data: any
): Promise<boolean> {
  if (!conditions) return true

  const { operator, rules } = conditions

  // Operador lógico: AND ou OR
  if (operator === 'AND') {
    return rules.every((rule: any) => evaluateRule(rule, data))
  } else if (operator === 'OR') {
    return rules.some((rule: any) => evaluateRule(rule, data))
  }

  // Condição única
  return evaluateRule(conditions, data)
}

function evaluateRule(rule: any, data: any): boolean {
  const { field, operator, value } = rule

  // Obter valor do campo
  const fieldValue = getNestedValue(data, field)

  // Avaliar operador
  switch (operator) {
    case 'equals':
    case '==':
      return fieldValue == value

    case 'not_equals':
    case '!=':
      return fieldValue != value

    case 'greater_than':
    case '>':
      return Number(fieldValue) > Number(value)

    case 'greater_than_or_equal':
    case '>=':
      return Number(fieldValue) >= Number(value)

    case 'less_than':
    case '<':
      return Number(fieldValue) < Number(value)

    case 'less_than_or_equal':
    case '<=':
      return Number(fieldValue) <= Number(value)

    case 'contains':
      return String(fieldValue).includes(String(value))

    case 'not_contains':
      return !String(fieldValue).includes(String(value))

    case 'starts_with':
      return String(fieldValue).startsWith(String(value))

    case 'ends_with':
      return String(fieldValue).endsWith(String(value))

    case 'in':
      return Array.isArray(value) && value.includes(fieldValue)

    case 'not_in':
      return Array.isArray(value) && !value.includes(fieldValue)

    case 'is_empty':
      return !fieldValue || fieldValue === '' || fieldValue === null

    case 'is_not_empty':
      return !!fieldValue && fieldValue !== '' && fieldValue !== null

    default:
      console.warn(`Operador desconhecido: ${operator}`)
      return false
  }
}

function getNestedValue(obj: any, path: string): any {
  return path.split('.').reduce((current, key) => current?.[key], obj)
}

import { prisma } from '@/lib/prisma'
import crypto from 'crypto'

class WebhookManager {
  private static instance: WebhookManager

  private constructor() {}

  static getInstance(): WebhookManager {
    if (!WebhookManager.instance) {
      WebhookManager.instance = new WebhookManager()
    }
    return WebhookManager.instance
  }

  // Disparar webhook
  async trigger(event: string, tenantId: string, payload: any) {
    try {
      // Buscar webhooks ativos para este evento
      const webhooks = await prisma.webhook.findMany({
        where: {
          tenant_id: tenantId,
          is_active: true,
          events: {
            has: event,
          },
        },
      })

      if (webhooks.length === 0) {
        return
      }

      console.log(
        `[WEBHOOK] Disparando ${webhooks.length} webhook(s) para evento ${event}`
      )

      // Disparar cada webhook
      for (const webhook of webhooks) {
        await this.callWebhook(webhook, event, payload)
      }
    } catch (error) {
      console.error('[WEBHOOK] Erro ao disparar webhooks:', error)
    }
  }

  // Chamar webhook
  async callWebhook(webhook: any, event: string, payload: any, attempt: number = 1) {
    const startTime = Date.now()

    try {
      // Preparar headers
      const headers: Record<string, string> = {
        'Content-Type': 'application/json',
        'X-Webhook-Event': event,
        'X-Webhook-ID': webhook.id,
        'User-Agent': 'VisiteCRM-Webhook/1.0',
        ...webhook.headers,
      }

      // Adicionar assinatura se houver secret
      if (webhook.secret) {
        const signature = this.generateSignature(payload, webhook.secret)
        headers['X-Webhook-Signature'] = signature
      }

      // Fazer requisição
      const response = await fetch(webhook.url, {
        method: webhook.method,
        headers,
        body: JSON.stringify(payload),
        signal: AbortSignal.timeout(30000), // 30 segundos timeout
      })

      const duration = Date.now() - startTime
      const responseBody = await response.text()

      // Registrar log
      await prisma.webhookLog.create({
        data: {
          webhook_id: webhook.id,
          event,
          payload,
          status_code: response.status,
          response_body: responseBody.substring(0, 5000), // Limitar tamanho
          duration_ms: duration,
          attempt,
        },
      })

      // Atualizar estatísticas
      await prisma.webhook.update({
        where: { id: webhook.id },
        data: {
          total_calls: { increment: 1 },
          success_calls: response.ok ? { increment: 1 } : undefined,
          failed_calls: !response.ok ? { increment: 1 } : undefined,
          last_called: new Date(),
          last_status: response.status,
          last_error: !response.ok ? `HTTP ${response.status}` : null,
        },
      })

      // Retry se falhou e retry está habilitado
      if (!response.ok && webhook.retry_enabled && attempt < webhook.max_retries) {
        console.log(
          `[WEBHOOK] Tentativa ${attempt} falhou, agendando retry...`
        )
        setTimeout(() => {
          this.callWebhook(webhook, event, payload, attempt + 1)
        }, webhook.retry_delay * 1000)
      }

      return response.ok
    } catch (error: any) {
      const duration = Date.now() - startTime

      console.error(`[WEBHOOK] Erro ao chamar ${webhook.url}:`, error)

      // Registrar log de erro
      await prisma.webhookLog.create({
        data: {
          webhook_id: webhook.id,
          event,
          payload,
          error_message: error.message,
          duration_ms: duration,
          attempt,
        },
      })

      // Atualizar estatísticas
      await prisma.webhook.update({
        where: { id: webhook.id },
        data: {
          total_calls: { increment: 1 },
          failed_calls: { increment: 1 },
          last_called: new Date(),
          last_error: error.message,
        },
      })

      // Retry se retry está habilitado
      if (webhook.retry_enabled && attempt < webhook.max_retries) {
        console.log(
          `[WEBHOOK] Tentativa ${attempt} falhou, agendando retry...`
        )
        setTimeout(() => {
          this.callWebhook(webhook, event, payload, attempt + 1)
        }, webhook.retry_delay * 1000)
      }

      return false
    }
  }

  // Gerar assinatura HMAC
  generateSignature(payload: any, secret: string): string {
    const hmac = crypto.createHmac('sha256', secret)
    hmac.update(JSON.stringify(payload))
    return hmac.digest('hex')
  }

  // Validar assinatura
  validateSignature(payload: any, signature: string, secret: string): boolean {
    const expectedSignature = this.generateSignature(payload, secret)
    return crypto.timingSafeEqual(
      Buffer.from(signature),
      Buffer.from(expectedSignature)
    )
  }
}

// GET - Listar Automações
// ═══════════════════════════════════════════════════════════════

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const { searchParams } = new URL(req.url)
    const page = parseInt(searchParams.get('page') || '1')
    const limit = parseInt(searchParams.get('limit') || '20')
    const search = searchParams.get('search') || ''
    const triggerType = searchParams.get('trigger_type') || ''
    const isActive = searchParams.get('is_active')

    const skip = (page - 1) * limit

    // Filtros
    const where: any = {
      tenant_id: session.user.tenantId,
    }

    if (search) {
      where.OR = [
        { name: { contains: search, mode: 'insensitive' } },
        { description: { contains: search, mode: 'insensitive' } },
      ]
    }

    if (triggerType) {
      where.trigger_type = triggerType
    }

    if (isActive !== null) {
      where.is_active = isActive === 'true'
    }

    // Buscar automações
    const [automations, total] = await Promise.all([
      prisma.automation.findMany({
        where,
        skip,
        take: limit,
        orderBy: { created_at: 'desc' },
        include: {
          _count: {
            select: { executions: true },
          },
        },
      }),
      prisma.automation.count({ where }),
    ])

    return NextResponse.json({
      automations,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
    })
  } catch (error: any) {
    console.error('Erro ao buscar automações:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar automações' },
      { status: 500 }
    )
  }
}

      data: {
        tenant_id: session.user.tenantId,
        user_id: session.user.id,
        action: 'create',
        resource: 'automation',
        resource_id: automation.id,
        metadata: {
          name: automation.name,
          trigger_type: automation.trigger_type,

    })

    console.error('Erro ao criar automação:', error)
    return NextResponse.json(
      { error: 'Erro ao criar automação' },
      { status: 500 }
    )
  }

  searchParams: { page?: string; search?: string }
}) {
  const session = await getServerSession(authOptions)

  if (!session) {
    return null
  }

  const page = parseInt(searchParams.page || '1')
  const limit = 20
  const skip = (page - 1) * limit
  const search = searchParams.search || ''

  const where: any = {
    tenant_id: session.user.tenantId,
  }

  if (search) {
    where.OR = [
      { name: { contains: search, mode: 'insensitive' } },
      { description: { contains: search, mode: 'insensitive' } },
    ]
  }

  const [automations, total] = await Promise.all([
    prisma.automation.findMany({
      where,
      skip,
      take: limit,
      orderBy: { created_at: 'desc' },
      include: {
        _count: {
          select: { executions: true },
        },
      },
    }),
    prisma.automation.count({ where }),
  ])

  // Estatísticas
  const stats = await prisma.automation.aggregate({
    where: { tenant_id: session.user.tenantId },
    _count: { id: true },
    _sum: {
      execution_count: true,
      success_count: true,
      error_count: true,
    },
  })

  return (
    <div className="container mx-auto py-8 px-4">
      <div className="flex items-center justify-between mb-6">
        <div>
          <h1 className="text-3xl font-bold">Automações</h1>
          <p className="text-gray-600">
            Automatize tarefas e processos do seu negócio
          </p>
        </div>
        <Link href="/dashboard/automations/new">
          <Button>
            <Plus className="h-4 w-4 mr-2" />
            Nova Automação
          </Button>
        </Link>
      </div>

      {/* Estatísticas */}
      <div className="grid grid-cols-1 md:grid-cols-4 gap-4 mb-6">
        <div className="bg-white rounded-lg border p-4">
          <p className="text-sm text-gray-600">Total de Automações</p>
          <p className="text-2xl font-bold">{stats._count.id}</p>
        </div>
        <div className="bg-white rounded-lg border p-4">
          <p className="text-sm text-gray-600">Execuções</p>
          <p className="text-2xl font-bold">{stats._sum.execution_count || 0}</p>
        </div>
        <div className="bg-white rounded-lg border p-4">
          <p className="text-sm text-gray-600">Sucessos</p>
          <p className="text-2xl font-bold text-green-600">
            {stats._sum.success_count || 0}
          </p>
        </div>
        <div className="bg-white rounded-lg border p-4">
          <p className="text-sm text-gray-600">Erros</p>
          <p className="text-2xl font-bold text-red-600">
            {stats._sum.error_count || 0}
          </p>
        </div>
      </div>

      <AutomationsList
        automations={automations}
        total={total}
        page={page}
        limit={limit}
      />
    </div>
  )
}

import { format } from 'date-fns'
import { ptBR } from 'date-fns/locale'

interface AutomationsListProps {
  automations: any[]
  total: number
  page: number
  limit: number
}

export function AutomationsList({
  automations: initialAutomations,
  total,
  page,
  limit,
}: AutomationsListProps) {
  const router = useRouter()
  const { toast } = useToast()
  const [automations, setAutomations] = useState(initialAutomations)

  const handleToggle = async (id: string, currentStatus: boolean) => {
    try {
      const response = await fetch(`/api/dashboard/automations/${id}/toggle`, {
        method: 'POST',
      })

      if (!response.ok) {
        throw new Error('Erro ao alternar automação')
      }

      const { automation } = await response.json()

      setAutomations((prev) =>
        prev.map((a) => (a.id === id ? automation : a))
      )

      toast({
        title: automation.is_active ? 'Automação ativada' : 'Automação desativada',
      })
    } catch (error) {
      toast({
        title: 'Erro ao alternar automação',
        variant: 'destructive',
      })
    }
  }

  const handleDelete = async (id: string) => {
    if (!confirm('Tem certeza que deseja deletar esta automação?')) {
      return
    }

    try {
      const response = await fetch(`/api/dashboard/automations/${id}`, {
        method: 'DELETE',
      })

      if (!response.ok) {
        throw new Error('Erro ao deletar automação')
      }

      setAutomations((prev) => prev.filter((a) => a.id !== id))

      toast({
        title: 'Automação deletada com sucesso',
      })
    } catch (error) {
      toast({
        title: 'Erro ao deletar automação',
        variant: 'destructive',
      })
    }
  }

  const getTriggerLabel = (type: string) => {
    const labels: Record<string, string> = {
      reservation_created: 'Reserva Criada',
      reservation_confirmed: 'Reserva Confirmada',
      payment_received: 'Pagamento Recebido',
      deal_stage_changed: 'Estágio do Deal Alterado',
      deal_won: 'Deal Ganho',
      client_created: 'Cliente Criado',
      trip_published: 'Viagem Publicada',
    }
    return labels[type] || type
  }

  const getSuccessRate = (automation: any) => {
    if (automation.execution_count === 0) return 0
    return Math.round(
      (automation.success_count / automation.execution_count) * 100
    )
  }

  return (
    <div className="space-y-4">
      {automations.length === 0 ? (
        <Card className="p-12 text-center">
          <Zap className="h-12 w-12 mx-auto mb-4 text-gray-400" />
          <h3 className="text-lg font-semibold mb-2">Nenhuma automação criada</h3>
          <p className="text-gray-600 mb-4">
            Crie sua primeira automação para começar
          </p>
          <Button onClick={() => router.push('/dashboard/automations/new')}>
            <Plus className="h-4 w-4 mr-2" />
            Nova Automação
          </Button>
        </Card>
      ) : (
        <>
          {automations.map((automation) => (
            <Card key={automation.id} className="p-6">
              <div className="flex items-start justify-between">
                <div className="flex-1">
                  <div className="flex items-center gap-3 mb-2">
                    <h3 className="text-lg font-bold">{automation.name}</h3>
                    <Badge
                      variant={automation.is_active ? 'default' : 'secondary'}
                    >
                      {automation.is_active ? 'Ativa' : 'Inativa'}
                    </Badge>
                  </div>

                  {automation.description && (
                    <p className="text-sm text-gray-600 mb-3">
                      {automation.description}
                    </p>
                  )}

                  <div className="flex flex-wrap gap-4 text-sm">
                    <div className="flex items-center gap-2">
                      <Zap className="h-4 w-4 text-blue-600" />
                      <span className="text-gray-600">Trigger:</span>
                      <span className="font-medium">
                        {getTriggerLabel(automation.trigger_type)}
                      </span>
                    </div>

                    <div className="flex items-center gap-2">
                      <BarChart3 className="h-4 w-4 text-green-600" />
                      <span className="text-gray-600">Execuções:</span>
                      <span className="font-medium">
                        {automation.execution_count}
                      </span>
                    </div>

                    <div className="flex items-center gap-2">
                      <span className="text-gray-600">Taxa de Sucesso:</span>
                      <span
                        className={`font-medium ${
                          getSuccessRate(automation) >= 80
                            ? 'text-green-600'
                            : getSuccessRate(automation) >= 50
                            ? 'text-yellow-600'
                            : 'text-red-600'
                        }`}
                      >
                        {getSuccessRate(automation)}%
                      </span>
                    </div>

                    {automation.last_executed && (
                      <div className="flex items-center gap-2">
                        <span className="text-gray-600">Última execução:</span>
                        <span className="font-medium">
                          {format(
                            new Date(automation.last_executed),
                            "dd/MM/yyyy 'às' HH:mm",
                            { locale: ptBR }
                          )}
                        </span>
                      </div>
                    )}
                  </div>
                </div>

                <div className="flex items-center gap-2">
                  <Switch
                    checked={automation.is_active}
                    onCheckedChange={() =>
                      handleToggle(automation.id, automation.is_active)
                    }
                  />

                  <DropdownMenu>
                    <DropdownMenuTrigger asChild>
                      <Button variant="ghost" size="sm">
                        <MoreVertical className="h-4 w-4" />
                      </Button>
                    </DropdownMenuTrigger>
                    <DropdownMenuContent align="end">
                      <DropdownMenuItem
                        onClick={() =>
                          router.push(`/dashboard/automations/${automation.id}`)
                        }
                      >
                        <Edit className="h-4 w-4 mr-2" />
                        Editar

                          router.push(`/dashboard/automations/${automation.id}/logs`)
                        }
                      >
                        <BarChart3 className="h-4 w-4 mr-2" />
                        Ver Logs
                      </DropdownMenuItem>

                        onClick={() => handleDelete(automation.id)}

                      >
                        <Trash2 className="h-4 w-4 mr-2" />

                      </DropdownMenuItem>
                    </DropdownMenuContent>

                </div>

            <div className="flex justify-center gap-2 mt-6">

      )}

export function ActionsBuilder({
  actions,
  onChange,
  users,
  pipelines,
  triggerType,
}: ActionsBuilderProps) {
  const [selectedAction, setSelectedAction] = useState<string | null>(null)
  const [editingIndex, setEditingIndex] = useState<number | null>(null)
  const [dialogOpen, setDialogOpen] = useState(false)

  const addAction = (actionType: string) => {
    const newAction = {
      type: actionType,
      config: getDefaultConfig(actionType),
    }

    if (editingIndex !== null) {
      const newActions = [...actions]
      newActions[editingIndex] = newAction
      onChange(newActions)
      setEditingIndex(null)
    } else {
      onChange([...actions, newAction])
    }

    setDialogOpen(false)
    setSelectedAction(null)
  }

  const removeAction = (index: number) => {
    onChange(actions.filter((_, i) => i !== index))
  }

  const updateActionConfig = (index: number, config: any) => {
    const newActions = [...actions]
    newActions[index].config = config
    onChange(newActions)
  }

  const handleDragEnd = (result: any) => {
    if (!result.destination) return

    const items = Array.from(actions)
    const [reorderedItem] = items.splice(result.source.index, 1)
    items.splice(result.destination.index, 0, reorderedItem)

    onChange(items)
  }

  const getDefaultConfig = (actionType: string) => {
    const defaults: Record<string, any> = {
      send_email: {
        to: '{{client.email}}',
        subject: '',
        template: 'default',
        variables: {},
      },
      send_whatsapp: {
        phone: '{{client.phone}}',
        message: '',
      },
      create_task: {
        title: '',
        description: '',
        assigned_to: '',
        priority: 'medium',

      update_field: {

        record_id: '{{client.id}}',
        field: '',

      add_tag: {
        client_id: '{{client.id}}',
        tag: '',
      },
      webhook: {
        url: '',
        method: 'POST',
        headers: {},
        body: {},
      },
      send_notification: {
        user_id: '',
        title: '',
        message: '',
      },
      create_activity: {
        type: 'note',
        subject: '',
        description: '',
        client_id: '{{client.id}}',
        user_id: '',
      },
    }

    return defaults[actionType] || {}
  }

  const getActionTypeInfo = (type: string) => {
    return ACTION_TYPES.find((a) => a.id === type)
  }

  if (actions.length === 0) {
    return (
      <div className="text-center py-8">
        <div className="inline-flex items-center justify-center w-16 h-16 bg-gray-100 rounded-full mb-4">
          <Plus className="h-8 w-8 text-gray-400" />
        </div>
        <p className="text-gray-600 mb-4">

          <DialogTrigger asChild>

              <Plus className="h-4 w-4 mr-2" />
              Adicionar Ação
            </Button>
          </DialogTrigger>
          <DialogContent className="max-w-3xl">
            <DialogHeader>
              <DialogTitle>Selecione uma Ação</DialogTitle>
            </DialogHeader>
            <ActionSelector
              onSelect={(type) => {
                setSelectedAction(type)
                addAction(type)
              }}
            />
          </DialogContent>
        </Dialog>
      </div>
    )
  }

  return (
    <div className="space-y-4">
      <DragDropContext onDragEnd={handleDragEnd}>

                          >
                            <GripVertical className="h-5 w-5 text-gray-400" />
                          </div>

                                <h4 className="font-semibold">

                                <p className="text-sm text-gray-600">
                                  {actionInfo.description}
                                </p>
                              </div>

                                variant="ghost"
                                size="sm"
                                onClick={() => removeAction(index)}
                                className="text-red-600"
                              >
                                <Trash2 className="h-4 w-4" />
                              </Button>
                            </div>

                            {/* Action Config */}
                            <ActionConfig
                              type={action.type}
                              config={action.config}
                              onChange={(config) =>
                                updateActionConfig(index, config)
                              }
                              users={users}
                              pipelines={pipelines}
                              triggerType={triggerType}
                            />
                          </div>
                        </div>
                      </Card>
                    )}
                  </Draggable>
                )
              })}
              {provided.placeholder}
            </div>
          )}
        </Droppable>
      </DragDropContext>

      {/* Adicionar Nova Ação */}

        </DialogTrigger>
        <DialogContent className="max-w-3xl">
          <DialogHeader>
            <DialogTitle>Selecione uma Ação</DialogTitle>
          </DialogHeader>

              setSelectedAction(type)
              addAction(type)
            }}
          />
        </DialogContent>
      </Dialog>
    </div>
  )
}

import { Clock, Zap, Calendar } from 'lucide-react'

interface DelayConfigProps {
  delayType: string
  delayValue: number | null
  delayUnit: string
  onChange: (delayType: string, delayValue: number | null, delayUnit: string) => void
}

export function DelayConfig({
  delayType,
  delayValue,
  delayUnit,
  onChange,
}: DelayConfigProps) {
  return (
    <div className="space-y-4">
      <RadioGroup
        value={delayType}
        onValueChange={(value) => onChange(value, delayValue, delayUnit)}
      >
        {/* Imediato */}
        <Card
          className={`p-4 cursor-pointer transition-all ${
            delayType === 'immediate'
              ? 'ring-2 ring-blue-500 bg-blue-50'
              : 'hover:bg-gray-50'

          onClick={() => onChange('immediate', null, 'minutes')}
        >
          <div className="flex items-start gap-3">
            <RadioGroupItem value="immediate" id="immediate" className="mt-1" />
            <div className="flex-1">

                <Label htmlFor="immediate" className="cursor-pointer font-semibold">
                  Executar Imediatamente
                </Label>
              </div>
              <p className="text-sm text-gray-600">
                As ações serão executadas assim que o trigger ocorrer
              </p>
            </div>
          </div>
        </Card>

        {/* Com Delay */}
        <Card
          className={`p-4 cursor-pointer transition-all ${
            delayType === 'delay'
              ? 'ring-2 ring-blue-500 bg-blue-50'
              : 'hover:bg-gray-50'
          }`}
          onClick={() => onChange('delay', delayValue || 1, delayUnit)}
        >
          <div className="flex items-start gap-3">
            <RadioGroupItem value="delay" id="delay" className="mt-1" />
            <div className="flex-1">
              <div className="flex items-center gap-2 mb-1">
                <Clock className="h-5 w-5 text-orange-600" />
                <Label htmlFor="delay" className="cursor-pointer font-semibold">
                  Aguardar um Período
                </Label>
              </div>
              <p className="text-sm text-gray-600 mb-3">
                Executar após um tempo específico
              </p>

              {delayType === 'delay' && (
                <div className="flex gap-3">
                  <div className="flex-1">
                    <Label className="text-xs">Quantidade</Label>
                    <Input
                      type="number"
                      min="1"
                      value={delayValue || 1}
                      onChange={(e) =>
                        onChange(delayType, parseInt(e.target.value), delayUnit)
                      }
                      onClick={(e) => e.stopPropagation()}
                    />
                  </div>
                  <div className="flex-1">
                    <Label className="text-xs">Unidade</Label>
                    <Select
                      value={delayUnit}
                      onValueChange={(value) =>
                        onChange(delayType, delayValue, value)
                      }
                    >
                      <SelectTrigger onClick={(e) => e.stopPropagation()}>
                        <SelectValue />
                      </SelectTrigger>
                      <SelectContent>
                        <SelectItem value="minutes">Minutos</SelectItem>
                        <SelectItem value="hours">Horas</SelectItem>
                        <SelectItem value="days">Dias</SelectItem>
                      </SelectContent>
                    </Select>
                  </div>
                </div>
              )}
            </div>
          </div>
        </Card>

        {/* Agendado */}
        <Card
          className={`p-4 cursor-pointer transition-all ${
            delayType === 'scheduled'
              ? 'ring-2 ring-blue-500 bg-blue-50'
              : 'hover:bg-gray-50'
          }`}
          onClick={() => onChange('scheduled', delayValue, delayUnit)}
        >
          <div className="flex items-start gap-3">
            <RadioGroupItem value="scheduled" id="scheduled" className="mt-1" />
            <div className="flex-1">
              <div className="flex items-center gap-2 mb-1">
                <Calendar className="h-5 w-5 text-purple-600" />
                <Label htmlFor="scheduled" className="cursor-pointer font-semibold">
                  Agendar para Data/Hora Específica
                </Label>
              </div>
              <p className="text-sm text-gray-600">
                Em desenvolvimento - Executar em data e hora específicas
              </p>
            </div>
          </div>
        </Card>
      </RadioGroup>

      {/* Preview do Delay */}
      {delayType === 'delay' && delayValue && (
        <div className="bg-blue-50 border border-blue-200 rounded-lg p-3">
          <p className="text-sm text-blue-900">
            ⏱️ As ações serão executadas{' '}
            <strong>
              {delayValue} {delayUnit === 'minutes' ? 'minuto(s)' : delayUnit === 'hours' ? 'hora(s)' : 'dia(s)'}
            </strong>{' '}
            após o trigger ocorrer
          </p>
        </div>
      )}
    </div>
  )
}

                </div>
              </div>
            </>
          ) : (
            <div className="text-center py-8 text-gray-500">
              <p className="text-sm">Nenhuma ação configurada</p>
            </div>
          )}
        </div>
      </div>
    </Card>
  )
}
