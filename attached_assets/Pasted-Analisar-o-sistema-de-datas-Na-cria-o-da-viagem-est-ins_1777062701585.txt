// lib/date-utils.ts

import { differenceInHours, differenceInMinutes, format } from 'date-fns'
import { ptBR } from 'date-fns/locale'

/**
 * Calcula a duração entre duas datas em dias e horas
 */
export function calculateTripDuration(
  departureDate: Date,
  returnDate: Date
): {
  totalHours: number
  days: number
  hours: number
  formatted: string
} {
  // Calcular diferença total em horas
  const totalHours = differenceInHours(returnDate, departureDate)
  
  // Calcular dias completos
  const days = Math.floor(totalHours / 24)
  
  // Calcular horas restantes
  const hours = totalHours % 24

  // Formatar string de duração
  let formatted = ''
  
  if (days > 0) {
    formatted += `${days} ${days === 1 ? 'dia' : 'dias'}`
  }
  
  if (hours > 0) {
    if (formatted) formatted += ' e '
    formatted += `${hours} ${hours === 1 ? 'hora' : 'horas'}`
  }
  
  if (!formatted) {
    formatted = 'Menos de 1 hora'
  }

  return {
    totalHours,
    days,
    hours,
    formatted,
  }
}

/**
 * Formata data e hora para exibição
 */
export function formatDateTime(date: Date): string {
  return format(date, "dd/MM/yyyy 'às' HH:mm", { locale: ptBR })
}

/**
 * Formata apenas a data
 */
export function formatDate(date: Date): string {
  return format(date, 'dd/MM/yyyy', { locale: ptBR })
}

/**
 * Formata apenas a hora
 */
export function formatTime(date: Date): string {
  return format(date, 'HH:mm', { locale: ptBR })
}

/**
 * Formata data de forma extensa
 */
export function formatDateLong(date: Date): string {
  return format(date, "dd 'de' MMMM 'de' yyyy", { locale: ptBR })
}

/**
 * Formata data e hora de forma extensa
 */
export function formatDateTimeLong(date: Date): string {
  return format(date, "dd 'de' MMMM 'de' yyyy 'às' HH:mm", { locale: ptBR })
}

/**
 * Combina data e hora em um único DateTime
 */
export function combineDateAndTime(date: string, time: string): Date {
  // date formato: "2026-12-17"
  // time formato: "20:00"
  return new Date(`${date}T${time}:00`)
}

/**
 * Valida se a data de retorno é posterior à data de saída
 */
export function validateTripDates(
  departureDate: Date,
  returnDate: Date
): { valid: boolean; error?: string } {
  if (returnDate <= departureDate) {
    return {
      valid: false,
      error: 'A data de retorno deve ser posterior à data de saída',
    }
  }

  const duration = differenceInHours(returnDate, departureDate)
  
  if (duration < 1) {
    return {
      valid: false,
      error: 'A viagem deve ter no mínimo 1 hora de duração',
    }
  }

  return { valid: true }
}

export default async function PackageDetailsPage({
  params,
}: {
  params: { slug: string; id: string }
}) {
  const tenant = await prisma.tenant.findUnique({
    where: { slug: params.slug },
  })

  if (!tenant) {
    notFound()
  }

  const trip = await prisma.trip.findUnique({
    where: {
      id: params.id,
      tenant_id: tenant.id,
    },
    include: {
      destination: true,
      vehicle: {
        include: {
          bus_model: true,
        },
      },
    },
  })

  if (!trip) {
    notFound()
  }

  const duration = calculateTripDuration(
    new Date(trip.departure_date),
    new Date(trip.return_date)
  )

    return new Intl.NumberFormat('pt-BR', {
      style: 'currency',
      currency: 'BRL',
    }).format(Number(value))
