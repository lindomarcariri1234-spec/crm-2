// lib/auth.ts

import { NextAuthOptions } from 'next-auth'
import GoogleProvider from 'next-auth/providers/google'
import { PrismaAdapter } from '@next-auth/prisma-adapter'
import { prisma } from './prisma'

export const authOptions: NextAuthOptions = {
  adapter: PrismaAdapter(prisma),
  providers: [
    GoogleProvider({
      clientId: process.env.GOOGLE_CLIENT_ID!,
      clientSecret: process.env.GOOGLE_CLIENT_SECRET!,
      authorization: {
        params: {
          scope: [
            'openid',
            'email',
            'profile',
            'https://www.googleapis.com/auth/calendar',
            'https://www.googleapis.com/auth/calendar.events',
          ].join(' '),
          access_type: 'offline',
          prompt: 'consent',
        },
      },
    }),
    // ... outros providers
  ],
  callbacks: {
    async signIn({ user, account, profile }) {
      if (account?.provider === 'google' && account.access_token) {
        // Salvar tokens do Google
        await prisma.user.update({
          where: { id: user.id },
          data: {
            google_access_token: account.access_token,
            google_refresh_token: account.refresh_token,
            google_token_expiry: account.expires_at
              ? new Date(account.expires_at * 1000)
              : null,
            google_calendar_enabled: true,
          },
        })
      }
      return true
    },
    async session({ session, user }) {
      if (session.user) {
        session.user.id = user.id
        session.user.tenantId = user.tenant_id
        session.user.role = user.role
        
        // Adicionar informações do Google Calendar
        const userData = await prisma.user.findUnique({
          where: { id: user.id },
          select: {
            google_calendar_enabled: true,
            google_calendar_id: true,
          },
        })
        
        session.user.googleCalendarEnabled = userData?.google_calendar_enabled || false
        session.user.googleCalendarId = userData?.google_calendar_id || null
      }
      return session
    },
  },
  // ... outras configurações
}

// lib/google-calendar/calendar-service.ts

import { google } from 'googleapis'
import { prisma } from '@/lib/prisma'

export class GoogleCalendarService {
  private calendar

  constructor(accessToken: string) {
    const auth = new google.auth.OAuth2(
      process.env.GOOGLE_CLIENT_ID,
      process.env.GOOGLE_CLIENT_SECRET,
      process.env.GOOGLE_CALENDAR_REDIRECT_URI
    )

    auth.setCredentials({
      access_token: accessToken,
    })

    this.calendar = google.calendar({ version: 'v3', auth })
  }

  // ═══════════════════════════════════════════════════════════
  // CRIAR EVENTO
  // ═══════════════════════════════════════════════════════════

  async createEvent(data: {
    summary: string
    description?: string
    location?: string
    startDateTime: Date
    endDateTime?: Date
    attendees?: string[]
    reminders?: {
      useDefault: boolean
      overrides?: Array<{
        method: 'email' | 'popup'
        minutes: number
      }>
    }
  }) {
    try {
      const event = {
        summary: data.summary,
        description: data.description,
        location: data.location,
        start: {
          dateTime: data.startDateTime.toISOString(),
          timeZone: 'America/Sao_Paulo',
        },
        end: {
          dateTime: (data.endDateTime || data.startDateTime).toISOString(),
          timeZone: 'America/Sao_Paulo',
        },
        attendees: data.attendees?.map((email) => ({ email })),
        reminders: data.reminders || {
          useDefault: false,
          overrides: [
            { method: 'email', minutes: 24 * 60 }, // 1 dia antes
            { method: 'popup', minutes: 30 }, // 30 minutos antes
          ],
        },
        colorId: '9', // Azul
      }

      const response = await this.calendar.events.insert({
        calendarId: 'primary',
        requestBody: event,
        sendUpdates: 'all',
      })

      return response.data
    } catch (error) {
      console.error('Erro ao criar evento no Google Calendar:', error)
      throw error
    }
  }

  // ═══════════════════════════════════════════════════════════
  // ATUALIZAR EVENTO
  // ═══════════════════════════════════════════════════════════

  async updateEvent(
    eventId: string,
    data: {
      summary?: string
      description?: string
      location?: string
      startDateTime?: Date
      endDateTime?: Date
      attendees?: string[]
    }
  ) {
    try {
      const event: any = {}

      if (data.summary) event.summary = data.summary
      if (data.description) event.description = data.description
      if (data.location) event.location = data.location

      if (data.startDateTime) {
        event.start = {
          dateTime: data.startDateTime.toISOString(),
          timeZone: 'America/Sao_Paulo',
        }
      }

      if (data.endDateTime) {
        event.end = {
          dateTime: data.endDateTime.toISOString(),
          timeZone: 'America/Sao_Paulo',
        }
      }

      if (data.attendees) {
        event.attendees = data.attendees.map((email) => ({ email }))
      }

      const response = await this.calendar.events.patch({
        calendarId: 'primary',
        eventId,
        requestBody: event,
        sendUpdates: 'all',
      })

      return response.data
    } catch (error) {
      console.error('Erro ao atualizar evento no Google Calendar:', error)
      throw error
    }
  }

  // ═══════════════════════════════════════════════════════════
  // DELETAR EVENTO
  // ═══════════════════════════════════════════════════════════

  async deleteEvent(eventId: string) {
    try {
      await this.calendar.events.delete({
        calendarId: 'primary',
        eventId,
        sendUpdates: 'all',
      })

      return true
    } catch (error) {
      console.error('Erro ao deletar evento no Google Calendar:', error)
      throw error
    }
  }

  // ═══════════════════════════════════════════════════════════
  // LISTAR EVENTOS
  // ═══════════════════════════════════════════════════════════

  async listEvents(startDate?: Date, endDate?: Date) {
    try {
      const response = await this.calendar.events.list({
        calendarId: 'primary',
        timeMin: startDate?.toISOString(),
        timeMax: endDate?.toISOString(),
        singleEvents: true,
        orderBy: 'startTime',
      })

      return response.data.items || []
    } catch (error) {
      console.error('Erro ao listar eventos do Google Calendar:', error)
      throw error
    }
  }

  // ═══════════════════════════════════════════════════════════
  // REFRESH TOKEN
  // ═══════════════════════════════════════════════════════════

  static async refreshAccessToken(userId: string) {
    try {
      const user = await prisma.user.findUnique({
        where: { id: userId },
        select: {
          google_refresh_token: true,
          google_token_expiry: true,
        },
      })

      if (!user?.google_refresh_token) {
        throw new Error('Refresh token não encontrado')
      }

      // Verificar se o token ainda é válido
      if (
        user.google_token_expiry &&
        user.google_token_expiry > new Date()
      ) {
        return // Token ainda válido
      }

      const auth = new google.auth.OAuth2(
        process.env.GOOGLE_CLIENT_ID,
        process.env.GOOGLE_CLIENT_SECRET,
        process.env.GOOGLE_CALENDAR_REDIRECT_URI
      )

      auth.setCredentials({
        refresh_token: user.google_refresh_token,
      })

      const { credentials } = await auth.refreshAccessToken()

      // Atualizar tokens no banco
      await prisma.user.update({
        where: { id: userId },
        data: {
          google_access_token: credentials.access_token,
          google_token_expiry: credentials.expiry_date
            ? new Date(credentials.expiry_date)
            : null,
        },
      })

      return credentials.access_token
    } catch (error) {
      console.error('Erro ao renovar access token:', error)
      throw error
    }
  }
}

import { GoogleCalendarService } from './calendar-service'

export class CalendarSyncService {
  // ═══════════════════════════════════════════════════════════
  // SINCRONIZAR VIAGEM
  // ═══════════════════════════════════════════════════════════

  static async syncTrip(tripId: string) {
    try {
      const trip = await prisma.trip.findUnique({
        where: { id: tripId },
        include: {
          tenant: true,
          reservations: {
            where: { status: 'confirmed' },
            include: {
              customer: true,
            },
          },
        },
      })

      if (!trip) {
        throw new Error('Viagem não encontrada')
      }

      // 1. Sincronizar para a AGÊNCIA
      await this.syncTripForTenant(trip)

      // 2. Sincronizar para VENDEDORES
      const salesUsers = await prisma.user.findMany({
        where: {
          tenant_id: trip.tenant_id,
          role: 'sales',
          google_calendar_enabled: true,
        },
      })

      for (const user of salesUsers) {
        // Verificar se o vendedor tem clientes nesta viagem
        const hasCustomers = trip.reservations.some(
          (r) => r.customer.assigned_to === user.id
        )

        if (hasCustomers) {
          await this.syncTripForUser(trip, user.id)
        }
      }

      // 3. Sincronizar para CLIENTES
      for (const reservation of trip.reservations) {
        if (reservation.customer.google_calendar_enabled) {
          await this.syncTripForCustomer(trip, reservation.customer.id)
        }
      }

      console.log(`✅ Viagem ${trip.id} sincronizada com sucesso!`)
    } catch (error) {
      console.error('Erro ao sincronizar viagem:', error)
      throw error
    }
  }

  // ═══════════════════════════════════════════════════════════
  // SINCRONIZAR VIAGEM PARA AGÊNCIA
  // ═══════════════════════════════════════════════════════════

  private static async syncTripForTenant(trip: any) {
    try {
      const adminUser = await prisma.user.findFirst({
        where: {
          tenant_id: trip.tenant_id,
          role: 'admin',
          google_calendar_enabled: true,
        },
      })

      if (!adminUser?.google_access_token) {
        console.log('Admin não tem Google Calendar habilitado')
        return
      }

      const calendarService = new GoogleCalendarService(
        adminUser.google_access_token
      )

      // Verificar se já existe evento
      const existingEvent = await prisma.calendarEvent.findFirst({
        where: {
          trip_id: trip.id,
          user_id: adminUser.id,
          event_type: 'trip',
        },
      })

      const eventData = {
        summary: `🚌 ${trip.name || trip.title}`,
        description: this.buildTripDescription(trip),
        location: `${trip.origin_city || ''} → ${trip.destination}`,
        startDateTime: trip.departure_date || new Date(),
        endDateTime: trip.return_date || addHours(trip.departure_date || new Date(), 12),
        attendees: trip.reservations
          .map((r: any) => r.customer.email)
          .filter(Boolean),
      }

      if (existingEvent) {
        // Atualizar evento existente
        const googleEvent = await calendarService.updateEvent(
          existingEvent.google_event_id,
          eventData
        )

        await prisma.calendarEvent.update({
          where: { id: existingEvent.id },
          data: {
            title: eventData.summary,
            description: eventData.description,
            start_date: eventData.startDateTime,
            end_date: eventData.endDateTime,
            location: eventData.location,
            synced_at: new Date(),
          },
        })
      } else {
        // Criar novo evento
        const googleEvent = await calendarService.createEvent(eventData)

        await prisma.calendarEvent.create({
          data: {
            tenant_id: trip.tenant_id,
            user_id: adminUser.id,
            trip_id: trip.id,
            google_event_id: googleEvent.id!,
            calendar_id: 'primary',
            event_type: 'trip',
            title: eventData.summary,
            description: eventData.description,
            start_date: eventData.startDateTime,
            end_date: eventData.endDateTime,
            location: eventData.location,
          },
        })
      }

      console.log(`✅ Viagem sincronizada para agência`)
    } catch (error) {
      console.error('Erro ao sincronizar viagem para agência:', error)
    }
  }

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const oauth2Client = new google.auth.OAuth2(
      process.env.GOOGLE_CLIENT_ID,
      process.env.GOOGLE_CLIENT_SECRET,
      process.env.GOOGLE_CALENDAR_REDIRECT_URI
    )

    const scopes = [

      'https://www.googleapis.com/auth/calendar.events',
    ]

    const url = oauth2Client.generateAuthUrl({
      access_type: 'offline',
      scope: scopes,
      prompt: 'consent',
      state: session.user.id, // Passar o ID do usuário
    })

    return NextResponse.json({ url })
  } catch (error) {
    console.error('Erro ao gerar URL de autorização:', error)
    return NextResponse.json(
      { error: 'Erro ao conectar com Google Calendar' },
      { status: 500 }
    )
  }
}

// app/api/google-calendar/callback/route.ts

import { NextResponse } from 'next/server'
import { google } from 'googleapis'
import { prisma } from '@/lib/prisma'

export async function GET(req: Request) {
  try {
    const { searchParams } = new URL(req.url)
    const code = searchParams.get('code')
    const state = searchParams.get('state') // userId
    const error = searchParams.get('error')

    if (error) {
      return NextResponse.redirect(
        `${process.env.NEXTAUTH_URL}/dashboard/settings?error=google_calendar_denied`
      )
    }

    if (!code || !state) {
      return NextResponse.redirect(
        `${process.env.NEXTAUTH_URL}/dashboard/settings?error=invalid_callback`
      )
    }

    const oauth2Client = new google.auth.OAuth2(
      process.env.GOOGLE_CLIENT_ID,
      process.env.GOOGLE_CLIENT_SECRET,
      process.env.GOOGLE_CALENDAR_REDIRECT_URI
    )

    // Trocar código por tokens
    const { tokens } = await oauth2Client.getToken(code)

    // Salvar tokens no banco
    await prisma.user.update({
      where: { id: state },
      data: {
        google_access_token: tokens.access_token,
        google_refresh_token: tokens.refresh_token,
        google_token_expiry: tokens.expiry_date
          ? new Date(tokens.expiry_date)
          : null,
        google_calendar_enabled: true,
      },
    })

    return NextResponse.redirect(
      `${process.env.NEXTAUTH_URL}/dashboard/settings?success=google_calendar_connected`
    )
  } catch (error) {
    console.error('Erro no callback do Google Calendar:', error)
    return NextResponse.redirect(
      `${process.env.NEXTAUTH_URL}/dashboard/settings?error=google_calendar_error`
    )
  }
}

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { google } from 'googleapis'

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const user = await prisma.user.findUnique({
      where: { id: session.user.id },
      select: { google_access_token: true },
    })

    // Revogar token no Google
    if (user?.google_access_token) {
      try {
        const oauth2Client = new google.auth.OAuth2(
          process.env.GOOGLE_CLIENT_ID,
          process.env.GOOGLE_CLIENT_SECRET,
          process.env.GOOGLE_CALENDAR_REDIRECT_URI
        )

        await oauth2Client.revokeToken(user.google_access_token)
      } catch (error) {
        console.error('Erro ao revogar token:', error)
      }
    }

    // Remover tokens do banco
    await prisma.user.update({
      where: { id: session.user.id },
      data: {
        google_access_token: null,
        google_refresh_token: null,
        google_token_expiry: null,
        google_calendar_enabled: false,
      },
    })

    // Deletar eventos sincronizados
    await prisma.calendarEvent.deleteMany({
      where: {
        user_id: session.user.id,
      },
    })

    return NextResponse.json({
      success: true,
      message: 'Google Calendar desconectado com sucesso',
    })
  } catch (error) {
    console.error('Erro ao desconectar Google Calendar:', error)
    return NextResponse.json(
      { error: 'Erro ao desconectar Google Calendar' },
      { status: 500 }
    )
  }

API de Sincronização Manual

// app/api/google-calendar/sync/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { CalendarSyncService } from '@/lib/google-calendar/sync-service'

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const body = await req.json()
    const { type, id } = body // type: 'trip' | 'payment' | 'birthday' | 'all'

    let synced = 0

    if (type === 'trip' && id) {
      await CalendarSyncService.syncTrip(id)
      synced = 1
    } else if (type === 'payment' && id) {
      await CalendarSyncService.syncPayment(id)
      synced = 1
    } else if (type === 'birthday' && id) {
      await CalendarSyncService.syncBirthday(id)
      synced = 1
    } else if (type === 'all') {
      // Sincronizar tudo
      const tenantId = session.user.tenantId

      // 1. Sincronizar todas as viagens ativas
      const trips = await prisma.trip.findMany({
        where: {
          tenant_id: tenantId,
          status: 'active',
        },
      })

      for (const trip of trips) {
        try {
          await CalendarSyncService.syncTrip(trip.id)
          synced++
        } catch (error) {
          console.error(`Erro ao sincronizar viagem ${trip.id}:`, error)
        }
      }

      // 2. Sincronizar todos os pagamentos pendentes
      const payments = await prisma.revenue.findMany({
        where: {
          tenant_id: tenantId,
          status: 'pending',
        },
      })

      for (const payment of payments) {
        try {
          await CalendarSyncService.syncPayment(payment.id)
          synced++
        } catch (error) {
          console.error(`Erro ao sincronizar pagamento ${payment.id}:`, error)
        }
      }

      // 3. Sincronizar aniversários
      const customers = await prisma.customer.findMany({
        where: {
          tenant_id: tenantId,
          birth_date: {
            not: null,
          },
        },
      })

      for (const customer of customers) {
        try {
          await CalendarSyncService.syncBirthday(customer.id)
          synced++
        } catch (error) {
          console.error(`Erro ao sincronizar aniversário ${customer.id}:`, error)
        }
      }
    }

    return NextResponse.json({
      success: true,
      message: `${synced} evento(s) sincronizado(s) com sucesso`,
      synced,
    })
  } catch (error) {
    console.error('Erro ao sincronizar:', error)
    return NextResponse.json(
      { error: 'Erro ao sincronizar eventos' },
      { status: 500 }
    )
  }
}

Componente de Botão de Sincronização Rápida

// components/google-calendar/sync-button.tsx

'use client'

import { useState } from 'react'
import { Button } from '@/components/ui/button'
import { Calendar, Loader2, CheckCircle } from 'lucide-react'
import { toast } from 'sonner'

interface SyncButtonProps {
  type: 'trip' | 'payment' | 'birthday'
  id: string
  variant?: 'default' | 'outline' | 'ghost'
  size?: 'default' | 'sm' | 'lg'
  showLabel?: boolean
}

export function GoogleCalendarSyncButton({
  type,
  id,
  variant = 'outline',
  size = 'sm',
  showLabel = true,
}: SyncButtonProps) {
  const [isSyncing, setIsSyncing] = useState(false)
  const [synced, setSynced] = useState(false)

  const handleSync = async () => {
    try {
      setIsSyncing(true)

      const response = await fetch('/api/google-calendar/sync', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ type, id }),
      })

      const data = await response.json()

      if (response.ok) {
        toast.success('Sincronizado com Google Calendar!')
        setSynced(true)
        setTimeout(() => setSynced(false), 3000)
      } else {
        toast.error(data.error || 'Erro ao sincronizar')
      }
    } catch (error) {
      console.error('Erro ao sincronizar:', error)
      toast.error('Erro ao sincronizar com Google Calendar')
    } finally {
      setIsSyncing(false)
    }
  }

  return (
    <Button
      onClick={handleSync}
      disabled={isSyncing || synced}
      variant={variant}
      size={size}
    >
      {isSyncing ? (
        <>
          <Loader2 className="h-4 w-4 animate-spin" />
          {showLabel && <span className="ml-2">Sincronizando...</span>}
        </>
      ) : synced ? (
        <>
          <CheckCircle className="h-4 w-4 text-green-600" />
          {showLabel && <span className="ml-2">Sincronizado!</span>}
        </>
      ) : (
        <>
          <Calendar className="h-4 w-4" />
          {showLabel && <span className="ml-2">Sincronizar</span>}
        </>
      )}
    </Button>
  )
}

Hook Automático de Sincronização

// hooks/use-google-calendar-sync.ts

import { useEffect } from 'react'
import { useSession } from 'next-auth/react'

export function useGoogleCalendarSync(
  type: 'trip' | 'payment' | 'birthday',
  id: string,
  enabled: boolean = true
) {
  const { data: session } = useSession()

  useEffect(() => {
    if (!enabled || !session?.user?.googleCalendarEnabled || !id) {
      return
    }

    const syncEvent = async () => {
      try {
        await fetch('/api/google-calendar/sync', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ type, id }),
        })
      } catch (error) {
        console.error('Erro ao sincronizar automaticamente:', error)
      }
    }

    // Sincronizar após 2 segundos (debounce)
    const timer = setTimeout(syncEvent, 2000)

    return () => clearTimeout(timer)
  }, [type, id, enabled, session])
}

// lib/google-calendar/auto-sync.ts

import { CalendarSyncService } from './sync-service'
import { prisma } from '@/lib/prisma'

export class AutoSyncService {
  // Sincronizar quando uma viagem for criada/atualizada
  static async onTripChange(tripId: string) {
    try {
      await CalendarSyncService.syncTrip(tripId)
      console.log(`✅ Viagem ${tripId} sincronizada automaticamente`)
    } catch (error) {
      console.error('Erro na sincronização automática:', error)
    }
  }

  // Sincronizar quando um pagamento for criado/atualizado
  static async onPaymentChange(revenueId: string) {
    try {
      await CalendarSyncService.syncPayment(revenueId)
      console.log(`✅ Pagamento ${revenueId} sincronizado automaticamente`)
    } catch (error) {
      console.error('Erro na sincronização automática:', error)
    }
  }

  // Sincronizar quando um cliente for criado/atualizado
  static async onCustomerChange(customerId: string) {
    try {
      const customer = await prisma.customer.findUnique({
        where: { id: customerId },
        select: { birth_date: true },
      })

      if (customer?.birth_date) {
        await CalendarSyncService.syncBirthday(customerId)
        console.log(`✅ Aniversário sincronizado automaticamente`)
      }
    } catch (error) {
      console.error('Erro na sincronização automática:', error)
    }
  }
}
