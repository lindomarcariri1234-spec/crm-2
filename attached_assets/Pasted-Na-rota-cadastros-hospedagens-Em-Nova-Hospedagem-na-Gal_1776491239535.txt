import { useState } from 'react'

interface GalleryImage {
  url: string

interface AccommodationGalleryUploadProps {
  value: GalleryImage[]
  onChange: (images: GalleryImage[]) => void
  disabled?: boolean
  maxImages?: number
}

// Componente de item sortable
function SortableImageItem({
  image,
  onRemove,
  disabled,
}: {
  image: GalleryImage
  onRemove: () => void
  disabled?: boolean
}) {
  const {
    attributes,
    listeners,
    setNodeRef,
    transform,
    transition,
    isDragging,
  } = useSortable({ id: image.key })

  const style = {
    transform: CSS.Transform.toString(transform),
    transition,
    opacity: isDragging ? 0.5 : 1,
  }

  return (
    <div
      ref={setNodeRef}
      style={style}
      className="relative group bg-white rounded-lg border-2 border-gray-200 overflow-hidden hover:border-blue-400 transition"
    >
      {/* Handle de arrastar */}
      <div
        {...attributes}
        {...listeners}
        className="absolute top-2 left-2 z-10 p-1 bg-white/90 rounded cursor-move hover:bg-white"
      >
        <GripVertical className="h-5 w-5 text-gray-600" />
      </div>

      {/* Botão remover */}
      <button
        type="button"
        onClick={onRemove}
        disabled={disabled}
        className="absolute top-2 right-2 z-10 p-1 bg-red-600 text-white rounded-full hover:bg-red-700 transition"
      >
        <X className="h-4 w-4" />
      </button>

      {/* Imagem */}
      <div className="relative h-40 w-full">
        <Image
          src={image.url}
          alt={`Imagem ${image.order + 1}`}
          fill
          className="object-cover"
        />
      </div>

      {/* Ordem */}
      <div className="absolute bottom-2 left-2 px-2 py-1 bg-blue-600 text-white text-xs font-bold rounded">
        #{image.order + 1}
      </div>
    </div>
  )
}

export function AccommodationGalleryUpload({
  value,
  onChange,
  disabled,
  maxImages = 10,
}: AccommodationGalleryUploadProps) {
  const [isUploading, setIsUploading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const sensors = useSensors(
    useSensor(PointerSensor),
    useSensor(KeyboardSensor, {
      coordinateGetter: sortableKeyboardCoordinates,
    })
  )

  const handleDragEnd = (event: DragEndEvent) => {
    const { active, over } = event

    if (over && active.id !== over.id) {
      const oldIndex = value.findIndex((img) => img.key === active.id)
      const newIndex = value.findIndex((img) => img.key === over.id)

      const reorderedImages = arrayMove(value, oldIndex, newIndex).map(
        (img, index) => ({
          ...img,
          order: index,
        })
      )

      onChange(reorderedImages)
    }
  }

  const handleRemove = (key: string) => {
    const updatedImages = value
      .filter((img) => img.key !== key)
      .map((img, index) => ({
        ...img,
        order: index,
      }))

    onChange(updatedImages)
  }

  const canUploadMore = value.length < maxImages

  return (
    <div className="space-y-6">
      {/* Header */}
      <div>
        <label className="block text-sm font-medium mb-2">
          📸 Galeria de Fotos
        </label>
        <p className="text-sm text-gray-600">
          Faça upload de até {maxImages} imagens da hospedagem (máx. 5MB cada)
        </p>
      </div>

      {/* Galeria Atual */}
      {value.length > 0 && (
        <div>
          <div className="flex items-center justify-between mb-4">
            <span className="text-sm font-semibold">
              {value.length} de {maxImages} imagens
            </span>
            <span className="text-xs text-gray-500">
              Arraste para reordenar
            </span>
          </div>

          <DndContext
            sensors={sensors}
            collisionDetection={closestCenter}
            onDragEnd={handleDragEnd}
          >
            <SortableContext
              items={value.map((img) => img.key)}
              strategy={verticalListSortingStrategy}
            >
              <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-4">
                {value.map((image) => (
                  <SortableImageItem
                    key={image.key}
                    image={image}
                    onRemove={() => handleRemove(image.key)}
                    disabled={disabled || isUploading}
                  />
                ))}
              </div>
            </SortableContext>
          </DndContext>
        </div>
      )}

      {/* Upload Zone */}
      {canUploadMore && (
        <Card className="p-6 border-2 border-dashed border-gray-300 hover:border-blue-400 transition bg-gray-50">
          <UploadDropzone
            endpoint="accommodationGallery"
            onClientUploadComplete={(res) => {
              if (res) {
                console.log('✅ Upload completo:', res)

                const newImages: GalleryImage[] = res.map((file, index) => ({
                  url: file.url,
                  key: file.key || '',
                  order: value.length + index,
                  name: file.name,
                }))

                onChange([...value, ...newImages])
                setError(null)
              }
              setIsUploading(false)
            }}
            onUploadError={(error: Error) => {
              console.error('❌ Erro no upload:', error)
              setError(error.message)
              setIsUploading(false)
            }}
            onUploadBegin={() => {
              console.log('🔄 Iniciando upload...')
              setIsUploading(true)
              setError(null)
            }}
            appearance={{
              container: 'w-full cursor-pointer',
              uploadIcon: 'text-blue-600 w-12 h-12',
              label:
                'text-blue-600 hover:text-blue-700 font-bold text-base mt-4',
              allowedContent: 'text-gray-500 text-sm mt-2',
              button:
                'bg-blue-600 hover:bg-blue-700 text-white px-6 py-2 rounded-lg font-semibold transition ut-ready:bg-blue-600 ut-uploading:bg-blue-400',
            }}
            content={{
              label: isUploading
                ? 'Enviando imagens...'
                : 'Clique ou arraste as imagens aqui',
              allowedContent: `PNG, JPG, JPEG ou WEBP (máx. 5MB cada) - ${
                maxImages - value.length
              } restantes`,
              button: isUploading ? (
                <span className="flex items-center gap-2">
                  <Loader2 className="h-4 w-4 animate-spin" />
                  Enviando...
                </span>
              ) : (
                'Selecionar imagens'
              ),
            }}
          />
        </Card>
      )}

      {/* Limite atingido */}
      {!canUploadMore && (
        <Alert>
          <CheckCircle className="h-4 w-4" />
          <AlertDescription>
            <strong>Limite de imagens atingido</strong>
            <br />
            Você já adicionou o máximo de {maxImages} imagens. Remova alguma
            para adicionar novas.
          </AlertDescription>
        </Alert>
      )}

      {/* Erro */}
      {error && (
        <Alert variant="destructive">
          <AlertCircle className="h-4 w-4" />
          <AlertDescription>
            <strong>Erro ao fazer upload:</strong> {error}
          </AlertDescription>
        </Alert>
      )}

      {/* Loading State */}
      {isUploading && (
        <div className="flex items-center justify-center gap-3 p-4 bg-blue-50 rounded-lg border-2 border-blue-200 animate-pulse">
          <Loader2 className="h-5 w-5 animate-spin text-blue-600" />
          <span className="text-sm font-semibold text-blue-700">
            Fazendo upload das imagens... Por favor, aguarde.
          </span>
        </div>
      )}

      {/* Dica */}
      <div className="bg-blue-50 border border-blue-200 rounded-lg p-4">
        <p className="text-sm text-blue-700">
          💡 <strong>Dica:</strong> A primeira imagem será usada como capa.
          Arraste as imagens para reordenar.
        </p>
      </div>
    </div>
  )
}

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    if (!hasPermission(session.user.role as any, 'accommodations.create')) {
      return NextResponse.json(
        { error: 'Sem permissão para criar hospedagens' },
        { status: 403 }
      )
    }

    const body = await req.json()

    console.log('📥 Dados recebidos:', body)

    // Validações
    if (!body.name) {
      return NextResponse.json(
        { error: 'Nome da hospedagem é obrigatório' },
        { status: 400 }
      )
    }

    const accommodation = await prisma.accommodation.create({
      data: {
        tenant_id: session.user.tenantId,
        created_by: session.user.id,
        
        // Dados básicos
        name: body.name,
        description: body.description || null,
        address: body.address || null,
        city: body.city || null,
        state: body.state || null,
        country: body.country || 'Brasil',
        
        // Contato
        phone: body.phone || null,
        email: body.email || null,
        website: body.website || null,
        
        // Acomodação
        total_rooms: body.total_rooms ? parseInt(body.total_rooms) : null,
        daily_rate: body.daily_rate ? parseFloat(body.daily_rate) : null,
        
        // Comodidades
        amenities: body.amenities || [],
        
        // GALERIA
        gallery_images: body.gallery_images || [],
        
        // Status
        status: 'active',
      },
    })

    console.log('✅ Hospedagem criada:', accommodation)

    return NextResponse.json({
      success: true,
      accommodation,
      message: 'Hospedagem cadastrada com sucesso!',
    })
  } catch (error: any) {
    console.error('❌ Erro ao criar hospedagem:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao criar hospedagem' },
      { status: 500 }
    )
  }
}
