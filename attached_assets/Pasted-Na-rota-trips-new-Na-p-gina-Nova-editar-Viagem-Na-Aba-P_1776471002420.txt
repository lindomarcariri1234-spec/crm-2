// components/dashboard/trips/financial/financial-summary.tsx

'use client'

import { Card } from '@/components/ui/card'

import {
  TrendingUp,

} from 'lucide-react'

interface FinancialSummaryProps {
  // Receita
  pricePerPax: number
  totalPax: number
  
  // Custos
  fixedCosts: Array<{ value: number }>
  variableCosts: Array<{ value_per_pax: number }>
  
  // Descontos
  discounts?: Array<{ value: number }>
  
  // Impostos
  taxes?: Array<{ percentage: number }>
  
  // Taxas financeiras
  financialFees?: Array<{ percentage: number }>
}

export function FinancialSummary({
  pricePerPax,
  totalPax,
  fixedCosts,
  variableCosts,
  discounts = [],
  taxes = [],
  financialFees = [],
}: FinancialSummaryProps) {
  // ═══════════════════════════════════════════════════════════
  // CÁLCULOS
  // ═══════════════════════════════════════════════════════════

  // 1. RECEITA BRUTA
  const grossRevenue = pricePerPax * totalPax

  // 2. CUSTOS FIXOS
  const totalFixedCost = fixedCosts.reduce((sum, cost) => sum + cost.value, 0)

  // 3. CUSTOS VARIÁVEIS
  const totalVariableCost = variableCosts.reduce(
    (sum, cost) => sum + cost.value_per_pax * totalPax,
    0
  )

  // 4. CUSTO TOTAL OPERACIONAL
  const totalOperationalCost = totalFixedCost + totalVariableCost

  // 5. DESCONTOS
  const totalDiscounts = discounts.reduce((sum, d) => sum + d.value, 0)

  // 6. RECEITA LÍQUIDA (após descontos)
  const netRevenue = grossRevenue - totalDiscounts

  // 7. IMPOSTOS (sobre receita líquida)
  const totalTaxPercentage = taxes.reduce((sum, t) => sum + t.percentage, 0)
  const totalTaxes = (netRevenue * totalTaxPercentage) / 100

  // 8. TAXAS FINANCEIRAS (sobre receita líquida)
  const totalFeePercentage = financialFees.reduce(
    (sum, f) => sum + f.percentage,
    0
  )
  const totalFinancialFees = (netRevenue * totalFeePercentage) / 100

  // 9. CUSTO TOTAL (operacional + impostos + taxas)
  const totalCost =
    totalOperationalCost + totalTaxes + totalFinancialFees + totalDiscounts

  // 10. LUCRO/PREJUÍZO
  const profit = netRevenue - totalCost + totalDiscounts

  // 11. MARGEM DE LUCRO
  const profitMargin = grossRevenue > 0 ? (profit / grossRevenue) * 100 : 0

  // 12. CUSTO POR PASSAGEIRO
  const costPerPax = totalPax > 0 ? totalCost / totalPax : 0

  // 13. PONTO DE EQUILÍBRIO
  const breakEvenPax =
    pricePerPax > 0
      ? Math.ceil(
          totalFixedCost /
            (pricePerPax -
              variableCosts.reduce((sum, c) => sum + c.value_per_pax, 0))
        )
      : 0

  // ═══════════════════════════════════════════════════════════
  // ANÁLISE
  // ═══════════════════════════════════════════════════════════

  const isProfit = profit > 0
  const isBreakEven = Math.abs(profit) < 10
  const isLoss = profit < 0

  const marginStatus =
    profitMargin >= 30
      ? 'excellent'
      : profitMargin >= 15
      ? 'good'
      : profitMargin >= 0
      ? 'low'
      : 'loss'

  return (
    <div className="space-y-6">
      <div>
        <h3 className="text-lg font-bold mb-2">📊 Resumo Financeiro</h3>
        <p className="text-sm text-gray-600">
          Análise completa de custos e rentabilidade
        </p>
      </div>

      {/* Cards Principais */}
      <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
        {/* Receita */}

                    <td className="text-right">R$ {simTotalCost.toFixed(2)}</td>
                    <td
                      className={`text-right ${
                        simProfit > 0 ? 'text-green-600' : 'text-red-600'
                      }`}
                    >
                      R$ {simProfit.toFixed(2)}
                    </td>
                    <td
                      className={`text-right ${
                        simMargin > 0 ? 'text-green-600' : 'text-red-600'
                      }`}
                    >
                      {simMargin.toFixed(1)}%
                    </td>
                  </tr>
                )
              })}
            </tbody>
          </table>
        </div>
      </Card>
    </div>
  )
}

// app/api/dashboard/trips/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { hasPermission } from '@/lib/permissions/check-permissions'

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    if (!hasPermission(session.user.role as any, 'trips.create')) {
      return NextResponse.json(
        { error: 'Sem permissão para criar viagens' },
        { status: 403 }
      )
    }

    const body = await req.json()

    console.log('📥 Dados recebidos:', body)

    // Validações
    if (!body.title || !body.destination) {
      return NextResponse.json(
        { error: 'Título e destino são obrigatórios' },
        { status: 400 }
      )
    }

    if (!body.image_url) {
      return NextResponse.json(
        { error: 'Imagem de capa é obrigatória' },
        { status: 400 }
      )
    }

    if (!body.price_adult) {
      return NextResponse.json(
        { error: 'Preço adulto é obrigatório' },
        { status: 400 }
      )
    }

    // ═══════════════════════════════════════════════════════════
    // CALCULAR TOTAIS
    // ═══════════════════════════════════════════════════════════

    const pricePerPax = parseFloat(body.price_adult)
    const totalPax = parseInt(body.max_passengers) || 0

    // Custos Fixos
    const totalFixedCost = (body.fixed_costs || []).reduce(
      (sum: number, cost: any) => sum + cost.value,
      0
    )

    // Custos Variáveis
    const totalVariableCost = (body.variable_costs || []).reduce(
      (sum: number, cost: any) => sum + cost.value_per_pax * totalPax,
      0
    )

    // Custo Total
    const totalCost = totalFixedCost + totalVariableCost

    // Receita
    const totalRevenue = pricePerPax * totalPax

    // Lucro
    const totalProfit = totalRevenue - totalCost

    // Margem
    const profitMargin = totalRevenue > 0 ? (totalProfit / totalRevenue) * 100 : 0

    // ═══════════════════════════════════════════════════════════
    // CRIAR VIAGEM
    // ═══════════════════════════════════════════════════════════

    const trip = await prisma.trip.create({
      data: {
        tenant_id: session.user.tenantId,
        created_by: session.user.id,
        
        // Dados básicos
        title: body.title,
        destination: body.destination,
        description: body.description || null,
        
        // IMAGENS
        image_url: body.image_url,
        image_key: body.image_key || null,
        gallery_images: body.gallery_images || [],
        
        // PREÇOS
        price_adult: parseFloat(body.price_adult),
        price_child: body.price_child ? parseFloat(body.price_child) : null,
        price_senior: body.price_senior ? parseFloat(body.price_senior) : null,
        price_launch: body.price_launch ? parseFloat(body.price_launch) : null,
        price_early: body.price_early ? parseFloat(body.price_early) : null,
        price_standard: body.price_standard
          ? parseFloat(body.price_standard)
          : null,
        price_last: body.price_last ? parseFloat(body.price_last) : null,
        
        // CAPACIDADE
        min_passengers: body.min_passengers
          ? parseInt(body.min_passengers)
          : null,
        max_passengers: body.max_passengers
          ? parseInt(body.max_passengers)
          : null,
        
        // CUSTOS
        fixed_costs: body.fixed_costs || [],
        variable_costs: body.variable_costs || [],
        discounts: body.discounts || [],
        taxes: body.taxes || [],
        financial_fees: body.financial_fees || [],
        
        // TOTAIS CALCULADOS
        total_fixed_cost: totalFixedCost,
        total_variable_cost: totalVariableCost,
        total_cost: totalCost,
        total_revenue: totalRevenue,
        total_profit: totalProfit,
        profit_margin: profitMargin,
        
        // Datas
        departure_date: body.departure_date
          ? new Date(body.departure_date)
          : null,
        return_date: body.return_date ? new Date(body.return_date) : null,
        duration_days: body.duration_days
          ? parseInt(body.duration_days)
          : null,
        
        // TRANSPORTE
        vehicle_type: body.vehicle_type || null,
        vehicle_plate: body.vehicle_plate || null,
        driver_name: body.driver_name || null,
        tour_guide: body.tour_guide || null,
        trip_organizer: body.trip_organizer || null,
        
        // Status
        status: 'active',
      },
    })

    console.log('✅ Viagem criada:', trip)

    return NextResponse.json({
      success: true,
      trip,
      message: 'Viagem criada com sucesso!',
      financial: {
        totalCost,
        totalRevenue,
        totalProfit,
        profitMargin,
      },
    })
  } catch (error: any) {
    console.error('❌ Erro ao criar viagem:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao criar viagem' },
      { status: 500 }
    )
  }
}
