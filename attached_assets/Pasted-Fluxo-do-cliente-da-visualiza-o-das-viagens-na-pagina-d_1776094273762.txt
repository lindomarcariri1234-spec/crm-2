import { prisma } from '@/lib/prisma'
import { ReservationWizard } from '@/components/public/reservation-wizard'
import { redirect } from 'next/navigation'

interface ReservationPageProps {
  params: { tripId: string }
  searchParams: { itinerary?: string }
}

export default async function ReservationPage({
  params,
  searchParams,
}: ReservationPageProps) {
  const { tripId } = params
  const { itinerary: itineraryId } = searchParams

  // Buscar viagem
  const trip = await prisma.trip.findUnique({
    where: { id: tripId },
    include: {
      images: true,
      itineraries: {
        where: {
          status: 'open',
          departure_date: { gte: new Date() },
        },
        orderBy: { departure_date: 'asc' },
      },
      included_items: true,
      not_included_items: true,
      seats: true,
    },
  })

  if (!trip || !trip.is_published) {
    redirect('/loja')
  }

  // Buscar itinerário específico ou usar o primeiro disponível
  const selectedItinerary = itineraryId
    ? trip.itineraries.find((it) => it.id === itineraryId)
    : trip.itineraries[0]

  if (!selectedItinerary) {
    redirect('/loja')
  }

  // Buscar assentos ocupados
  const occupiedSeats = await prisma.reservationSeat.findMany({
    where: {
      trip_id: tripId,
      itinerary_id: selectedItinerary.id,
      status: { in: ['reserved', 'confirmed'] },
    },
    select: { seat_number: true },
  })

  const occupiedSeatNumbers = occupiedSeats.map((s) => s.seat_number)

  return (
    <div className="min-h-screen bg-gray-50">
      <ReservationWizard
        trip={trip}
        itinerary={selectedItinerary}
        occupiedSeats={occupiedSeatNumbers}
      />
    </div>
  )
}

// components/public/reservation-wizard.tsx

'use client'

import { useState } from 'react'
import { Card } from '@/components/ui/card'
import { Button } from '@/components/ui/button'
import { Progress } from '@/components/ui/progress'
import { ArrowLeft, ArrowRight, Check } from 'lucide-react'
import { StepIndicator } from './wizard/step-indicator'
import { StepDados } from './wizard/step-dados'
import { StepRevisao } from './wizard/step-revisao'
import { StepAssentos } from './wizard/step-assentos'
import { StepPagamento } from './wizard/step-pagamento'
import { StepConfirmacao } from './wizard/step-confirmacao'
import { useRouter } from 'next/navigation'

interface ReservationWizardProps {
  trip: any
  itinerary: any
  occupiedSeats: string[]
}

const STEPS = [
  { id: 1, name: 'Dados', icon: '👤' },
  { id: 2, name: 'Revisão', icon: '📋' },
  { id: 3, name: 'Selecionar assento', icon: '💺' },
  { id: 4, name: 'Pagamento', icon: '💳' },
  { id: 5, name: 'Confirmação', icon: '✅' },
]

export function ReservationWizard({
  trip,
  itinerary,
  occupiedSeats,
}: ReservationWizardProps) {
  const router = useRouter()
  const [currentStep, setCurrentStep] = useState(1)
  const [loading, setLoading] = useState(false)

  // Estado do formulário
  const [formData, setFormData] = useState({
    // Dados pessoais
    name: '',
    cpf: '',
    email: '',
    phone: '',
    birthdate: '',
    
    // Quantidade de passageiros
    passengers_count: 1,
    
    // Assentos selecionados
    selected_seats: [] as string[],
    
    // Códigos e cupons
    referral_code: '',
    coupon_code: '',
    
    // Descontos aplicados
    referral_discount: 0,
    coupon_discount: 0,
    
    // Pagamento
    payment_method: '',
    
    // Observações
    observations: '',
  })

  const [reservation, setReservation] = useState<any>(null)

  const updateFormData = (data: Partial<typeof formData>) => {
    setFormData((prev) => ({ ...prev, ...data }))
  }

  const canProceed = () => {
    switch (currentStep) {
      case 1:
        return (
          formData.name &&
          formData.cpf &&
          formData.email &&
          formData.phone &&
          formData.birthdate
        )
      case 2:
        return true
      case 3:
        return formData.selected_seats.length === formData.passengers_count
      case 4:
        return formData.payment_method !== ''
      case 5:
        return true
      default:
        return false
    }
  }

  const handleNext = async () => {
    if (currentStep === 4) {
      // Finalizar reserva
      await handleFinishReservation()
    } else {
      setCurrentStep((prev) => Math.min(prev + 1, STEPS.length))
    }
  }

  const handleBack = () => {
    setCurrentStep((prev) => Math.max(prev - 1, 1))
  }

  const handleFinishReservation = async () => {
    setLoading(true)

    try {
      const response = await fetch('/api/public/reservations', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          trip_id: trip.id,
          itinerary_id: itinerary.id,
          ...formData,
        }),
      })

      const data = await response.json()

      if (!response.ok) {
        throw new Error(data.error || 'Erro ao criar reserva')
      }

      setReservation(data.reservation)
      setCurrentStep(5)
    } catch (error: any) {
      alert(error.message)
    } finally {
      setLoading(false)
    }
  }

  const calculateTotal = () => {
    const basePrice = trip.promotional_price || trip.price_per_person
    const subtotal = basePrice * formData.passengers_count
    const totalDiscount = formData.referral_discount + formData.coupon_discount
    return subtotal - totalDiscount
  }

  const progress = (currentStep / STEPS.length) * 100

  return (
    <div className="min-h-screen bg-gray-50 py-8">
      <div className="container mx-auto px-4 max-w-5xl">
        {/* Header */}
        <div className="mb-8">
          <Button
            variant="ghost"
            onClick={() => router.back()}
            className="mb-4"
          >
            <ArrowLeft className="h-4 w-4 mr-2" />
            Voltar
          </Button>

          <h1 className="text-3xl font-bold mb-2">Finalizar Reserva</h1>
          <p className="text-gray-600">{trip.title}</p>
        </div>

        {/* Progress Bar */}
        <div className="mb-8">
          <Progress value={progress} className="h-2" />
        </div>

        {/* Step Indicator */}
        <StepIndicator steps={STEPS} currentStep={currentStep} />

        {/* Content */}
        <div className="mt-8">
          {currentStep === 1 && (
            <StepDados
              formData={formData}
              updateFormData={updateFormData}
              trip={trip}
              itinerary={itinerary}
            />
          )}

          {currentStep === 2 && (
            <StepRevisao
              formData={formData}
              updateFormData={updateFormData}
              trip={trip}
              itinerary={itinerary}
              calculateTotal={calculateTotal}
            />
          )}

          {currentStep === 3 && (
            <StepAssentos
              formData={formData}
              updateFormData={updateFormData}
              trip={trip}
              occupiedSeats={occupiedSeats}
            />
          )}

          {currentStep === 4 && (
            <StepPagamento
              formData={formData}
              updateFormData={updateFormData}
              trip={trip}
              calculateTotal={calculateTotal}
            />
          )}

          {currentStep === 5 && reservation && (
            <StepConfirmacao reservation={reservation} trip={trip} />
          )}
        </div>

        {/* Navigation Buttons */}
        {currentStep < 5 && (
          <div className="flex justify-between mt-8">
            <Button
              variant="outline"
              onClick={handleBack}
              disabled={currentStep === 1 || loading}
            >
              <ArrowLeft className="h-4 w-4 mr-2" />
              Voltar
            </Button>

            <Button
              onClick={handleNext}
              disabled={!canProceed() || loading}
              className="bg-orange-500 hover:bg-orange-600"
            >
              {loading ? (
                'Processando...'
              ) : currentStep === 4 ? (
                <>
                  <Check className="h-4 w-4 mr-2" />
                  Confirmar Reserva
                </>
              ) : (
                <>
                  Próximo
                  <ArrowRight className="h-4 w-4 ml-2" />
                </>
              )}
            </Button>
          </div>
        )}
      </div>
    </div>
  )
}

// components/public/wizard/step-indicator.tsx

'use client'

import { Check } from 'lucide-react'

interface Step {
  id: number
  name: string
  icon: string
}

interface StepIndicatorProps {
  steps: Step[]
  currentStep: number
}

export function StepIndicator({ steps, currentStep }: StepIndicatorProps) {
  return (
    <div className="flex items-center justify-between">
      {steps.map((step, index) => {
        const isCompleted = currentStep > step.id
        const isCurrent = currentStep === step.id
        const isUpcoming = currentStep < step.id

        return (
          <div key={step.id} className="flex items-center flex-1">
            {/* Step Circle */}
            <div className="flex flex-col items-center">
              <div
                className={`
                  flex items-center justify-center w-12 h-12 rounded-full border-2 transition-all
                  ${
                    isCompleted
                      ? 'bg-green-500 border-green-500 text-white'
                      : isCurrent
                      ? 'bg-blue-500 border-blue-500 text-white'
                      : 'bg-white border-gray-300 text-gray-400'
                  }
                `}
              >
                {isCompleted ? (
                  <Check className="h-6 w-6" />
                ) : (
                  <span className="text-2xl">{step.icon}</span>
                )}
              </div>
              <p
                className={`
                  mt-2 text-sm font-medium text-center
                  ${isCurrent ? 'text-blue-600' : 'text-gray-600'}
                `}
              >
                {step.name}
              </p>
            </div>

            {/* Connector Line */}
            {index < steps.length - 1 && (
              <div
                className={`
                  flex-1 h-0.5 mx-2 transition-all
                  ${isCompleted ? 'bg-green-500' : 'bg-gray-300'}
                `}
              />
            )}
          </div>
        )
      })}
    </div>
  )
}

import { Card } from '@/components/ui/card'

import { useState } from 'react'
import { Tag, Percent, Loader2 } from 'lucide-react'
import { format } from 'date-fns'
import { ptBR } from 'date-fns/locale'

interface StepRevisaoProps {
  formData: any
  updateFormData: (data: any) => void
  trip: any
  itinerary: any
  calculateTotal: () => number
}

export function StepRevisao({
  formData,
  updateFormData,
  trip,
  itinerary,
  calculateTotal,
}: StepRevisaoProps) {
  const [loadingReferral, setLoadingReferral] = useState(false)
  const [loadingCoupon, setLoadingCoupon] = useState(false)

  const basePrice = trip.promotional_price || trip.price_per_person
  const subtotal = basePrice * formData.passengers_count

  const handleApplyReferral = async () => {
    if (!formData.referral_code) return

    setLoadingReferral(true)

    try {
      const response = await fetch('/api/public/validate-referral', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ code: formData.referral_code }),
      })

      const data = await response.json()

      if (response.ok) {
        const discount = subtotal * 0.05 // 5% de desconto
        updateFormData({ referral_discount: discount })
        alert('✅ Código de indicação aplicado! Você ganhou 5% de desconto')
      } else {
        alert('❌ ' + data.error)
      }
    } catch (error) {
      alert('Erro ao validar código')
    } finally {
      setLoadingReferral(false)
    }
  }

  const handleApplyCoupon = async () => {
    if (!formData.coupon_code) return

    setLoadingCoupon(true)

    try {
      const response = await fetch('/api/public/validate-coupon', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          code: formData.coupon_code,
          trip_id: trip.id,
        }),
      })

      const data = await response.json()

      if (response.ok) {
        const discount = data.coupon.discount_type === 'percentage'
          ? (subtotal * data.coupon.discount_value) / 100
          : data.coupon.discount_value

        updateFormData({ coupon_discount: discount })
        alert(`✅ Cupom aplicado! Desconto de R$ ${discount.toFixed(2)}`)
      } else {
        alert('❌ ' + data.error)
      }
    } catch (error) {
      alert('Erro ao validar cupom')
    } finally {
      setLoadingCoupon(false)
    }
  }

  return (
    <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
      {/* Resumo da Viagem */}
      <div className="lg:col-span-2 space-y-6">
        <Card className="p-6">
          <h2 className="text-2xl font-bold mb-6">Resumo da Reserva</h2>

          {/* Viagem */}

              <p className="text-gray-600">📍 {trip.destination}</p>
            </div>

            <Separator />

            {/* Data */}
            <div>
              <p className="text-sm text-gray-600">Data de Saída</p>
              <p className="font-semibold">
                {format(new Date(itinerary.departure_date), "dd 'de' MMMM 'de' yyyy", {
                  locale: ptBR,
                })}
              </p>
            </div>

            <Separator />

            {/* Passageiro */}
            <div>
              <p className="text-sm text-gray-600">Passageiro Principal</p>
              <p className="font-semibold">{formData.name}</p>
              <p className="text-sm text-gray-600">CPF: {formData.cpf}</p>
              <p className="text-sm text-gray-600">Email: {formData.email}</p>
              <p className="text-sm text-gray-600">Telefone: {formData.phone}</p>
            </div>

            <Separator />

            {/* Quantidade */}
            <div>
              <p className="text-sm text-gray-600">Quantidade de Passageiros</p>
              <p className="font-semibold">
                {formData.passengers_count}{' '}
                {formData.passengers_count === 1 ? 'passageiro' : 'passageiros'}
              </p>
            </div>
          </div>
        </Card>

        {/* Códigos e Cupons */}
        <Card className="p-6">
          <h3 className="text-lg font-bold mb-4">Códigos de Desconto</h3>

          <div className="space-y-4">
            {/* Código de Indicação */}
            <div>
              <Label htmlFor="referral_code">Código de Indicação</Label>
              <div className="flex gap-2">
                <div className="relative flex-1">
                  <Tag className="absolute left-3 top-3 h-5 w-5 text-gray-400" />
                  <Input
                    id="referral_code"
                    value={formData.referral_code}
                    onChange={(e) =>
                      updateFormData({ referral_code: e.target.value.toUpperCase() })
                    }
                    placeholder="Ex: MARIA2024"
                    className="pl-10"
                    disabled={formData.referral_discount > 0}
                  />
                </div>
                <Button
                  onClick={handleApplyReferral}
                  disabled={
                    !formData.referral_code ||
                    formData.referral_discount > 0 ||
                    loadingReferral
                  }
                  variant="outline"
                >
                  {loadingReferral ? (
                    <Loader2 className="h-4 w-4 animate-spin" />
                  ) : (
                    'Aplicar'
                  )}
                </Button>
              </div>
              {formData.referral_discount > 0 && (
                <p className="text-sm text-green-600 mt-1">
                  ✅ Desconto de 5% aplicado (R$ {formData.referral_discount.toFixed(2)})
                </p>
              )}
            </div>

            {/* Cupom de Desconto */}
            <div>
              <Label htmlFor="coupon_code">Cupom de Desconto</Label>
              <div className="flex gap-2">
                <div className="relative flex-1">
                  <Percent className="absolute left-3 top-3 h-5 w-5 text-gray-400" />
                  <Input
                    id="coupon_code"
                    value={formData.coupon_code}
                    onChange={(e) =>
                      updateFormData({ coupon_code: e.target.value.toUpperCase() })
                    }
                    placeholder="Ex: PROMO10"
                    className="pl-10"
                    disabled={formData.coupon_discount > 0}
                  />
                </div>
                <Button
                  onClick={handleApplyCoupon}
                  disabled={
                    !formData.coupon_code ||
                    formData.coupon_discount > 0 ||
                    loadingCoupon
                  }

// components/public/wizard/step-assentos.tsx

'use client'

import { Card } from '@/components/ui/card'
import { Badge } from '@/components/ui/badge'
import { Alert, AlertDescription } from '@/components/ui/alert'
import { Info, Check } from 'lucide-react'
import { useState } from 'react'

interface StepAssentosProps {
  formData: any
  updateFormData: (data: any) => void
  trip: any
  occupiedSeats: string[]
}

export function StepAssentos({
  formData,
  updateFormData,
  trip,
  occupiedSeats,
}: StepAssentosProps) {
  const [selectedSeats, setSelectedSeats] = useState<string[]>(
    formData.selected_seats || []
  )

  // Configuração do layout (2x2 padrão)
  const seatsPerRow = 4
  const totalSeats = trip.total_seats || 46
  const rows = Math.ceil(totalSeats / seatsPerRow)

  const handleSeatClick = (seatNumber: string) => {
    // Verifica se o assento está ocupado
    if (occupiedSeats.includes(seatNumber)) {
      return
    }

    let newSelectedSeats = [...selectedSeats]

    if (newSelectedSeats.includes(seatNumber)) {
      // Remove assento
      newSelectedSeats = newSelectedSeats.filter((s) => s !== seatNumber)
    } else {
      // Adiciona assento (se não exceder a quantidade)
      if (newSelectedSeats.length < formData.passengers_count) {
        newSelectedSeats.push(seatNumber)
      } else {
        alert(
          `Você só pode selecionar ${formData.passengers_count} assento(s)`
        )
        return
      }
    }

    setSelectedSeats(newSelectedSeats)
    updateFormData({ selected_seats: newSelectedSeats })
  }

  const getSeatStatus = (seatNumber: string) => {
    if (occupiedSeats.includes(seatNumber)) return 'occupied'
    if (selectedSeats.includes(seatNumber)) return 'selected'
    return 'available'
  }

  const getSeatColor = (status: string) => {
    switch (status) {
      case 'occupied':
        return 'bg-gray-300 cursor-not-allowed'
      case 'selected':
        return 'bg-orange-500 text-white cursor-pointer hover:bg-orange-600'
      case 'available':
        return 'bg-white border-2 border-gray-300 cursor-pointer hover:border-orange-500 hover:bg-orange-50'
      default:
        return 'bg-white'
    }
  }

  // Gerar números dos assentos
  const generateSeatNumber = (rowIndex: number, seatIndex: number) => {
    return (rowIndex * seatsPerRow + seatIndex + 1).toString()
  }

  return (
    <div className="space-y-6">
      {/* Informações */}
      <Alert>
        <Info className="h-4 w-4" />
        <AlertDescription>
          Selecione <strong>{formData.passengers_count}</strong>{' '}
          {formData.passengers_count === 1 ? 'assento' : 'assentos'} para sua
          viagem. Os assentos em cinza já estão ocupados.
        </AlertDescription>
      </Alert>

      {/* Legenda */}
      <Card className="p-4">
        <div className="flex items-center justify-center gap-8">
          <div className="flex items-center gap-2">
            <div className="w-8 h-8 bg-white border-2 border-gray-300 rounded" />
            <span className="text-sm">Disponível</span>
          </div>
          <div className="flex items-center gap-2">
            <div className="w-8 h-8 bg-orange-500 rounded" />
            <span className="text-sm">Selecionado</span>
          </div>
          <div className="flex items-center gap-2">
            <div className="w-8 h-8 bg-gray-300 rounded" />
            <span className="text-sm">Ocupado</span>
          </div>
        </div>
      </Card>

      {/* Mapa de Assentos */}
      <Card className="p-6">
        <div className="max-w-2xl mx-auto">
          {/* Frente do Ônibus */}
          <div className="mb-6">
            <div className="bg-gray-800 text-white text-center py-3 rounded-t-lg font-semibold">
              🚌 FRENTE DO ÔNIBUS
            </div>
          </div>

          {/* Grid de Assentos */}
          <div className="space-y-3">
            {Array.from({ length: rows }, (_, rowIndex) => (
              <div key={rowIndex} className="flex justify-center gap-3">
                {/* Lado Esquerdo (2 assentos) */}
                <div className="flex gap-2">
                  {Array.from({ length: 2 }, (_, seatIndex) => {
                    const seatNumber = generateSeatNumber(rowIndex, seatIndex)
                    const status = getSeatStatus(seatNumber)

                    if (parseInt(seatNumber) > totalSeats) return null

                    return (
                      <button
                        key={seatNumber}
                        onClick={() => handleSeatClick(seatNumber)}
                        disabled={status === 'occupied'}
                        className={`
                          w-14 h-14 rounded-lg font-semibold text-sm
                          transition-all duration-200 relative
                          ${getSeatColor(status)}
                        `}
                      >
                        {seatNumber}
                        {status === 'selected' && (
                          <Check className="absolute top-1 right-1 h-4 w-4" />
                        )}
                      </button>
                    )
                  })}
                </div>

                {/* Corredor */}
                <div className="w-8" />

                {/* Lado Direito (2 assentos) */}
                <div className="flex gap-2">
                  {Array.from({ length: 2 }, (_, seatIndex) => {
                    const seatNumber = generateSeatNumber(
                      rowIndex,
                      seatIndex + 2
                    )
                    const status = getSeatStatus(seatNumber)

                    if (parseInt(seatNumber) > totalSeats) return null

                    return (
                      <button
                        key={seatNumber}
                        onClick={() => handleSeatClick(seatNumber)}
                        disabled={status === 'occupied'}
                        className={`
                          w-14 h-14 rounded-lg font-semibold text-sm
                          transition-all duration-200 relative
                          ${getSeatColor(status)}
                        `}
                      >
                        {seatNumber}
                        {status === 'selected' && (
                          <Check className="absolute top-1 right-1 h-4 w-4" />
                        )}
                      </button>
                    )
                  })}
                </div>
              </div>
            ))}
          </div>
        </div>
      </Card>

      {/* Resumo da Seleção */}
      <Card className="p-4">
        <div className="flex items-center justify-between">
          <div>
            <p className="text-sm text-gray-600">Assentos Selecionados</p>
            <div className="flex gap-2 mt-2">
              {selectedSeats.length > 0 ? (
                selectedSeats.map((seat) => (
                  <Badge key={seat} className="bg-orange-500">
                    Assento {seat}
                  </Badge>
                ))
              ) : (
                <p className="text-gray-400">Nenhum assento selecionado</p>
              )}
            </div>
          </div>

          <div className="text-right">
            <p className="text-sm text-gray-600">Progresso</p>
            <p className="text-2xl font-bold text-orange-500">
              {selectedSeats.length} / {formData.passengers_count}
            </p>
          </div>
        </div>
      </Card>
    </div>
  )
}

API DE CRIAÇÃO DE RESERVA

// app/api/public/reservations/route.ts

import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { eventEmitter } from '@/lib/events/event-emitter'

export async function POST(req: Request) {
  try {
    const body = await req.json()

    const {
      trip_id,
      itinerary_id,
      name,
      cpf,
      email,
      phone,
      birthdate,
      passengers_count,
      selected_seats,
      referral_code,
      coupon_code,
      referral_discount,
      coupon_discount,
      payment_method,
      observations,
    } = body

    // Validações
    if (!trip_id || !itinerary_id || !name || !cpf || !email || !phone) {
      return NextResponse.json(
        { error: 'Campos obrigatórios faltando' },
        { status: 400 }
      )
    }

    // Buscar viagem e itinerário
    const trip = await prisma.trip.findUnique({
      where: { id: trip_id },
      include: { itineraries: true },
    })

    if (!trip) {
      return NextResponse.json(
        { error: 'Viagem não encontrada' },
        { status: 404 }
      )
    }

    const itinerary = trip.itineraries.find((it) => it.id === itinerary_id)

    if (!itinerary) {
      return NextResponse.json(
        { error: 'Itinerário não encontrado' },
        { status: 404 }
      )
    }

    // Verificar disponibilidade
    if (itinerary.available_seats < passengers_count) {
      return NextResponse.json(
        { error: 'Vagas insuficientes' },
        { status: 400 }
      )
    }

    // Verificar se CPF já tem reserva nesta viagem
    const existingReservation = await prisma.reservation.findFirst({
      where: {
        trip_id,
        itinerary_id,
        client_cpf: cpf,
        status: { not: 'cancelled' },
      },
    })

    if (existingReservation) {
      return NextResponse.json(
        { error: 'CPF já possui reserva nesta viagem' },
        { status: 400 }
      )
    }

    // Calcular valores
    const basePrice = trip.promotional_price || trip.price_per_person
    const subtotal = basePrice * passengers_count
    const totalDiscount = (referral_discount || 0) + (coupon_discount || 0)
    const totalAmount = subtotal - totalDiscount

    // Buscar ou criar cliente
    let client = await prisma.client.findFirst({
      where: {
        tenant_id: trip.tenant_id,
        OR: [{ cpf }, { email }],
      },
    })

    if (!client) {
      client = await prisma.client.create({
        data: {
          tenant_id: trip.tenant_id,
          name,
          cpf,
          email,
          phone,
          birthdate: new Date(birthdate),
          source: 'website',
          segment: 'individual',
        },
      })
    }

    // Gerar número de reserva e voucher
    const reservationNumber = await generateReservationNumber(trip.tenant_id)
    const voucherCode = generateVoucherCode()

    // Criar reserva
    const reservation = await prisma.reservation.create({
      data: {
        tenant_id: trip.tenant_id,
        reservation_number: reservationNumber,
        voucher_code: voucherCode,
        trip_id,
        itinerary_id,
        client_id: client.id,
        client_name: name,
        client_cpf: cpf,
        client_email: email,
        client_phone: phone,
        total_passengers: passengers_count,
        adults_count: passengers_count,
        subtotal_amount: subtotal,
        discount_amount: totalDiscount,
        total_amount: totalAmount,
        amount_pending: totalAmount,
        payment_method,
        payment_status: 'pending',
        status: 'pending',
        source: 'website',
        observations,
        departure_date: itinerary.departure_date,
      },
      include: {
        seats: true,
      },
    })

    // Criar assentos
    for (const seatNumber of selected_seats) {
      await prisma.reservationSeat.create({
        data: {
          reservation_id: reservation.id,
          trip_id,
          itinerary_id,
          seat_number: seatNumber,
          status: 'reserved',
        },
      })
    }

    // Processar código de indicação
    if (referral_code) {
      const referral = await prisma.referral.findFirst({
        where: { code: referral_code, status: 'active' },
      })

      if (referral) {
        await prisma.reservation.update({
          where: { id: reservation.id },
          data: { referral_id: referral.id },
        })

        await prisma.referral.update({
          where: { id: referral.id },
          data: { status: 'pending' },
        })
      }
    }

    // Atualizar vagas do itinerário
    await prisma.itinerary.update({
      where: { id: itinerary_id },
      data: {
        available_seats: { decrement: passengers_count },
      },
    })

    // Emitir evento
    eventEmitter.emitEvent('reservation.created', {
      tenant_id: trip.tenant_id,
      timestamp: new Date(),
      data: reservation,
    })

    // Buscar reserva completa
    const fullReservation = await prisma.reservation.findUnique({
      where: { id: reservation.id },
      include: {
        seats: true,
        client: true,
      },
    })

    return NextResponse.json({
      success: true,
      reservation: fullReservation,
    })
  } catch (error: any) {
    console.error('Erro ao criar reserva:', error)
    return NextResponse.json(
      { error: 'Erro ao criar reserva' },
      { status: 500 }
    )
  }
}

// Utilitários
async function generateReservationNumber(tenantId: string): Promise<string> {
  const lastReservation = await prisma.reservation.findFirst({
    where: { tenant_id: tenantId },
    orderBy: { created_at: 'desc' },
    select: { reservation_number: true },
  })

  const lastNumber = lastReservation
    ? parseInt(lastReservation.reservation_number.split('-').pop() || '0')
    : 0

  return `RES-${new Date().getFullYear()}-${String(lastNumber + 1).padStart(
    5,
    '0'
  )}`
}

function generateVoucherCode(): string {
  const characters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
  let code = 'VC-'

  for (let i = 0; i < 6; i++) {
    code += characters.charAt(Math.floor(Math.random() * characters.length))
  }

  return code
}
