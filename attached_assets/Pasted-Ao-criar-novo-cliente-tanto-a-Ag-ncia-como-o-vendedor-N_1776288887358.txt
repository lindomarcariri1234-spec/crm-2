interface Trip {

  totalSeats: number
  layoutId: string

interface TripSelectorWithDetailsProps {
  value: string
  onChange: (tripId: string) => void
  onTripLoaded: (trip: Trip) => void
}

export function TripSelectorWithDetails({
  value,
  onChange,
  onTripLoaded,
}: TripSelectorWithDetailsProps) {
  const [trips, setTrips] = useState<Trip[]>([])
  const [selectedTrip, setSelectedTrip] = useState<Trip | null>(null)
  const [isLoading, setIsLoading] = useState(true)

  // Carregar viagens disponíveis
  useEffect(() => {
    loadTrips()
  }, [])

  // Carregar detalhes quando selecionar
  useEffect(() => {
    if (value) {
      loadTripDetails(value)
    }
  }, [value])

  const loadTrips = async () => {
    try {
      setIsLoading(true)
      const response = await fetch('/api/dashboard/trips/available')
      const data = await response.json()
      setTrips(data.trips)
    } catch (error) {
      console.error('Erro ao carregar viagens:', error)
    } finally {
      setIsLoading(false)
    }
  }

  const loadTripDetails = async (tripId: string) => {
    try {
      const response = await fetch(`/api/dashboard/trips/${tripId}/details`)
      const data = await response.json()
      setSelectedTrip(data.trip)
      onTripLoaded(data.trip)
    } catch (error) {
      console.error('Erro ao carregar detalhes da viagem:', error)
    }

    onChange(tripId)
    loadTripDetails(tripId)

// app/api/dashboard/trips/[tripId]/details/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function GET(
  req: Request,
  { params }: { params: { tripId: string } }
) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const trip = await prisma.trip.findUnique({
      where: {
        id: params.tripId,
        tenant_id: session.user.tenantId,
      },
      include: {
        layout: true,
        itineraries: {
          where: {
            status: 'active',
          },
          orderBy: {
            departure_date: 'asc',
          },
          take: 1,
        },
      },
    })

    if (!trip) {
      return NextResponse.json(
        { error: 'Viagem não encontrada' },
        { status: 404 }
      )
    }

    const itinerary = trip.itineraries[0]

    if (!itinerary) {
      return NextResponse.json(
        { error: 'Nenhum itinerário disponível' },
        { status: 404 }
      )
    }

    // Calcular assentos ocupados
    const occupiedSeats = await prisma.reservationSeat.count({
      where: {
        trip_id: trip.id,
        itinerary_id: itinerary.id,
        status: {
          in: ['reserved', 'confirmed', 'checked_in'],
        },
      },
    })

    const totalSeats = trip.layout?.total_seats || itinerary.total_seats
    const availableSeats = totalSeats - occupiedSeats

    return NextResponse.json({
      trip: {
        id: trip.id,
        title: trip.title,
        destination: trip.destination,
        departureDate: itinerary.departure_date,
        pricePerPerson: Number(trip.promotional_price || trip.price_per_person),
        availableSeats,
        totalSeats,
        layoutId: trip.layout_id,
        itineraryId: itinerary.id,
      },
    })
  } catch (error) {
    console.error('Erro ao buscar detalhes da viagem:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar detalhes' },
      { status: 500 }
    )
  }
}

// app/api/dashboard/clients/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { findOrCreateClient } from '@/lib/clients/find-or-create'
import { generateReservationNumber } from '@/lib/reservations/generate-number-complete'
import { prisma } from '@/lib/prisma'

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const body = await req.json()

    const {
      name,
      cpf,
      email,
      phone,
      birthdate,
      tripId,
      itineraryId,
      selectedSeats,
      totalPrice,
      hasInsurance,
      hasChild,
      observations,
    } = body

    // ═══════════════════════════════════════════════════════════
    // 1. BUSCAR OU CRIAR CLIENTE
    // ═══════════════════════════════════════════════════════════
    
    const { client } = await findOrCreateClient({
      tenantId: session.user.tenantId,
      data: {
        name,
        cpf,
        email,
        phone,
        birthdate: new Date(birthdate),
      },
      userId: session.user.id,
      source: 'dashboard',
    })

    // ═══════════════════════════════════════════════════════════
    // 2. CRIAR RESERVA
    // ═══════════════════════════════════════════════════════════
    
    const trip = await prisma.trip.findUnique({
      where: { id: tripId },
    })

    if (!trip) {
      return NextResponse.json(
        { error: 'Viagem não encontrada' },
        { status: 404 }
      )
    }

    const reservationNumberData = await generateReservationNumber(
      session.user.tenantId,
      trip.type as any
    )

    const reservation = await prisma.reservation.create({
      data: {
        tenant_id: session.user.tenantId,
        reservation_number: reservationNumberData.full,
        number_prefix: reservationNumberData.prefix,
        number_type: reservationNumberData.typeCode,
        number_yearmonth: reservationNumberData.yearMonth,
        number_sequence: reservationNumberData.sequence,
        voucher_code: generateVoucherCode(),
        trip_id: tripId,
        itinerary_id: itineraryId,
        client_id: client.id,
        client_name: client.name,
        client_cpf: client.cpf,
        client_email: client.email,
        client_phone: client.phone,
        total_passengers: selectedSeats.length,
        adults_count: selectedSeats.length,
        subtotal_amount: totalPrice,
        total_amount: totalPrice,
        amount_pending: totalPrice,
        payment_status: 'pending',
        status: 'pending',
        source: 'dashboard',
        observations,
        seats_summary: selectedSeats.join(', '),
        created_by: session.user.id,
      },
    })

    // ═══════════════════════════════════════════════════════════
    // 3. CRIAR ASSENTOS
    // ═══════════════════════════════════════════════════════════
    
    for (const seatNumber of selectedSeats) {
      await prisma.reservationSeat.create({
        data: {
          reservation_id: reservation.id,
          trip_id: tripId,
          itinerary_id: itineraryId,
          seat_number: seatNumber,
          passenger_name: client.name,
          passenger_cpf: client.cpf,
          status: 'confirmed',
        },
      })
    }

    return NextResponse.json({
      success: true,
      client,
      reservation,
    })
  } catch (error: any) {
    console.error('Erro ao criar cliente:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao criar cliente' },
      { status: 500 }
    )
  }
}

function generateVoucherCode(): string {
  const characters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
  let code = 'VC-'
  for (let i = 0; i < 6; i++) {
    code += characters.charAt(Math.floor(Math.random() * characters.length))
  }
  return code
}
