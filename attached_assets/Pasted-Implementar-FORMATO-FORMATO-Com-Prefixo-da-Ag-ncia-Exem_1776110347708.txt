  // TIPO DA VIAGEM (para numeração)

  // ... resto do modelo ...
  
  @@map("trips")
}

model Reservation {
  id              String    @id @default(cuid())
  tenant_id       String
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)
  
  // NÚMERO DA RESERVA (formato completo)
  reservation_number String @unique // CHQ-EXC-202604-00001
  
  // COMPONENTES DO NÚMERO (para busca e análise)
  number_prefix   String    // CHQ
  number_type     String    // EXC
  number_yearmonth String   // 202604
  number_sequence Int       // 1
  
  // ... resto do modelo ...
  
  @@index([tenant_id, number_yearmonth, number_type])
  @@index([reservation_number])
  @@map("reservations")
}

model ReservationSequence {
  id              String    @id @default(cuid())
  tenant_id       String
  tenant          Tenant    @relation(fields: [tenant_id], references: [id], onDelete: Cascade)

  @@unique([tenant_id, year_month, trip_type])
  @@index([tenant_id, year_month])
  @@map("reservation_sequences")

// lib/reservations/generate-number-complete.ts

import { prisma } from '@/lib/prisma'

/**
 * Mapeamento de tipos de viagem para códigos
 */
const TRIP_TYPE_CODES = {
  excursion: 'EXC',   // Excursão
  package: 'PCT',     // Pacote
  day_trip: 'BTV',    // Bate-volta
} as const

type TripType = keyof typeof TRIP_TYPE_CODES

/**
 * Gera número completo de reserva no formato:
 * [PREFIXO]-[TIPO]-[AAAAMM]-[NNNNN]
 * 
 * Exemplo: CHQ-EXC-202604-00001
 * 
 * @param tenantId - ID do tenant (agência)
 * @param tripType - Tipo da viagem (excursion, package, day_trip)
 * @returns Número único da reserva
 */
export async function generateReservationNumber(
  tenantId: string,
  tripType: TripType
): Promise<{
  full: string
  prefix: string
  typeCode: string
  yearMonth: string
  sequence: number
}> {
  // Buscar tenant para pegar o prefixo
  const tenant = await prisma.tenant.findUnique({
    where: { id: tenantId },
    select: { reservation_prefix: true, name: true },
  })

  if (!tenant) {
    throw new Error('Tenant não encontrado')
  }

  // Prefixo da agência (padrão: 3 primeiras letras maiúsculas)
  const prefix = tenant.reservation_prefix || 
    tenant.name.substring(0, 3).toUpperCase().replace(/[^A-Z]/g, '')

  // Código do tipo de viagem
  const typeCode = TRIP_TYPE_CODES[tripType]

  // Ano e mês atual (YYYYMM)
  const now = new Date()
  const year = now.getFullYear()
  const month = String(now.getMonth() + 1).padStart(2, '0')
  const yearMonth = `${year}${month}` // "202604"

  // Usar transação para garantir unicidade
  const result = await prisma.$transaction(async (tx) => {
    // Buscar ou criar sequência
    const sequence = await tx.reservationSequence.upsert({
      where: {
        tenant_id_year_month_trip_type: {
          tenant_id: tenantId,
          year_month: yearMonth,
          trip_type: tripType,
        },
      },
      update: {
        last_number: {
          increment: 1,
        },
      },
      create: {
        tenant_id: tenantId,
        year_month: yearMonth,
        trip_type: tripType,
        type_code: typeCode,
        last_number: 1,
      },
    })

    // Formatar número sequencial com 5 dígitos
    const formattedSequence = String(sequence.last_number).padStart(5, '0')

    // Montar número completo
    const fullNumber = `${prefix}-${typeCode}-${yearMonth}-${formattedSequence}`

    // Verificar se já existe (segurança extra)
    const existing = await tx.reservation.findUnique({
      where: { reservation_number: fullNumber },
    })

    if (existing) {
      // Se existir, incrementar novamente
      const nextSequence = await tx.reservationSequence.update({
        where: {
          tenant_id_year_month_trip_type: {
            tenant_id: tenantId,
            year_month: yearMonth,
            trip_type: tripType,
          },
        },
        data: {
          last_number: {
            increment: 1,
          },
        },
      })

      const nextFormattedSequence = String(nextSequence.last_number).padStart(5, '0')
      const nextFullNumber = `${prefix}-${typeCode}-${yearMonth}-${nextFormattedSequence}`

      return {
        full: nextFullNumber,
        prefix,
        typeCode,
        yearMonth,
        sequence: nextSequence.last_number,
      }
    }

    return {
      full: fullNumber,
      prefix,
      typeCode,
      yearMonth,
      sequence: sequence.last_number,
    }
  })

  return result
}

/**
 * Valida formato do número de reserva
 */
export function isValidReservationNumber(number: string): boolean {
  // Formato: XXX-XXX-YYYYMM-NNNNN
  const pattern = /^[A-Z]{3}-[A-Z]{3}-\d{6}-\d{5}$/
  return pattern.test(number)
}

/**
 * Extrai componentes do número de reserva
 */
export function parseReservationNumber(number: string) {
  if (!isValidReservationNumber(number)) {
    throw new Error('Formato de número inválido')
  }

  const parts = number.split('-')

  const yearMonth = parts[2]
  const year = parseInt(yearMonth.substring(0, 4))
  const month = parseInt(yearMonth.substring(4, 6))

  return {
    full: number,
    prefix: parts[0],        // "CHQ"
    typeCode: parts[1],      // "EXC"
    yearMonth: parts[2],     // "202604"
    sequence: parseInt(parts[3]), // 1
    year,                    // 2026
    month,                   // 4
    monthName: getMonthName(month),
    typeName: getTypeName(parts[1]),
  }
}

/**
 * Retorna nome do mês em português
 */
function getMonthName(month: number): string {
  const months = [
    'Janeiro', 'Fevereiro', 'Março', 'Abril', 'Maio', 'Junho',
    'Julho', 'Agosto', 'Setembro', 'Outubro', 'Novembro', 'Dezembro'
  ]
  return months[month - 1] || 'Desconhecido'
}

/**
 * Retorna nome do tipo de viagem
 */
function getTypeName(code: string): string {
  const types: Record<string, string> = {
    'EXC': 'Excursão',
    'PCT': 'Pacote',
    'BTV': 'Bate-volta',
  }
  return types[code] || 'Desconhecido'
}

/**
 * Gera próximos números (preview)
 */
export async function previewNextNumbers(
  tenantId: string,
  tripType: TripType,
  count: number = 5
): Promise<string[]> {
  const tenant = await prisma.tenant.findUnique({
    where: { id: tenantId },
    select: { reservation_prefix: true, name: true },
  })

  if (!tenant) {
    throw new Error('Tenant não encontrado')
  }

  const prefix = tenant.reservation_prefix || 
    tenant.name.substring(0, 3).toUpperCase().replace(/[^A-Z]/g, '')

  const typeCode = TRIP_TYPE_CODES[tripType]

  const now = new Date()
  const year = now.getFullYear()
  const month = String(now.getMonth() + 1).padStart(2, '0')
  const yearMonth = `${year}${month}`

  // Buscar última sequência
  const sequence = await prisma.reservationSequence.findUnique({
    where: {
      tenant_id_year_month_trip_type: {
        tenant_id: tenantId,
        year_month: yearMonth,
        trip_type: tripType,
      },
    },
  })

  const lastNumber = sequence?.last_number || 0
  const preview: string[] = []

  for (let i = 1; i <= count; i++) {
    const nextNumber = lastNumber + i
    const formattedNumber = String(nextNumber).padStart(5, '0')
    preview.push(`${prefix}-${typeCode}-${yearMonth}-${formattedNumber}`)
  }

  return preview
}

import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { generateReservationNumber } from '@/lib/reservations/generate-number-complete'
import { createDealFromReservation } from '@/lib/deals/create-from-reservation'

export async function POST(req: Request) {
  try {
    const body = await req.json()

    const {
      trip_id,
      itinerary_id,
      name,
      cpf,
      email,
      phone,
      birthdate,
      passengers_count,
      selected_seats,
      referral_code,
      coupon_code,
      referral_discount,
      coupon_discount,
      payment_method,
      observations,
    } = body

    // Validações básicas
    if (!trip_id || !itinerary_id || !name || !cpf || !email || !phone) {
      return NextResponse.json(
        { error: 'Campos obrigatórios faltando' },
        { status: 400 }
      )
    }

    // Buscar viagem
    const trip = await prisma.trip.findUnique({
      where: { id: trip_id },
      include: { 
        itineraries: true,
        tenant: {
          select: {
            id: true,
            reservation_prefix: true,
            name: true,
          },
        },
      },
    })

    if (!trip) {
      return NextResponse.json(
        { error: 'Viagem não encontrada' },
        { status: 404 }
      )
    }

    const itinerary = trip.itineraries.find((it) => it.id === itinerary_id)

    if (!itinerary) {
      return NextResponse.json(
        { error: 'Itinerário não encontrado' },
        { status: 404 }
      )
    }

    // Verificar disponibilidade de assentos
    if (itinerary.available_seats < passengers_count) {
      return NextResponse.json(
        { error: 'Vagas insuficientes' },
        { status: 400 }
      )
    }

    // Verificar se assentos estão disponíveis
    const occupiedSeats = await prisma.reservationSeat.findMany({
      where: {
        trip_id,
        itinerary_id,
        seat_number: { in: selected_seats },
        status: { in: ['reserved', 'confirmed', 'checked_in'] },
      },
    })

    if (occupiedSeats.length > 0) {
      return NextResponse.json(
        {
          error: 'Um ou mais assentos já estão ocupados',
          occupied: occupiedSeats.map((s) => s.seat_number),
        },
        { status: 400 }
      )
    }

    // Calcular valores
    const basePrice = trip.promotional_price || trip.price_per_person
    const subtotal = basePrice * passengers_count
    const totalDiscount = (referral_discount || 0) + (coupon_discount || 0)
    const totalAmount = subtotal - totalDiscount

    // Criar ou atualizar cliente
    let client = await prisma.client.findFirst({
      where: {
        tenant_id: trip.tenant_id,
        OR: [{ cpf }, { email }],
      },
    })

    if (!client) {
      client = await prisma.client.create({
        data: {
          tenant_id: trip.tenant_id,
          name,
          cpf,
          email,
          phone,
          birthdate: new Date(birthdate),
          source: 'website',
          segment: 'individual',
          status: 'active',
        },
      })
    } else {
      client = await prisma.client.update({
        where: { id: client.id },
        data: { name, email, phone, birthdate: new Date(birthdate) },
      })
    }

    // ═══════════════════════════════════════════════════════════
    // GERAR NÚMERO DE RESERVA COMPLETO
    // ═══════════════════════════════════════════════════════════

    const reservationNumberData = await generateReservationNumber(
      trip.tenant_id,
      trip.type as 'excursion' | 'package' | 'day_trip'
    )

    console.log('📋 Número de reserva gerado:', reservationNumberData.full)

    // Gerar código de voucher
    const voucherCode = generateVoucherCode()

    // Criar resumo dos assentos
    const seatsSummary = selected_seats
      .sort((a: string, b: string) => parseInt(a) - parseInt(b))
      .join(', ')

    // ═══════════════════════════════════════════════════════════
    // CRIAR RESERVA COM NÚMERO COMPLETO
    // ═══════════════════════════════════════════════════════════

    const reservation = await prisma.reservation.create({
      data: {
        tenant_id: trip.tenant_id,
        
        // NÚMERO COMPLETO
        reservation_number: reservationNumberData.full,
        number_prefix: reservationNumberData.prefix,
        number_type: reservationNumberData.typeCode,
        number_yearmonth: reservationNumberData.yearMonth,
        number_sequence: reservationNumberData.sequence,
        
        voucher_code: voucherCode,
        trip_id,
        itinerary_id,
        client_id: client.id,
        client_name: name,
        client_cpf: cpf,
        client_email: email,
        client_phone: phone,
        total_passengers: passengers_count,
        adults_count: passengers_count,
        subtotal_amount: subtotal,
        discount_amount: totalDiscount,
        total_amount: totalAmount,
        amount_pending: totalAmount,
        payment_method,
        payment_status: 'pending',
        status: 'pending',
        source: 'website',
        observations,
        departure_date: itinerary.departure_date,
        seats_summary: seatsSummary,
      },
    })

    // Criar assentos
    for (const seatNumber of selected_seats) {
      await prisma.reservationSeat.create({
        data: {
          reservation_id: reservation.id,
          trip_id,
          itinerary_id,
          seat_number: seatNumber,
          passenger_name: name,
          passenger_cpf: cpf,
          status: 'reserved',
        },
      })
    }

    // Processar referral
    if (referral_code) {
      const referral = await prisma.referral.findFirst({
        where: { code: referral_code, status: 'active' },
      })

      if (referral) {
        await prisma.reservation.update({
          where: { id: reservation.id },
          data: { referral_id: referral.id, referral_code },
        })

        await applyReferralBenefits({
          referralCode: referral_code,
          reservationId: reservation.id,
          totalAmount,
          tenantId: trip.tenant_id,
        })
      }
    }

    // Atualizar vagas
    await prisma.itinerary.update({
      where: { id: itinerary_id },
      data: { available_seats: { decrement: passengers_count } },
    })

    // Criar deal no pipeline
    const deal = await createDealFromReservation({
      tenantId: trip.tenant_id,
      clientId: client.id,
      reservationId: reservation.id,
      tripId: trip_id,
      totalAmount,
      seatsSummary,
      tripTitle: trip.title,
      departureDate: itinerary.departure_date,
    })

    // Vincular deal à reserva
    await prisma.reservation.update({
      where: { id: reservation.id },
      data: { deal_id: deal.id },
    })

    // Buscar reserva completa
    const fullReservation = await prisma.reservation.findUnique({
      where: { id: reservation.id },
      include: {
        seats: true,
        client: true,
        deal: true,
        trip: true,
        itinerary: true,
      },
    })

    return NextResponse.json({
      success: true,
      reservation: fullReservation,
      reservation_number: reservationNumberData.full,
    })
  } catch (error: any) {
    console.error('Erro ao criar reserva:', error)
    return NextResponse.json(
      { error: 'Erro ao criar reserva' },
      { status: 500 }
    )
  }
}

function generateVoucherCode(): string {
  const characters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
  let code = 'VC-'

  for (let i = 0; i < 6; i++) {
    code += characters.charAt(Math.floor(Math.random() * characters.length))
  }

  return code
}

  showDetails = false,

  const [copied, setCopied] = useState(false)

    navigator.clipboard.writeText(number)

    sm: 'text-xs px-2 py-1',

    details = null
  }

  return (

      {showCopy && (
        <Button

import { prisma } from '@/lib/prisma'

/**
 * Estatísticas detalhadas por tipo e mês
 */
export async function getReservationStatsByTypeAndMonth(
  tenantId: string,
  year: number,
  month: number

  const sequences = await prisma.reservationSequence.findMany({
    where: {
      tenant_id: tenantId,
      year_month: yearMonth,
    },
  })

  const stats = {
    yearMonth,
    year,
    month,
    byType: {} as Record<string, {
      code: string
      name: string
      count: number
      lastNumber: number
      nextNumber: number
    }>,
    total: 0,
  }

  const typeNames: Record<string, string> = {
    'EXC': 'Excursões',
    'PCT': 'Pacotes',
    'BTV': 'Bate-volta',
  }

  for (const seq of sequences) {
    stats.byType[seq.type_code] = {
      code: seq.type_code,
      name: typeNames[seq.type_code] || seq.type_code,
      count: seq.last_number,
      lastNumber: seq.last_number,
      nextNumber: seq.last_number + 1,
    }
    stats.total += seq.last_number
  }

  return stats
}

/**
 * Relatório completo do ano
 */
export async function getAnnualReservationReport(
  tenantId: string,
  year: number
) {
  const sequences = await prisma.reservationSequence.findMany({
    where: {
      tenant_id: tenantId,
      year_month: {
        startsWith: String(year),
      },
    },
    orderBy: {
      year_month: 'asc',
    },
  })

  const report = {
    year,
    totalReservations: 0,
    byMonth: {} as Record<number, {
      month: number,
      monthName: string,
      total: number,
      byType: Record<string, number>
    }>,
    byType: {
      EXC: 0,
      PCT: 0,
      BTV: 0,
    },
  }

  const monthNames = [
    'Janeiro', 'Fevereiro', 'Março', 'Abril', 'Maio', 'Junho',
    'Julho', 'Agosto', 'Setembro', 'Outubro', 'Novembro', 'Dezembro'
  ]

  for (const seq of sequences) {
    const month = parseInt(seq.year_month.substring(4, 6))
    
    if (!report.byMonth[month]) {
      report.byMonth[month] = {
        month,
        monthName: monthNames[month - 1],
        total: 0,
        byType: {},
      }
    }

    report.byMonth[month].total += seq.last_number
    report.byMonth[month].byType[seq.type_code] = seq.last_number
    report.byType[seq.type_code as keyof typeof report.byType] += seq.last_number
    report.totalReservations += seq.last_number
  }

  return report
}
