model Passenger {
  id                String    @id @default(cuid())
  reservation_id    String
  reservation       Reservation @relation(fields: [reservation_id], references: [id], onDelete: Cascade)
  
  // Dados pessoais
  full_name         String
  cpf               String
  rg                String?
  birth_date        DateTime
  phone             String?
  email             String?
  
  // Categoria
  category          String    // ADULT, CHILD, BABY, SENIOR, PCD
  is_under_7        Boolean   @default(false)
  is_titular        Boolean   @default(false)
  
  // Assento
  seat_number       Int?
  
  // ═══════════════════════════════════════════════════════════
  // NOVO: LOCAL DE EMBARQUE
  // ═══════════════════════════════════════════════════════════
  boarding_location_id String?
  boarding_location    BoardingLocation? @relation(fields: [boarding_location_id], references: [id])
  
  // ═══════════════════════════════════════════════════════════
  // NOVO: LOCAL DE DESEMBARQUE
  // ═══════════════════════════════════════════════════════════
  disembark_location_id String?
  disembark_location    BoardingLocation? @relation("DisembarkLocation", fields: [disembark_location_id], references: [id])
  
  // Check-in
  checked_in        Boolean   @default(false)
  checked_in_at     DateTime?
  
  // ═══════════════════════════════════════════════════════════
  // NOVO: OBSERVAÇÕES E NECESSIDADES ESPECIAIS
  // ═══════════════════════════════════════════════════════════
  observations      String?   @db.Text
  special_needs     String?   @db.Text
  has_companion     Boolean   @default(false)
  companion_name    String?
  
  // ═══════════════════════════════════════════════════════════
  // NOVO: DOCUMENTOS ADICIONAIS
  // ═══════════════════════════════════════════════════════════
  document_type     String?   // RG, CPF, CNH, PASSPORT, BIRTH_CERTIFICATE
  document_issuer   String?   // SSP/SP, SSP/RJ, etc
  
  // ═══════════════════════════════════════════════════════════
  // NOVO: GRATUIDADE E DESCONTOS
  // ═══════════════════════════════════════════════════════════
  is_free           Boolean   @default(false)
  discount_percent  Decimal?  @db.Decimal(5, 2)
  discount_reason   String?
  
  // ═══════════════════════════════════════════════════════════
  // NOVO: ASSINATURA DIGITAL
  // ═══════════════════════════════════════════════════════════
  signature         String?   @db.Text // Base64 da assinatura
  signed_at         DateTime?
  
  // Auditoria
  created_at        DateTime  @default(now())
  updated_at        DateTime  @updatedAt
  
  @@index([reservation_id])
  @@index([cpf])
  @@map("passengers")
}

'use client'

import { useState, useEffect } from 'react'
import { Label } from '@/components/ui/label'
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from '@/components/ui/select'
import { MapPin, Clock } from 'lucide-react'

interface BoardingLocation {
  id: string
  name: string
  address: string
  city: string
  state: string
  reference?: string
  departure_time: string
}

interface BoardingLocationSelectorProps {
  tripId: string
  value?: string
  onChange: (locationId: string) => void
  type?: 'boarding' | 'disembark'
  disabled?: boolean
}

export function BoardingLocationSelector({
  tripId,
  value,
  onChange,
  type = 'boarding',
  disabled,
}: BoardingLocationSelectorProps) {
  const [locations, setLocations] = useState<BoardingLocation[]>([])
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    loadLocations()
  }, [tripId])

  const loadLocations = async () => {
    try {
      setIsLoading(true)
      const response = await fetch(`/api/dashboard/trips/${tripId}/boarding-locations`)
      const data = await response.json()

      if (response.ok) {
        setLocations(data.locations || [])
      }
    } catch (error) {
      console.error('Erro ao carregar locais:', error)
    } finally {
      setIsLoading(false)
    }
  }

  const selectedLocation = locations.find((l) => l.id === value)

  return (
    <div className="space-y-2">
      <Label>
        {type === 'boarding' ? '📍 Local de Embarque' : '🎯 Local de Desembarque'}
      </Label>

      <Select value={value} onValueChange={onChange} disabled={disabled || isLoading}>
        <SelectTrigger>
          <SelectValue placeholder={isLoading ? 'Carregando...' : 'Selecione o local'} />
        </SelectTrigger>
        <SelectContent>
          {locations.map((location) => (
            <SelectItem key={location.id} value={location.id}>
              <div className="flex items-center gap-2">
                <MapPin className="h-4 w-4 text-blue-600" />
                <div>
                  <p className="font-semibold">{location.name}</p>
                  <p className="text-xs text-gray-500">
                    {location.city}-{location.state} • {location.departure_time}
                  </p>
                </div>
              </div>
            </SelectItem>
          ))}
        </SelectContent>
      </Select>

      {selectedLocation && (
        <div className="p-3 bg-blue-50 border border-blue-200 rounded-lg text-sm">
          <div className="flex items-start gap-2">
            <MapPin className="h-4 w-4 text-blue-600 mt-0.5" />
            <div className="flex-1">
              <p className="font-semibold text-blue-900">{selectedLocation.name}</p>
              <p className="text-blue-700">{selectedLocation.address}</p>
              <p className="text-blue-600">
                {selectedLocation.city}-{selectedLocation.state}
              </p>
              {selectedLocation.reference && (
                <p className="text-blue-500 text-xs mt-1">
                  📌 {selectedLocation.reference}
                </p>
              )}
            </div>
            <div className="flex items-center gap-1 text-blue-700">
              <Clock className="h-4 w-4" />
              <span className="font-semibold">{selectedLocation.departure_time}</span>
            </div>
          </div>
        </div>
      )}
    </div>
  )
}

  tripId,
  reservationId,
  passenger,
  onSuccess,
  onCancel,
}: PassengerFormProps) {
  const [isSaving, setIsSaving] = useState(false)

  const [formData, setFormData] = useState({
    // Dados pessoais
    full_name: passenger?.full_name || '',
    cpf: passenger?.cpf || '',
    rg: passenger?.rg || '',
    birth_date: passenger?.birth_date
      ? new Date(passenger.birth_date).toISOString().split('T')[0]
      : '',
    phone: passenger?.phone || '',
    email: passenger?.email || '',
    
    // Categoria
    category: passenger?.category || 'ADULT',
    
    // Assento
    seat_number: passenger?.seat_number?.toString() || '',
    
    // Locais
    boarding_location_id: passenger?.boarding_location_id || '',
    disembark_location_id: passenger?.disembark_location_id || '',
    
    // Documentos
    document_type: passenger?.document_type || 'RG',
    document_issuer: passenger?.document_issuer || '',
    
    // Observações
    observations: passenger?.observations || '',
    special_needs: passenger?.special_needs || '',
    
    // Acompanhante
    has_companion: passenger?.has_companion || false,
    companion_name: passenger?.companion_name || '',
    
    // Gratuidade
    is_free: passenger?.is_free || false,
    discount_percent: passenger?.discount_percent?.toString() || '',
    discount_reason: passenger?.discount_reason || '',

    signature: passenger?.signature || '',

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()

    if (!formData.full_name || !formData.cpf) {
      toast.error('Nome e CPF são obrigatórios')
      return
    }

    try {
      setIsSaving(true)

      const url = passenger
        ? `/api/dashboard/passengers/${passenger.id}`
        : `/api/dashboard/passengers`

      const method = passenger ? 'PUT' : 'POST'

      const response = await fetch(url, {
        method,
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          ...formData,
          reservation_id: reservationId,
          trip_id: tripId,
        }),
      })

      const data = await response.json()

      if (response.ok) {
        toast.success(
          passenger
            ? 'Passageiro atualizado com sucesso!'
            : 'Passageiro cadastrado com sucesso!'
        )
        onSuccess()
      } else {
        toast.error(data.error || 'Erro ao salvar passageiro')
      }
    } catch (error) {
      console.error('Erro ao salvar passageiro:', error)
      toast.error('Erro ao salvar passageiro')
    } finally {
      setIsSaving(false)
    }
  }

    <form onSubmit={handleSubmit} className="space-y-6">
      {/* DADOS PESSOAIS */}

          {/* Nome Completo */}

            <Label htmlFor="full_name">
              Nome Completo <span className="text-red-500">*</span>
            </Label>
            <Input
              id="full_name"
              value={formData.full_name}
              onChange={(e) =>
                setFormData({ ...formData, full_name: e.target.value })
              }
              placeholder="Nome completo sem abreviações"
              required
              disabled={isSaving}
              className="mt-2"
            />
          </div>

          {/* CPF */}
          <div>
            <Label htmlFor="cpf">
              CPF <span className="text-red-500">*</span>
            </Label>
            <Input
              id="cpf"
              value={formData.cpf}
              onChange={(e) => setFormData({ ...formData, cpf: e.target.value })}
              placeholder="000.000.000-00"
              required
              disabled={isSaving}
              className="mt-2"
            />
          </div>

          {/* Data de Nascimento */}
          <div>
            <Label htmlFor="birth_date">
              Data de Nascimento <span className="text-red-500">*</span>
            </Label>
            <Input
              id="birth_date"
              type="date"
              value={formData.birth_date}
              onChange={(e) =>

// components/dashboard/trips/passengers/signature-pad.tsx

'use client'

import { useRef, useState, useEffect } from 'react'
import { Button } from '@/components/ui/button'
import { Card } from '@/components/ui/card'
import { Trash2, Check, X } from 'lucide-react'

interface SignaturePadProps {
  value?: string
  onChange: (signature: string) => void
  disabled?: boolean
  width?: number
  height?: number
}

export function SignaturePad({
  value,
  onChange,
  disabled,
  width = 500,
  height = 200,
}: SignaturePadProps) {
  const canvasRef = useRef<HTMLCanvasElement>(null)
  const [isDrawing, setIsDrawing] = useState(false)
  const [isEmpty, setIsEmpty] = useState(true)

  useEffect(() => {
    if (value && canvasRef.current) {
      const canvas = canvasRef.current
      const ctx = canvas.getContext('2d')
      if (ctx) {
        const img = new Image()
        img.onload = () => {
          ctx.drawImage(img, 0, 0)
          setIsEmpty(false)
        }
        img.src = value
      }
    }
  }, [value])

  const startDrawing = (e: React.MouseEvent<HTMLCanvasElement>) => {
    if (disabled) return

    const canvas = canvasRef.current
    if (!canvas) return

    const rect = canvas.getBoundingClientRect()
    const x = e.clientX - rect.left
    const y = e.clientY - rect.top

    const ctx = canvas.getContext('2d')
    if (ctx) {
      ctx.beginPath()
      ctx.moveTo(x, y)
      setIsDrawing(true)
      setIsEmpty(false)
    }
  }

  const draw = (e: React.MouseEvent<HTMLCanvasElement>) => {
    if (!isDrawing || disabled) return

    const canvas = canvasRef.current
    if (!canvas) return

    const rect = canvas.getBoundingClientRect()
    const x = e.clientX - rect.left
    const y = e.clientY - rect.top

    const ctx = canvas.getContext('2d')
    if (ctx) {
      ctx.lineTo(x, y)
      ctx.strokeStyle = '#000'
      ctx.lineWidth = 2
      ctx.lineCap = 'round'
      ctx.lineJoin = 'round'
      ctx.stroke()
    }
  }

  const stopDrawing = () => {
    if (!isDrawing) return

    setIsDrawing(false)

    const canvas = canvasRef.current
    if (canvas) {
      const signature = canvas.toDataURL('image/png')
      onChange(signature)
    }
  }

  const clear = () => {
    const canvas = canvasRef.current
    if (!canvas) return

    const ctx = canvas.getContext('2d')
    if (ctx) {
      ctx.clearRect(0, 0, canvas.width, canvas.height)
      setIsEmpty(true)
      onChange('')
    }
  }

  // Touch support
  const handleTouchStart = (e: React.TouchEvent<HTMLCanvasElement>) => {
    if (disabled) return
    e.preventDefault()

    const canvas = canvasRef.current
    if (!canvas) return

    const rect = canvas.getBoundingClientRect()
    const touch = e.touches[0]
    const x = touch.clientX - rect.left
    const y = touch.clientY - rect.top

    const ctx = canvas.getContext('2d')
    if (ctx) {
      ctx.beginPath()
      ctx.moveTo(x, y)
      setIsDrawing(true)
      setIsEmpty(false)
    }
  }

  const handleTouchMove = (e: React.TouchEvent<HTMLCanvasElement>) => {
    if (!isDrawing || disabled) return
    e.preventDefault()

    const canvas = canvasRef.current
    if (!canvas) return

    const rect = canvas.getBoundingClientRect()
    const touch = e.touches[0]
    const x = touch.clientX - rect.left
    const y = touch.clientY - rect.top

    const ctx = canvas.getContext('2d')
    if (ctx) {
      ctx.lineTo(x, y)
      ctx.strokeStyle = '#000'
      ctx.lineWidth = 2
      ctx.lineCap = 'round'
      ctx.lineJoin = 'round'
      ctx.stroke()
    }
  }

  const handleTouchEnd = () => {
    if (!isDrawing) return

    setIsDrawing(false)

    const canvas = canvasRef.current
    if (canvas) {
      const signature = canvas.toDataURL('image/png')
      onChange(signature)
    }
  }

  return (
    <div className="space-y-4">
      <Card className="p-4 bg-gray-50">
        <div className="flex items-center justify-between mb-3">
          <p className="text-sm font-semibold text-gray-700">
            {disabled ? '✍️ Assinatura Registrada' : '✍️ Assine abaixo'}
          </p>
          {!disabled && (
            <Button
              type="button"
              variant="outline"
              size="sm"
              onClick={clear}
              disabled={isEmpty}
            >
              <Trash2 className="h-4 w-4 mr-2" />
              Limpar
            </Button>
          )}
        </div>

        <div className="relative">
          <canvas
            ref={canvasRef}
            width={width}
            height={height}
            onMouseDown={startDrawing}
            onMouseMove={draw}
            onMouseUp={stopDrawing}
            onMouseLeave={stopDrawing}
            onTouchStart={handleTouchStart}
            onTouchMove={handleTouchMove}
            onTouchEnd={handleTouchEnd}
            className={`border-2 border-dashed rounded-lg bg-white w-full ${
              disabled ? 'cursor-not-allowed opacity-60' : 'cursor-crosshair'
            }`}
            style={{ touchAction: 'none', maxWidth: '100%', height: 'auto' }}
          />

          {isEmpty && !disabled && (
            <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
              <p className="text-gray-400 text-sm">
                Clique e arraste para assinar
              </p>
            </div>
          )}
        </div>

        {!isEmpty && (
          <div className="mt-3 flex items-center gap-2 text-sm text-green-600">
            <Check className="h-4 w-4" />
            <span>Assinatura capturada</span>
          </div>
        )}
      </Card>

      {!disabled && (
        <p className="text-xs text-gray-500">
          💡 Dica: Use um tablet ou celular para uma assinatura mais natural
        </p>
      )}
    </div>
  )
}

    total_adults: number
    total_children: number
    total_seniors: number
    total_pcd: number
    total_free: number
  }
}

export class ManifestPDFGenerator {
  private doc: typeof PDFDocument
  private pageWidth = 595.28 // A4 width in points
  private pageHeight = 841.89 // A4 height in points
  private margin = 50
  private currentY = 0

  async generate(data: ManifestData): Promise<Buffer> {
    return new Promise(async (resolve, reject) => {
      try {
        this.doc = new PDFDocument({
          size: 'A4',
          margins: {
            top: this.margin,
            bottom: this.margin,
            left: this.margin,
            right: this.margin,
          },
          info: {
            Title: `Manifesto de Passageiros - ${data.manifest_number}`,
            Author: data.company_name,
            Subject: 'Manifesto de Passageiros ANTT',
            Keywords: 'manifesto, passageiros, ANTT, transporte',
          },
        })

        const chunks: Buffer[] = []

        this.doc.on('data', (chunk) => chunks.push(chunk))
        this.doc.on('end', () => resolve(Buffer.concat(chunks)))
        this.doc.on('error', reject)

        // Gerar PDF
        await this.addHeader(data)
        this.addCompanyInfo(data)
        this.addTripInfo(data)
        this.addVehicleInfo(data)
        this.addCrewInfo(data)
        this.addPassengerTable(data)
        this.addRoute(data)
        this.addStatistics(data)
        await this.addFooter(data)

        this.doc.end()
      } catch (error) {
        reject(error)
      }
    })
  }

  private async addHeader(data: ManifestData) {
    this.currentY = this.margin

    // Logo (se houver)
    // this.doc.image('logo.png', this.margin, this.currentY, { width: 100 })

    // Título
    this.doc
      .fontSize(20)
      .font('Helvetica-Bold')
      .text('MANIFESTO DE PASSAGEIROS', this.margin + 120, this.currentY, {
        align: 'center',
      })

    this.currentY += 25

    // Subtítulo
    this.doc
      .fontSize(10)
      .font('Helvetica')
      .fillColor('#666')
      .text(
        'Conforme Resolução ANTT nº 4.777/2015',
        this.margin,
        this.currentY,
        {
          align: 'center',
        }
      )

    this.currentY += 15

    // Número do Manifesto
    this.doc
      .fontSize(12)
      .font('Helvetica-Bold')
      .fillColor('#000')
      .text(
        `Manifesto Nº: ${data.manifest_number}`,
        this.margin,
        this.currentY,
        {
          align: 'center',
        }
      )

    this.currentY += 20

    // QR Code
    const qrCodeDataURL = await QRCode.toDataURL(
      `https://sistema.com/manifesto/${data.manifest_number}`,
      {
        width: 100,
        margin: 1,
      }
    )

    const qrCodeBuffer = Buffer.from(
      qrCodeDataURL.replace(/^data:image\/png;base64,/, ''),
      'base64'
    )

    this.doc.image(qrCodeBuffer, this.pageWidth - this.margin - 100, this.margin, {
      width: 80,
    })

    this.currentY += 10
    this.addLine()
  }

  private addPassengerTable(data: ManifestData) {
    this.currentY += 15

    const tableTop = this.currentY
    const col1X = this.margin
    const col2X = this.margin + 40
    const col3X = this.margin + 180
    const col4X = this.margin + 280
    const col5X = this.margin + 360

    this.doc
      .fontSize(8)
      .font('Helvetica-Bold')
      .text('Assento', col1X, tableTop)
      .text('Nome Completo', col2X, tableTop)
      .text('Documento', col3X, tableTop)
      .text('Nascimento', col4X, tableTop)
      .text('Embarque', col5X, tableTop)

    this.currentY += 15

    // Linha separadora
    this.doc
      .moveTo(this.margin, this.currentY)
      .lineTo(this.pageWidth - this.margin, this.currentY)
      .stroke()

    this.currentY += 5

    // Linhas de passageiros
    data.passengers.forEach((passenger, index) => {
      // Verificar se precisa de nova página
      if (this.currentY > this.pageHeight - 100) {
        this.doc.addPage()
        this.currentY = this.margin
      }

      const rowY = this.currentY

      // Assento
      this.doc
        .fontSize(8)
        .font('Helvetica-Bold')
        .text(passenger.seat_number.toString().padStart(2, '0'), col1X, rowY)

      // Nome
      this.doc
        .font('Helvetica')
        .text(passenger.full_name.substring(0, 25), col2X, rowY, {
          width: 130,
        })

        passenger.document_issuer ? ' ' + passenger.document_issuer : ''
      }`
      this.doc.text(docText, col3X, rowY, { width: 90 })

      // Data de Nascimento
      this.doc.text(
        format(passenger.birth_date, 'dd/MM/yyyy'),
        col4X,
        rowY
      )

      // Local de Embarque
      if (passenger.boarding_location) {
        this.doc.text(
          `${passenger.boarding_location.city}-${passenger.boarding_location.state}`,
          col5X,
          rowY,
          { width: 100 }
        )
      }

      this.currentY += 20

      // Linha separadora
      if (index < data.passengers.length - 1) {
        this.doc
          .strokeColor('#ddd')
          .moveTo(this.margin, this.currentY)
          .lineTo(this.pageWidth - this.margin, this.currentY)
          .stroke()
          .strokeColor('#000')

      }
    })

    this.addLine()
  }

      .font('Helvetica-Bold')

    data.route.forEach((point, index) => {
      const icon =

          ? '🎯'
          : '🛑'

      this.doc
        .fontSize(9)
        .font('Helvetica-Bold')
        .text(
          `${icon} ${point.point}`,
          this.margin,
          this.currentY,
          { continued: true }
        )
        .font('Helvetica')
        .text(
          ` - ${point.city}-${point.state} • ${format(
            point.estimated_time,
            'HH:mm'
          )}`
        )

      this.currentY += 15
    })

    this.addLine()
  }

  private addStatistics(data: ManifestData) {
    this.currentY += 15

    this.doc
      .fontSize(12)
      .font('Helvetica-Bold')
      .text('RESUMO', this.margin, this.currentY)

    this.currentY += 20

    const stats = [
      { label: 'Total de Passageiros:', value: data.stats.total_passengers },
      { label: 'Adultos:', value: data.stats.total_adults },
      { label: 'Crianças:', value: data.stats.total_children },
      { label: 'Idosos:', value: data.stats.total_seniors },
      { label: 'PCD:', value: data.stats.total_pcd },
      { label: 'Gratuidades:', value: data.stats.total_free },
    ]

    stats.forEach((stat) => {
      this.doc
        .fontSize(10)
        .font('Helvetica-Bold')
        .text(stat.label, this.margin, this.currentY, { continued: true })
        .font('Helvetica')
        .text(` ${stat.value}`)

      this.currentY += 15
    })

    this.addLine()
  }

  private async addFooter(data: ManifestData) {
    // Verificar se precisa de nova página
    if (this.currentY > this.pageHeight - 200) {
      this.doc.addPage()
      this.currentY = this.margin
    }

    this.currentY += 30

    // Declaração
    this.doc
      .fontSize(9)
      .font('Helvetica')
      .text(
        'Declaro que as informações acima são verdadeiras e que estou ciente das normas de transporte rodoviário de passageiros.',
        this.margin,
        this.currentY,
        {
          align: 'justify',
          width: this.pageWidth - 2 * this.margin,
        }
      )

    this.currentY += 40

    // Assinatura do Responsável
    if (data.responsible) {
      this.doc
        .fontSize(10)
        .font('Helvetica-Bold')
        .text('Responsável pela Viagem:', this.margin, this.currentY)

      this.currentY += 20

      // Se houver assinatura digital
      if (data.responsible.signature) {
        const signatureBuffer = Buffer.from(
          data.responsible.signature.replace(/^data:image\/png;base64,/, ''),
          'base64'
        )

        this.doc.image(signatureBuffer, this.margin, this.currentY, {
          width: 200,
          height: 60,
        })

        this.currentY += 70
      } else {
        // Linha para assinatura manual
        this.doc
          .moveTo(this.margin, this.currentY + 40)
          .lineTo(this.margin + 200, this.currentY + 40)
          .stroke()

        this.currentY += 50
      }

      this.doc
        .fontSize(9)
        .font('Helvetica')
        .text(data.responsible.name, this.margin, this.currentY)
        .text(
          `CPF: ${this.formatCPF(data.responsible.cpf)}`,
          this.margin,
          this.currentY + 12
        )
    }

    this.currentY += 40

    // Data e hora de emissão
    this.doc
      .fontSize(8)
      .font('Helvetica')
      .fillColor('#666')
      .text(
        `Documento emitido em ${format(new Date(), "dd/MM/yyyy 'às' HH:mm", {
          locale: ptBR,
        })}`,
        this.margin,
        this.currentY,
        {
          align: 'center',
        }
      )

    this.currentY += 15

    // Rodapé com número de página
    const pageCount = (this.doc as any).bufferedPageRange().count
    for (let i = 0; i < pageCount; i++) {
      this.doc.switchToPage(i)
      this.doc
        .fontSize(8)
        .font('Helvetica')
        .fillColor('#666')
        .text(
          `Página ${i + 1} de ${pageCount}`,
          this.margin,
          this.pageHeight - this.margin + 20,
          {
            align: 'center',
          }
        )
    }
  }

  private addLine() {
    this.currentY += 10
    this.doc
      .moveTo(this.margin, this.currentY)
      .lineTo(this.pageWidth - this.margin, this.currentY)
      .stroke()
  }

  private formatCPF(cpf: string): string {
    return cpf.replace(/(\d{3})(\d{3})(\d{3})(\d{2})/, '$1.$2.$3-$4')
  }

  private formatCNPJ(cnpj: string): string {
    return cnpj.replace(
      /(\d{2})(\d{3})(\d{3})(\d{4})(\d{2})/,
      '$1.$2.$3/$4-$5'
    )
  }
}

// lib/validators/manifest-validator.ts

import { differenceInYears, isPast, isFuture } from 'date-fns'

interface ValidationResult {
  valid: boolean
  errors: string[]
  warnings: string[]
}

interface Passenger {
  id: string
  full_name: string
  cpf: string
  rg?: string
  birth_date: Date
  category: string
  seat_number?: number
  boarding_location_id?: string
  has_companion: boolean
  companion_name?: string
  is_free: boolean
  discount_percent?: number
}

interface Manifest {
  vehicle_capacity: number
  passengers: Passenger[]
  driver1_cnh_expiry?: Date
  driver2_cnh_expiry?: Date
}

export class ManifestValidator {
  // ═══════════════════════════════════════════════════════════
  // VALIDAÇÃO COMPLETA DO MANIFESTO
  // ═══════════════════════════════════════════════════════════
  
  static validateManifest(manifest: Manifest): ValidationResult {
    const errors: string[] = []
    const warnings: string[] = []

    // 1. Validar capacidade
    const capacityValidation = this.validateCapacity(
      manifest.passengers.length,
      manifest.vehicle_capacity
    )
    if (!capacityValidation.valid) {
      errors.push(...capacityValidation.errors)
    }

    // 2. Validar documentos
    manifest.passengers.forEach((passenger) => {
      const docValidation = this.validateDocument(passenger.cpf, 'CPF')
      if (!docValidation.valid) {
        errors.push(`${passenger.full_name}: ${docValidation.errors.join(', ')}`)
      }
    })

    // 3. Validar idades e categorias
    manifest.passengers.forEach((passenger) => {
      const ageValidation = this.validateAgeCategory(
        passenger.birth_date,
        passenger.category
      )
      if (!ageValidation.valid) {
        errors.push(`${passenger.full_name}: ${ageValidation.errors.join(', ')}`)
      }
    })

    // 4. Validar gratuidades
    const gratuityValidation = this.validateGratuities(manifest.passengers)
    if (!gratuityValidation.valid) {
      errors.push(...gratuityValidation.errors)
    }
    warnings.push(...gratuityValidation.warnings)

    // 5. Validar menores desacompanhados
    const minorsValidation = this.validateMinors(manifest.passengers)
    warnings.push(...minorsValidation.warnings)

    // 6. Validar assentos duplicados
    const seatsValidation = this.validateSeats(manifest.passengers)
    if (!seatsValidation.valid) {
      errors.push(...seatsValidation.errors)
    }

    // 7. Validar CNH dos motoristas
    if (manifest.driver1_cnh_expiry) {
      const cnhValidation = this.validateCNHExpiry(manifest.driver1_cnh_expiry)
      if (!cnhValidation.valid) {
        errors.push('CNH do Motorista 1 vencida ou próxima do vencimento')
      }
    }

    if (manifest.driver2_cnh_expiry) {
      const cnhValidation = this.validateCNHExpiry(manifest.driver2_cnh_expiry)
      if (!cnhValidation.valid) {
        warnings.push('CNH do Motorista 2 vencida ou próxima do vencimento')
      }
    }

    // 8. Validar locais de embarque
    const boardingValidation = this.validateBoardingLocations(manifest.passengers)
    warnings.push(...boardingValidation.warnings)

    return {
      valid: errors.length === 0,
      errors,
      warnings,
    }
  }

  // ═══════════════════════════════════════════════════════════
  // VALIDAÇÃO DE CPF
  // ═══════════════════════════════════════════════════════════
  
  static validateCPF(cpf: string): boolean {
    // Remove caracteres não numéricos
    cpf = cpf.replace(/[^\d]/g, '')

    // Verifica se tem 11 dígitos
    if (cpf.length !== 11) return false

    // Verifica se todos os dígitos são iguais
    if (/^(\d)\1{10}$/.test(cpf)) return false

    // Validação do primeiro dígito verificador
    let sum = 0
    for (let i = 0; i < 9; i++) {
      sum += parseInt(cpf.charAt(i)) * (10 - i)
    }
    let digit = 11 - (sum % 11)
    if (digit >= 10) digit = 0
    if (digit !== parseInt(cpf.charAt(9))) return false

    // Validação do segundo dígito verificador
    sum = 0
    for (let i = 0; i < 10; i++) {
      sum += parseInt(cpf.charAt(i)) * (11 - i)
    }
    digit = 11 - (sum % 11)
    if (digit >= 10) digit = 0
    if (digit !== parseInt(cpf.charAt(10))) return false

    return true
  }

  // ═══════════════════════════════════════════════════════════
  // VALIDAÇÃO DE DOCUMENTO
  // ═══════════════════════════════════════════════════════════
  
  static validateDocument(
    document: string,
    type: 'CPF' | 'RG' | 'CNH'
  ): ValidationResult {
    const errors: string[] = []

    if (!document || document.trim() === '') {
      errors.push('Documento não informado')
      return { valid: false, errors, warnings: [] }
    }

    if (type === 'CPF') {
      if (!this.validateCPF(document)) {
        errors.push('CPF inválido')
      }
    }

    return {
      valid: errors.length === 0,
      errors,
      warnings: [],
    }
  }

  // ═══════════════════════════════════════════════════════════
  // VALIDAÇÃO DE IDADE E CATEGORIA
  // ═══════════════════════════════════════════════════════════
  
  static validateAgeCategory(
    birthDate: Date,
    category: string
  ): ValidationResult {
    const errors: string[] = []
    const age = differenceInYears(new Date(), birthDate)

    const categoryRules: Record<string, { min: number; max: number }> = {
      BABY: { min: 0, max: 5 },
      CHILD: { min: 6, max: 11 },
      ADULT: { min: 12, max: 59 },
      SENIOR: { min: 60, max: 120 },
    }

    const rule = categoryRules[category]

    if (rule) {
      if (age < rule.min || age > rule.max) {
        errors.push(
          `Idade (${age} anos) não corresponde à categoria ${category}`
        )
      }
    }

    return {
      valid: errors.length === 0,
      errors,
      warnings: [],
    }
  }

  // ═══════════════════════════════════════════════════════════
  // VALIDAÇÃO DE CAPACIDADE
  // ═══════════════════════════════════════════════════════════
  
  static validateCapacity(
    passengers: number,
    capacity: number
  ): ValidationResult {
    const errors: string[] = []

    if (passengers > capacity) {
      errors.push(
        `Número de passageiros (${passengers}) excede a capacidade do veículo (${capacity})`
      )
    }

    return {
      valid: errors.length === 0,
      errors,
      warnings: [],
    }
  }

  // ═══════════════════════════════════════════════════════════
  // VALIDAÇÃO DE GRATUIDADES
  // ═══════════════════════════════════════════════════════════
  
  static validateGratuities(passengers: Passenger[]): ValidationResult {
    const errors: string[] = []
    const warnings: string[] = []

    // Contar gratuidades por categoria
    const freeSeats = {
      seniors: passengers.filter(
        (p) => p.category === 'SENIOR' && p.is_free
      ).length,
      pcd: passengers.filter((p) => p.category === 'PCD' && p.is_free).length,
      children: passengers.filter(
        (p) => p.category === 'BABY' && p.is_free
      ).length,
    }

    // Limites legais
    const limits = {
      seniors: 2, // 2 vagas gratuitas para idosos
      pcd: 1, // 1 vaga gratuita para PCD
    }

    // Validar limites
    if (freeSeats.seniors > limits.seniors) {
      errors.push(
        `Limite de gratuidades para idosos excedido (${freeSeats.seniors}/${limits.seniors})`
      )
    }

    if (freeSeats.pcd > limits.pcd) {
      errors.push(
        `Limite de gratuidades para PCD excedido (${freeSeats.pcd}/${limits.pcd})`
      )
    }

    // Avisos
    if (freeSeats.seniors === limits.seniors) {
      warnings.push('Limite de gratuidades para idosos atingido')
    }

    if (freeSeats.pcd === limits.pcd) {
      warnings.push('Limite de gratuidades para PCD atingido')
    }

    return {
      valid: errors.length === 0,
      errors,
      warnings,
    }
  }

  // ═══════════════════════════════════════════════════════════
  // VALIDAÇÃO DE MENORES DESACOMPANHADOS
  // ═══════════════════════════════════════════════════════════
  
  static validateMinors(passengers: Passenger[]): ValidationResult {
    const warnings: string[] = []

    passengers.forEach((passenger) => {
      const age = differenceInYears(new Date(), passenger.birth_date)

      // Menor de 12 anos sem acompanhante
      if (age < 12 && !passenger.has_companion) {
        warnings.push(
          `${passenger.full_name} (${age} anos) está desacompanhado. Necessária autorização judicial.`
        )
      }

      // Menor de 16 anos sem acompanhante
      if (age >= 12 && age < 16 && !passenger.has_companion) {
        warnings.push(
          `${passenger.full_name} (${age} anos) está desacompanhado. Necessária autorização dos pais ou judicial.`
        )
      }

      // Menor de 18 anos sem acompanhante (viagem interestadual)
      if (age >= 16 && age < 18 && !passenger.has_companion) {
        warnings.push(
          `${passenger.full_name} (${age} anos) está desacompanhado. Recomendada autorização dos pais para viagem interestadual.`
        )
      }
    })

    return {
      valid: true,
      errors: [],
      warnings,
    }
  }

  // ═══════════════════════════════════════════════════════════
  // VALIDAÇÃO DE ASSENTOS DUPLICADOS
  // ═══════════════════════════════════════════════════════════
  
  static validateSeats(passengers: Passenger[]): ValidationResult {
    const errors: string[] = []
    const seatMap = new Map<number, string[]>()

    passengers.forEach((passenger) => {
      if (passenger.seat_number) {
        const occupants = seatMap.get(passenger.seat_number) || []
        occupants.push(passenger.full_name)
        seatMap.set(passenger.seat_number, occupants)
      }
    })

    // Verificar assentos duplicados
    seatMap.forEach((occupants, seat) => {
      if (occupants.length > 1) {
        errors.push(
          `Assento ${seat} duplicado: ${occupants.join(', ')}`
        )
      }
    })

    return {
      valid: errors.length === 0,
      errors,
      warnings: [],
    }
  }

  // ═══════════════════════════════════════════════════════════
  // VALIDAÇÃO DE VALIDADE DA CNH
  // ═══════════════════════════════════════════════════════════
  
  static validateCNHExpiry(expiryDate: Date): ValidationResult {
    const errors: string[] = []

    // CNH vencida
    if (isPast(expiryDate)) {
      errors.push('CNH vencida')
    }

    // CNH vence em menos de 30 dias
    const thirtyDaysFromNow = new Date()
    thirtyDaysFromNow.setDate(thirtyDaysFromNow.getDate() + 30)

    if (expiryDate < thirtyDaysFromNow) {
      errors.push('CNH vence em menos de 30 dias')
    }

    return {
      valid: errors.length === 0,
      errors,
      warnings: [],
    }
  }

  // ═══════════════════════════════════════════════════════════
  // VALIDAÇÃO DE LOCAIS DE EMBARQUE
  // ═══════════════════════════════════════════════════════════
  
  static validateBoardingLocations(passengers: Passenger[]): ValidationResult {
    const warnings: string[] = []

    const withoutBoarding = passengers.filter(
      (p) => !p.boarding_location_id
    ).length

    if (withoutBoarding > 0) {
      warnings.push(
        `${withoutBoarding} passageiro(s) sem local de embarque definido`
      )
    }

    return {
      valid: true,
      errors: [],
      warnings,
    }
  }

  // ═══════════════════════════════════════════════════════════
  // CALCULAR IDADE
  // ═══════════════════════════════════════════════════════════
  
  static calculateAge(birthDate: Date): number {
    return differenceInYears(new Date(), birthDate)
  }
}

// app/api/dashboard/trips/[id]/manifest/generate-pdf/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { ManifestPDFGenerator } from '@/lib/pdf/manifest-generator'
import { uploadToS3 } from '@/lib/storage/s3'

export async function POST(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    // Buscar dados completos
    const manifest = await prisma.manifest.findUnique({
      where: { trip_id: params.id },
      include: {
        trip: {
          include: {
            reservations: {
              where: { status: 'confirmed' },
              include: {
                passengers: {
                  include: {
                    boarding_location: true,
                    disembark_location: true,
                  },
                },
              },
            },
          },
        },
      },
    })

    if (!manifest) {
      return NextResponse.json(
        { error: 'Manifesto não encontrado' },
        { status: 404 }
      )
    }

    // Preparar dados para o PDF
    const passengers = manifest.trip.reservations.flatMap((r) =>
      r.passengers.map((p) => ({
        seat_number: p.seat_number || 0,
        full_name: p.full_name,
        document_type: p.document_type || 'CPF',
        document_number: p.cpf,
        document_issuer: p.document_issuer,
        birth_date: p.birth_date,
        category: p.category,
        boarding_location: p.boarding_location
          ? {
              name: p.boarding_location.name,
              city: p.boarding_location.city,
              state: p.boarding_location.state,
              departure_time: p.boarding_location.departure_time,
            }
          : undefined,
        disembark_location: p.disembark_location
          ? {
              name: p.disembark_location.name,
              city: p.disembark_location.city,
              state: p.disembark_location.state,
            }
          : undefined,
        observations: p.observations,
        signature: p.signature,
      }))
    )

    // Calcular estatísticas
    const stats = {
      total_passengers: passengers.length,
      total_adults: passengers.filter((p) => p.category === 'ADULT').length,
      total_children: passengers.filter((p) => p.category === 'CHILD').length,
      total_seniors: passengers.filter((p) => p.category === 'SENIOR').length,
      total_pcd: passengers.filter((p) => p.category === 'PCD').length,
      total_free: passengers.filter((p) => p.is_free).length,
    }

    // Gerar PDF
    const generator = new ManifestPDFGenerator()
    const pdfBuffer = await generator.generate({
      manifest_number: manifest.manifest_number,
      company_name: manifest.company_name,
      company_cnpj: manifest.company_cnpj,
      antt_authorization: manifest.antt_authorization || undefined,
      
      trip: {
        name: manifest.trip.name || manifest.trip.title,
        origin: manifest.trip.origin_city || '',
        destination: manifest.trip.destination,
        departure_date: manifest.trip.departure_date || new Date(),
      },
      
      vehicle: {
        plate: manifest.vehicle_plate,
        type: manifest.vehicle_type,
        capacity: manifest.vehicle_capacity,
        year: manifest.vehicle_year || undefined,
        fleet_number: manifest.vehicle_fleet || undefined,
      },
      
      crew: {
        driver1: {
          name: manifest.driver1_name,
          cpf: manifest.driver1_cpf,
          cnh: manifest.driver1_cnh,
          cnh_category: manifest.driver1_cnh_category,
        },
        driver2: manifest.driver2_name
          ? {
              name: manifest.driver2_name,
              cpf: manifest.driver2_cpf!,
              cnh: manifest.driver2_cnh!,
              cnh_category: manifest.driver2_cnh_category!,
            }
          : undefined,
        tour_guide: manifest.tour_guide_name
          ? {
              name: manifest.tour_guide_name,
              cpf: manifest.tour_guide_cpf!,
              registration: manifest.tour_guide_registration || undefined,
            }
          : undefined,
      },
      
      passengers,
      route: [], // Adicionar pontos de parada se houver
      
      responsible: manifest.responsible_name
        ? {
            name: manifest.responsible_name,
            cpf: manifest.responsible_cpf!,
            signature: manifest.responsible_signature || undefined,
          }
        : undefined,
      
      stats,
    })

    // Upload para S3 (opcional)
    // const s3Url = await uploadToS3(pdfBuffer, `manifests/${manifest.manifest_number}.pdf`)

    // Atualizar manifesto com URL do PDF
    // await prisma.manifest.update({
    //   where: { id: manifest.id },
    //   data: { pdf_url: s3Url }
    // })

    // Retornar PDF
    return new NextResponse(pdfBuffer, {
      headers: {
        'Content-Type': 'application/pdf',
        'Content-Disposition': `attachment; filename="manifesto-${manifest.manifest_number}.pdf"`,
      },
    })
  } catch (error: any) {
    console.error('Erro ao gerar PDF:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao gerar PDF' },
      { status: 500 }
    )
  }
}

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { ManifestValidator } from '@/lib/validators/manifest-validator'

export async function POST(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    // Buscar dados do manifesto
    const manifest = await prisma.manifest.findUnique({
      where: { trip_id: params.id },
      include: {
        trip: {
          include: {
            reservations: {
              where: { status: 'confirmed' },
              include: {
                passengers: {
                  include: {
                    boarding_location: true,
                  },
                },
              },
            },
          },
        },
      },
    })

    if (!manifest) {
      return NextResponse.json(
        { error: 'Manifesto não encontrado' },
        { status: 404 }
      )
    }

    // Preparar dados para validação
    const passengers = manifest.trip.reservations.flatMap((r) =>
      r.passengers.map((p) => ({
        id: p.id,
        full_name: p.full_name,
        cpf: p.cpf,
        rg: p.rg || undefined,
        birth_date: p.birth_date,
        category: p.category,
        seat_number: p.seat_number || undefined,
        boarding_location_id: p.boarding_location_id || undefined,
        has_companion: p.has_companion,
        companion_name: p.companion_name || undefined,
        is_free: p.is_free,
        discount_percent: p.discount_percent
          ? parseFloat(p.discount_percent.toString())
          : undefined,
      }))
    )

    // Validar manifesto
    const validation = ManifestValidator.validateManifest({
      vehicle_capacity: manifest.vehicle_capacity,
      passengers,
      driver1_cnh_expiry: manifest.driver1_cnh_expiry || undefined,
      driver2_cnh_expiry: manifest.driver2_cnh_expiry || undefined,
    })

    return NextResponse.json({
      valid: validation.valid,
      errors: validation.errors,
      warnings: validation.warnings,
      total_passengers: passengers.length,
      vehicle_capacity: manifest.vehicle_capacity,
    })
  } catch (error: any) {
    console.error('Erro ao validar manifesto:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao validar manifesto' },
      { status: 500 }
    )
  }

API de Envio de E-mails

// app/api/dashboard/trips/[id]/manifest/send-email/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { ManifestEmailService } from '@/lib/email/manifest-email'
import { format } from 'date-fns'
import { ptBR } from 'date-fns/locale'

export async function POST(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 })
    }

    const body = await req.json()
    const { type, recipients } = body // type: 'driver' | 'guide' | 'passengers'

    // Buscar dados do manifesto
    const manifest = await prisma.manifest.findUnique({
      where: { trip_id: params.id },
      include: {
        trip: {
          include: {
            reservations: {
              where: { status: 'confirmed' },
              include: {
                passengers: {
                  include: {
                    boarding_location: true,
                  },
                },
              },
            },
          },
        },
      },
    })

    if (!manifest || !manifest.pdf_url) {
      return NextResponse.json(
        { error: 'Manifesto ou PDF não encontrado' },
        { status: 404 }
      )
    }

    const passengers = manifest.trip.reservations.flatMap((r) => r.passengers)

    const emailData = {
      to: recipients,
      tripName: manifest.trip.name || manifest.trip.title,
      departureDate: format(
        manifest.trip.departure_date || new Date(),
        "dd/MM/yyyy 'às' HH:mm",
        { locale: ptBR }
      ),
      manifestNumber: manifest.manifest_number,
      pdfUrl: manifest.pdf_url,
      totalPassengers: passengers.length,
    }

    let success = false

    if (type === 'driver') {
      success = await ManifestEmailService.sendToDriver(emailData)
    } else if (type === 'guide') {
      success = await ManifestEmailService.sendToGuide(emailData)
    } else if (type === 'passengers') {
      const passengersData = passengers
        .filter((p) => p.email)
        .map((p) => ({
          email: p.email!,
          name: p.full_name,
          seatNumber: p.seat_number || 0,
          boardingLocation: p.boarding_location?.name || '',
          boardingTime: p.boarding_location?.departure_time || '',
        }))

      success = await ManifestEmailService.sendToPassengers(passengersData, {
        tripName: manifest.trip.name || manifest.trip.title,
        departureDate: emailData.departureDate,
        origin: manifest.trip.origin_city || '',
        destination: manifest.trip.destination,
      })
    }

    if (success) {
      return NextResponse.json({
        success: true,
        message: 'E-mails enviados com sucesso!',
      })
    } else {
      return NextResponse.json(
        { error: 'Erro ao enviar e-mails' },
        { status: 500 }
      )
    }
  } catch (error: any) {
    console.error('Erro ao enviar e-mails:', error)
    return NextResponse.json(
      { error: error.message || 'Erro ao enviar e-mails' },
      { status: 500 }
    )
  }
}
