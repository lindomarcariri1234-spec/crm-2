💳 API - UPGRADE DE PLANO
// app/api/subscriptions/upgrade/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { addDays, addMonths, addYears } from 'date-fns'

export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)

    if (!session || !['admin', 'owner'].includes(session.user.role)) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    const { planSlug, billingCycle } = await req.json()

    // Buscar plano
    const plan = await prisma.plan.findUnique({
      where: { slug: planSlug },
    })

    if (!plan) {
      return NextResponse.json({ error: 'Plano não encontrado' }, { status: 404 })
    }

    // Buscar tenant
    const tenant = await prisma.tenant.findUnique({
      where: { id: session.user.tenantId },
      include: { subscription: true },
    })

    if (!tenant) {
      return NextResponse.json({ error: 'Agência não encontrada' }, { status: 404 })
    }

    const now = new Date()

    // ═══════════════════════════════════════════════════════════
    // PLANO STARTER (GRATUITO)
    // ═══════════════════════════════════════════════════════════
    if (plan.plan_type === 'STARTER') {
      // Cancelar assinatura existente se houver
      if (tenant.subscription) {
        await prisma.subscription.update({
          where: { id: tenant.subscription.id },
          data: {
            status: 'CANCELED',
            canceled_at: now,
          },
        })
      }

      // Atualizar tenant
      await prisma.tenant.update({
        where: { id: tenant.id },
        data: {
          current_plan: 'STARTER',
          subscription_status: 'ACTIVE',
        },
      })

      return NextResponse.json({
        success: true,
        message: 'Plano Starter ativado com sucesso!',
        requiresPayment: false,
      })
    }

    // ═══════════════════════════════════════════════════════════
    // PLANOS PRO E ENTERPRISE (PAGOS)
    // ═══════════════════════════════════════════════════════════

    const amount =
      billingCycle === 'yearly'
        ? plan.price_yearly.toNumber()
        : plan.price_monthly.toNumber()

    const periodEnd =
      billingCycle === 'yearly' ? addYears(now, 1) : addMonths(now, 1)

    // Verificar se já tem assinatura
    let subscription = tenant.subscription

    if (subscription) {
      // Atualizar assinatura existente
      subscription = await prisma.subscription.update({
        where: { id: subscription.id },
        data: {
          plan_id: plan.id,
          billing_cycle: billingCycle,
          current_period_start: now,
          current_period_end: periodEnd,
          status: plan.trial_days > 0 ? 'TRIAL' : 'ACTIVE',
          trial_start: plan.trial_days > 0 ? now : null,
          trial_end: plan.trial_days > 0 ? addDays(now, plan.trial_days) : null,
        },
      })
    } else {
      // Criar nova assinatura
      subscription = await prisma.subscription.create({
        data: {
          tenant_id: tenant.id,
          plan_id: plan.id,
          billing_cycle: billingCycle,
          current_period_start: now,
          current_period_end: periodEnd,
          status: plan.trial_days > 0 ? 'TRIAL' : 'ACTIVE',
          trial_start: plan.trial_days > 0 ? now : null,
          trial_end: plan.trial_days > 0 ? addDays(now, plan.trial_days) : null,
        },
      })
    }

    // Gerar fatura
    const invoiceNumber = `INV-${Date.now()}-${tenant.id.slice(0, 8)}`
    const dueDate = plan.trial_days > 0 ? addDays(now, plan.trial_days) : addDays(now, 7)

    const invoice = await prisma.invoice.create({
      data: {
        subscription_id: subscription.id,
        tenant_id: tenant.id,
        invoice_number: invoiceNumber,
        amount: amount,
        tax_amount: 0,
        total_amount: amount,
        status: plan.trial_days > 0 ? 'PENDING' : 'PENDING',
        issue_date: now,
        due_date: dueDate,
        description: `Assinatura ${plan.name} - ${
          billingCycle === 'yearly' ? 'Anual' : 'Mensal'
        }`,
      },
    })

    // Atualizar tenant
    await prisma.tenant.update({
      where: { id: tenant.id },
      data: {
        current_plan: plan.plan_type,
        subscription_status: subscription.status,
      },
    })

    // Criar tracking de uso
    await prisma.usageTracking.create({
      data: {
        tenant_id: tenant.id,
        period_start: now,
        period_end: periodEnd,
        users_limit: plan.max_users,
        clients_limit: plan.max_clients,
        trips_limit: plan.max_trips,
        storage_limit_gb: plan.max_storage_gb,
      },
    })

    return NextResponse.json({
      success: true,
      message: plan.trial_days > 0
        ? `Trial de ${plan.trial_days} dias iniciado!`
        : 'Upgrade realizado! Aguardando pagamento.',
      requiresPayment: true,
      subscription,
      invoice: {
        id: invoice.id,
        number: invoiceNumber,
        amount: amount,
        dueDate: dueDate,
      },
    })
  } catch (error) {
    console.error('Erro ao fazer upgrade:', error)
    return NextResponse.json(
      { error: 'Erro ao fazer upgrade' },
      { status: 500 }
    )
  }
}

// app/api/invoices/[id]/pix/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import crypto from 'crypto'

export async function POST(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)

    if (!session) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    const invoice = await prisma.invoice.findUnique({
      where: { id: params.id },
      include: {
        tenant: true,
        subscription: {
          include: { plan: true },
        },
      },
    })

    if (!invoice) {
      return NextResponse.json({ error: 'Fatura não encontrada' }, { status: 404 })
    }

    if (invoice.tenant_id !== session.user.tenantId) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 403 })
    }

    // ═══════════════════════════════════════════════════════════
    // GERAR CÓDIGO PIX (Simulado - Integrar com API real)
    // ═══════════════════════════════════════════════════════════

    const pixKey = process.env.PIX_KEY || 'contato@visitecrm.com'
    const merchantName = 'VISITECRM LTDA'
    const merchantCity = 'SAO PAULO'
    const txid = crypto.randomBytes(16).toString('hex')

    // Formato EMV do PIX (simplificado)
    const pixCode = `00020126580014br.gov.bcb.pix0136${pixKey}520400005303986540${invoice.total_amount.toFixed(
      2
    )}5802BR5913${merchantName}6009${merchantCity}62070503${txid}6304`

    // Calcular CRC16
    const crc = calculateCRC16(pixCode.slice(0, -4))
    const finalPixCode = pixCode.slice(0, -4) + crc

    // URL do QR Code
    const qrCodeUrl = `https://api.qrserver.com/v1/create-qr-code/?size=300x300&data=${encodeURIComponent(
      finalPixCode
    )}`

    // Atualizar fatura
    await prisma.invoice.update({
      where: { id: params.id },
      data: {
        pix_code: finalPixCode,
        pix_qr_code_url: qrCodeUrl,
        pix_expires_at: new Date(Date.now() + 24 * 60 * 60 * 1000), // 24h
        payment_method: 'PIX',
        status: 'PROCESSING',
      },
    })

    return NextResponse.json({
      success: true,
      pix: {
        code: finalPixCode,
        qrCodeUrl,
        expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000),
      },
    })
  } catch (error) {
    console.error('Erro ao gerar PIX:', error)
    return NextResponse.json(
      { error: 'Erro ao gerar PIX' },
      { status: 500 }
    )
  }
}

// Função auxiliar para calcular CRC16
function calculateCRC16(str: string): string {
  let crc = 0xffff
  for (let i = 0; i < str.length; i++) {
    crc ^= str.charCodeAt(i) << 8
    for (let j = 0; j < 8; j++) {
      if ((crc & 0x8000) !== 0) {
        crc = (crc << 1) ^ 0x1021
      } else {
        crc = crc << 1
      }
    }
  }
  return (crc & 0xffff).toString(16).toUpperCase().padStart(4, '0')
}
